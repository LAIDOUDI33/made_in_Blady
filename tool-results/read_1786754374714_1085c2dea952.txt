   588→            </TabsTrigger>
   589→          </TabsList>
   590→
   591→          <TabsContent value="description" className="mt-6">

   593→              <CardContent className="p-6">

   597→                    dangerouslySetInnerHTML={{ __html: product.description }}
   598→                  />

   602→                  </p>

   604→              </CardContent>
   605→            </Card>
   606→          </TabsContent>
   607→
   608→          <TabsContent value="specifications" className="mt-6">

   612→                  <table className="w-full">
   613→                    <tbody>
   614→                      <tr className="border-b">
   615→                        <td className="py-3 px-4 font-medium bg-gray-50 w-1/3">SKU / Référence</td>
   616→                        <td className="py-3 px-4">{product.sku || "N/A"}</td>

   661→                          )}
   662→                        </td>
   663→                      </tr>
   664→                    </tbody>

   668→            </Card>
   669→          </TabsContent>
   670→
   671→          <TabsContent value="reviews" className="mt-6">
   672→            {/* Review Form Modal */}
   673→            <Dialog open={isReviewModalOpen} onOpenChange={setIsReviewModalOpen}>

   677→                </DialogHeader>
   678→                {product && (
   679→                  <ReviewForm
   680→                    targetId={product.id}
   681→                    reviewType="product"
   682→                    slug={slug}
   683→                    onSuccess={() => {
   684→                      setIsReviewModalOpen(false);
   685→                      setReviewSubmitted(true);
   686→                    }}
   687→                    onCancel={() => setIsReviewModalOpen(false)}
   688→                  />
   689→                )}
   690→              </DialogContent>
   691→            </Dialog>
   692→
   693→            {/* Review List */}
   694→            <ReviewList
   695→              apiUrl={`/api/products/${slug}/reviews?XTransformPort=3000`}
   696→              showStats={true}
   697→              allowWriteReview={!reviewSubmitted}
   698→              onWriteReview={() => setIsReviewModalOpen(true)}
   699→            />
   700→          </TabsContent>
   701→        </Tabs>
   702→
   703→        {/* Similar Products */}
   704→        {product.similarProducts && product.similarProducts.length > 0 && (
   705→          <section className="mt-12">
   706→            <div className="flex items-center justify-between mb-6">
   707→              <h2 className="text-xl font-semibold">Produits Similaires</h2>
   708→              <Link
   709→                href={`/categories/${product.category.slug}`}
   710→                className="text-green-600 hover:text-green-700 text-sm flex items-center gap-1"
   711→              >
   712→                Voir tous <ChevronRight className="h-4 w-4" />
   713→              </Link>
   714→            </div>
   715→
   716→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   717→              {product.similarProducts.map((similarProduct) => (
   718→                <ProductCard
   719→                  key={similarProduct.id}
   720→                  product={similarProduct as unknown as Product}
   721→                  onAddToRFQ={() => {}}
   722→                  onToggleFavorite={() => {}}
   723→                />
   724→              ))}
   725→            </div>
   726→          </section>
   727→        )}
   728→      </div>
   729→    </div>
   730→  );
   731→}
