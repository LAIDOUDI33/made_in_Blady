   200→export default function ShipmentTrackerScreen() {
   201→  const route = useRoute<ShipmentRouteProp>();
   202→  const navigation = useNavigation();
   203→  
   204→  const { trackingNumber: initialTrackingNumber } = route.params || {};
   205→  
   206→  // State
   207→  const [loading, setLoading] = useState(true);
   208→  const [refreshing, setRefreshing] = useState(false);
   209→  const [shipment, setShipment] = useState<ShipmentData | null>(null);
   210→  const [trackingInput, setTrackingInput] = useState(initialTrackingNumber || '');
   211→  const [showRatingModal, setShowRatingModal] = useState(false);
   212→  const [rating, setRating] = useState(0);
   213→  const [ratingComment, setRatingComment] = useState('');
   214→  const [categoryRatings, setCategoryRatings] = useState({
   215→    deliverySpeed: 5,
   216→    packageCondition: 5,
   217→    courierBehavior: 5,
   218→  });
   219→  const [submittingRating, setSubmittingRating] = useState(false);
   220→
   221→  // Load shipment data
   222→  const loadShipment = useCallback(async (trackingNum?: string) => {
   223→    const num = trackingNum || trackingInput;
   224→    
   225→    if (!num) return;
   226→
   227→    try {
   228→      const data = await apiService.trackShipment(num);
   229→      setShipment(data);
   230→      
   231→      // Subscribe to push notifications for this shipment
   232→      pushNotificationService.subscribeToTopic(`shipment_${data.id}`);
   233→    } catch (error) {
   234→      console.error('[ShipmentTrackerScreen] Error loading:', error);
   235→      Alert.alert('Erreur', 'Impossible de trouver cette expédition. Vérifiez le numéro de suivi.');
   236→    } finally {
   237→      setLoading(false);
   238→      setRefreshing(false);
   239→    }
   240→  }, [trackingInput]);
   241→
   242→  useEffect(() => {
   243→    if (initialTrackingNumber) {
   244→      loadShipment(initialTrackingNumber);
   245→    } else {
   246→      setLoading(false);
   247→    }
   248→  }, []);
   249→
   250→  // Pull to refresh
   251→  const onRefresh = async () => {
   252→    setRefreshing(true);
   253→    await loadShipment();
   254→  };
   255→
   256→  // Search for shipment
   257→  const handleSearch = () => {
   258→    if (!trackingInput.trim()) {
   259→      Alert.alert('Erreur', 'Veuillez entrer un numéro de suivi.');
   260→      return;
   261→    }
   262→    
   263→    setLoading(true);
   264→    loadShipment(trackingInput.trim());
   265→  };
   266→
   267→  // Contact driver
   268→  const handleContactDriver = (type: 'call' | 'message') => {
   269→    if (!shipment?.driverInfo) return;
   270→
   271→    if (type === 'call') {
   272→      Alert.alert(
   273→        'Appeler le livreur',
   274→        `Voulez-vous appeler ${shipment.driverInfo.name} au ${shipment.driverInfo.phone} ?`,
   275→        [
   276→          { text: 'Annuler', style: 'cancel' },
   277→          { 
   278→            text: 'Appeler', 
   279→            onPress: () => {
   280→              // In real app, would use Linking.openURL(`tel:${phone}`)
   281→              console.log('[ShipmentTracker] Calling driver');
   282→            }
   283→          }
   284→        ]
   285→      );
   286→    } else {
   287→      // Navigate to chat with driver
   288→      navigation.navigate('Chat' as any, {
   289→        conversationId: `delivery_${shipment.id}`,
   290→        userName: `${shipment.driverInfo.name} (Livreur)`,
   291→      });
   292→    }
   293→  };
   294→
   295→  // Submit rating
   296→  const handleSubmitRating = async () => {
   297→    if (rating === 0) {
   298→      Alert.alert('Erreur', 'Veuillez sélectionner une note.');
   299→      return;
   300→    }
   301→
   302→    setSubmittingRating(true);
   303→
   304→    try {
   305→      await apiService.rateDelivery(shipment!.id, {
   306→        rating,
   307→        comment: ratingComment || undefined,
   308→        categories: categoryRatings,
   309→      });
   310→
   311→      Alert.alert('Merci !', 'Votre avis a été enregistré.', [
   312→        { text: 'OK', onPress: () => setShowRatingModal(false) }
   313→      ]);
   314→    } catch (error) {
   315→      Alert.alert('Erreur', 'Impossible d\'enregistrer votre avis.');
   316→    } finally {
   317→      setSubmittingRating(false);
   318→    }
   319→  };
   320→
   321→  // Get current progress step
   322→  const getCurrentStepIndex = (): number => {
   323→    if (!shipment) return -1;
   324→    return STATUS_ORDER.indexOf(shipment.status);
   325→  };
   326→
   327→  // Render loading state
   328→  if (loading) {
   329→    return (
   330→      <View style={styles.loadingContainer}>
   331→        <ActivityIndicator size="large" color={Colors.primary} />
   332→        <Text style={styles.loadingText}>Recherche de l'expédition...</Text>
   333→      </View>
   334→    );
   335→  }
   336→
   337→  // Render search view (no shipment loaded)
   338→  if (!shipment) {
   339→    return (
   340→      <View style={styles.container}>
   341→        {/* Header */}
   342→        <View style={styles.header}>
   343→          <TouchableOpacity 
   344→            style={styles.backButton}
   345→            onPress={() => navigation.goBack()}
   346→          >
   347→            <Ionicons name="arrow-back" size={24} color={Colors.text} />
   348→          </TouchableOpacity>
   349→          <Text style={styles.headerTitle}>Suivi d'expédition</Text>
   350→          <View style={{ width: 40 }} />
   351→        </View>
   352→
   353→        <View style={styles.searchContainer}>
   354→          {/* Illustration */}
   355→          <View style={styles.searchIllustration}>
   356→            <Ionicons name="locate-outline" size={80} color={Colors.primary} />
   357→          </View>
   358→
   359→          <Text style={styles.searchTitle}>Suivez votre colis</Text>
   360→          <Text style={styles.searchSubtitle}>
   361→            Entrez votre numéro de suivi pour connaître l'état de votre livraison
   362→          </Text>
   363→
   364→          {/* Search Input */}
   365→          <View style={styles.searchInputContainer}>
   366→            <Ionicons name="search-outline" size={20} color={Colors.textTertiary} />
   367→            <TextInput
   368→              style={styles.searchInput}
   369→              value={trackingInput}
   370→              onChangeText={setTrackingInput}
   371→              placeholder="Numéro de suivi"
   372→              placeholderTextColor={Colors.textTertiary}
   373→              autoCapitalize="characters"
   374→              autoCorrect={false}
   375→              onSubmitEditing={handleSearch}
   376→            />
   377→            <TouchableOpacity style={styles.searchButton} onPress={handleSearch}>
   378→              <Ionicons name="arrow-forward" size={20} color={Colors.white} />
   379→            </TouchableOpacity>
   380→          </View>
   381→
   382→          {/* Recent searches would go here */}
   383→          <View style={styles.recentSection}>
   384→            <Text style={styles.recentTitle}>Recherches récentes</Text>
   385→            <Text style={styles.noRecentText}>Aucune recherche récente</Text>
   386→          </View>
   387→
   388→          {/* Help */}
   389→          <TouchableOpacity style={styles.helpCard}>
   390→            <Ionicons name="help-circle-outline" size={24} color={Colors.primary} />
   391→            <View style={styles.helpContent}>
   392→              <Text style={styles.helpTitle}>Besoin d'aide ?</Text>
   393→              <Text style={styles.helpDescription}>
   394→                Consultez notre FAQ ou contactez le support client
   395→              </Text>
   396→            </View>
   397→            <Ionicons name="chevron-forward" size={20} color={Colors.textTertiary} />
   398→          </TouchableOpacity>
   399→        </View>
   400→      </View>
   401→    );
   402→  }
   403→
   404→  // Main tracker view
   405→  const currentStep = getCurrentStepIndex();
   406→  const statusConfig = SHIPMENT_STATUS_CONFIG[shipment.status];
   407→  const methodConfig = SHIPPING_METHOD_CONFIG[shipment.shippingMethod];
   408→
   409→  return (
   410→    <View style={styles.container}>
   411→      {/* Header */}
   412→      <View style={styles.header}>
   413→        <TouchableOpacity 
   414→          style={styles.backButton}
   415→          onPress={() => navigation.goBack()}
   416→        >
   417→          <Ionicons name="arrow-back" size={24} color={Colors.text} />
   418→        </TouchableOpacity>
   419→        <Text style={styles.headerTitle}>Suivi d'expédition</Text>
   420→        <TouchableOpacity style={styles.shareButton}>
   421→          <Ionicons name="share-social-outline" size={20} color={Colors.text} />
   422→        </TouchableOpacity>
   423→      </View>
   424→
   425→      <ScrollView
   426→        style={styles.scrollView}
   427→        refreshControl={
   428→          <RefreshControl refreshing={refreshing} onRefresh={onRefresh} />
   429→        }
   430→        showsVerticalScrollIndicator={false}
   431→      >
   432→        {/* Status Card */}
   433→        <View style={[styles.statusCard, { borderLeftColor: statusConfig.color }]}>
   434→          <View style={styles.statusHeader}>
   435→            <View style={[styles.statusIconContainer, { backgroundColor: statusConfig.color + '15' }]}>
   436→              <Ionicons name={statusConfig.icon as any} size={28} color={statusConfig.color} />
   437→            </View>
   438→            
   439→            <View style={styles.statusInfo}>
   440→              <Text style={[styles.statusText, { color: statusConfig.color }]}>
   441→                {statusConfig.label}
   442→              </Text>
   443→              <Text style={styles.statusDescription}>{statusConfig.description}</Text>
   444→            </View>
   445→          </View>
   446→
   447→          {/* Tracking Number */}
   448→          <View style={styles.trackingNumberContainer}>
   449→            <Text style={styles.trackingNumberLabel}>N° de suivi</Text>
   450→            <View style={styles.trackingNumberRow}>
   451→              <Text style={styles.trackingNumberValue}>{shipment.trackingNumber}</Text>
   452→              <TouchableOpacity onPress={() => {
   453→                // Copy to clipboard
   454→                Alert.alert('Copié', 'Le numéro de suivi a été copié.');
   455→              }}>
   456→                <Ionicons name="copy-outline" size={18} color={Colors.primary} />
   457→              </TouchableOpacity>
   458→            </View>
   459→          </View>
   460→
   461→          {/* Delivery Estimate */}
   462→          {(shipment.estimatedDeliveryStart || shipment.estimatedDeliveryEnd) && (
   463→            <View style={styles.deliveryEstimate}>
   464→              <Ionicons name="calendar-outline" size={16} color={Colors.textSecondary} />
   465→              <Text style={styles.deliveryEstimateText}>
   466→                Livraison estimée:{' '}
   467→                {shipment.estimatedDeliveryStart?.toLocaleDateString('fr-FR', {
   468→                  day: 'numeric',
   469→                  month: 'short',
   470→                })}
   471→                {shipment.estimatedDeliveryEnd && shipment.estimatedDeliveryStart?.getTime() !== shipment.estimatedDeliveryEnd?.getTime()
   472→                  ? ` - ${shipment.estimatedDeliveryEnd.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })}`
   473→                  : ''
   474→                }
   475→              </Text>
   476→            </View>
   477→          )}
   478→        </View>
   479→
   480→        {/* Progress Steps */}
   481→        <View style={styles.progressSection}>
   482→          <Text style={styles.sectionTitle}>Progression</Text>
   483→          
   484→          <View style={styles.stepsContainer}>
   485→            {STATUS_ORDER.map((status, index) => {
   486→              const config = SHIPMENT_STATUS_CONFIG[status];
   487→              const isCompleted = index <= currentStep;
   488→              const isCurrent = index === currentStep;
   489→              
   490→              return (
   491→                <React.Fragment key={status}>
   492→                  {/* Step Circle */}
   493→                  <View style={styles.stepItem}>
   494→                    <View style={[
   495→                      styles.stepCircle,
   496→                      isCompleted && styles.stepCircleCompleted,
   497→                      isCurrent && styles.stepCircleCurrent,
   498→                    ]}>
   499→                      {isCompleted ? (
   500→                        <Ionicons name="checkmark" size={14} color={Colors.white} />
   501→                      ) : (
   502→                        <View style={styles.stepCircleInner} />
   503→                      )}
   504→                    </View>
   505→                    
   506→                    <Text style={[
   507→                      styles.stepLabel,
   508→                      isCompleted && styles.stepLabelCompleted,
   509→                      isCurrent && styles.stepLabelCurrent,
   510→                    ]}>
   511→                      {config.label}
   512→                    </Text>
   513→                  </View>
   514→
   515→                  {/* Connector Line */}
   516→                  {index < STATUS_ORDER.length - 1 && (
   517→                    <View style={[
   518→                      styles.stepConnector,
   519→                      index < currentStep && styles.stepConnectorCompleted,
   520→                    ]} />
   521→                  )}
   522→                </React.Fragment>
   523→              );
   524→            })}
   525→          </View>
   526→        </View>
   527→
   528→        {/* Map Placeholder (would be real map in production) */}
   529→        <View style={styles.mapContainer}>
   530→          <View style={styles.mapPlaceholder}>
   531→            <Ionicons name="map-outline" size={48} color={Colors.textTertiary} />
   532→            <Text style={styles.mapPlaceholderText}>Carte de suivi en temps réel</Text>
   533→            
   534→            {/* Simulated route visualization */}
   535→            <View style={styles.routeVisualization}>
   536→              <View style={styles.routeOrigin}>
   537→                <View style={styles.routeDotOrigin} />
   538→                <Text style={styles.routeOriginLabel}>Origine</Text>
   539→              </View>
   540→              
   541→              <View style={styles.routeLine}>
   542→                {[...Array(5)].map((_, i) => (

   546→                      styles.routeCheckpoint,
   547→                      i <= Math.floor(currentStep / 2) && styles.routeCheckpointActive
   548→                    ]} 
   549→                  />
   550→                ))}
   551→              </View>
   552→              
   553→              <View style={styles.routeDestination}>
   554→                <View style={[
   555→                  styles.routeDotDestination,
   556→                  currentStep >= STATUS_ORDER.length - 2 && styles.routeDotDestinationActive
   557→                ]} />
   558→                <Text style={styles.routeDestinationLabel}>Destination</Text>
   559→              </View>
   560→            </View>
   561→          </View>
   562→
   563→          {/* Address Cards */}
   564→          <View style={styles.addressesContainer}>
   565→            <View style={styles.addressCard}>
   566→              <View style={styles.addressHeader}>
   567→                <View style={[styles.addressIcon, { backgroundColor: Colors.primaryLight }]}>
   568→                  <Ionicons name="arrow-up" size={16} color={Colors.primary} />
   569→                </View>
   570→                <Text style={styles.addressType}>Expéditeur</Text>
   571→              </View>
   572→              <Text style={styles.addressName}>{shipment.originAddress.name}</Text>
   573→              <Text style={styles.addressDetail}>
   574→                {shipment.originAddress.street}, {shipment.originAddress.city}
   575→              </Text>
   576→              <Text style={styles.addressDetail}>
   577→                {shipment.originAddress.wilaya}, {shipment.originAddress.postalCode}
   578→              </Text>
   579→            </View>
   580→
   581→            <View style={styles.addressCard}>
   582→              <View style={styles.addressHeader}>
   583→                <View style={[styles.addressIcon, { backgroundColor: Colors.successLight }]}>
   584→                  <Ionicons name="arrow-down" size={16} color={Colors.success} />
   585→                </View>
   586→                <Text style={styles.addressType}>Destinataire</Text>
   587→              </View>
   588→              <Text style={styles.addressName}>{shipment.destinationAddress.name}</Text>
   589→              <Text style={styles.addressDetail}>
   590→                {shipment.destinationAddress.street}, {shipment.destinationAddress.city}
   591→              </Text>

   593→                {shipment.destinationAddress.wilaya}, {shipment.destinationAddress.postalCode}
