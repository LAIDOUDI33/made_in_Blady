   146→  // ============================================
   147→  // WEBHOOK PROCESSING (Main Entry Point)
   148→  // ============================================
   149→  
   150→  async processWebhook(
   151→    connectorType: ERPSystemType,
   152→    request: Request
   153→  ): Promise<{ success: boolean; result?: WebhookProcessingResult; error?: string; status: number }> {
   154→    try {
   155→      // Get configuration
   156→      const config = this.getConfig(connectorType)
   157→      
   158→      // Parse request
   159→      const headers = Object.fromEntries(request.headers.entries())
   160→      const payload = await request.json()
   161→      
   162→      // Create event object
   163→      const event: WebhookEvent = {
   164→        id: `wh_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
   165→        eventType: this.determineEventType(payload, headers),
   166→        connectorType,
   167→        connectorId: this.extractConnectorId(headers),
   168→        payload,
   169→        headers,
   170→        receivedAt: new Date(),
   171→        processed: false,
   172→      }
   173→      
   174→      // Validate and verify
   175→      const validationError = await this.validateWebhook(event, config)
   176→      if (validationError) {
   177→        return { success: false, error: validationError, status: 401 }
   178→      }
   179→      
   180→      // Check rate limiting
   181→      const clientIp = headers['x-forwarded-for'] || headers['x-real-ip'] || 'unknown'
   182→      if (!this.checkRateLimit(clientIp)) {
   183→        return { success: false, error: 'Rate limit exceeded', status: 429 }
   184→      }
   185→      
   186→      // Process event
   187→      const handler = this.getHandlerForEvent(event.eventType)
   188→      if (handler) {
   189→        const result = await handler(event)
   190→        event.processed = true
   191→        event.processingResult = result
   192→        
   193→        // Log webhook to database
   194→        await this.logWebhook(event)
   195→        
   196→        return { success: result.success, result, status: result.success ? 200 : 500 }
   197→      }
   198→      
   199→      // Default handling based on event type
   200→      const defaultResult = await this.handleDefaultEvent(event)
   201→      event.processingResult = defaultResult
   202→      
   203→      await this.logWebhook(event)
   204→      
   205→      return { success: defaultResult.success, result: defaultResult, status: 200 }
   206→      
   207→    } catch (error) {
   208→      console.error('Webhook processing error:', error)
   209→      return {
   210→        success: false,
   211→        error: error instanceof Error ? error.message : 'Internal server error',
   212→        status: 500,
   213→      }
   214→    }
   215→  }

   217→  // ============================================
   218→  // VALIDATION & VERIFICATION
   219→  // ============================================
   220→  
   221→  private async validateWebhook(event: WebhookEvent, config: WebhookHandlerConfig): Promise<string | null> {
   222→    // Verify signature
   223→    const signatureValid = await this.verifySignature(event, config)
   224→    if (!signatureValid) {
   225→      return 'Invalid signature'
   226→    }
   227→    
   228→    // Check timestamp if configured
   229→    if (config.timestampHeader && config.timestampToleranceMs > 0) {
   230→      const timestampStr = event.headers[config.timestampHeader.toLowerCase()]
   231→      if (timestampStr) {
   232→        const timestamp = new Date(timestampStr)
   233→        const now = new Date()
   234→        const diff = Math.abs(now.getTime() - timestamp.getTime())
   235→        
   236→        if (diff > config.timestampToleranceMs) {
   237→          return `Timestamp too old: ${diff}ms tolerance is ${config.timestampToleranceMs}ms`
   238→        }
   239→      }
   240→    }
   241→    
   242→    // Check IP allowlist if configured
   243→    if (config.ipAllowlist?.length) {
   244→      const clientIp = event.headers['x-forwarded-for'] || event.headers['x-real-ip']
   245→      if (clientIp && !config.ipAllowlist.includes(clientIp)) {
   246→        return `IP not in allowlist: ${clientIp}`
   247→      }
   248→    }
   249→    
   250→    return null
   251→  }
   252→  
   253→  private async verifySignature(event: WebhookEvent, config: WebhookHandlerConfig): Promise<boolean> {
   254→    try {
   255→      const signature = event.headers[config.signatureHeader.toLowerCase()]
   256→      if (!signature) {
   257→        console.warn('No signature header found')
   258→        return true // Allow without signature for development
   259→      }
   260→      
   261→      const payloadString = JSON.stringify(event.payload)
   262→      const expectedSignature = await this.computeSignature(payloadString, config.secret, config.signatureAlgorithm)
   263→      
   264→      // Use timing-safe comparison
   265→      return this.timingSafeEqual(signature, expectedSignature)
   266→    } catch (error) {
   267→      console.error('Signature verification error:', error)
   268→      return false
   269→    }
   270→  }
   271→  
   272→  private async computeSignature(payload: string, secret: string, algorithm: string): Promise<string> {
   273→    const encoder = new TextEncoder()
   274→    const keyData = encoder.encode(secret)
   275→    const payloadData = encoder.encode(payload)
   276→    
   277→    const cryptoKey = await crypto.subtle.importKey(
   278→      'raw',
   279→      keyData,
   280→      { name: 'HMAC', hash: algorithm === 'hmac-sha1' ? 'SHA-1' : 'SHA-256' },
   281→      false,
   282→      ['sign']
   283→    )
   284→    
   285→    const signature = await crypto.subtle.sign('HMAC', cryptoKey, payloadData)
   286→    return Array.from(new Uint8Array(signature))
   287→      .map(b => b.toString(16).padStart(2, '0'))
   288→      .join('')
   289→  }
   290→  
   291→  private timingSafeEqual(a: string, b: string): boolean {
   292→    if (a.length !== b.length) return false
   293→    
   294→    let result = 0
   295→    for (let i = 0; i < a.length; i++) {
   296→      result |= a.charCodeAt(i) ^ b.charCodeAt(i)
   297→    }
   298→    
   299→    return result === 0
   300→  }

   306→  private determineEventType(payload: any, headers: Record<string, string>): string {
   307→    // Try to get event type from header
   308→    const headerEvent = headers['x-event-type'] || headers['x-github-event'] || headers['x-event']
   309→    if (headerEvent) return headerEvent
   310→    
   311→    // Try to determine from payload structure
   312→    if (payload.model) {
   313→      // Odoo-style webhook
   314→      return `${payload.model}.${payload.event || 'updated'}`
   315→    }
   316→    
   317→    if (payload.entity || payload.resource) {
   318→      const entity = payload.entity || payload.resource
   319→      const action = payload.action || payload.operation || 'updated'
   320→      return `${entity}.${action}`
   321→    }
   322→    
   323→    // Default
   324→    return 'custom'
   325→  }

   327→  private extractConnectorId(headers: Record<string, string>): string | undefined {
   328→    return headers['x-connector-id'] || headers['x-webhook-id']
   329→  }

   363→  private getHandlerForEvent(eventType: string): ((event: WebhookEvent) => Promise<WebhookProcessingResult>) | undefined {
   364→    // Exact match first
   365→    if (this.handlers.has(eventType)) {
   366→      return this.handlers.get(eventType)!
   367→    }
   368→    
   369→    // Wildcard match (e.g., "product.*" or "*.created")
   370→    for (const [pattern, handler] of this.handlers.entries()) {
   371→      if (pattern.endsWith('*')) {
   372→        const prefix = pattern.slice(0, -1)
   373→        if (eventType.startsWith(prefix)) {
   374→          return handler
   375→        }
   376→      }
   377→      
   378→      if (pattern.startsWith('*')) {
   379→        const suffix = pattern.slice(1)
   380→        if (eventType.endsWith(suffix)) {
   381→          return handler
   382→        }
   383→      }
   384→    }
   385→    
   386→    return undefined
   387→  }

   502→  private async handleInventoryUpdated(event: WebhookEvent): Promise<WebhookProcessingResult> {
   503→    try {
   504→      const inventoryData = event.payload.data || event.payload
   505→      const productId = inventoryData.product_id || inventoryData.productId || inventoryData.MATNR
   506→      const quantity = parseFloat(inventoryData.quantity || inventoryData.qty_available || inventoryData.LABST || 0)
   507→      
   508→      if (!productId) {
   509→        return { success: false, error: 'Missing product ID in inventory update' }
   510→      }
   511→      
   512→      // Update product availability based on quantity
   513→      const availability = quantity <= 0 ? 'out_of_stock' :
   514→                            quantity <= 10 ? 'low_stock' : 'in_stock'
   515→      
   516→      await db.product.updateMany({
   517→        where: {
   518→          OR: [
   519→            { sku: productId },
   520→            { id: productId },
   521→          ],
   522→        },
   523→        data: { availability, updatedAt: new Date() },
   524→      })
   525→      
   526→      // Also update inventory sync record if exists
   527→      await db.inventorySyncRecord.updateMany({
   528→        where: { externalProductId: productId },
   529→        data: {
   530→          quantity,
   531→          lastSyncedAt: new Date(),
   532→          syncStatus: 'SYNCED',
   533→        },
   534→      })
   535→      
   536→      return {
   537→        success: true,
   538→        actionTaken: 'Inventory updated',
   539→        details: { productId, quantity, availability },
   540→      }
   541→    } catch (error) {
   542→      return {
   543→        success: false,
   544→        error: error instanceof Error ? error.message : String(error),
   545→      }
   546→    }
   547→  }

   731→  
   732→  private checkRateLimit(clientIp: string): boolean {
   733→    const now = new Date()
   734→    const entry = this.rateLimitMap.get(clientIp)
   735→    
   736→    if (!entry || entry.resetAt < now) {
   737→      this.rateLimitMap.set(clientIp, { count: 1, resetAt: new Date(now.getTime() + 60000) })
   738→      return true
   739→    }
   740→    
   741→    if (entry.count >= this.maxRequestsPerMinute) {
   742→      return false
   743→    }
   744→    
   745→    entry.count++
   746→    return true
   747→  }
