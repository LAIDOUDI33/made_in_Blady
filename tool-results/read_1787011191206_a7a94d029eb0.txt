   801→function escapeHtml(text: string): string {
   802→  return text
   803→    .replace(/&/g, '&amp;')
   804→    .replace(/</g, '&lt;')
   805→    .replace(/>/g, '&gt;')
   806→    .replace(/"/g, '&quot;')
   807→    .replace(/'/g, '&#039;');
   808→}
