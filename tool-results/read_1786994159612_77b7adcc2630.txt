   183→
   184→export default function InspectionBookingScreen() {
   185→  const route = useRoute<InspectionRouteProp>();
   186→  const navigation = useNavigation();
   187→  
   188→  const { productId } = route.params || {};
   189→  
   190→  // State
   191→  const [loading, setLoading] = useState(true);
   192→  const [services, setServices] = useState<InspectionService[]>([]);
   193→  const [selectedService, setSelectedService] = useState<InspectionService | null>(null);
   194→  const [selectedDate, setSelectedDate] = useState<Date | null>(null);
   195→  const [selectedTimeSlot, setSelectedTimeSlot] = useState<string | null>(null);
   196→  const [step, setStep] = useState(1); // 1: Select type, 2: Details, 3: Confirm, 4: Result
   197→  
   198→  // Form state
   199→  const [street, setStreet] = useState('');
   200→  const [city, setCity] = useState('');
   201→  const [wilaya, setWilaya] = useState('');
   202→  const [postalCode, setPostalCode] = useState('');
   203→  const [contactPerson, setContactPerson] = useState('');
   204→  const [phone, setPhone] = useState('');
   205→  const [specialInstructions, setSpecialInstructions] = useState('');
   206→  const [isUrgent, setIsUrgent] = useState(false);
   207→  
   208→  // Booking & Result state
   209→  const [currentBooking, setCurrentBooking] = useState<BookingInfo | null>(null);
   210→  const [submitting, setSubmitting] = useState(false);
   211→  const [showCalendarModal, setShowCalendarModal] = useState(false);
   212→  const [showWilayaModal, setShowWilayaModal] = useState(false);
   213→
   214→  // Load services
   215→  const loadServices = useCallback(async () => {
   216→    try {
   217→      const data = await apiService.getInspectionServices();
   218→      setServices(data);
   219→    } catch (error) {
   220→      console.error('[InspectionBookingScreen] Error loading services:', error);
   221→      Alert.alert('Erreur', 'Impossible de charger les services d\'inspection.');
   222→    } finally {
   223→      setLoading(false);
   224→    }
   225→  }, []);
   226→
   227→  useEffect(() => {
   228→    loadServices();
   229→  }, [loadServices]);
   230→
   231→  // Select service and move to next step
   232→  const handleSelectService = (service: InspectionService) => {
   233→    setSelectedService(service);
   234→    setStep(2);
   235→  };
   236→
   237→  // Generate calendar days
   238→  const generateCalendarDays = () => {
   239→    const days = [];
   240→    const today = new Date();
   241→    
   242→    for (let i = 1; i <= 30; i++) {
   243→      const date = new Date(today);
   244→      date.setDate(today.getDate() + i);
   245→      
   246→      // Skip weekends (optional)
   247→      // if (date.getDay() === 0 || date.getDay() === 6) continue;
   248→      
   249→      days.push(date);
   250→    }
   251→    
   252→    return days;
   253→  };
   254→
   255→  // Handle date selection
   256→  const handleDateSelect = (date: Date) => {
   257→    setSelectedDate(date);
   258→    setShowCalendarModal(false);
   259→  };
   260→
   261→  // Validate form
   262→  const validateForm = (): boolean => {
   263→    if (!street.trim()) {
   264→      Alert.alert('Erreur', 'Veuillez entrer l\'adresse.');
   265→      return false;
   266→    }
   267→    if (!city.trim()) {
   268→      Alert.alert('Erreur', 'Veuillez entrer la ville.');
   269→      return false;
   270→    }
   271→    if (!wilaya) {
   272→      Alert.alert('Erreur', 'Veuillez sélectionner la wilaya.');
   273→      return false;
   274→    }
   275→    if (!contactPerson.trim()) {
   276→      Alert.alert('Erreur', 'Veuillez entrer le nom du contact.');
   277→      return false;
   278→    }
   279→    if (!phone.trim()) {
   280→      Alert.alert('Erreur', 'Veuillez entrer un numéro de téléphone.');
   281→      return false;
   282→    }
   283→    if (!selectedDate) {
   284→      Alert.alert('Erreur', 'Veuillez sélectionner une date.');
   285→      return false;
   286→    }
   287→    if (!selectedTimeSlot) {
   288→      Alert.alert('Erreur', 'Veuillez sélectionner un créneau horaire.');
   289→      return false;
   290→    }
   291→    return true;
   292→  };
   293→
   294→  // Submit booking
   295→  const handleSubmitBooking = async () => {
   296→    if (!validateForm() || !selectedService) return;
   297→
   298→    setSubmitting(true);
   299→
   300→    try {
   301→      const booking = await apiService.bookInspection({
   302→        productId,
   303→        serviceId: selectedService.id,
   304→        inspectionType: selectedService.type,
   305→        preferredDate: selectedDate!,
   306→        address: {
   307→          street,
   308→          city,
   309→          wilaya,
   310→          postalCode,
   311→          contactPerson,
   312→          phone,
   313→        },
   314→        specialInstructions: specialInstructions || undefined,
   315→        isUrgent,
   316→      });
   317→
   318→      setCurrentBooking(booking);
   319→      setStep(3); // Confirmation step
   320→      
   321→      Alert.alert(
   322→        'Réservation confirmée !',
   323→        `Votre réservation ${booking.bookingNumber} a été enregistrée.`,
   324→        [{ text: 'OK' }]
   325→      );
   326→    } catch (error) {
   327→      console.error('[InspectionBookingScreen] Error booking:', error);
   328→      Alert.alert('Erreur', 'Impossible de finaliser la réservation. Veuillez réessayer.');
   329→    } finally {
   330→      setSubmitting(false);
   331→    }
   332→  };
   333→
   334→  // Calculate total price
   335→  const calculateTotal = (): number => {
   336→    if (!selectedService) return 0;
   337→    
   338→    let total = selectedService.basePrice;
   339→    
   340→    if (isUrgent) {
   341→      total *= 1.5; // 50% surcharge for urgent
   342→    }
   343→    
   344→    return Math.round(total * 100) / 100;
   345→  };
   346→
   347→  // Render loading
   348→  if (loading) {
   349→    return (
   350→      <View style={styles.loadingContainer}>
   351→        <ActivityIndicator size="large" color={Colors.primary} />
   352→        <Text style={styles.loadingText}>Chargement...</Text>
   353→      </View>
   354→    );
   355→  }
   356→
   357→  // Step 1: Select Service Type
   358→  if (step === 1) {
   359→    return (
   360→      <View style={styles.container}>
   361→        {/* Header */}
   362→        <View style={styles.header}>
   363→          <TouchableOpacity 
   364→            style={styles.backButton}
   365→            onPress={() => navigation.goBack()}
   366→          >
   367→            <Ionicons name="arrow-back" size={24} color={Colors.text} />
   368→          </TouchableOpacity>
   369→          <Text style={styles.headerTitle}>Réserver une inspection</Text>
   370→          <View style={{ width: 40 }} />
   371→        </View>
   372→
   373→        <ScrollView 
   374→          style={styles.scrollView}
   375→          showsVerticalScrollIndicator={false}
   376→        >
   377→          {/* Info Banner */}
   378→          <View style={styles.infoBanner}>
   379→            <Ionicons name="shield-checkmark" size={24} color={Colors.primary} />
   380→            <View style={styles.infoBannerContent}>
   381→              <Text style={styles.infoBannerTitle}>Inspection professionnelle</Text>
   382→              <Text style={styles.infoBannerText}>
   383→                Nos inspecteurs certifiés vérifient la qualité de vos produits avant expédition.
   384→              </Text>
   385→            </View>
   386→          </View>
   387→
   388→          {/* Service Types */}
   389→          <View style={styles.section}>
   390→            <Text style={styles.sectionTitle}>Choisissez le type d'inspection</Text>
   391→            
   392→            {services.map(service => {
   393→              const config = INSPECTION_TYPES[service.type];
   394→              
   395→              return (
   396→                <TouchableOpacity
   397→                  key={service.id}
   398→                  style={styles.serviceCard}
   399→                  onPress={() => handleSelectService(service)}
   400→                  activeOpacity={0.7}
   401→                >
   402→                  <View style={[styles.serviceIconContainer, { backgroundColor: config.color + '15' }]}>
   403→                    <Ionicons name={config.icon as any} size={28} color={config.color} />
   404→                  </View>
   405→                  
   406→                  <View style={styles.serviceContent}>
   407→                    <Text style={styles.serviceName}>{config.label}</Text>
   408→                    <Text style={styles.serviceDescription} numberOfLines={2}>
   409→                      {config.description}
   410→                    </Text>
   411→                    
   412→                    <View style={styles.serviceMeta}>
   413→                      <View style={styles.metaItem}>
   414→                        <Ionicons name="time-outline" size={14} color={Colors.textSecondary} />
   415→                        <Text style={styles.metaText}>
   416→                          ~{service.typicalLeadTime} jours
   417→                        </Text>
   418→                      </View>
   419→                      
   420→                      {service.includesPhotoReport && (
   421→                        <View style={[styles.metaItem, styles.metaItemSuccess]}>
   422→                          <Ionicons name="camera-outline" size={14} color={Colors.success} />
   423→                          <Text style={[styles.metaText, styles.metaTextSuccess]}>
   424→                            Photos incluses
   425→                          </Text>
   426→                        </View>
   427→                      )}
   428→                    </View>
   429→                  </View>
   430→
   431→                  <View style={styles.priceContainer}>
   432→                    <Text style={styles.priceValue}>
   433→                      {service.basePrice.toLocaleString('fr-FR')} {service.currency}
   434→                    </Text>
   435→                    <Ionicons name="chevron-forward" size={20} color={Colors.textTertiary} />
   436→                  </View>
   437→                </TouchableOpacity>
   438→              );
   439→            })}
   440→          </View>
   441→
   442→          {/* How it works */}
   443→          <View style={styles.howItWorksSection}>
   444→            <Text style={styles.sectionTitle}>Comment ça marche ?</Text>
   445→            
   446→            <View style={styles.stepsList}>
   447→              {[
   448→                { step: 1, title: 'Choisissez', desc: 'Sélectionnez le type d\'inspection' },
   449→                { step: 2, title: 'Planifiez', desc: 'Définissez la date et le lieu' },
   450→                { step: 3, title: 'Payez', desc: 'Effectuez le paiement sécurisé' },
   451→                { step: 4, title: 'Recevez', desc: 'Obtenez votre rapport détaillé' },
   452→              ].map(item => (
   453→                <View key={item.step} style={styles.stepItem}>
   454→                  <View style={styles.stepNumber}>
   455→                    <Text style={styles.stepNumberText}>{item.step}</Text>
   456→                  </View>
   457→                  <View style={styles.stepContent}>
   458→                    <Text style={styles.stepTitle}>{item.title}</Text>
   459→                    <Text style={styles.stepDescription}>{item.desc}</Text>
   460→                  </View>
   461→                </View>
   462→              ))}
   463→            </View>
   464→          </View>
   465→        </ScrollView>
   466→      </View>
   467→    );
   468→  }
   469→
   470→  // Step 2: Details Form
   471→  if (step === 2) {
   472→    return (
   473→      <View style={styles.container}>
   474→        {/* Header */}
   475→        <View style={styles.header}>
   476→          <TouchableOpacity 
   477→            style={styles.backButton}
   478→            onPress={() => setStep(1)}
   479→          >
   480→            <Ionicons name="arrow-back" size={24} color={Colors.text} />
   481→          </TouchableOpacity>
   482→          <Text style={styles.headerTitle}>Détails de la réservation</Text>
   483→          <View style={{ width: 40 }} />
   484→        </View>
   485→
   486→        <ScrollView 
   487→          style={styles.scrollView}
   488→          showsVerticalScrollIndicator={false}
   489→        >
   490→          {/* Selected Service Summary */}
   491→          {selectedService && (
   492→            <View style={styles.selectedServiceBanner}>
   493→              <View style={[
   494→                styles.selectedServiceIcon, 
   495→                { backgroundColor: INSPECTION_TYPES[selectedService.type].color + '15' }
   496→              ]}>
   497→                <Ionicons 
   498→                  name={INSPECTION_TYPES[selectedService.type].icon as any} 
   499→                  size={20} 
   500→                  color={INSPECTION_TYPES[selectedService.type].color} 
   501→                />
   502→              </View>
   503→              <View style={styles.selectedServiceInfo}>
   504→                <Text style={styles.selectedServiceName}>
   505→                  {INSPECTION_TYPES[selectedService.type].label}
   506→                </Text>
   507→                <Text style={styles.selectedServicePrice}>
   508→                  À partir de {selectedService.basePrice.toLocaleString('fr-FR')} {selectedService.currency}
   509→                </Text>
   510→              </View>
   511→              <TouchableOpacity onPress={() => setStep(1)}>
   512→                <Ionicons name="create-outline" size={20} color={Colors.primary} />
   513→              </TouchableOpacity>
   514→            </View>
   515→          )}
   516→
   517→          {/* Address Section */}
   518→          <View style={styles.formSection}>
   519→            <Text style={styles.formSectionTitle}>Adresse d'inspection</Text>
   520→            
   521→            <TextInput
   522→              style={styles.textInput}
   523→              value={street}
   524→              onChangeText={setStreet}
   525→              placeholder="Adresse *"
   526→              placeholderTextColor={Colors.textTertiary}
   527→            />
   528→
   529→            <View style={styles.rowInputs}>
   530→              <TextInput
   531→                style={[styles.textInput, styles.flexInput]}
   532→                value={city}
   533→                onChangeText={setCity}
   534→                placeholder="Ville *"
   535→                placeholderTextColor={Colors.textTertiary}
   536→              />
   537→              
   538→              <TouchableOpacity
   539→                style={[styles.textInput, styles.selectInput]}
   540→                onPress={() => setShowWilayaModal(true)}
   541→              >
   542→                <Text style={[styles.selectText, wilaya ? styles.selectTextFilled : {}]}>
   543→                  {wilaya || 'Wilaya *'}
   544→                </Text>
   545→                <Ionicons name="chevron-down" size={16} color={Colors.textTertiary} />
   546→              </TouchableOpacity>
   547→            </View>
   548→
   549→            <TextInput
   550→              style={styles.textInput}
   551→              value={postalCode}
   552→              onChangeText={setPostalCode}
   553→              placeholder="Code postal"
   554→              placeholderTextColor={Colors.textTertiary}
   555→              keyboardType="number-pad"
   556→            />
   557→          </View>
   558→
   559→          {/* Contact Section */}
   560→          <View style={styles.formSection}>
   561→            <Text style={styles.formSectionTitle}>Contact sur place</Text>
   562→            
   563→            <TextInput
   564→              style={styles.textInput}
   565→              value={contactPerson}
   566→              onChangeText={setContactPerson}
   567→              placeholder="Nom du contact *"
   568→              placeholderTextColor={Colors.textTertiary}
   569→            />
   570→
   571→            <TextInput
   572→              style={styles.textInput}
   573→              value={phone}
   574→              onChangeText={setPhone}
   575→              placeholder="Téléphone *"
   576→              placeholderTextColor={Colors.textTertiary}
   577→              keyboardType="phone-pad"
   578→            />
   579→          </View>
   580→
   581→          {/* Date & Time */}
   582→          <View style={styles.formSection}>
   583→            <Text style={styles.formSectionTitle}>Date et heure préférées</Text>

   587→              onPress={() => setShowCalendarModal(true)}
   588→            >

   590→              <Text style={styles.dateText}>
   591→                {selectedDate

   604→            <View style={styles.timeSlotsContainer}>

   609→                    styles.timeSlot,

   948→// Helper functions
   949→function getStatusColor(status: InspectionStatus): string {
   950→  switch (status) {
   951→    case 'scheduled': return Colors.info;
   952→    case 'in_progress': return Colors.warning;
   953→    case 'completed':
   954→    case 'pass': return Colors.success;
   955→    case 'failed':
   956→    case 'fail': return Colors.error;
   957→    case 'cancelled': return Colors.textTertiary;
   958→    default: return Colors.textTertiary;
   959→  }
   960→}
   961→
   962→function getStatusIcon(status: InspectionStatus): string {
   963→  switch (status) {
   964→    case 'pending': return 'time-outline';
   965→    case 'scheduled': return 'calendar-outline';
   966→    case 'in_progress': return 'sync-outline';
   967→    case 'completed':
   968→    case 'pass': return 'checkmark-circle';
   969→    case 'failed':
   970→    case 'fail': return 'close-circle';
   971→    case 'cancelled': return 'ban-outline';
   972→    default: return 'ellipsis-circle';
   973→  }
   974→}
   975→
   976→function getStatusText(status: InspectionStatus): string {
   977→  switch (status) {
   978→    case 'pending': return 'En attente';
   979→    case 'scheduled': return 'Planifiée';
   980→    case 'in_progress': return 'En cours';
   981→    case 'completed': return 'Terminée';
   982→    case 'failed': return 'Échouée';
   983→    case 'cancelled': return 'Annulée';
   984→    default: return status;
   985→  }
   986→}
