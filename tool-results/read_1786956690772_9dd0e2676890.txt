   245→// ===========================================
   246→// Main Security Auditor Class

   248→
   249→class SecurityAuditor {
   250→  private config: AuditConfig;
   251→  private currentAudit: SecurityAudit | null = null;
   252→
   253→  constructor(config?: Partial<AuditConfig>) {
   254→    this.config = { ...DEFAULT_CONFIG, ...config };
   255→  }
   256→
   257→  /**
   258→   * Run comprehensive security audit
   259→   */
   260→  async runAudit(options?: {
   261→    targetUrl?: string;
   262→    customChecks?: Array<{
   263→      name: string;
   264→      category: string;
   265→      test: () => Promise<{ passed: boolean; details?: any }>;
   266→      severity: Finding['severity'];
   267→    }>;
   268→  }): Promise<SecurityAudit> {
   269→    const startTime = Date.now();
   270→    
   271→    const audit: SecurityAudit = {
   272→      id: `audit_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
   273→      initiatedAt: new Date().toISOString(),
   274→      status: 'running',
   275→      overallScore: 0,
   276→      grade: 'F',
   277→      categories: [],
   278→      summary: {
   279→        totalFindings: 0,
   280→        criticalFindings: 0,
   281→        highFindings: 0,
   282→        mediumFindings: 0,
   283→        lowFindings: 0,
   284→        infoFindings: 0,
   285→        passedChecks: 0,
   286→        failedChecks: 0,
   287→        skippedChecks: 0,
   288→      },
   289→      recommendations: [],
   290→      metadata: {
   291→        auditorVersion: '2.4.1',
   292→        scannerVersion: '2.4.1',
   293→        environment: process.env.NODE_ENV || 'unknown',
   294→        targetUrl: options?.targetUrl,
   295→        durationMs: 0,
   296→      },
   297→    };
   298→
   299→    this.currentAudit = audit;
   300→
   301→    try {
   302→      // Run all enabled scan categories
   303→      if (this.config.scans.owaspTop10) {
   304→        await this.runOWASP10Scan(audit);
   305→      }
   306→
   307→      if (this.config.scans.configurationSecurity) {
   308→        await this.runConfigurationScan(audit);
   309→      }
   310→
   311→      if (this.config.scans.headerSecurity) {
   312→        await this.runHeaderSecurityScan(audit);
   313→      }
   314→
   315→      if (this.config.scans.authenticationSecurity) {
   316→        await this.runAuthenticationScan(audit);
   317→      }
   318→
   319→      if (this.config.scans.dataProtection) {
   320→        await this.runDataProtectionScan(audit);
   321→      }
   322→
   323→      if (this.config.scans.complianceChecks) {
   324→        await this.runComplianceScan(audit);
   325→      }
   326→
   327→      // Run custom checks if provided
   328→      if (options?.customChecks) {
   329→        await this.runCustomChecks(audit, options.customChecks);
   330→      }
   331→
   332→      // Calculate final scores and grades
   333→      this.calculateFinalScores(audit);
   334→
   335→      audit.status = 'completed';
   336→      audit.completedAt = new Date().toISOString();
   337→      audit.metadata.durationMs = Date.now() - startTime;
   338→
   339→      // Generate recommendations
   340→      audit.recommendations = this.generateRecommendations(audit);
   341→
   342→    } catch (error) {
   343→      console.error('[Security Audit] Error during audit:', error);
   344→      audit.status = 'failed';
   345→      
   346→      if (this.config.thresholds.failOnCritical && 
   347→          audit.summary.criticalFindings > 0) {
   348→        throw error; // Re-throw on critical failures if configured
   349→      }
   350→    }
   351→
   352→    this.currentAudit = null;
   353→    return audit;
   354→  }

   356→  // ===========================================
   357→  // OWASP Top 10 Scan
   358→  // ===========================================
   359→
   360→  private async runOWASP10Scan(audit: SecurityAudit): Promise<void> {
   361→    const category = this.createCategory(
   362→      'owasp-top10',
   363→      'OWASP Top 10 Security Risks',
   364→      'Detection of common web application vulnerabilities as defined by OWASP Top 10 (2021)',
   365→      25 // High importance
   366→    );
   367→
   368→    try {
   369→      // A01:2021 - Broken Access Control
   370→      await this.checkBrokenAccessControl(category);
   371→
   372→      // A02:2021 - Cryptographic Failures
   373→      await this.checkCryptographicFailures(category);
   374→
   375→      // A03:2021 - Injection
   376→      await this.checkInjection(category);
   377→
   378→      // A04:2021 - Insecure Design
   379→      await this.checkInsecureDesign(category);
   380→
   381→      // A05:2021 - Security Misconfiguration
   382→      await this.checkSecurityMisconfiguration(category);
   383→
   384→      // A06:2021 - Vulnerable Components
   385→      await this.checkVulnerableComponents(category);
   386→
   387→      // A07:2021 - Identification and Authentication Failures
   388→      await this.checkAuthenticationFailures(category);
   389→
   390→      // A08:2021 - Software and Data Integrity Failures
   391→      await this.checkDataIntegrityFailures(category);
   392→
   393→      // A09:2021 - Security Logging and Monitoring Failures
   394→      await this.checkLoggingFailures(category);
   395→
   396→      // A10:2021 - Server-Side Request Forgery
   397→      await this.checkSSRF(category);
   398→
   399→      // Additional important checks
   400→      await this.checkXSS(category);
   401→      await this.checkCSRF(category);
   402→      await this.checkPathTraversal(category);
   403→
   404→    } catch (error) {
   405→      console.error('[Security Audit] OWASP Scan error:', error);
   406→      category.findings.push({
   407→        id: `owasp-error-${Date.now()}`,
   408→        categoryId: category.id,
   409→        categoryName: category.name,
   410→        severity: 'high',
   411→        title: 'OWASP Scan Error',
   412→        description: `Error during OWASP Top 10 scanning: ${error.message}`,
   413→        location: { type: 'system', value: 'scanner' },
   414→        impact: 'Complete OWASP scan may not have completed',
   415→        remediation: [],
   416→        references: [],
   417→        discoveredAt: new Date().toISOString(),
   418→      });
   419→    }
   420→
   421→    this.calculateCategoryScore(category);
   422→    audit.categories.push(category);
   423→  }

   435→        id: `a01-access-${path}`,
   436→        checkName: 'Admin Panel Access Control',
   437→        category: 'access-control',
   438→        description: `Verify ${path} has proper access control`,
   439→        status: 'pass',
   440→        severity: 'high',
   441→        message: `${path} appears to have access control implemented`,
   442→      };

  1189→  private async runCustomChecks(audit: SecurityAudit, checks: Array<{
  1190→    name: string;
  1191→    category: string;
  1192→    test: () => Promise<{ passed: boolean; details?: any }>;
  1193→    severity: Finding['severity'];
  1194→  }>): Promise<void> {
  1195→    for (const check of checks) {
  1196→      try {
  1197→        const result = await check.test();
  1198→        
  1199→        const checkResult: CheckResult = {
  1200→          id: `custom-${Date.now()}-${Math.random().toString(36).substring(2, 8)}`,
  1201→          checkName: check.name,
  1202→          category: check.category,
  1203→          description: 'Custom security check',
  1204→          status: result.passed ? 'pass' : 'fail',
  1205→          severity: check.severity,
  1206→          message: result.passed ? 'Check passed' : 'Check failed',
  1207→          details: result.details,
  1208→        };
  1209→
  1210→        if (!result.passed) {
  1211→          audit.categories
  1212→            .find(c => c.id === 'custom') || c.name === 'Custom Checks')
  1213→            ?.findings.push({
  1214→              id: checkResult.id,
  1215→              categoryId: 'custom',
  1216→              categoryName: 'Custom Checks',
  1217→              severity: check.severity,
  1218→              title: `Custom Check Failed: ${check.name}`,
  1219→              description: checkResult.message || 'Custom security check did not pass',
  1220→              location: { type: 'custom', value: check.name },
  1221→              impact: 'Custom security check failure',
  1222→              remediation: [],
  1223→              references: [],
  1224→              discoveredAt: new Date().toISOString(),
  1225→            }) ||
  1226→            category.findings.push({
  1227→              id: checkResult.id,
  1228→              categoryId: 'custom',
  1229→              categoryName: 'Custom Checks',
  1230→              severity: check.severity,
  1231→              title: `Custom Check Failed: ${check.name}`,
  1232→              description: checkResult.message || 'Custom security check did not pass',
  1233→              location: { type: 'custom', value: check.name },
  1234→              impact: 'Custom security check failure',
  1235→              remediation: [],
  1236→              references: [],
  1237→              discoveredAt: new Date().toISOString(),
  1238→            });
  1239→        }
  1240→
  1241→        category.passedChecks.push(checkResult);
  1242→      } catch (error) {
  1243→        category.failedChecks.push({
  1244→          id: `custom-error-${Date.now()}`,
  1245→          checkName: check.name,
  1246→          category: check.category,

  1248→          status: 'error',
  1249→          severity: check.severity,

  1252→      }
  1253→    }
  1254→  }

  1260→  private calculateCategoryScore(category: AuditCategory): void {
  1261→    const totalChecks = category.passedChecks.length + 
  1262→                     category.failedChecks.length + 
  1263→                     category.skippedChecks.length;
  1264→    
  1265→    if (totalChecks === 0) {
  1266→      category.score = 50; // Neutral if no checks
  1267→      category.grade = 'N/A';
  1268→      return;
  1269→    }
  1270→
  1271→    const passedCount = category.passedChecks.length;
  1272→    const rawScore = (passedCount / totalChecks) * 100;
  1273→
  1274→    category.score = Math.round(rawScore);
  1275→    category.grade = this.scoreToGrade(category.score);
  1276→    
  1277→    // Move findings from checks to category findings
  1278→    for (const check of [...category.passedChecks, ...category.failedChecks, ...category.skippedChecks]) {
  1279→      if (check.status === 'fail') {
  1280→        // Convert failed check to finding
  1281→        category.findings.push({
  1282→          id: check.id,
  1283→          categoryId: category.id,
  1284→          categoryName: category.name,
  1285→          severity: check.severity,
  1286→          title: check.checkName,
  1287→          description: check.message,
  1288→          location: { type: 'check', value: check.checkName },
  1289→          impact: 'Security check failure',
  1290→          remediation: [],
  1291→          references: [],
  1292→          discoveredAt: new Date().toISOString(),
  1293→        });
  1294→      }
  1295→    }
  1296→
  1297→    // Limit findings per config
  1298→    if (category.findings.length > this.config.thresholds.maxFindingsPerCategory) {
  1299→      category.findings = category.findings.slice(0, this.config.thresholds.maxFindingsPerCategory);
  1300→    }
  1301→  }

  1303→  private calculateFinalScores(audit: SecurityAudit): void {
  1304→    let weightedSum = 0;
  1305→    let totalWeight = 0;
  1306→
  1307→    for (const category of audit.categories) {
  1308→      weightedSum += category.score * category.weight;
  1309→      totalWeight += category.weight;
  1310→    }
  1311→
  1312→    audit.overallScore = totalWeight > 0 ? Math.round(weightedSum / totalWeight) : 0;
  1313→    audit.grade = this.scoreToGrade(audit.overallScore);
  1314→
  1315→    // Calculate summary statistics
  1316→    audit.summary.totalFindings = audit.categories.reduce((sum, cat) => sum + cat.findings.length, 0);
  1317→    audit.summary.criticalFindings = audit.categories.reduce((sum, cat) => 
  1318→      sum + cat.findings.filter(f => f.severity === 'critical').length, 0);
  1319→    audit.summary.highFindings = audit.categories.reduce((sum, cat) => 
  1320→      sum + cat.findings.filter(f => f.severity === 'high').length, 0);
  1321→    audit.summary.mediumFindings = audit资源配置
  1322→    audit.summary.lowFindings = audit.categories.reduce((sum, cat) => 
  1323→      sum + cat.findings.filter(f => f.severity === 'low').length, 0);
  1324→    audit.summary.infoFindings = audit.categories.reduce((sum, cat) => 
  1325→      sum + cat.findings.filter(f => f.severity === 'info').length, 0);
  1326→    
  1327→    audit.summary.passedChecks = audit.categories.reduce((sum, cat) => sum + cat.passedChecks.length, 0);
  1328→    audit.summary.failedChecks = audit.categories.reduce((sum, cat) => sum + cat.failedChecks.length, 0);
  1329→    audit.summary.skippedChecks = audit.categories.reduce((sum, cat) => sum + cat.skippedChecks.length, 0);

  1332→  private scoreToGrade(score: number): SecurityAudit['grade'] {
  1333→    if (score >= 95) return 'A+';
  1334→    if (score >= 90) return 'A';
  1335→    if (score >= 85) return 'A-';
  1336→    if (score >= 80) return 'B+';
  1337→    if (score >= 75) return 'B';
  1338→    if (score >= 70) return 'B-';
  1339→    if (score >= 60) return 'C+';
  1340→    if (score >= 50) return 'C';
  1341→    if (score >= 40) return 'C-';
  1342→    if (score >= 25) return 'D';
  1343→    return 'F';
  1344→  }

  1350→  private generateRecommendations(audit: SecurityAudit): Recommendation[] {
  1351→    const recommendations: Recommendation[] = [];
  1352→
  1353→    // Critical findings need immediate attention
  1354→    if (audit.summary.criticalFindings > 0) {
  1355→      recommendations.push({
  1356→        id: 'rec-immediate-critical',
  1357→        priority: 'immediate',
  1358→        title: 'Address Critical Security Vulnerabilities',
  1359→        description: `${audit.summary.criticalFindings} critical severity issues found that require immediate attention to prevent potential exploitation.`,
  1360→        affectedCategories: audit.categories.filter(c => 
  1361→          c.findings.some(f => f.severity === 'critical')
  1362→        ).map(c => c.id),
  1363→        businessImpact: 'High risk of security breach, data compromise, or service disruption',
  1364→        effort: 'high',
  1365→        findings: audit.categories.flatMap(c => 
  1366→          c.findings.filter(f => f.severity === 'critical').map(f => f.id)
  1367→        ),
  1368→      });
  1369→    }
  1370→
  1371→    // High severity findings
  1372→    if (audit.summary.highFindings > 3) {
  1373→      recommendations.push({
  1374→        id: 'rec-short-term-high',
  1375→        priority: 'short_term',
  1376→        title: 'Address High-Severity Security Issues',
  1377→        description: `${audit.summary.highFindings} high severity issues identified that should be addressed within 30 days.`,
  1378→        affectedCategories: audit.categories.filter(c => 
  1379→          c.findings.some(f => f.severity === 'high')
  1380→        ).map(c => c.id),
  1381→        businessImpact: 'Moderate risk if exploited, could lead to data exposure or service issues',
  1382→        effort: 'medium',
  1383→        findings: audit.categories.flatMap(c => 
  1384→          c.findings.filter(f => f.severity === 'high').map(f => f.id).slice(0, 10)
  1385→        ),
  1386→      });
  1387→    }
  1388→
  1389→    // General improvements
  1390→    if (audit.overallScore < 80) {
  1391→      recommendations.push({
  1392→        id: 'rec-long-term-general',
  1393→        priority: 'long_term',
  1394→        title: 'Improve Overall Security Posture',
  1395→        description: `Current security score is ${audit.overallScore}/100 (${audit.grade}). Implement recommended security hardening measures to improve posture.`,
  1396→        affectedCategories: audit.categories.map(c => c.id),
  1397→        businessImpact: 'Improved security posture reduces long-term risk and builds customer trust',
  1398→        effort: 'medium',
  1399→        findings: ['general-hardening'],
  1400→      });
  1401→    }
  1402→
  1403→    // Specific recommendations based on findings patterns
  1404→    const sslFinding = audit.categories.find(c => 
  1405→      c.findings.some(f => f.title.includes('TLS') || f.title.includes('SSL') || f.title.includes('HSTS'))
  1406→    );
  1407→    if (sslFinding) {
  1408→      recommendations.push({
  1409→        id: 'rec-ssl-hardening',
  1410→        priority: 'short_term',
  1411→        title: 'Strengthen TLS Configuration',
  1412→        description: 'Implement HSTS, upgrade to TLS 1.3, disable weak ciphers, and enable OC Stapling.',
  1413→        affectedCategories: [sslFinding.id],
  1414→        businessImpact: 'Stronger encryption protects data in transit',
  1415→        effort: 'medium',
  1416→        findings: [sslFinding.findings[0]?.id],
  1417→      });
  1418→    }
  1419→
  1420→    return recommendations;
  1421→  }
