    23→  /**
    24→   * Generate a complete report based on configuration
    25→   * @param config - Report configuration
    26→   * @param userId - ID of user requesting the report
    27→   * @returns Complete report result with data and metadata
    28→   */
    29→  async generateReport(config: ReportConfig, userId: string): Promise<ReportResult> {
    30→    const startTime = Date.now();
    31→    
    32→    try {
    33→      let data: ReportData;
    34→      
    35→      // Route to appropriate generator based on report type
    36→      switch (config.type) {
    37→        case 'sales_overview':
    38→          data = await this.generateSalesOverview(config);
    39→          break;
    40→        case 'product_performance':
    41→          data = await this.generateProductPerformance(config);
    42→          break;
    43→        case 'supplier_analytics':
    44→          data = await this.generateSupplierAnalytics(config);
    45→          break;
    46→        case 'buyer_behavior':
    47→          data = await this.generateBuyerBehavior(config);
    48→          break;
    49→        case 'rfq_analysis':
    50→          data = await this.generateRFQAnalysis(config);
    51→          break;
    52→        case 'revenue_by_category':
    53→          data = await this.generateRevenueByCategory(config);
    54→          break;
    55→        case 'geographic_distribution':
    56→          data = await this.generateGeographicDistribution(config);
    57→          break;
    58→        case 'payment_methods':
    59→          data = await this.generatePaymentMethodsReport(config);
    60→          break;
    61→        case 'user_growth':
    62→          data = await this.generateUserGrowthReport(config);
    63→          break;
    64→        case 'inventory_status':
    65→          data = await this.generateInventoryStatusReport(config);
    66→          break;
    67→        case 'custom':
    68→          data = await this.generateCustomReport(config);
    69→          break;
    70→        default:
    71→          throw new Error(`Unknown report type: ${config.type}`);
    72→      }
    73→      
    74→      const processingTime = Date.now() - startTime;
    75→      
    76→      // Apply limit to table rows if specified
    77→      if (config.limit && config.limit < data.table.rows.length) {
    78→        data.table.rows = data.table.rows.slice(0, config.limit);
    79→      }
    80→      
    81→      return {
    82→        id: randomUUID(),
    83→        config,
    84→        generatedAt: new Date(),
    85→        generatedBy: userId,
    86→        status: 'completed',
    87→        expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000), // 24 hours expiry
    88→        metadata: {
    89→          recordCount: data.table.rows.length,
    90→          processingTimeMs: processingTime,
    91→        },
    92→        data,
    93→      };
    94→    } catch (error) {
    95→      console.error('Report generation error:', error);
    96→      
    97→      return {
    98→        id: randomUUID(),
    99→        config,
   100→        generatedAt: new Date(),
   101→        generatedBy: userId,
   102→        status: 'failed',
   103→        expiresAt: new Date(),
   104→        metadata: {
   105→          recordCount: 0,
   106→          processingTimeMs: Date.now() - startTime,
   107→        },
   108→        data: this.emptyReport(),
   109→      };
   110→    }
   111→  }

   117→  /**
   118→   * Generate sales overview report
   119→   * Includes revenue trends, order statistics, and conversion metrics
   120→   */
   121→  private async generateSalesOverview(config: ReportConfig): Promise<ReportData> {
   122→    const { startDate, endDate } = this.getDateRange(config.period, config.dateRange);
   123→    
   124→    // Query orders with related data
   125→    const orders = await db.order.findMany({
   126→      where: {
   127→        createdAt: { gte: startDate, lte: endDate },
   128→        ...(config.filters?.status?.length && { status: { in: config.filters.status as any[] } }),
   129→      },
   130→      include: {
   131→        items: { include: { product: { include: { category: true } } } },
   132→        buyer: { select: { id: true, firstName: true, lastName: true, wilaya: true, city: true } },
   133→      },
   134→      orderBy: { createdAt: 'desc' },
   135→      take: config.limit || 500,
   136→    });
   137→    
   138→    // Calculate key metrics
   139→    const totalRevenue = orders.reduce((sum, o) => sum + Number(o.totalAmount), 0);
   140→    const avgOrderValue = orders.length > 0 ? totalRevenue / orders.length : 0;
   141→    const completedOrders = orders.filter(o => o.status === 'DELIVERED').length;
   142→    const pendingOrders = orders.filter(o => o.status === 'PENDING' || o.status === 'CONFIRMED').length;
   143→    const cancelledOrders = orders.filter(o => o.status === 'CANCELLED').length;
   144→    
   145→    // Group daily sales for trend analysis
   146→    const dailySales = this.groupByDateField(orders, 'createdAt', 'totalAmount');
   147→    const dailyOrders = this.groupByDateCount(orders, 'createdAt');
   148→    
   149→    // Status distribution
   150→    const statusDistribution = this.countByField(orders, 'status');
   151→    
   152→    // Revenue by payment method
   153→    const revenueByPayment = this.aggregateByField(orders, 'paymentMethod', 'totalAmount');
   154→    
   155→    return {
   156→      summary: {
   157→        totalRevenue: `${totalRevenue.toFixed(2)} DZD`,
   158→        totalOrders: orders.length.toString(),
   159→        averageOrderValue: `${avgOrderValue.toFixed(2)} DZD`,
   160→        completionRate: orders.length > 0 ? `${((completedOrders / orders.length) * 100).toFixed(1)}%` : 'N/A',
   161→        completedOrders: completedOrders.toString(),
   162→        pendingOrders: pendingOrders.toString(),
   163→        cancelledOrders: cancelledOrders.toString(),
   164→        periodStart: startDate.toLocaleDateString('fr-FR'),
   165→        periodEnd: endDate.toLocaleDateString('fr-FR'),
   166→      },
   167→      charts: [
   168→        {
   169→          type: 'line',
   170→          title: "Évolution des Ventes Quotidiennes",
   171→          data: Object.entries(dailySales).map(([date, amount]) => ({
   172→            date,
   173→            revenu: Number(amount),
   174→            commandes: dailyOrders[date] || 0,
   175→          })),
   176→          options: { 
   177→            yAxis: { label: 'Montant (DZD)' },
   178→            xAxis: { label: 'Date' },
   179→          },
   180→        },
   181→        {
   182→          type: 'pie',
   183→          title: 'Répartition par Statut de Commande',
   184→          data: statusDistribution.map(([name, value]) => ({ name, value })),
   185→        },
   186→        {
   187→          type: 'bar',
   188→          title: 'Revenus par Mode de Paiement',
   189→          data: revenueByPayment.map(([method, amount]) => ({
   190→            name: method || 'Inconnu',
   191→            montant: Number(amount),
   192→          })),
   193→        },
   194→      ],
   195→      table: {
   196→        headers: ['N° Commande', 'Date', 'Client', 'Montant', 'Statut', 'Paiement', 'Wilaya'],
   197→        rows: orders.map(o => [
   198→          o.orderNumber,
   199→          o.createdAt.toLocaleDateString('fr-FR'),
   200→          `${o.buyer.firstName} ${o.buyer.lastName}`,
   201→          `${Number(o.totalAmount).toFixed(2)} DZD`,
   202→          o.status,
   203→          o.paymentMethod || '-',
   204→          o.buyer.wilaya || '-',
   205→        ]),
   206→      },
   207→      insights: this.generateSalesInsights(orders, totalRevenue, avgOrderValue),
   208→    };
   209→  }

   211→  /**
   212→   * Generate product performance report
   213→   * Analyzes product views, favorites, orders, and ratings
   214→   */
   215→  private async generateProductPerformance(config: ReportConfig): Promise<ReportData> {
   216→    const { startDate, endDate } = this.getDateRange(config.period, config.dateRange);
   217→    
   218→    const products = await db.product.findMany({
   219→      where: {
   220→        ...(config.filters?.categories?.length && { categoryId: { in: config.filters.categories } }),
   221→        status: 'published',
   222→      },
   223→      include: {
   224→        _count: { select: { orderItems: true, favorites: true, reviews: true } },
   225→        category: { select: { name: true } },
   226→        company: { select: { name: true } },
   227→      },
   228→      take: config.limit || 100,
   229→      orderBy: { viewCount: 'desc' },
   230→    });
   231→    
   232→    // Get order items for revenue calculation
   233→    const productIds = products.map(p => p.id);
   234→    const orderItems = await db.orderItem.findMany({
   235→      where: {
   236→        productId: { in: productIds },
   237→        order: { createdAt: { gte: startDate, lte: endDate } },
   238→      },
   239→      include: { product: { select: { id: true } } },
   240→    });
   241→    
   242→    // Calculate revenue per product
   243→    const productRevenue: Record<string, number> = {};
   244→    orderItems.forEach(item => {
   245→      productRevenue[item.productId] = (productRevenue[item.productId] || 0) + Number(item.totalPrice);
   246→    });
   247→    
   248→    // Category distribution
   249→    const categoryStats = this.countByField(products, 'category.name');
   250→    
   251→    // Top performers
   252→    const topProducts = products.slice(0, 10).map(p => ({
   253→      name: p.name.length > 30 ? p.name.substring(0, 30) + '...' : p.name,
   254→      vues: p.viewCount,
   255→      favoris: p._count.favorites,
   256→      commandes: p._count.orderItems,
   257→      revenu: productRevenue[p.id] || 0,
   258→    }));
   259→    
   260→    return {
   261→      summary: {
   262→        totalProduits: products.length.toString(),
   263→        vuesTotales: products.reduce((s, p) => s + p.viewCount, 0).toString(),
   264→        favorisTotaux: products.reduce((s, p) => s + p._count.favorites, 0).toString(),
   265→        revenuTotal: `${Object.values(productRevenue).reduce((a, b) => a + b, 0).toFixed(2)} DZD`,
   266→        moyenneVues: Math.round(products.reduce((s, p) => s + p.viewCount, 0) / products.length).toString(),
   267→      },
   268→      charts: [
   269→        {
   270→          type: 'bar',
   271→          title: 'Top 10 Produits les Plus Vus',
   272→          data: topProducts.map(p => ({ ...p, revenu: Number(p.revenu) })),
   273→          options: { horizontal: true },
   274→        },
   275→        {
   276→          type: 'pie',
   277→          title: 'Répartition par Catégorie',
   278→          data: categoryStats.map(([name, value]) => ({ name, value })),
   279→        },
   280→        {
   281→          type: 'scatter',
   282→          title: 'Correlation Vues vs Commandes',
   283→          data: products.slice(0, 50).map(p => ({
   284→            x: p.viewCount,
   285→            y: p._count.orderItems,
   286→            name: p.name.substring(0, 20),
   287→          })),
   288→        },
   289→      ],
   290→      table: {
   291→        headers: ['Produit', 'Catégorie', 'Fournisseur', 'Vues', 'Favoris', 'Commandes', 'Revenu'],
   292→        rows: products.map(p => [
   293→          p.name,
   294→          p.category.name,
   295→          p.company.name,
   296→          p.viewCount.toString(),
   297→          p._count.favorites.toString(),
   298→          p._count.orderItems.toString(),
   299→          `${(productRevenue[p.id] || 0).toFixed(2)} DZD`,
   300→        ]),
   301→      },
   302→      insights: this.generateProductInsights(products, productRevenue),
   303→    };
   304→  }

   306→  /**
   307→   * Generate supplier analytics report
   308→   * Analyzes supplier performance, response times, and catalog quality
   309→   */
   310→  private async generateSupplierAnalytics(config: ReportConfig): Promise<ReportData> {
   311→    const { startDate, endDate } = this.getDateRange(config.period, config.dateRange);
   312→    
   313→    // Get suppliers (users with SUPPLIER role who have companies)
   314→    const suppliers = await db.user.findMany({
   315→      where: {
   316→        role: 'SUPPLIER',
   317→        isActive: true,
   318→        ...(config.filters?.suppliers?.length && { 
   319→          OR: [
   320→            { id: { in: config.filters.suppliers } },
   321→            { company: { name: { in: config.filters.suppliers } } },
   322→          ]
   323→        }),
   324→      },
   325→      include: {
   326→        company: {
   327→          include: {
   328→            _count: { select: { products: true } },
   329→            products: {
   330→              where: { status: 'published' },
   331→              select: { id: true, viewCount: true },
   332→              take: 100,
   333→            },
   334→          },
   335→        },
   336→        _count: {
   337→          select: {
   338→            quotationsSent: true,
   339→          },
   340→        },
   341→      },
   342→      take: config.limit || 50,
   343→    });
   344→    
   345→    // Get quotation stats per supplier
   346→    const supplierIds = suppliers.map(s => s.id);
   347→    const quotations = await db.quotation.findMany({
   348→      where: {
   349→        supplierId: { in: supplierIds },
   350→        createdAt: { gte: startDate, lte: endDate },
   351→      },
   352→      select: { supplierId: true, id: true, status: true, totalPrice: true, createdAt: true },
   353→    });
   354→    
   355→    // Aggregate supplier performance
   356→    const supplierStats = suppliers.map(supplier => {
   357→      const supplierQuotations = quotations.filter(q => q.supplierId === supplier.id);
   358→      const acceptedQuots = supplierQuotations.filter(q => q.status === 'ACCEPTED');
   359→      const totalQuotationValue = supplierQuotations.reduce((sum, q) => sum + Number(q.totalPrice || 0), 0);
   360→      const avgResponseTime = this.calculateAvgResponseTime(supplierQuotations);
   361→      
   362→      return {
   363→        id: supplier.id,
   364→        name: supplier.company?.name || `${supplier.firstName} ${supplier.lastName}`,
   365→        produits: supplier.company?._count.products || 0,
   366→        vuesTotales: supplier.company?.products.reduce((s, p) => s + p.viewCount, 0) || 0,
   367→        devisEnvoyes: supplierQuotations.length,
   368→        devisAcceptes: acceptedQuots.length,
   369→        tauxAcceptation: supplierQuotations.length > 0 
   370→          ? ((acceptedQuots.length / supplierQuotations.length) * 100).toFixed(1)
   371→          : 'N/A',
   372→        valeurTotale: totalQuotationValue,
   373→        tempsReponseMoyen: avgResponseTime,
   374→      };
   375→    }).sort((a, b) => b.valeurTotale - a.valeurTotale);
   376→    
   377→    return {
   378→      summary: {
   379→        totalFournisseurs: suppliers.length.toString(),
   380→        totalDevis: quotations.length.toString(),
   381→        valeurTotaleDevis: `${quotations.reduce((s, q) => s + Number(q.totalPrice || 0), 0).toFixed(2)} DZD`,
   382→        produitsActifs: suppliers.reduce((s, sup) => s + (sup.company?._count.products || 0), 0).toString(),
   383→      },
   384→      charts: [
   385→        {
   386→          type: 'bar',
   387→          title: 'Top Fournisseurs par Valeur de Devis',
   388→          data: supplierStats.slice(0, 15).map(s => ({
   389→            name: s.name.length > 25 ? s.name.substring(0, 25) + '...' : s.name,
   390→            valeur: s.valeurTotale,
   391→            acceptes: s.devisAcceptes,
   392→          })),
   393→          options: { horizontal: true },
   394→        },
   395→        {
   396→          type: 'radar',
   397→          title: 'Performance des Top 5 Fournisseurs',
   398→          data: supplierStats.slice(0, 5).map(s => ({
   399→            name: s.name.substring(0, 15),
   400→            produits: s.produits,
   401→            devis: s.devisEnvoyes,
   402→            acceptes: s.devisAcceptes,
   403→            vues: Math.round(s.vuesTotales / 100),
   404→          })),
   405→        },
   406→      ],
   407→      table: {
   408→        headers: ['Fournisseur', 'Produits', 'Vues', 'Devis Envoyés', 'Acceptés', 'Taux Accept.', 'Valeur Totale'],
   409→        rows: supplierStats.map(s => [
   410→          s.name,
   411→          s.produits.toString(),
   412→          s.vuesTotales.toString(),
   413→          s.devisEnvoyes.toString(),
   414→          s.devisAcceptes.toString(),
   415→          `${s.tauxAcceptation}%`,
   416→          `${s.valeurTotale.toFixed(2)} DZD`,
   417→        ]),
   418→      },
   419→      insights: this.generateSupplierInsights(supplierStats),
   420→    };
   421→  }

   425→   * Analyzes purchasing patterns, preferences, and engagement
   426→   */
   427→  private async generateBuyerBehavior(config: ReportConfig): Promise<ReportData> {
   428→    const { startDate, endDate } = this.getDateRange(config.period, config.dateRange);
   429→    
   430→    // Get buyers with their activity
   431→    const buyers = await db.user.findMany({
   432→      where: {
   433→        role: 'BUYER',
   434→        isActive: true,
   435→        ...(config.filters?.buyers?.length && { 
   436→          OR: [
   437→            { id: { in: config.filters.buyers } },
   438→            { firstName: { in: config.filters.buyers } },
   439→          ]
   440→        }),
   441→      },
   442→      include: {
   443→        _count: {
   444→          select: {
   445→            ordersPlaced: true,
   446→            rfqsCreated: true,
   447→            favorites: true,
   448→            reviews: true,
   449→          },
   450→        },
   451→        ordersPlaced: {
   452→          where: { createdAt: { gte: startDate, lte: endDate } },
   453→          select: { totalAmount: true, createdAt: true, status: true },
   454→          take: 50,
   455→        },
   456→      },
   457→      take: config.limit || 100,
   458→    });
   459→    
   460→    // Analyze buying patterns
   461→    const buyerAnalysis = buyers.map(buyer => {
   462→      const orders = buyer.ordersPlaced;
   463→      const totalSpent = orders.reduce((sum, o) => sum + Number(o.totalAmount), 0);
   464→      const avgOrderValue = orders.length > 0 ? totalSpent / orders.length : 0;
   465→      
   466→      // Determine most active time of day
   467→      const hourCounts: Record<number, number> = {};
   468→      orders.forEach(o => {
   469→        const hour = new Date(o.createdAt).getHours();
   470→        hourCounts[hour] = (hourCounts[hour] || 0) + 1;
   471→      });
   472→      const peakHour = Object.entries(hourCounts).sort((a, b) => b[1] - a[1])[0]?.[0];
   473→      
   474→      return {
   475→        id: buyer.id,
   476→        name: `${buyer.firstName} ${buyer.lastName}`,
   477→        wilaya: (buyer as any).wilaya || 'Inconnu',
   478→        commandes: orders.length,
   479→        depensesTotales: totalSpent,
   480→        moyennePanier: avgOrderValue,
   481→        rfqs: buyer._count.rfqsCreated,
   482→        favoris: buyer._count.favorites,
   483→        avis: buyer._count.reviews,
   484→        heurePic: peakHour ? `${peakHour}h` : 'N/A',
   485→      };
   486→    }).sort((a, b) => b.depensesTotales - a.depensesTotales);
   487→    
   488→    // Wilaya distribution of buyers
   489→    const buyersByWilaya = this.countByField(buyers as any[], 'wilaya');
   490→    
   491→    // Spending segments
   492→    const spendingSegments = {
   493→      'Petits acheteurs (<10k DZD)': buyerAnalysis.filter(b => b.depensesTotales < 10000).length,
   494→      'Acheteurs moyens (10k-50k)': buyerAnalysis.filter(b => b.depensesTotales >= 10000 && b.depensesTotales < 50000).length,
   495→      'Gros acheteurs (50k-100k)': buyerAnalysis.filter(b => b.depensesTotales >= 50000 && b.depensesTotales < 100000).length,
   496→      'VIP (>100k DZD)': buyerAnalysis.filter(b => b.depensesTotales >= 100000).length,
   497→    };
   498→    
   499→    return {
   500→      summary: {
   501→        totalAcheteurs: buyers.length.toString(),
   502→        depensesTotales: `${buyerAnalysis.reduce((s, b) => s + b.depensesTotales, 0).toFixed(2)} DZD`,
   503→        panierMoyen: `${(buyerAnalysis.reduce((s, b) => s + b.moyennePanier, 0) / buyerAnalysis.length || 0).toFixed(2)} DZD`,
   504→        rfqTotal: buyerAnalysis.reduce((s, b) => s + b.rfqs, 0).toString(),
   505→      },
   506→      charts: [
   507→        {
   508→          type: 'bar',
   509→          title: 'Top 15 Acheteurs par Dépenses',
   510→          data: buyerAnalysis.slice(0, 15).map(b => ({
   511→            name: b.name,
   512→            depenses: b.depensesTotales,
   513→            commandes: b.commandes,
   514→          })),
   515→          options: { horizontal: true },
   516→        },
   517→        {
   518→          type: 'pie',
   519→          title: 'Segments de Dépense',
   520→          data: Object.entries(spendingSegments).map(([name, value]) => ({ name, value })),
   521→        },
   522→        {
   523→          type: 'funnel',
   524→          title: "Entonnoir d'Engagement Acheteur",
   525→          data: [
   526→            { stage: 'Inscrits', value: buyers.length },
   527→            { stage: 'Avec Favoris', value: buyers.filter(b => b._count.favorites > 0).length },
   528→            { stage: 'RFQ Créés', value: buyers.filter(b => b._count.rfqsCreated > 0).length },
   529→            { stage: 'Commandes', value: buyerAnalysis.filter(b => b.commandes > 0).length },
   530→            { stage: 'Récurrents (>1 cmd)', value: buyerAnalysis.filter(b => b.commandes > 1).length },
   531→          ],
   532→        },
   533→      ],
   534→      table: {
   535→        headers: ['Acheteur', 'Wilaya', 'Commandes', 'Dépenses Totales', 'Panier Moyen', 'RFQs', 'Favoris', 'Heure Pic'],
   536→        rows: buyerAnalysis.map(b => [
   537→          b.name,
   538→          b.wilaya,
   539→          b.commandes.toString(),
   540→          `${b.depensesTotales.toFixed(2)} DZD`,
   541→          `${b.moyennePanier.toFixed(2)} DZD`,
   542→          b.rfqs.toString(),
   543→          b.favoris.toString(),
   544→          b.heurePic,
   545→        ]),
   546→      },
   547→      insights: this.generateBuyerInsights(buyerAnalysis),
   548→    };
   549→  }

   551→  /**
   552→   * Generate RFQ (Request for Quotation) analysis report
   553→   */
   554→  private async generateRFQAnalysis(config: ReportConfig): Promise<ReportData> {
   555→    const { startDate, endDate } = this.getDateRange(config.period, config.dateRange);
   556→

  1240→   * Get date range from period configuration
  1241→   */
  1242→  private getDateRange(period: ReportPeriod, customRange?: { start: Date; end: Date }): { startDate: Date; endDate: Date } {
  1243→    const now = new Date();
  1244→    const endDate = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59);
  1245→    
  1246→    switch (period) {
  1247→      case 'today':
  1248→        return { 
  1249→          startDate: new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0), 
  1250→          endDate 
  1251→        };
  1252→      case 'week': {
  1253→        const weekStart = new Date(now);
  1254→        weekStart.setDate(weekStart.getDate() - 7);
  1255→        return { startDate: weekStart, endDate };
  1256→      }
  1257→      case 'month': {
  1258→        const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
  1259→        return { startDate: monthStart, endDate };
  1260→      }
  1261→      case 'quarter': {
  1262→        const quarterStart = new Date(now);
  1263→        quarterStart.setMonth(quarterStart.getMonth() - 3);
  1264→        return { startDate: quarterStart, endDate };
  1265→      }
  1266→      case 'year': {
  1267→        const yearStart = new Date(now.getFullYear(), 0, 1);
  1268→        return { startDate: yearStart, endDate };
  1269→      }
  1270→      case 'custom':
  1271→        return { startDate: customRange!.start, endDate: customRange!.end };
  1272→      default:
  1273→        const defaultStart = new Date(now);
  1274→        defaultStart.setMonth(defaultStart.getMonth() - 1);
  1275→        return { startDate: defaultStart, endDate };
  1276→    }

  1585→    
  1586→    if (allProducts.length > 0) {
  1587→      const stockIssueRate = ((outOfStock.length + lowStock.length) / allProducts.length) * 100;
  1588→      insights.push(`${outOfStock.length} produits sont en rupture de stock`);
  1589→      insights.push(`${lowStock.length} produits ont un stock faible (≤10 unités)`);
  1590→      
  1591→      if (stockIssueRate > 20) {
  1592→        insights.push(`⚠️ Attention: ${stockIssueRate.toFixed(1)}% des produits ont des problèmes de stock`);
  1593→      }
  1594→      
  1595→      // High value items at risk
  1596→      const highValueAtRisk = outOfStock.filter(p => p.price > 10000);
  1597→      if (highValueAtRisk.length > 0) {
  1598→        insights.push(`${highValueAtRisk.length} produits à haute valeur (>10k DZD) sont en rupture - priorité de réapprovisionnement`);
  1599→      }
  1600→    }
  1601→    
  1602→    return insights;
  1603→  }
