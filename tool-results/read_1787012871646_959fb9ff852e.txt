   122→  async pullEntities(entityType: EntityType, options?: SyncOptions): Promise<SyncResult> {
   123→    const result: SyncResult = {
   124→      success: true,
   125→      processed: 0,
   126→      created: 0,
   127→      updated: 0,
   128→      failed: 0,
   129→      skipped: 0,
   130→      errors: [],
   131→    }
   132→
   133→    switch (entityType) {
   134→      case 'PRODUCTS':
   135→        await this.pullProducts(result, options)
   136→        break
   137→        
   138→      case 'INVENTORY':
   139→        await this.pullInventory(result, options)
   140→        break
   141→        
   142→      case 'CUSTOMERS':
   143→        await this.pullCustomers(result, options)
   144→        break
   145→        
   146→      case 'ORDERS':
   147→        await this.pullOrders(result, options)
   148→        break
   149→        
   150→      case 'CATEGORIES':
   151→        await this.pullCategories(result, options)
   152→        break
   153→        
   154→      case 'SUPPLIERS':
   155→        await this.pullSuppliers(result, options)
   156→        break
   157→        
   158→      case 'INVOICES':
   159→        await this.pullInvoices(result, options)
   160→        break
   161→        
   162→      default:
   163→        result.errors.push({
   164→          message: `Entity type ${entityType} not supported for Odoo pull`,
   165→          code: 'UNSUPPORTED_ENTITY',
   166→        })
   167→        result.success = false
   168→    }
   169→
   170→    return result
   171→  }

   173→  async pushEntities(entityType: EntityType, data: any[], options?: SyncOptions): Promise<SyncResult> {
   174→    const result: SyncResult = {
   175→      success: true,
   176→      processed: data.length,
   177→      created: 0,
   178→      updated: 0,
   179→      failed: 0,
   180→      skipped: 0,
   181→      errors: [],
   182→    }
   183→
   184→    if (options?.dryRun) {
   185→      result.created = data.length
   186→      return result
   187→    }
   188→
   189→    for (const item of data) {
   190→      try {
   191→        const odooData = this.transformData(item, this.config.fieldMappings, 'toERP')
   192→        
   193→        switch (entityType) {
   194→          case 'PRODUCTS':
   195→            await this.pushProduct(odooData, item.id)
   196→            result.updated++
   197→            break
   198→            
   199→          case 'INVENTORY':
   200→            await this.pushInventoryUpdate(odooData, item.id)
   201→            result.updated++
   202→            break
   203→            
   204→          case 'ORDERS':
   205→            await this.pushOrder(odooData, item.id)
   206→            result.created++
   207→            break
   208→            
   209→          case 'CUSTOMERS':
   210→            await this.pushCustomer(odooData, item.id)
   211→            result.created++
   212→            break
   213→            
   214→          default:
   215→            result.skipped++
   216→        }
   217→        
   218→        result.processed++
   219→      } catch (error: any) {
   220→        result.failed++
   221→        result.errors.push({
   222→          recordId: item.id,
   223→          message: error.message || 'Failed to push entity to Odoo',
   224→          code: 'PUSH_ERROR',
   225→          details: error,
   226→        })
   227→      }
   228→    }
   229→
   230→    result.success = result.failed === 0
   231→    return result
   232→  }

   304→  private async pullProducts(result: SyncResult, options?: SyncOptions): Promise<void> {
   305→    try {
   306→      let domain: any[] = [['sale_ok', '=', true]]
   307→      
   308→      if (options?.fromDate) {
   309→        domain.push(['write_date', '>=', options.fromDate.toISOString()])
   310→      }
   311→      
   312→      if (options?.entityIds?.length) {
   313→        domain.push(['id', 'in', options.entityIds.map(id => parseInt(id))])
   314→      }
   315→      
   316→      const response = await this.makeRESTRequest('GET', '/product.product', {
   317→        domain,
   318→        fields: ['id', 'name', 'default_code', 'list_price', 'description_sale', 'type', 
   319→                'sale_ok', 'categ_id', 'barcode', 'weight', 'image_1920'],
   320→        limit: options?.batchSize || 100,
   321→      })
   322→      
   323→      const products: OdooProduct[] = response || []
   324→      
   325→      for (const product of products) {
   326→        try {
   327→          const localData = this.transformData(product, this.config.fieldMappings, 'fromERP')
   328→          
   329→          const existing = await db.product.findFirst({
   330→            where: { 
   331→              OR: [
   332→                { sku: String(product.default_code || product.id) },
   333→                { name: product.name },
   334→              ]
   335→            }
   336→          })
   337→          
   338→          if (existing) {
   339→            await db.product.update({
   340→              where: { id: existing.id },
   341→              data: {
   342→                ...localData,
   343→                sku: String(product.default_code || product.id),
   344→                price: parseFloat(String(product.list_price)) || undefined,
   345→                updatedAt: new Date(),
   346→              }
   347→            })
   348→            result.updated++
   349→          } else {
   350→            await db.product.create({
   351→              data: {
   352→                id: `odoo_${product.id}`,
   353→                name: product.name,
   354→                slug: `${product.name}`.toLowerCase().replace(/[^a-z0-9]+/g, '-'),
   355→                sku: String(product.default_code || product.id),
   356→                shortDescription: product.description_sale?.substring(0, 200),
   357→                price: parseFloat(String(product.list_price)),
   358→                currency: 'DZD',
   359→                status: 'published',
   360→                availability: product.type === 'service' ? 'on_order' : 'in_stock',
   361→              }
   362→            })
   363→            result.created++
   364→          }
   365→          
   366→          result.processed++
   367→        } catch (error: any) {
   368→          result.failed++
   369→          result.errors.push({
   370→            recordId: String(product.id),
   371→            message: error.message || 'Failed to sync product from Odoo',
   372→          })
   373→        }
   374→      }
   375→    } catch (error: any) {
   376→      result.success = false
   377→      result.errors.push({
   378→        message: error.message || 'Failed to fetch products from Odoo',
   379→        code: 'FETCH_ERROR',
   380→      })
   381→    }
   382→  }

   806→  private async makeRESTRequest(
   807→    method: string,
   808→    path: string,
   809→    params?: Record<string, any>
   810→  ): Promise<any> {
   811→    const apiVersion = this.config.apiVersion || 16
   812→    const url = new URL(`${this.config.endpoint}/web/api/${apiVersion}${path}`)
   813→    
   814→    const headers = this.buildAuthHeaders()
   815→    
   816→    // Add Odoo-specific headers
   817→    headers['Content-Type'] = 'application/json'
   818→    
   819→    // For GET requests with params, we need to handle them specially
   820→    let body: any = undefined
   821→    
   822→    if (method === 'GET' && params) {
   823→      // Convert params to query string for GET requests
   824→      Object.entries(params).forEach(([key, value]) => {
   825→        if (key === 'domain') {
   826→          url.searchParams.set(key, JSON.stringify(value))
   827→        } else if (key === 'fields') {
   828→          url.searchParams.set(key, JSON.stringify(value))
   829→        } else {
   830→          url.searchParams.set(key, String(value))
   831→        }
   832→      })
   833→    } else if (params) {
   834→      body = params
   835→    }
   836→    
   837→    const response = await fetch(url.toString(), {
   838→      method,
   839→      headers,
   840→      body: body ? JSON.stringify(body) : undefined,
   841→    })
   842→    
   843→    if (!response.ok) {
   844→      const errorText = await response.text()
   845→      let errorMessage = `Odoo API error: ${response.status}`
   846→      
   847→      try {
   848→        const errorJson = JSON.parse(errorText)
   849→        errorMessage = errorJson.error?.message || errorMessage
   850→      } catch {}
   851→      
   852→      throw new Error(errorMessage)
   853→    }
   854→    
   855→    return response.json()
   856→  }
