   498→    this.loadBuiltInRules();

   501→  /**
   502→   * Main method: Assess a fraud event
   503→   */
   504→  async assessEvent(event: Partial<FraudEvent>): Promise<FraudAssessment> {
   505→    if (!this.config.enabled) {
   506→      return this.createSafeAssessment(event);
   507→    }
   508→
   509→    const completeEvent: FraudEvent = {
   510→      id: `fraud_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`,
   511→      timestamp: new Date().toISOString(),
   512→      eventType: event.eventType || 'api_call',
   513→      sessionId: event.sessionId || 'unknown',
   514→      ipAddress: event.ipAddress || 'unknown',
   515→      userAgent: event.userAgent || '',
   516→      ...event,
   517→      data: event.data || {},
   518→      context: event.context || {},
   519→    };
   520→
   521→    // Store event
   522→    this.events.push(completeEvent);
   523→    this.trimOldEvents();
   524→
   525→    // Initialize assessment
   526→    const assessment: FraudAssessment = {
   527→      eventId: completeEvent.id,
   528→      score: 0,
   529→      riskLevel: 'low',
   530→      decision: 'approve',
   531→      reasons: [],
   532→      recommendations: [],
   533→      reviewRequired: false,
   534→    };
   535→
   536→    // Run all enabled checks
   537→    await Promise.all([
   538→      this.checkVelocityRules(completeEvent, assessment),
   539→      this.checkLocationAnomalies(completeEvent, assessment),
   540→      this.checkDeviceRisk(completeEvent, assessment),
   541→      this.checkBehavioralAnomalies(completeEvent, assessment),
   542→      this.checkIdentityRisks(completeEvent, assessment),
   543→      this.checkPaymentRisks(completeEvent, assessment),
   544→      this.checkAccountRisks(completeEvent, assessment),
   545→      this.checkNetworkRisks(completeEvent, assessment),
   546→      this.evaluateCustomRules(completeEvent, assessment),
   547→    ]);
   548→
   549→    // Apply ML-style composite scoring
   550→    if (this.config.features.mlScoring) {
   551→      this.applyCompositeScoring(completeEvent, assessment);
   552→    }
   553→
   554→    // Determine final decision
   555→    this.makeDecision(assessment);
   556→
   557→    // Create case if needed
   558→    if (assessment.decision === 'review' || assessment.decision === 'block') {
   559→      if (this.config.features.caseManagement) {
   560→        await this.createFraudCase(completeEvent, assessment);
   561→      }
   562→    }
   563→
   564→    // Update statistics
   565→    this.updateStatistics(assessment);
   566→
   567→    // Trigger alerts
   568→    if (this.config.alerting.enabled) {
   569→      this.triggerAlerts(completeEvent, assessment);
   570→    }
   571→
   572→    return assessment;
   573→  }

   575→  /**
   576→   * Generate device fingerprint from client data
   577→   */
   578→  generateDeviceFingerprint(data: Partial<DeviceFingerprint>): DeviceFingerprint {
   579→    const components = [
   580→      data.userAgent,
   581→      data.language,
   582→      data.platform,
   583→      data.vendor,
   584→      `${data.screenWidth}x${data.screenHeight}`,
   585→      `${data.colorDepth}`,
   586→      `${data.pixelRatio}`,
   587→      String(data.deviceMemory),
   588→      String(data.hardwareConcurrency),
   589→      String(data.maxTouchPoints),
   590→      data.connectionType,
   591→      String(data.downlink),
   592→      String(data.rtt),
   593→      String(data.cookiesEnabled),
   594→      String(data.localStorage),
   595→      String(data.sessionStorage),
   596→      String(data.indexedDB),
   597→      data.webGL,
   598→      data.canvas,
   599→      data.fonts?.join(','),
   600→      data.plugins?.join(','),
   601→    ].filter(Boolean);
   602→
   603→    const hash = this.simpleHash(components.join('|'));
   604→
   605→    const now = new Date().toISOString();
   606→
   607→    // Check if we've seen this device before
   608→    let existingProfile = this.deviceProfiles.get(hash);
   609→    
   610→    if (existingProfile) {
   611→      existingProfile.lastSeen = now;
   612→      existingProfile.seenCount++;
   613→      
   614→      // Update risk assessment
   615→      existingProfile.riskScore = this.calculateDeviceRisk({ ...existingProfile, ...data });
   616→    } else {
   617→      existingProfile = {
   618→        id: `device_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
   619→        hash,
   620→        
   621→        userAgent: data.userAgent || '',
   622→        language: data.language || '',
   623→        languages: data.languages || [],
   624→        platform: data.platform || '',
   625→        vendor: data.vendor || '',
   626→        
   627→        screenWidth: data.screenWidth || 0,
   628→        screenHeight: data.screenHeight || 0,
   629→        colorDepth: data.colorDepth || 0,
   630→        pixelRatio: data.pixelRatio || 0,
   631→        
   632→        deviceMemory: data.deviceMemory,
   633→        hardwareConcurrency: data.hardwareConcurrency,
   634→        maxTouchPoints: data.maxTouchPoints,
   635→        
   636→        connectionType: data.connectionType,
   637→        downlink: data.downlink,
   638→        rtt: data.rtt,
   639→        
   640→        cookiesEnabled: data.cookiesEnabled ?? true,
   641→        localStorage: data.localStorage ?? true,
   642→        sessionStorage: data.sessionStorage ?? true,
   643→        indexedDB: data.indexedDB ?? true,
   644→        webGL: data.webGL,
   645→        canvas: data.canvas,
   646→        fonts: data.fonts,
   647→        plugins: data.plugins,
   648→        
   649→        firstSeen: now,
   650→        lastSeen: now,
   651→        seenCount: 1,
   652→        
   653→        riskScore: 0,
   654→        isEmulator: false,
   655→        isBot: false,
   656→        isVPN: false,
   657→        isTor: false,
   658→      };
   659→
   660→      // Analyze device characteristics
   661→      this.analyzeDeviceCharacteristics(existingProfile);
   662→      
   663→      this.deviceProfiles.set(hash, existingProfile);
   664→    }
   665→
   666→    return existingProfile;
   667→  }

   928→  private async checkVelocityRules(event: FraudEvent, assessment: FraudAssessment): Promise<void> {
   929→    if (!this.config.features.velocityChecking) return;
   930→
   931→    const now = Date.now();
   932→
   933→    // Check login velocity
   934→    if (event.eventType === 'login') {
   935→      const key = `login:${event.ipAddress}:${event.userId || 'anonymous'}`;
   936→      const recentLogins = this.getVelocityEntries(key, 3600000); // Last hour
   937→      
   938→      if (recentLogins.length >= this.config.velocityDefaults.maxLoginsPerHour) {
   939→        assessment.reasons.push({
   940→          ruleId: 'velocity_login_ip',
   941→          ruleName: 'Excessive Login Attempts from IP',
   942→          severity: 'warning',
   943→          scoreImpact: 25,
   944→          description: `${recentLogins.length} login attempts from this IP in the last hour`,
   945→        });
   946→      }
   947→
   948→      // Check per-user velocity
   949→      if (event.userId) {
   950→        const userKey = `login:user:${event.userId}`;
   951→        const userLogins = this.getVelocityEntries(userKey, 3600000);
   952→        
   953→        if (userLogins.length >= this.config.velocityDefaults.maxLoginsPerHour) {
   954→          assessment.reasons.push({
   955→            ruleId: 'velocity_login_user',
   956→            ruleName: 'Excessive Login Attempts by User',
   957→            severity: 'warning',
   958→            scoreImpact: 20,
   959→            description: `${userLogins.length} login attempts by this user in the last hour`,
   960→          });
   961→        }
   962→      }
   963→    }
   964→
   965→    // Check transaction velocity
   966→    if (event.eventType === 'transaction' || event.eventType === 'payment') {
   967→      const txKey = `transaction:${event.userId || event.ipAddress}`;
   968→      const recentTxs = this.getVelocityEntries(txKey, 3600000);
   969→      
   970→      if (recentTxs.length >= this.config.velocityDefaults.maxTransactionsPerHour) {
   971→        assessment.reasons.push({
   972→          ruleId: 'velocity_transaction',
   973→          ruleName: 'High Transaction Velocity',
   974→          severity: 'danger',
   975→          scoreImpact: 35,
   976→          description: `${recentTxs.length} transactions in the last hour`,
   977→        });
   978→      }
   979→
   980→      // High-value transaction velocity
   981→      const highValueTxs = recentTxs.filter(t => t.data.amount > 2000);
   982→      if (highValueTxs.length >= 3) {
   983→        assessment.reasons.push({
   984→          ruleId: 'velocity_high_value',
   985→          ruleName: 'Multiple High-Value Transactions',
   986→          severity: 'danger',
   987→          scoreImpact: 40,
   988→          description: `${highValueTxs.length} high-value transactions (>2000 DZD) in the last hour`,
   989→        });
   990→      }
   991→    }
   992→
   993→    // Registration velocity
   994→    if (event.eventType === 'registration') {
   995→      const regKey = `registration:${event.ipAddress}`;
   996→      const recentRegs = this.getVelocityEntries(regKey, 86400000); // Last day
   997→      
   998→      if (recentRegs.length >= this.config.velocityDefaults.maxRegistrationsPerIPPerDay) {
   999→        assessment.reasons.push({
  1000→          ruleId: 'velocity_registration',
  1001→          ruleName: 'Excessive Registrations from IP',
  1002→          severity: 'danger',
  1003→          scoreImpact: 45,
  1004→          description: `${recentRegs.length} registrations from this IP today`,
  1005→        });
  1006→      }
  1007→    }
  1008→
  1009→    // Password reset velocity
  1010→    if (event.eventType === 'password_reset') {
  1011→      const resetKey = `password_reset:${event.userId || event.data.email}`;
  1012→      const recentResets = this.getVelocityEntries(resetKey, 86400000);
  1013→      
  1014→      if (recentResets.length >= this.config.velocityDefaults.maxPasswordResetsPerDay) {
  1015→        assessment.reasons.push({
  1016→          ruleId: 'velocity_password_reset',
  1017→          ruleName: 'Excessive Password Resets',
  1018→          severity: 'warning',
  1019→          scoreImpact: 30,
  1020→          description: `${recentResets.length} password reset requests today`,
  1021→        });
  1022→      }
  1023→    }
  1024→
  1025→    // Record this event for future velocity checks
  1026→    const recordKey = `${event.eventType}:${event.userId || event.ipAddress}`;
  1027→    this.addVelocityEntry(recordKey, { timestamp: now, data: event.data });
  1028→  }

  1532→  private makeDecision(assessment: FraudAssessment): void {
  1533→    // Sum up all score impacts
  1534→    assessment.score = Math.min(100, assessment.reasons.reduce((sum, r) => sum + r.scoreImpact, 0));
  1535→
  1536→    // Determine risk level
  1537→    if (assessment.score < this.config.thresholds.lowRisk) {
  1538→      assessment.riskLevel = 'low';
  1539→    } else if (assessment.score < this.config.thresholds.mediumRisk) {
  1540→      assessment.riskLevel = 'medium';
  1541→    } else if (assessment.score < this.config.thresholds.highRisk) {
  1542→      assessment.riskLevel = 'high';
  1543→    } else {
  1544→      assessment.riskLevel = 'critical';
  1545→    }
  1546→
  1547→    // Determine decision based on mode and thresholds
  1548→    if (this.config.mode === 'detect') {
  1549→      assessment.decision = 'approve'; // Log only, don't block
  1550→    } else if (this.config.mode === 'prevent') {
  1551→      if (assessment.score >= this.config.thresholds.autoBlockThreshold) {
  1552→        assessment.decision = 'block';
  1553→      } else if (assessment.score >= this.config.thresholds.autoChallengeThreshold) {
  1554→        assessment.decision = 'challenge';
  1555→      } else if (assessment.score >= this.config.thresholds.autoReviewThreshold) {
  1556→        assessment.decision = 'review';
  1557→        assessment.reviewRequired = true;
  1558→      } else {
  1559→        assessment.decision = 'approve';
  1560→      }
  1561→    }
  1562→
  1563→    // Add recommendations based on findings
  1564→    if (assessment.reasons.some(r => r.ruleId.includes('device')) && event => event.fingerprint) {
  1565→      assessment.recommendations.push('Consider requiring MFA for this device');
  1566→    }
  1567→    if (assessment.reasons.some(r => r.ruleId.includes('geo'))) {
  1568→      assessment.recommendations.push('Verify user location via additional means');
  1569→    }
  1570→    if (assessment.reasons.some(r => r.severity === 'danger')) {
  1571→      assessment.recommendations.push('Manual review recommended due to high-severity indicators');
  1572→    }
  1573→  }

  1613→  private evaluateFraudRuleConditions(
  1614→    conditions: FraudRuleCondition[],
  1615→    event: FraudEvent,
  1616→    logic: 'AND' | 'OR'
  1617→  ): boolean {
  1618→    if (logic === 'AND') {
  1619→      return conditions.every(cond => this.evaluateFraudCondition(cond, event));
  1620→    } else {
  1621→      return conditions.some(cond => this.evaluateFraudCondition(cond, event));
  1622→    }
  1623→  }

  1625→  private evaluateFraudCondition(condition: FraudRuleCondition, event: FraudEvent): boolean {
  1626→    const value = this.getEventFieldValue(event, condition.field);
  1627→    
  1628→    switch (condition.operator) {
  1629→      case 'equals': return value === condition.value;
  1630→      case 'not_equals': return value !== condition.value;
  1631→      case 'contains': return typeof value === 'string' && value.includes(condition.value);
  1632→      case 'not_contains': return typeof value === 'string' && !value.includes(condition.value);
  1633→      case 'matches': return typeof value === 'string' && new RegExp(condition.value).test(value);
  1634→      case 'greater_than': return typeof value === 'number' && value > condition.value;
  1635→      case 'less_than': return typeof value === 'number' && value < condition.value;
  1636→      case 'in': return Array.isArray(condition.value) && condition.value.includes(value);
  1637→      case 'not_in': return !Array.isArray(condition.value) || !condition.value.includes(value);
  1638→      default:
  1639→        console.warn(`Unknown fraud rule operator: ${condition.operator}`);
  1640→        return false;
  1641→    }
  1642→  }
