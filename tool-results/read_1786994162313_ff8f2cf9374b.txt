   161→// Main Component
   162→// ============================================
   163→
   164→export default function ExhibitionScreen(props: ExhibitionScreenProps) {
   165→  const route = useRoute<ExhibitionRouteProp>();
   166→  const navigation = useNavigation();
   167→  
   168→  const exhibitionId = props.exhibitionId || (route.params?.id as string);
   169→  
   170→  // State
   171→  const [loading, setLoading] = useState(true);
   172→  const [refreshing, setRefreshing] = useState(false);
   173→  const [exhibitions, setExhibitions] = useState<Exhibition[]>([]);
   174→  const [selectedExhibition, setSelectedExhibition] = useState<Exhibition | null>(null);
   175→  const [showRegistrationModal, setShowRegistrationModal] = useState(false);
   176→  const [selectedRegistrationType, setSelectedRegistrationType] = useState<RegistrationType>('visitor');
   177→  const [companyName, setCompanyName] = useState('');
   178→  const [jobTitle, setJobTitle] = useState('');
   179→  const [interests, setInterests] = useState('');
   180→  const [registering, setRegistering] = useState(false);
   181→  const [activeTab, setActiveTab] = useState<'info' | 'booths' | 'events' | 'networking'>('info');
   182→  const [isOffline] = useOfflineStatus();
   183→
   184→  // Load exhibitions or single exhibition
   185→  const loadData = useCallback(async () => {
   186→    try {
   187→      if (exhibitionId) {
   188→        const data = await apiService.getExhibitionDetail(exhibitionId);
   189→        setSelectedExhibition(data);
   190→      } else {
   191→        const data = await apiService.getExhibitions();
   192→        setExhibitions(data);
   193→        
   194→        // Cache for offline
   195→        if (!isOffline) {
   196→          await offlineService.cacheExhibitions(data);
   197→        }
   198→      }
   199→    } catch (error) {
   200→      console.error('[ExhibitionScreen] Error loading data:', error);
   201→      
   202→      // Try cache if offline
   203→      if (isOffline || !exhibitionId) {
   204→        const cached = await offlineService.getCachedExhibitions();
   205→        if (cached.length > 0) {
   206→          if (exhibitionId) {
   207→            setSelectedExhibition(cached.find(e => e.id === exhibitionId) || null);
   208→          } else {
   209→            setExhibitions(cached);
   210→          }
   211→        }
   212→      }
   213→    } finally {
   214→      setLoading(false);
   215→      setRefreshing(false);
   216→    }
   217→  }, [exhibitionId, isOffline]);
   218→
   219→  useEffect(() => {
   220→    loadData();
   221→  }, [loadData]);
   222→
   223→  // Pull to refresh
   224→  const onRefresh = async () => {
   225→    setRefreshing(true);
   226→    await loadData();
   227→  };
   228→
   229→  // Register for exhibition
   230→  const handleRegister = async () => {
   231→    if (!selectedExhibition) return;
   232→
   233→    setRegistering(true);
   234→
   235→    try {
   236→      await apiService.registerForExhibition(selectedExhibition.id, {
   237→        type: selectedRegistrationType,
   238→        companyName: companyName || undefined,
   239→        jobTitle: jobTitle || undefined,
   240→        interests: interests || undefined,
   241→      });
   242→
   243→      Alert.alert(
   244→        'Inscription confirmée !',
   245→        `Vous êtes inscrit en tant que ${REGISTRATION_TYPES.find(r => r.value === selectedRegistrationType)?.label}.`,
   246→        [{ text: 'OK', onPress: () => {
   247→          setShowRegistrationModal(false);
   248→          loadData(); // Reload to show registration status
   249→        }}]
   250→      );
   251→    } catch (error) {
   252→      console.error('[ExhibitionScreen] Error registering:', error);
   253→      Alert.alert('Erreur', 'Impossible de finaliser l\'inscription.');
   254→    } finally {
   255→      setRegistering(false);
   256→    }
   257→  };
   258→
   259→  // Schedule meeting with exhibitor
   260→  const handleScheduleMeeting = (booth: ExhibitionBooth) => {
   261→    Alert.alert(
   262→      'Planifier une réunion',
   263→      `Souhaitez-vous planifier une réunion avec ${booth.companyName} ?`,
   264→      [
   265→        { text: 'Annuler', style: 'cancel' },
   266→        { text: 'Confirmer', onPress: () => {
   267→          navigation.navigate('Messages' as any, {
   268→            conversationId: `exhibition_${selectedExhibition?.id}_${booth.id}`,
   269→            userName: booth.companyName,
   270→          });
   271→        }}
   272→      ]
   273→    );
   274→  };
   275→
   276→  // Format date range
   277→  const formatDateRange = (start: Date, end: Date): string => {
   278→    const startDate = new Date(start);
   279→    const endDate = new Date(end);
   280→    
   281→    if (startDate.toDateString() === endDate.toDateString()) {
   282→      return startDate.toLocaleDateString('fr-FR', {
   283→        day: 'numeric',
   284→        month: 'long',
   285→        year: 'numeric',
   286→      });
   287→    }
   288→
   289→    return `${startDate.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })} - ${endDate.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short', year: 'numeric' })}`;
   290→  };
   291→
   292→  // Render loading
   293→  if (loading) {
   294→    return (
   295→      <View style={styles.loadingContainer}>
   296→        <ActivityIndicator size="large" color={Colors.primary} />
   297→        <Text style={styles.loadingText}>Chargement...</Text>
   298→      </View>
   299→    );
   300→  }
   301→
   302→  // List view (no specific exhibition selected)
   303→  if (!exhibitionId && !selectedExhibition) {
   304→    return (
   305→      <View style={styles.container}>
   306→        {/* Header */}
   307→        <View style={styles.header}>
   308→          <Text style={styles.headerTitle}>Expositions & Événements</Text>
   309→          <TouchableOpacity style={styles.filterButton}>
   310→            <Ionicons name="filter-outline" size={20} color={Colors.text} />
   311→          </TouchableOpacity>
   312→        </View>
   313→
   314→        <FlatList
   315→          data={exhibitions}
   316→          keyExtractor={(item) => item.id}
   317→          refreshControl={
   318→            <RefreshControl refreshing={refreshing} onRefresh={onRefresh} />
   319→          }
   320→          contentContainerStyle={styles.listContent}
   321→          renderItem={({ item }) => (
   322→            <TouchableOpacity
   323→              style={styles.exhibitionCard}
   324→              onPress={() => setSelectedExhibition(item)}
   325→              activeOpacity={0.7}
   326→            >
   327→              {/* Cover Image */}
   328→              <View style={styles.cardImageContainer}>
   329→                {item.coverImage ? (
   330→                  <Image source={{ uri: item.coverImage }} style={styles.cardImage} />
   331→                ) : (
   332→                  <View style={[styles.cardImagePlaceholder, { backgroundColor: EXHIBITION_TYPE_CONFIG[item.type]?.color + '20' }]}>
   333→                    <Ionicons 
   334→                      name={EXHIBITION_TYPE_CONFIG[item.type]?.icon as any} 
   335→                      size={48} 
   336→                      color={EXHIBITION_TYPE_CONFIG[item.type]?.color} 
   337→                    />
   338→                  </View>
   339→                )}
   340→                
   341→                {/* Status Badge */}
   342→                <View style={[
   343→                  styles.statusBadge,
   344→                  { backgroundColor: item.status === 'live' ? Colors.error : item.status === 'upcoming' ? Colors.primary : Colors.textTertiary }
   345→                ]}>
   346→                  <Text style={styles.statusText}>
   347→                    {item.status === 'live' ? 'EN DIRECT' : item.status === 'upcoming' ? 'À VENIR' : 'TERMINÉ'}
   348→                  </Text>
   349→                </View>
   350→
   351→                {/* Featured Badge */}
   352→                {item.featured && (
   353→                  <View style={styles.featuredBadge}>
   354→                    <Ionicons name="star" size={12} color="#F59E0B" />
   355→                    <Text style={styles.featuredBadgeText}>En vedette</Text>
   356→                  </View>
   357→                )}
   358→
   359→                {/* Type Badge */}
   360→                <View style={[styles.typeBadge, { backgroundColor: EXHIBITION_TYPE_CONFIG[item.type]?.color }]}>
   361→                  <Text style={styles.typeBadgeText}>
   362→                    {EXHIBITION_TYPE_CONFIG[item.type]?.label}
   363→                  </Text>
   364→                </View>
   365→              </View>
   366→
   367→              {/* Content */}
   368→              <View style={styles.cardContent}>
   369→                <Text style={styles.cardTitle} numberOfLines={2}>{item.title}</Text>
   370→                
   371→                <Text style={styles.cardDescription} numberOfLines={2}>
   372→                  {item.description}
   373→                </Text>
   374→
   375→                <View style={styles.cardMeta}>
   376→                  <View style={styles.metaItem}>
   377→                    <Ionicons name="calendar-outline" size={14} color={Colors.textSecondary} />
   378→                    <Text style={styles.metaText}>
   379→                      {formatDateRange(item.startDate, item.endDate)}
   380→                    </Text>
   381→                  </View>
   382→                  
   383→                  {item.location && (
   384→                    <View style={styles.metaItem}>
   385→                      <Ionicons name="location-outline" size={14} color={Colors.textSecondary} />
   386→                      <Text style={styles.metaText} numberOfLines={1}>
   387→                        {item.location.city}, {item.location.wilaya}
   388→                      </Text>
   389→                    </View>
   390→                  )}
   391→                </View>
   392→
   393→                <View style={styles.cardFooter}>
   394→                  <View style={styles.statsRow}>
   395→                    <View style={styles.statItem}>
   396→                      <Ionicons name="storefront-outline" size={14} color={Colors.textTertiary} />
   397→                      <Text style={styles.statText}>
   398→                        {item.currentExhibitors}/{item.maxExhibitors} exposants
   399→                      </Text>
   400→                    </View>
   401→                    <View style={styles.statItem}>
   402→                      <Ionicons name="people-outline" size={14} color={Colors.textTertiary} />
   403→                      <Text style={styles.statText}>
   404→                        {item.currentVisitors?.toLocaleString() || '0'} visiteurs
   405→                      </Text>
   406→                    </View>
   407→                  </View>
   408→
   409→                  {!item.userRegistration && item.status === 'upcoming' && (
   410→                    <TouchableOpacity 
   411→                      style={styles.registerButton}
   412→                      onPress={() => {
   413→                        setSelectedExhibition(item);
   414→                        setShowRegistrationModal(true);
   415→                      }}
   416→                    >
   417→                      <Text style={styles.registerButtonText}>S'inscrire</Text>
   418→                    </TouchableOpacity>
   419→                  )}
   420→
   421→                  {item.userRegistration && (
   422→                    <View style={styles.registeredBadge}>
   423→                      <Ionicons name="checkmark-circle" size={14} color={Colors.success} />
   424→                      <Text style={styles.registeredText}>Inscrit</Text>
   425→                    </View>
   426→                  )}
   427→                </View>
   428→              </View>
   429→            </TouchableOpacity>
   430→          )}
   431→          ListEmptyComponent={
   432→            <View style={styles.emptyContainer}>
   433→              <Ionicons name="calendar-outline" size={64} color={Colors.textTertiary} />
   434→              <Text style={styles.emptyTitle}>Aucun événement</Text>
   435→              <Text style={styles.emptySubtitle}>
   436→                Revenez plus tard pour découvrir les prochaines expositions.
   437→              </Text>
   438→            </View>
   439→          }
   440→        />
   441→
   442→        {/* Registration Modal */}
   443→        <RegistrationModal
   444→          visible={showRegistrationModal}
   445→          exhibition={selectedExhibition}
   446→          onClose={() => setShowRegistrationModal(false)}
   447→          onRegister={handleRegister}
   448→          loading={registering}
   449→        />
   450→      </View>
   451→    );
   452→  }
   453→
   454→  // Detail view (single exhibition)
   455→  if (!selectedExhibition) return null;
   456→
   457→  const typeConfig = EXHIBITION_TYPE_CONFIG[selectedExhibition.type];
   458→
   459→  return (
   460→    <View style={styles.container}>
   461→      {/* Header with Cover */}
   462→      <View style={styles.detailHeaderContainer}>
   463→        {selectedExhibition.coverImage ? (
   464→          <Image source={{ uri: selectedExhibition.coverImage }} style={styles.detailCoverImage} />
   465→        ) : (
   466→          <View style={[styles.detailCoverPlaceholder, { backgroundColor: typeConfig?.color + '30' }]}>
   467→            <Ionicons name={typeConfig?.icon as any} size={64} color={typeConfig?.color} />
   468→          </View>
   469→        )}
   470→        
   471→        <View style={styles.detailHeaderOverlay}>
   472→          <TouchableOpacity 
   473→            style={styles.backButton}
   474→            onPress={() => {
   475→              if (exhibitionId) {
   476→                navigation.goBack();
   477→              } else {
   478→                setSelectedExhibition(null);
   479→              }
   480→            }}
   481→          >
   482→            <Ionicons name="arrow-back" size={24} color={Colors.white} />
   483→          </TouchableOpacity>
   484→          
   485→          <View style={styles.detailHeaderActions}>
   486→            <TouchableOpacity style={styles.headerActionButton}>
   487→              <Ionicons name="share-social-outline" size={22} color={Colors.white} />
   488→            </TouchableOpacity>
   489→            <TouchableOpacity style={styles.headerActionButton}>
   490→              <Ionicons name="heart-outline" size={22} color={Colors.white} />
   491→            </TouchableOpacity>
   492→          </View>
   493→        </View>
   494→
   495→        {/* Gradient overlay */}
   496→        <View style={styles.gradientOverlay} />
   497→      </View>
   498→
   499→      {/* Title Section */}
   500→      <View style={styles.titleSection}>
   501→        <View style={styles.badgesRow}>
   502→          <View style={[
   503→            styles.detailTypeBadge, 
   504→            { backgroundColor: typeConfig?.color }
   505→          ]}>
   506→            <Text style={styles.detailTypeBadgeText}>{typeConfig?.label}</Text>
   507→          </View>
   508→          
   509→          <View style={[
   510→            styles.detailStatusBadge,
   511→            { 
   512→              backgroundColor: selectedExhibition.status === 'live' 
   513→                ? Colors.error 
   514→                : selectedExhibition.status === 'upcoming' 
   515→                  ? Colors.success 
   516→                  : Colors.textTertiary 
   517→            }
   518→          ]}>
   519→            <View style={[
   520→              styles.statusDot,
   521→              { backgroundColor: selectedExhibition.status === 'live' ? Colors.error : Colors.success }
   522→            ]} />
   523→            <Text style={styles.detailStatusText}>
   524→              {selectedExhibition.status === 'live' ? 'En direct' : selectedExhibition.status === 'upcoming' ? 'À venir' : 'Terminé'}
   525→            </Text>
   526→          </View>
   527→        </View>
   528→
   529→        <Text style={styles.detailTitle}>{selectedExhibition.title}</Text>
   530→        
   531→        <Text style={styles.detailOrganizer}>
   532→          Organisé par {selectedExhibition.organizerName}
   533→        </Text>
   534→
   535→        <View style={styles.dateLocationRow}>
   536→          <View style={styles.dateLocationItem}>
   537→            <Ionicons name="calendar-outline" size={18} color={Colors.primary} />
   538→            <Text style={styles.dateLocationText}>
   539→              {formatDateRange(selectedExhibition.startDate, selectedExhibition.endDate)}
   540→            </Text>
   541→          </View>
   542→          
   543→          {selectedExhibition.location && (
   544→            <View style={styles.dateLocationItem}>
   545→              <Ionicons name="location-outline" size={18} color={Colors.primary} />
   546→              <Text style={styles.dateLocationText} numberOfLines={1}>
   547→                {selectedExhibition.location.venue}, {selectedExhibition.location.city}
   548→              </Text>
   549→            </View>
   550→          )}
   551→        </View>
   552→      </View>
   553→
   554→      {/* Stats Bar */}
   555→      <View style={styles.statsBar}>
   556→        <View style={styles.statBox}>
   557→          <Text style={styles.statValue}>{selectedExhibition.currentExhibitors}</Text>
   558→          <Text style={styles.statLabel}>Exposants</Text>
   559→        </View>
   560→        <View style={styles.statDivider} />
   561→        <View style={styles.statBox}>
   562→          <Text style={styles.statValue}>{selectedExhibition.currentVisitors?.toLocaleString() || '0'}</Text>
   563→          <Text style={styles.statLabel}>Visiteurs</Text>
   564→        </View>
   565→        <View style={styles.statDivider} />
   566→        <View style={styles.statBox}>
   567→          <Text style={styles.statValue}>{selectedExhibition.events?.length || 0}</Text>
   568→          <Text style={styles.statLabel}>Événements</Text>
   569→        </View>
   570→      </View>
   571→
   572→      {/* Tabs */}
   573→      <View style={styles.tabBar}>
   574→        {(['info', 'booths', 'events', 'networking'] as const).map(tab => (
   575→          <TouchableOpacity
   576→            key={tab}
   577→            style={[styles.tab, activeTab === tab && styles.activeTab]}
   578→            onPress={() => setActiveTab(tab)}
