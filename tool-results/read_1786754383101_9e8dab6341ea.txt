    38→export interface WAFResponse {

    40→  blocked: boolean;
    41→  statusCode: number;
    42→  reason?: string;
    43→  ruleId?: string;
    44→  ruleName?: string;

    50→export interface WAFFirewallRule {
    51→  id: string;
    52→  name: string;
    53→  description: string;
    54→  enabled: boolean;
    55→  priority: number; // Higher = checked first
    56→  action: 'allow' | 'block' | 'challenge' | 'log' | 'captcha';
    57→  
    58→  // Conditions (AND logic)
    59→  conditions: Array<{
    60→    field: 'url' | 'headers' | 'query' | 'body' | 'ip' | 'userAgent' | 'country' | 'method' | 'contentType';
    61→    operator: 'equals' | 'contains' | 'matches' | 'startsWith' | 'endsWith' | 'in' | 'notIn' | 'exists' | 'notExists' | 'sizeGreaterThan' | 'sizeLessThan';
    62→    value: any;
    63→    caseSensitive?: boolean;
    64→  }>;
    65→  
    66→  // Metadata
    67→  category: 'sqli' | 'xss' | 'csrf' | 'path_traversal' | 'cmd_injection' | 'ssrf' | 'bot' | 'rate_limit' | 'geo' | 'ip_access' | 'custom';
    68→  severity: 'info' | 'low' | 'medium' | 'high' | 'critical';
    69→  tags: string[];
    70→  
    71→  // Rate limiting (for rate_limit action)
    72→  rateLimit?: {
    73→    windowMs: number;
    74→    maxRequests: number;
    75→  };
    76→  
    77→  // Statistics
    78→  hits: number;
    79→  lastTriggered?: string;
    80→  createdAt: string;
    81→  updatedAt: string;
    82→}

   394→      topAttackedEndpoints: [],

   396→      requestsByCountry: {},
   397→      averageResponseTime: 0,

   405→  /**
   406→   * Main request inspection method
   407→   */
   408→  async inspectRequest(request: Partial<WAFRequest>): Promise<WAFResponse> {
   409→    const startTime = Date.now();
   410→    
   411→    // Build complete request object
   412→    const wafRequest: WAFRequest = {
   413→      id: this.generateRequestId(),
   414→      timestamp: new Date().toISOString(),
   415→      method: request.method || 'GET',
   416→      url: request.url || '/',
   417→      headers: request.headers || {},
   418→      query: request.query || {},
   419→      body: request.body,
   420→      ip: request.ip || 'unknown',
   421→      userAgent: request.userAgent || '',
   422→      contentType: request.contentType,
   423→      contentLength: request.contentLength,
   424→    };
   425→
   426→    // Update statistics
   427→    this.statistics.totalRequests++;
   428→
   429→    // Quick checks first (fastest to evaluate)
   430→    
   431→    // 1. Check if WAF is enabled
   432→    if (!this.config.enabled) {
   433→      return this.createResponse('allow', wafRequest, startTime);
   434→    }
   435→
   436→    // 2. IP Allowlist/Blocklist check
   437→    const ipCheck = this.checkIPAccess(wafRequest);
   438→    if (!ipCheck.allowed) {
   439→      return ipCheck.response!;
   440→    }
   441→
   442→    // 3. Check if IP is temporarily blocked
   443→    const tempBlockCheck = this.checkTemporaryBlock(wafRequest);
   444→    if (!tempBlockCheck.allowed) {
   445→      return tempBlockCheck.response!;
   446→    }
   447→
   448→    // 4. Size validations
   449→    const sizeCheck = this.checkRequestSizes(wafRequest);
   450→    if (!sizeCheck.allowed) {
   451→      return sizeCheck.response!;
   452→    }
   453→
   454→    // 5. Geographic blocking
   455→    if (this.config.protections.geoBlocking && this.config.geoRules.length > 0) {
   456→      const geoCheck = this.checkGeoRules(wafRequest);
   457→      if (!geoCheck.allowed) {
   458→        return geoCheck.response!;
   459→      }
   460→    }
   461→
   462→    // 6. Bot detection
   463→    if (this.config.protections.botDetection && this.config.botDetection.enabled) {
   464→      const botCheck = this.checkBotDetection(wafRequest);
   465→      if (!botCheck.allowed) {
   466→        return botCheck.response!;
   467→      }
   468→    }
   469→
   470→    // 7. Rate limiting
   471→    if (this.config.protections.rateLimiting && this.config.rateLimiting.enabled) {
   472→      const rateLimitCheck = this.checkRateLimit(wafRequest);
   473→      if (!rateLimitCheck.allowed) {
   474→        return rateLimitCheck.response!;
   475→      }
   476→    }
   477→
   478→    // 8. Attack pattern detection (order matters for performance)
   479→    
   480→    // SQL Injection
   481→    if (this.config.protections.sqlInjection) {
   482→      const sqliCheck = this.checkSQLInjection(wafRequest);
   483→      if (!sqliCheck.allowed) {
   484→        return sqliCheck.response!;
   485→      }
   486→    }
   487→
   488→    // XSS
   489→    if (this.config.protections.xss) {
   490→      const xssCheck = this.checkXSS(wafRequest);
   491→      if (!xssCheck.allowed) {
   492→        return xssCheck.response!;
   493→      }
   494→    }
   495→
   496→    // Path Traversal
   497→    if (this.config.protections.pathTraversal) {
   498→      const pathCheck = this.checkPathTraversal(wafRequest);
   499→      if (!pathCheck.allowed) {
   500→        return pathCheck.response!;
   501→      }
   502→    }
   503→
   504→    // Command Injection
   505→    if (this.config.protections.commandInjection) {
   506→      const cmdCheck = this.checkCommandInjection(wafRequest);
   507→      if (!cmdCheck.allowed) {
   508→        return cmdCheck.response!;
   509→      }
   510→    }
   511→
   512→    // SSRF
   513→    if (this.config.protections.ssrf) {
   514→      const ssrfCheck = this.checkSSRF(wafRequest);
   515→      if (!ssrfCheck.allowed) {
   516→        return ssrfCheck.response!;
   517→      }
   518→    }
   519→
   520→    // 9. CSRF Token validation (for state-changing requests)
   521→    if (this.config.protections.csrf && this.config.csrf.enabled) {
   522→      const csrfCheck = this.checkCSRF(wafRequest);
   523→      if (!csrfCheck.allowed) {
   524→        return csrfCheck.response!;
   525→      }
   526→    }
   527→
   528→    // 10. Custom rules evaluation
   529→    for (const [ruleId, rule] of this.customRules) {
   530→      if (rule.enabled && this.evaluateRule(rule, wafRequest)) {
   531→        return this.handleRuleMatch(rule, wafRequest, startTime);
   532→      }
   533→    }
   534→
   535→    // Request passed all checks
   536→    return this.createResponse('allow', wafRequest, startTime);
   537→  }

   539→  // ===========================================
   540→  // CSRF Token Management
   541→  // ===========================================
   542→
   543→  generateCSRFToken(): string {
   544→    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
   545→    let token = '';
   546→    for (let i = 0; i < this.config.csrf.tokenLength; i++) {
   547→      token += chars.charAt(Math.floor(Math.random() * chars.length));
   548→    }
   549→    return token;
   550→  }
   551→
   552→  validateCSRFToken(token: string, sessionToken?: string): boolean {
   553→    if (!token) return false;
   554→    
   555→    // In production, compare against server-stored token
   556→    // For now, validate format
   557→    return token.length === this.config.csrf.tokenLength && /^[A-Za-z0-9]+$/.test(token);
   558→  }

   560→  getCSRFCookieOptions(): CookieOptions {
   561→    return this.config.csrf.cookieOptions;
   562→  }

   565→  // CAPTCHA Challenge Management
   566→  // ===========================================
   567→
   568→  generateCaptchaChallenge(ip: string): { challengeId: string; imageData: string } {
   569→    const challengeId = `captcha_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`;
   570→    const challenge = Math.random().toString(36).substring(2, 8); // Simple numeric captcha
   571→    
   572→    this.captchaChallenges.set(challengeId, {
   573→      challenge,
   574→      expires: Date.now() + 5 * 60 * 1000, // 5 minutes
   575→    });
   576→
   577→    // In production, generate actual image
   578→    return {
   579→      challengeId,
   580→      imageData: `data:image/svg+xml;base64,${Buffer.from(
   581→        `<svg xmlns="http://www.w3.org/2000/svg" width="200" height="80">
   582→          <text x="10" y="50" font-family="monospace" font-size="40" fill="#333">${challenge}</text>
   583→        </svg>`
   584→      ).toString('base64')}`,
   585→    };
   586→  }
   587→
   588→  validateCaptcha(challengeId: string, answer: string): boolean {
   589→    const stored = this.captchaChallenges.get(challengeId);
   590→    
   591→    if (!stored || Date.now() > stored.expires) {
   592→      this.captchaChallenges.delete(challengeId);
   593→      return false;
   594→    }
   595→
   596→    const valid = stored.challenge === answer.toLowerCase();
   597→    this.captchaChallenges.delete(challengeId);
   598→    
   599→    return valid;
   600→  }

   606→  addRule(rule: Omit<WAFFirewallRule, 'id' | 'hits' | 'createdAt' | 'updatedAt'>): WAFFirewallRule {
   607→    const newRule: WAFFirewallRule = {
   608→      ...rule,
   609→      id: `rule_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
   610→      hits: 0,
   611→      createdAt: new Date().toISOString(),
   612→      updatedAt: new Date().toISOString(),
   613→    };
   614→
   615→    this.customRules.set(newRule.id, newRule);
   616→    return newRule;
   617→  }

   619→  removeRule(ruleId: string): boolean {
   620→    return this.customRules.delete(ruleId);

   916→      if (action !== 'allow') {

   927→      }
   928→    }

   933→  private checkRateLimit(request: WAFRequest): { allowed: boolean; response?: WAFResponse } {
   934→    const now = Date.now();
   935→    const ipKey = `ip:${request.ip}`;
   936→    const globalKey = `global:${request.ip}`;
   937→    const endpointKey = `endpoint:${request.method}:${request.ip}:${request.url.split('?')[0]}`;
   938→
   939→    // Check global IP limit
   940→    const globalEntry = this.getOrCreateRateLimitEntry(
   941→      globalKey,
   942→      this.config.rateLimiting.perIPEWindowMs,
   943→      this.config.rateLimiting.perIPMaxRequests
   944→    );
   945→
   946→    if (globalEntry.count > globalEntry.max) {
   947→      // Auto-block for repeated violations
   948→      if (globalEntry.count > globalEntry.max * 2) {
   949→        this.blockIP(request.ip, 30, 'Excessive rate limiting violations');
   950→      }
   951→
   952→      return {
   953→        allowed: false,
   954→        response: this.createResponse('rate_limit', request, Date.now(), {
   955→          reason: 'IP rate limit exceeded',
   956→          ruleId: 'rate_limit_ip',
   957→          ruleName: 'IP Rate Limiting',
   958→          severity: 'low',
   959→          metadata: { 
   960→            current: globalEntry.count, 
   961→            max: globalEntry.max, 
   962→            resetIn: Math.ceil((globalEntry.resetTime - now) / 1000) 
   963→          },
   964→        }),
   965→      };
   966→    }
   967→
   968→    // Check per-endpoint limits
   969→    if (this.config.rateLimiting.perEndpointEnabled) {
   970→      for (const limit of this.config.rateLimiting.perEndpointLimits) {
   971→        if (limit.pattern.test(request.url)) {
   972→          const endpointEntry = this.getOrCreateRateLimitEntry(
   973→            endpointKey,
   974→            limit.windowMs,
   975→            limit.maxRequests
   976→          );
   977→
   978→          if (endpointEntry.count > endpointEntry.max) {
   979→            return {
   980→              allowed: false,
   981→              response: this.createResponse('rate_limit', request, Date.now(), {
   982→                reason: 'Endpoint rate limit exceeded',
   983→                ruleId: 'rate_limit_endpoint',
   984→                ruleName: 'Endpoint Rate Limiting',
   985→                severity: 'low',
   986→                metadata: {
   987→                  current: endpointEntry.count,
   988→                  max: endpointEntry.max,
   989→                  resetIn: Math.ceil((endpointEntry.resetTime - now) / 1000),
   990→                  endpoint: request.url.split('?')[0],
   991→                },
   992→              }),
   993→            };
   994→          }
   995→          
   996→          break;
   997→        }
   998→      }
   999→    }
  1000→
  1001→    // Increment counters
  1002→    globalEntry.count++;
  1003→    endpointKey && this.incrementRateLimit(endpointKey);
  1004→
  1005→    return { allowed: true };
  1006→  }

  1190→    // In a real implementation, we'd validate the token against the session
  1191→    // For now, just check presence for non-exempt methods

  1194→      request.headers['x-csrf-token'] ||
  1195→      request.query._csrf;

  1197→    if (!csrfToken) {
  1198→      return {
  1199→        allowed: false,

  1241→      
  1242→      // Track blocked IPs
  1243→      const existing = this.statistics.topBlockedIPs.find(entry => entry.ip === request.ip);
  1244→      if (existing) {
  1245→        existing.count++;
  1246→      } else {
  1247→        this.statistics.topBlockedIPs.push({ ip: request.ip, count: 1, reason: options?.reason || 'Unknown' });
  1248→      }
  1249→
  1250→      // Track attack types
  1251→      const attackType = options?.ruleName || 'Unknown';
  1252→      this.statistics.attackTypes[attackType] = (this.statistics.attackTypes[attackType] || 0) + 1;
  1253→
  1254→      // Track attacked endpoints
  1255→      const endpoint = `${request.method} ${request.url.split('?')[0]}`;
  1256→      const existingEndpoint = this.statistics.topAttackedEndpoints.find(e => e.endpoint === endpoint);
  1257→      if (existingEndpoint) {
  1258→        existingEndpoint.count++;
  1259→      } else {
  1260→        this.statistics.topAttackedEndpoints.push({ endpoint, count: 1 });
  1261→      }
  1262→    } else if (action === 'challenge') {
  1263→      this.statistics.challengedRequests++;
  1264→    } else {
  1265→      this.statistics.allowedRequests++;
  1266→    }
  1267→
  1268→    // Track by country
  1269→    if (request.countryCode) {
  1270→      this.statistics.requestsByCountry[request.countryCode] = 
  1271→        (this.statistics.requestsByCountry[request.countryCode] || 0) + 1;
  1272→    }
  1273→
  1274→    // Update average response time
  1275→    this.statistics.averageResponseTime = 
  1276→      (this.statistics.averageResponseTime + duration) / 2;
  1277→
  1278→    // Log if configured
  1279→    if (this.config.logging.enabled && ((blocked && this.config.logging.logBlockedRequests) || 
  1280→        (!blocked && this.config.logging.logAllowedRequests))) {
  1281→      this.logRequest(request, { ...options, action, duration } as WAFResponse);
  1282→    }
  1283→
  1284→    // Trigger alerts if needed
  1285→    if (this.config.alerting.enabled && blocked) {
  1286→      this.checkAndTriggerAlert(options?.severity);
  1287→    }
  1288→
  1289→    return {
  1290→      allowed: !blocked,
  1291→      blocked,
  1292→      statusCode: this.getStatusCode(action),
  1293→      reason: options?.reason,
  1294→      ruleId: options?.ruleId,
  1295→      ruleName: options?.ruleName,
  1296→      action,
  1297→      severity: options?.severity || 'low',
  1298→      metadata: options?.metadata,
  1299→    };
  1300→  }

  1302→  private getStatusCode(action: WAFResponse['action']): number {
  1303→    switch (action) {
  1304→      case 'block': return 403;
  1305→      case 'challenge': return 429;
  1306→      case 'rate_limit': return 429;
  1307→      case 'captcha': return 403;
  1308→      case 'allow': return 200;
  1309→      default: return 200;
  1310→    }
  1311→  }

  1331→  private evaluateRule(rule: WAFFirewallRule, request: WAFRequest): boolean {
  1332→    // All conditions must match (AND logic)
  1333→    return rule.conditions.every(condition => {
  1334→      const value = this.getFieldValue(request, condition.field);
  1335→      return this.evaluateCondition(value, condition);
  1336→    });
  1337→  }
  1338→
  1339→  private getFieldValue(request: WAFRequest, field: WAFFirewallRule['conditions'][0]['field']): any {
  1340→    switch (field) {
  1341→      case 'url': return request.url;
  1342→      case 'headers': return JSON.stringify(request.headers);
  1343→      case 'query': return JSON.stringify(request.query);
  1344→      case 'body': return typeof request.body === 'string' ? request.body : JSON.stringify(request.body);
  1345→      case 'ip': return request.ip;
  1346→      case 'userAgent': return request.userAgent;
  1347→      case 'country': return request.countryCode;
  1348→      case 'method': return request.method;
  1349→      case 'contentType': return request.contentType;
  1350→      default: return undefined;
  1351→    }
  1352→  }
  1353→
  1354→  private evaluateCondition(value: any, condition: WAFFirewallRule['conditions'][0]): boolean {
  1355→    const { operator, operand, caseSensitive = true } = condition;
  1356→    let compareValue = value;
  1357→    let conditionValue = operand;
  1358→
  1359→    if (!caseSensitive && typeof value === 'string' && typeof operand === 'string') {
  1360→      compareValue = value.toLowerCase();
  1361→      conditionValue = operand.toLowerCase();
  1362→    }
  1363→
  1364→    switch (operator) {
  1365→      case 'equals':
  1366→        return compareValue === conditionValue;
  1367→      case 'contains':
  1368→        return typeof compareValue === 'string' && compareValue.includes(conditionValue as string);
  1369→      case 'matches':
  1370→        return typeof compareValue === 'string' && new RegExp(conditionValue as string).test(compareValue);
  1371→      case 'startsWith':
  1372→        return typeof compareValue === 'string' && compareValue.startsWith(conditionValue as string);
  1373→      case 'endsWith':
  1374→        return typeof compareValue === 'string' && compareValue.endsWith(conditionValue as string);
  1375→      case 'in':
  1376→        return Array.isArray(conditionValue) && conditionValue.includes(compareValue);
  1377→      case 'notIn':
  1378→        return !Array.isArray(conditionValue) || !conditionValue.includes(compareValue);
  1379→      case 'exists':
  1380→        return value !== undefined && value !== null;
  1381→      case 'notExists':
  1382→        return value === undefined || value === null;
  1383→      case 'sizeGreaterThan':
  1384→        return typeof value === 'number' && value > (conditionValue as number);
  1385→      case 'sizeLessThan':
  1386→        return typeof value === 'number' && value < (conditionValue as number);
  1387→      default:
  1388→        return false;
  1389→    }
  1390→  }

  1392→  private getOrCreateRateLimitEntry(key: string, windowMs: number, max: number): { count: number; resetTime: number; max: number } {
  1393→    let entry = this.rateLimitStore.get(key);
  1394→    const now = Date.now();
  1395→
  1396→    if (!entry || entry.resetTime <= now) {
  1397→      entry = {
  1398→        count: 0,
  1399→        resetTime: now + windowMs,
  1400→        max,
  1401→      };
  1402→      this.rateLimitStore.set(key, entry);
  1403→
  1404→      // Clean up expired entries periodically
  1405→      if (Math.random() < 0.01) { // 1% chance to clean
  1406→        this.cleanExpiredRateLimitEntries();
  1407→      }
  1408→    }
  1409→
  1410→    return entry;
  1411→  }
  1412→
  1413→  private incrementRateLimit(key: string): void {
  1414→    const entry = this.rateLimitStore.get(key);
  1415→    if (entry) {
  1416→      entry.count++;
  1417→    }
  1418→  }
  1419→
  1420→  private cleanExpiredRateLimitEntries(): void {
  1421→    const now = Date.now();
  1422→    for (const [key, entry] of this.rateLimitStore.entries()) {
  1423→      if (entry.resetTime <= now) {
  1424→        this.rateLimitStore.delete(key);
  1425→      }
  1426→    }
  1427→  }

  1488→  private logRequest(request: WAFRequest, response: WAFResponse): void {
  1489→    // Mask sensitive fields
  1490→    const maskedRequest = { ...request };
  1491→    if (this.config.logging.sensitiveFields.length > 0 && maskedRequest.body) {
  1492→      maskedRequest.body = this.maskSensitiveData(maskedRequest.body);
  1493→    }
  1494→
  1495→    // In production, send to log aggregation service
  1496→    console.log(`[WAF] ${response.action.toUpperCase()} | ${request.method} ${request.url} | IP: ${request.ip} | Reason: ${response.reason}`);
  1497→  }

  1499→  private maskSensitiveData(data: any): any {
  1500→    if (typeof data !== 'object' || data === null) return data;
  1501→
  1502→    const masked = Array.isArray(data) ? [...data] : { ...data };
  1503→
  1504→    for (const key of Object.keys(masked)) {
  1505→      if (this.config.logging.sensitiveFields.some(field => 
  1506→          key.toLowerCase().includes(field.toLowerCase()))) {
  1507→        (masked as any)[key] = '[MASKED]';
  1508→      } else if (typeof (masked as any)[key] === 'object') {
  1509→        (masked as any)[key] = this.maskSensitiveData((masked as any)[key]);
  1510→      }
  1511→    }
  1512→
  1513→    return masked;
  1514→  }
