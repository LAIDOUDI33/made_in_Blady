   348→// Donut Chart Component (SVG-based)
   349→function DonutChart({ data }: { data: typeof revenueByPaymentMethod }) {
   350→  const radius = 80
   351→  const innerRadius = 50
   352→  const centerX = 100
   353→  const centerY = 100
   354→
   355→  // Calculate cumulative angles without mutation
   356→  const calculateArcs = () => {
   357→    let cumulativeAngle = 0
   358→    return data.map((item) => {
   359→      const angle = (item.percentage / 100) * 360
   360→      const startAngle = cumulativeAngle
   361→      const endAngle = cumulativeAngle + angle
   362→      cumulativeAngle = endAngle
   363→
   364→      const startRad = (startAngle - 90) * Math.PI / 180
   365→      const endRad = (endAngle - 90) * Math.PI / 180
   366→
   367→      const x1 = centerX + radius * Math.cos(startRad)
   368→      const y1 = centerY + radius * Math.sin(startRad)
   369→      const x2 = centerX + radius * Math.cos(endRad)
   370→      const y2 = centerY + radius * Math.sin(endRad)
   371→
   372→      const ix1 = centerX + innerRadius * Math.cos(startRad)
   373→      const iy1 = centerY + innerRadius * Math.sin(startRad)
   374→      const ix2 = centerX + innerRadius * Math.cos(endRad)
   375→      const iy2 = centerY + innerRadius * Math.sin(endRad)
   376→
   377→      const largeArc = angle > 180 ? 1 : 0
   378→
   379→      return {
   380→        color: item.color,
   381→        pathD: `M ${x1} ${y1} A ${radius} ${radius} 0 ${largeArc} 1 ${x2} ${y2} L ${ix2} ${iy2} A ${innerRadius} ${innerRadius} 0 ${largeArc} 0 ${ix1} ${iy1} Z`
   382→      }
   383→    })
   384→  }
   385→
   386→  const arcs = calculateArcs()
   387→
   388→  return (
   389→    <div className="flex items-center justify-center">
   390→      <svg width="200" height="200" viewBox="0 0 200 200">
   391→        {arcs.map((arc, index) => (
   392→          <g key={index}>
   393→            <path
   394→              d={arc.pathD}
   395→              fill={arc.color}
   396→              className="hover:opacity-80 transition-opacity cursor-pointer"
   397→            />
   398→          </g>
   399→        ))}
   400→        <text x={centerX} y={centerY - 5} textAnchor="middle" className="fill-gray-700 font-bold text-sm">Total</text>
   401→        <text x={centerX} y={centerY + 15} textAnchor="middle" className="fill-gray-500 text-xs">2.5M د.ج</text>
   402→      </svg>
   403→    </div>
   404→  )
   405→}
