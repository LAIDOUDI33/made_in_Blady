   234→export default function VerificationsManagementPage() {
   235→  const [verifications, setVerifications] = useState<VerificationRequest[]>(sampleVerifications);
   236→  const [activeTab, setActiveTab] = useState('all');
   237→  const [searchQuery, setSearchQuery] = useState('');
   238→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   239→  const [typeFilter, setTypeFilter] = useState<string>('ALL');
   240→  const [levelFilter, setLevelFilter] = useState<string>('ALL');
   241→  
   242→  // Detail dialog states
   243→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   244→  const [selectedVerification, setSelectedVerification] = useState<VerificationRequest | null>(null);
   245→  
   246→  // Action dialog states
   247→  const [actionDialogOpen, setActionDialogOpen] = useState(false);
   248→  const [actionType, setActionType] = useState<'approve' | 'reject'>('approve');
   249→  const [actionReason, setActionReason] = useState('');
   250→  const [actionNotes, setActionNotes] = useState('');
   251→  const [isLoading, setIsLoading] = useState(false);
   252→
   253→  // Filter verifications
   254→  const filteredVerifications = useMemo(() => {
   255→    let result = [...verifications];
   256→    
   257→    if (activeTab !== 'all') {
   258→      result = result.filter(v => v.status === activeTab.toUpperCase());
   259→    }
   260→    
   261→    if (statusFilter !== 'ALL') {
   262→      result = result.filter(v => v.status === statusFilter);
   263→    }
   264→    
   265→    if (typeFilter !== 'ALL') {
   266→      result = result.filter(v => v.type === typeFilter);
   267→    }
   268→    
   269→    if (levelFilter !== 'ALL') {
   270→      result = result.filter(v => v.level === levelFilter);
   271→    }
   272→    
   273→    if (searchQuery.trim()) {
   274→      const query = searchQuery.toLowerCase();
   275→      result = result.filter(v => 
   276→        v.userName.toLowerCase().includes(query) ||
   277→        v.userEmail.toLowerCase().includes(query) ||
   278→        v.companyName?.toLowerCase().includes(query) ||
   279→        v.id.toLowerCase().includes(query)
   280→      );
   281→    }
   282→    
   283→    // Sort: pending first, then by date
   284→    result.sort((a, b) => {
   285→      if (a.status === 'PENDING' && b.status !== 'PENDING') return -1;
   286→      if (a.status !== 'PENDING' && b.status === 'PENDING') return 1;
   287→      return new Date(b.submittedAt).getTime() - new Date(a.submittedAt).getTime();
   288→    });
   289→    
   290→    return result;
   291→  }, [verifications, activeTab, searchQuery, statusFilter, typeFilter, levelFilter]);
   292→
   293→  // Stats
   294→  const pendingCount = verifications.filter(v => v.status === 'PENDING').length;
   295→  const approvedToday = verifications.filter(v => 
   296→    v.status === 'APPROVED' && 
   297→    v.reviewedAt && 
   298→    new Date(v.reviewedAt).toDateString() === new Date().toDateString()
   299→  ).length;
   300→  const totalApproved = verifications.filter(v => v.status === 'APPROVED').length;
   301→  const rejectionRate = verifications.length > 0 
   302→    ? Math.round((verifications.filter(v => v.status === 'REJECTED').length / verifications.length) * 100) 
   303→    : 0;
   304→
   305→  // Handlers
   306→  const openDetailDialog = (verification: VerificationRequest) => {
   307→    setSelectedVerification(verification);
   308→    setDetailDialogOpen(true);
   309→  };
   310→
   311→  const openActionDialog = (verification: VerificationRequest, action: 'approve' | 'reject') => {
   312→    setSelectedVerification(verification);
   313→    setActionType(action);
   314→    setActionDialogOpen(true);
   315→    setActionReason('');
   316→    setActionNotes('');
   317→  };
   318→
   319→  const handleAction = async () => {
   320→    if (!selectedVerification) return;
   321→    
   322→    setIsLoading(true);
   323→    await new Promise(resolve => setTimeout(resolve, 800));
   324→    
   325→    setVerifications(prev => prev.map(v => 
   326→      v.id === selectedVerification.id 
   327→        ? { 
   328→            ...v, 
   329→            status: actionType === 'approve' ? 'APPROVED' as const : 'REJECTED' as const,
   330→            reviewedAt: new Date().toISOString(),
   331→            reviewedBy: 'Admin Current',
   332→            rejectionReason: actionType === 'reject' ? actionReason : undefined,
   333→            notes: actionType === 'approve' ? actionNotes : undefined,
   334→            badgeEarned: actionType === 'approve' ? getBadgeForType(v.type, v.level) : undefined,
   335→          }
   336→        : v
   337→    ));
   338→    
   339→    setIsLoading(false);
   340→    setActionDialogOpen(false);
   341→    setDetailDialogOpen(false);
   342→  };
   343→
   344→  const getBadgeForType = (type: VerificationType, level: VerificationLevel): string => {
   345→    if (type === 'IDENTITY') return 'Identité Vérifiée';
   346→    if (type === 'COMPANY' && level === 'ADVANCED') return 'Entreprise Certifiée Gold';
   347→    if (type === 'COMPANY') return 'Entreprise Vérifiée';
   348→    if (type === 'PROFESSIONNEL') return 'Pro Certifié';
   349→    return 'Membre Premium';
   350→  };
   351→
   352→  const formatDate = (dateString: string) => {
   353→    return new Date(dateString).toLocaleDateString('fr-FR', {
   354→      day: '2-digit',
   355→      month: '2-digit',
   356→      year: 'numeric',
   357→      hour: '2-digit',
   358→      minute: '2-digit'
   359→    });
   360→  };
   361→
   362→  const exportToCSV = () => {
   363→    const headers = ['ID', 'Utilisateur', 'Email', 'Entreprise', 'Type', 'Niveau', 'Statut', 'Date soumission', 'Date revue'];
   364→    const csvRows = filteredVerifications.map(v => [
   365→      v.id,
   366→      v.userName,
   367→      v.userEmail,
   368→      v.companyName || '',
   369→      typeConfig[v.type].label,
   370→      v.level,
   371→      statusConfig[v.status].label,
   372→      formatDate(v.submittedAt),
   373→      v.reviewedAt ? formatDate(v.reviewedAt) : ''
   374→    ]);
   375→    
   376→    const csvContent = [headers, ...csvRows]
   377→      .map(row => row.map(cell => `"${cell}"`).join(','))
   378→      .join('\n');
   379→    
   380→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   381→    const link = document.createElement('a');
   382→    link.href = URL.createObjectURL(blob);
   383→    link.download = `verifications_${new Date().toISOString().split('T')[0]}.csv`;
   384→    link.click();
   385→  };
   386→
   387→  return (
   388→    <div className="space-y-6">
   389→      {/* Header */}
   390→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   391→        <div>
   392→          <h1 className="text-2xl font-bold text-gray-900">Gestion des Vérifications</h1>
   393→          <p className="text-gray-500 mt-1">Traitez les demandes de vérification et gérez les badges</p>
   394→        </div>
   395→        <div className="flex items-center gap-2">
   396→          <Button variant="outline" size="sm" onClick={exportToCSV}>
   397→            <Download className="mr-2 h-4 w-4" /> Exporter CSV
   398→          </Button>
   399→        </div>
   400→      </div>
   401→
   402→      {/* Stats Cards */}
   403→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   404→        <StatsCard
   405→          title="En attente"
   406→          value={pendingCount}
   407→          description="Demandes à traiter"
   408→          icon={Clock}
   409→          iconClassName={pendingCount > 0 ? "bg-yellow-100 text-yellow-600 animate-pulse" : "bg-gray-100 text-gray-600"}
   410→          trend={{ value: 12, isPositive: false }}
   411→        />
   412→        <StatsCard
   413→          title="Approuvés aujourd'hui"
   414→          value={approvedToday}
   415→          description={`${totalApproved} total approuvés`}
   416→          icon={CheckCircle2}
   417→          iconClassName="bg-green-100 text-green-600"
   418→        />
   419→        <StatsCard
   420→          title="Taux de rejet"
   421→          value={`${rejectionRate}%`}
   422→          description="Sur le dernier mois"
   423→          icon={XCircle}
   424→          iconClassName="bg-red-100 text-red-600"
   425→          trend={{ value: rejectionRate > 15 ? 5 : -3, isPositive: rejectionRate <= 15 }}
   426→        />
   427→        <StatsCard
   428→          title="Badges attribués"
   429→          value={totalApproved}
   430→          description="Ce mois-ci"
   431→          icon={Award}
   432→          iconClassName="bg-purple-100 text-purple-600"
   433→        />
   434→      </div>
   435→
   436→      {/* Filters */}
   437→      <Card>
   438→        <CardHeader className="pb-4">
   439→          <CardTitle className="text-base flex items-center gap-2">
   440→            <Filter className="h-4 w-4" />
   441→            Filtres
   442→          </CardTitle>
   443→        </CardHeader>
   444→        <CardContent>
   445→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-5 gap-4">
   446→            <div className="lg:col-span-2 relative">
   447→              <Label htmlFor="search" className="sr-only">Rechercher</Label>
   448→              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-gray-400" />

   450→                id="search"
   451→                placeholder="Rechercher par nom, email ou ID..."
   452→                value={searchQuery}
   453→                onChange={(e) => setSearchQuery(e.target.value)}
   454→                className="pl-10"
   455→              />
   456→            </div>
   457→
   458→            <div>
   459→              <Label>Statut</Label>
   460→              <Select value={statusFilter} onValueChange={setStatusFilter}>
   461→                <SelectTrigger>
   462→                  <SelectValue placeholder="Tous les statuts" />
   463→                </SelectTrigger>
   464→                <SelectContent>
   465→                  <SelectItem value="ALL">Tous</SelectItem>
   466→                  <SelectItem value="PENDING">En attente</SelectItem>

   473→            </div>

   475→            <div>

   477→              <Select value={typeFilter} onValueChange={setTypeFilter}>
   478→                <SelectTrigger>
   479→                  <SelectValue placeholder="Tous les types" />
   480→                </SelectTrigger>
   481→                <SelectContent>
   482→                  <SelectItem value="ALL">Tous</SelectItem>
   483→                  <SelectItem value="IDENTITY">Identité</SelectItem>
   484→                  <SelectItem value="COMPANY">Entreprise</SelectItem>
   485→                  <SelectItem value="PROFESSIONAL">Professionnel</SelectItem>
   486→                  <SelectItem value="PREMIUM">Premium</SelectItem>
   487→                </SelectContent>
   488→              </Select>
   489→            </div>
   490→
   491→            <div>
   492→              <Label>Niveau</Label>
   493→              <Select value={levelFilter} onValueChange={setLevelFilter}>
   494→                <SelectTrigger>
   495→                  <SelectValue placeholder="Tous les niveaux" />
   496→                </SelectTrigger>
   497→                <SelectContent>
   498→                  <SelectItem value="ALL">Tous</SelectItem>
   499→                  <SelectItem value="BASIC">Basic</SelectItem>
   500→                  <SelectItem value="STANDARD">Standard</SelectItem>
   501→                  <SelectItem value="ADVANCED">Avancé</SelectItem>
   502→                </SelectContent>
   503→              </Select>
   504→            </div>
   505→          </div>
   506→        </CardContent>
   507→      </Card>
   508→
   509→      {/* Tabs and Table */}
   510→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   511→        <TabsList>
   512→          <TabsTrigger value="all">Toutes ({verifications.length})</TabsTrigger>
   513→          <TabsTrigger value="PENDING" className="gap-1">
   514→            <Clock className="h-3.5 w-3.5 hidden sm:inline" /> En attente ({pendingCount})
   515→          </TabsTrigger>
   516→          <TabsTrigger value="APPROVED" className="gap-1">
   517→            <CheckCircle2 className="h-3.5 w-3.5 hidden sm:inline" /> Approuvés
   518→          </TabsTrigger>
   519→          <TabsTrigger value="REJECTED" className="gap-1">
   520→            <XCircle className="h-3.5 w-3.5 hidden sm:inline" /> Rejetés

   523→
   524→        <TabsContent value={activeTab} className="mt-4">
   525→          <Card>
   526→            <CardContent className="p-0">
   527→              {filteredVerifications.length > 0 ? (
   528→                <div className="overflow-x-auto">
   529→                  <Table>
