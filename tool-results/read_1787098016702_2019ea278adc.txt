   100→// Helper functions
   101→function formatDZD(amount: number): string {
   102→  return new Intl.NumberFormat('fr-DZ', {
   103→    style: 'decimal',
   104→    minimumFractionDigits: 2,
   105→    maximumFractionDigits: 2,
   106→  }).format(amount) + ' د.ج'
   107→}
   108→
   109→function getStatusConfig(status: InvoiceStatus) {
   110→  switch (status) {
   111→    case 'draft':
   112→      return { label: 'Brouillon', color: 'bg-gray-100 text-gray-700 border-gray-200', icon: FileText }
   113→    case 'issued':
   114→      return { label: 'Émise', color: 'bg-blue-100 text-blue-700 border-blue-200', icon: Clock }
   115→    case 'paid':
   116→      return { label: 'Payée', color: 'bg-green-100 text-green-700 border-green-200', icon: CheckCircle }
   117→    case 'overdue':
   118→      return { label: 'En retard', color: 'bg-red-100 text-red-700 border-red-200', icon: AlertCircle }
   119→    case 'cancelled':
   120→      return { label: 'Annulée', color: 'bg-gray-100 text-gray-400 border-gray-200 line-through', icon: XCircle }
   121→  }
   122→}
