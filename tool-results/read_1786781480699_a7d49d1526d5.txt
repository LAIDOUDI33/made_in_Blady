   566→class PrivacyComplianceSystem {

   575→
   576→  constructor(config?: Partial<PrivacyConfig>) {
   577→    this.config = { ...DEFAULT_CONFIG, ...config };

   579→    // Load built-in privacy policy template
   580→    this.loadDefaultPrivacyPolicy();
   581→  }

   583→  // ===========================================
   584→  // Data Subject Operations
   585→  // ===========================================
   586→
   587→  /**
   588→   * Register a new data subject or update existing
   589→   */
   590→  registerDataSubject(subject: Partial<DataSubject>): DataSubject {
   591→    const id = subject.id || subject.userId || subject.email || `subject_${Date.now()}`;
   592→    
   593→    const now = new Date().toISOString();
   594→    
   595→    const dataSubject: DataSubject = {
   596→      id,
   597→      type: subject.type || 'visitor',
   598→      profileData: subject.profileData || {},
   599→      accountData: subject.accountData || {},
   600→      transactionData: subject.transactionData || {},
   601→      communicationsData: {
   602→        marketingConsent: false,
   603→        newsletterSubscribed: false,
   604→        smsConsent: false,
   605→        callConsent: false,
   606→        ...subject.communicationsData,
   607→      },
   608→      technicalData: subject.technicalData || {},
   609→      metadata: {
   610→        dataCollectedAt: now,
   611→        dataSources: [],
   612→        retentionCategory: 'medium',
   613→        legalBasis: '',
   614→        consentRecords: [],
   615→        dataProcessingActivities: [],
   616→        thirdPartySharing: [],
   617→      },
   618→      status: 'active',
   619→      ...subject,
   620→    };
   621→
   622→    this.dataSubjects.set(id, dataSubject);
   623→    return dataSubject;
   624→  }
   625→
   626→  /**
   627→   * Get data subject by ID
   628→   */
   629→  getDataSubject(id: string): DataSubject | undefined {
   630→    return this.dataSubjects.get(id);
   631→  }
   632→
   633→  /**
   634→   * Find data subject by various identifiers
   635→   */
   636→  findDataSubject(identifiers: { userId?: string; email?: string; phone?: string }): DataSubject | undefined {
   637→    for (const subject of this.dataSubjects.values()) {
   638→      if (identifiers.userId && subject.userId === identifiers.userId) return subject;
   639→      if (identifiers.email && subject.email === identifiers.email) return subject;
   640→      if (identifiers.phone && (subject as any).phone === identifiers.phone) return subject;
   641→    }
   642→    return undefined;
   643→  }
   644→
   645→  /**
   646→   * GDPR Article 15: Right of Access
   647→   * Provide copy of all personal data held
   648→   */
   649→  async fulfillAccessRequest(
   650→    subjectId: string,
   651→    requestFormat: 'json' | 'csv' = 'json',
   652→    identityVerified: boolean = false
   653→  ): Promise<{ success: boolean; data?: any; format: string; expiresAt: string }> {
   654→    const subject = this.dataSubjects.get(subjectId);
   655→    if (!subject) {
   656→      return { success: false, format: requestFormat, expiresAt: '' };
   657→    }
   658→
   659→    if (!identityVerified && this.config.rights.access.identityVerification !== 'none') {
   660→      return { success: false, error: 'Identity verification required', format: requestFormat, expiresAt: '' };
   661→    }
   662→
   663→    // Prepare data export (excluding sensitive operational data)
   664→    const accessibleData = {

   725→  /**
   726→   * GDPR Article 16: Right to Rectification
   727→   * Correct inaccurate personal data
   728→   */
   729→  async fulfillRectificationRequest(
   730→    subjectId: string,
   731→    corrections: Record<string, any>,
   732→    identityVerified: boolean = false
   733→  ): Promise<{ success: boolean; correctedFields: string[]; rejectionReason?: string }> {
   734→    const subject = this.dataSubjects.get(subjectId);
   735→    if (!subject) {
   736→      return { success: false, correctedFields: [], rejectionReason: 'Subject not found' };
   737→    }
   738→
   739→    if (!identityVerified && this.config.rights.rectification.requiresIdentityVerification) {
   740→      return { success: false, correctedFields: [], rejectionReason: 'Identity verification required' };
   741→    }
   742→
   743→    const correctedFields: string[] = [];
   744→    const allowedFields = this.config.rights.rectification.fieldsThatCanBeRectified;
   745→
   746→    for (const [field, value] of Object.entries(corrections)) {
   747→      if (!allowedFields.includes(field)) {
   748→        continue; // Skip fields that cannot be rectified
   749→      }
   750→
   751→      // Apply correction
   752→      if (field in subject.profileData) {
   753→        subject.profileData[field] = value;
   754→        correctedFields.push(`profileData.${field}`);
   755→      } else if (field in subject.accountData) {
   756→        subject.accountData[field] = value;
   757→        correctedFields.push(`accountData.${field}`);
   758→      }
   759→    }
   760→
   761→    // Log the rectification
   762→    this.addProcessingActivity(subjectId, {
   763→      purpose: 'Data rectification per GDPR Art. 16',
   764→      lawfulBasis: 'contract',
   765→      dataCategories: Object.keys(corrections),
   766→      processor: 'privacy-system',
   767→      retentionPeriod: 'per account lifetime',
   768→    });
   769→
   770→    return { success: true, correctedFields };
   771→  }

   777→  async initiateErasureRequest(
   778→    subjectId: string,
   779→    scope?: Partial<DataErasureRequest['scope']>,
   780→    reason?: string
   781→  ): Promise<DataErasureRequest> {
   782→    const subject = this.dataSubjects.get(subjectId);
   783→    if (!subject) {
   784→      throw new Error('Data subject not found');
   785→    }
   786→
   787→    const requestId = `erasure_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`;
   788→    const now = new Date().toISOString();
   789→
   790→    const request: DataErasureRequest = {
   791→      id: requestId,
   792→      dataSubjectId: subjectId,
   793→      requestedAt: now,
   794→      requestedVia: 'portal',
   795→      identityVerified: false,
   796→      verificationMethod: 'pending',
   797→      status: 'pending',
   798→      scope: {
   799→        profileData: true,
   800→        accountData: true,
   801→        transactionData: false, // Keep for legal requirements
   802→        communicationsData: true,
   803→        technicalData: true,
   804→        ...scope,
   805→      },
   806→      processingLog: [{
   807→        step: 'Request received',
   808→        timestamp: now,
   809→        performedBy: 'system',
   810→        result: reason || 'Standard erasure request',
   811→      }],
   812→      systemsProcessed: [],
   813→      dataDeleted: {},
   814→      dataRetained: {},
   815→      confirmationSent: false,
   816→    };
   817→
   818→    // Check for legal holds
   819→    const legalHolds = this.checkLegalHolds(subjectId);
   820→    if (legalHolds.hasHold) {
   821→      request.scope.customExclusions = legalHolds.reasons;
   822→      request.processingLog.push({
   823→        step: 'Legal hold check',
   824→        timestamp: now,
   825→        performedBy: 'system',
   826→        result: `Legal hold in effect: ${legalHolds.description}`,
   827→        details: JSON.stringify(legalHolds),
   828→      });
   829→    }
   830→
   831→    this.erasureRequests.set(requestId, request);
   832→
   833→    // Update subject status
   834→    subject.status = 'erasure_requested';
   835→    subject.erasureRequestDate = now;
   836→
   837→    return request;
   838→  }

   840→  /**
   841→   * Process an erasure request (actually delete data)
   842→   */
   843→  async processErasureRequest(requestId: string): Promise<void> {
   844→    const request = this.erasureRequests.get(requestId);
   845→    if (!request || request.status !== 'pending') {
   846→      throw new Error('Invalid or already processed request');
   847→    }
   848→
   849→    request.status = 'processing';
   850→    request.processingLog.push({
   851→      step: 'Processing started',
   852→      timestamp: new Date().toISOString(),
   853→      performedBy: 'system',
   854→      result: 'Beginning data deletion process',
   855→    });
   856→
   857→    const subject = this.dataSubjects.get(request.dataSubjectId);
   858→    if (!subject) return;
   859→
   860→    const systemsToDeleteFrom: string[] = [];
   861→    const deletedData: Record<string, { records: number; size: string }> = {};
   862→    const retainedData: Record<string, { reason: string; records: number }> = {};
   863→
   864→    // Delete profile data
   865→    if (request.scope.profileData) {
   866→      const keysBefore = Object.keys(subject.profileData).length;
   867→      subject.profileData = {};
   868→      deletedData['profileData'] = { records: keysBefore, size: '~small' };
   869→      systemsToDeleteFrom.push('profile_database');
   870→    }
   871→
   872→    // Delete account data (but keep ID for reference)
   873→    if (request.scope.accountData) {
   874→      const sensitiveFields = ['username', 'passwordHash', 'sessionTokens'];
   875→      const deletedFields: string[] = [];
   876→      
   877→      for (const field of Object.keys(subject.accountData)) {
   878→        if (!sensitiveFields.includes(field)) {
   879→          delete subject.accountData[field];
   880→          deletedFields.push(field);
   881→        }
   882→      }
   883→      
   884→      deletedData['accountData'] = { records: deletedFields.length, size: '~medium' };
   885→      systemsToDeleteFrom.push('auth_system');
   886→    }
   887→
   888→    // Delete communications preferences
   889→    if (request.scope.communicationsData) {
   890→      subject.communicationsData = {
   891→        marketingConsent: false,
   892→        newsletterSubscribed: false,
   893→        smsConsent: false,
   894→        callConsent: false,
   895→      };
   896→      deletedData['communicationsData'] = { records: 4, size: '~tiny' };
   897→      systemsToDeleteFrom.push('crm_system');
   898→    }
   899→
   900→    // Delete/anonymize technical data
   901→    if (request.scope.technicalData) {
   902→      if (subject.technicalData.ipAddress) {
   903→        subject.technicalData.ipAddress = this.anonymizeIP(subject.technicalData.ipAddress);
   904→      }
   905→      if (subject.technicalData.deviceId) {
   906→        subject.technicalData.deviceId = this.anonymizeValue(subject.technicalData.deviceId);
   907→      }
   908→      deletedData['technicalData'] = { records: 2, size: '~tiny' };
   909→      systemsToDeleteFrom.push('analytics_platform');
   910→    }
   911→
   912→    // Handle transaction data (usually retained for legal reasons)
   913→    if (request.scope.transactionData) {
   914→      retainedData['transactionData'] = {
   915→        reason: 'Legal requirement to retain financial records',
   916→        records: subject.transactionData.orderCount || 0,
   917→      };
   918→      request.processingLog.push({
   919→        step: 'Transaction data retention',
   920→        timestamp: new Date().toISOString(),
   921→        performedBy: 'system',
   922→        result: 'Retained due to legal requirements',
   923→      });
   924→    }
   925→
   926→    // Handle custom exclusions
   927→    if (request.scope.customExclusions?.length) {
   928→      for (const exclusion of request.scope.customExclusions) {
   929→        retainedData[exclusion] = {
   930→          reason: 'Custom exclusion during erasure',
   931→          records: 1,
   932→        };
   933→      }
   934→    }
   935→
   936→    // Update request
   937→    request.systemsProcessed = systemsToDeleteFrom;
   938→    request.dataDeleted = deletedData;
   939→    request.dataRetained = retainedData;
   940→    request.status = 'completed';
   941→    request.completedAt = new Date().toISOString();
   942→
   943→    // Update subject
   944→    subject.status = 'erased';
   945→    subject.erasureCompletionDate = request.completedAt;
   946→
   947→    // Schedule final deletion after backup period
   948→    setTimeout(() => {
   949→      this.permanentlyDeleteSubject(request.dataSubjectId);
   950→    }, this.config.rights.erasure.backupRetentionDays * 24 * 60 * 60 * 1000);
   951→  }

  1060→          break;

  1062→          subject.communicationsData.smsConsent = consent.given;
  1063→          break;
  1064→      }
  1065→

  1072→  /**
  1073→   * Withdraw consent
  1074→   */
  1075→  withdrawConsent(subjectId: string, consentType: string): void {
  1076→    const subjectConsents = this.consentRecords.get(subjectId);
  1077→    if (!subjectConsents) return;
  1078→
  1079→    const activeConsent = subjectConsents
  1080→      .filter(c => c.type === consentType && c.given && !c.withdrawnAt)
  1081→      .sort((a, b) => new Date(b.givenAt).getTime() - new Date(a.givenAt).getTime())[0];
  1082→
  1083→    if (activeConsent) {
  1084→      activeConsent.withdrawnAt = new Date().toISOString();
  1085→      activeConsent.given = false;
  1086→
  1087→      // Update subject preferences
  1088→      const subject = this.dataSubjects.get(subjectId);
  1089→      if (subject) {
  1090→        switch (consentType) {
  1091→          case 'marketing':
  1092→          case 'email':
  1093→            subject.communicationsData.marketingConsent = false;
  1094→            subject.communicationsData.lastMarketingOptOut = new Date().toISOString();
  1095→            break;
  1096→          case 'newsletter':
  1097→            subject.communicationsData.newsletterSubscribed = false;
  1098→            break;
  1099→          case 'sms':
  1100→            subject.communicationsData.smsConsent = false;
  1101→            break;
  1102→        }
  1103→      }
  1104→    }
  1105→  }

  1110→  getConsentStatus(subjectId: string): Record<string, boolean> {

  1123→      if (!(type in status)) {
  1124→        status[type] = false;
  1125→      }
  1126→    }
  1127→
  1128→    return status;
  1129→  }

  1430→  private anonymizeIP(ip: string): string {
  1431→    if (!ip || ip === 'unknown') return ip;
  1432→    // Keep first two octets, mask rest
  1433→    const parts = ip.split('.');
  1434→    if (parts.length === 4) {
  1435→      return `${parts[0]}.${parts[1]}.*.*`;
  1436→    }
  1437→    return '***.***.***.***';
  1438→  }
  1439→
  1440→  private anonymizeValue(value: string): string {
  1441→    if (!value) return value;
  1442→    return `${value.substring(0, 2)}${'*'.repeat(Math.min(value.length - 2, 8))}`;
  1443→  }

  1445→  private convertToCSV(data: any): string {
  1446→    // Simplified CSV conversion
  1447→    const flattenObject = (obj: any, prefix = ''): any[] => {
  1448→      const rows: any[] = [];
  1449→      
  1450→      for (const [key, value] of Object.entries(obj)) {
  1451→        const newKey = prefix ? `${prefix}.${key}` : key;
  1452→        
  1453→        if (value && typeof value === 'object' && !Array.isArray(value)) {
  1454→          rows.push(...flattenObject(value, newKey));
  1455→        } else {
  1456→          const stringValue = typeof value === 'object' ? JSON.stringify(value) : String(value ?? '');
  1457→          rows.push({ key: newKey, value: stringValue.replace(/"/g, '""').replace(/\n/g, ' ') });
  1458→        }
  1459→      }
  1460→      
  1461→      return rows;
  1462→    };
  1463→
  1464→    const flatData = flattenObject(data);
  1465→    if (flatData.length === 0) return '';
  1466→
  1467→    const headers = Object.keys(flatData[0]);
  1468→    const values = flatData.map(row => headers.map(h => row[h] || ''));
  1469→
  1470→    return [headers.join(','), ...values.map(v => v.join(','))].join('\n');
  1471→  }
