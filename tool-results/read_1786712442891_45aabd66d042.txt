   106→type HealthCheckFn = () => Promise<HealthCheckResult>;
   107→
   108→class HealthCheckRegistry {
   109→  private checks: Map<string, { fn: HealthCheckFn; config: HealthCheckConfig }> = new Map();
   110→  private results: Map<string, HealthCheckResult[]> = new Map();
   111→
   112→  register(name: string, fn: HealthCheckFn, config?: Partial<HealthCheckConfig>): void {
   113→    this.checks.set(name, {
   114→      fn,
   115→      config: { ...DEFAULT_CHECK_CONFIG, ...config, name },
   116→    });
   117→    
   118→    if (!this.results.has(name)) {
   119→      this.results.set(name, []);
   120→    }
   121→  }
   122→
   123→  unregister(name: string): void {
   124→    this.checks.delete(name);
   125→    this.results.delete(name);
   126→  }
   127→
   128→  getChecks(): Map<string, { fn: HealthCheckFn; config: HealthCheckConfig }> {
   129→    return new Map(this.checks);
   130→  }
   131→
   132→  storeResult(name: string, result: HealthCheckResult): void {
   133→    const results = this.results.get(name) || [];
   134→    results.push(result);
   135→    
   136→    // Keep only last 100 results per check
   137→    if (results.length > 100) {
   138→      results.shift();
   139→    }
   140→    
   141→    this.results.set(name, results);
   142→  }
   143→
   144→  getRecentResults(name: string, count: number = 10): HealthCheckResult[] {
   145→    const results = this.results.get(name) || [];
   146→    return results.slice(-count);
   147→  }
   148→
   149→  getAllRecentResults(count: number = 10): Map<string, HealthCheckResult[]> {
   150→    const allResults = new Map<string, HealthCheckResult[]>();
   151→    
   152→    for (const [name] of this.checks) {
   153→      allResults.set(name, this.getRecentResults(name, count));
   154→    }
   155→    
   156→    return allResults;
   157→  }
   158→}

   166→class DependencyTracker {
   167→  private dependencies: Map<string, DependencyHealth> = new Map();
   168→
   169→  register(name: string, type: DependencyHealth['type']): void {
   170→    this.dependencies.set(name, {
   171→      name,
   172→      type,
   173→      status: 'unknown',
   174→      lastChecked: new Date(0).toISOString(),
   175→      errorCount: 0,
   176→      consecutiveErrors: 0,
   177→    });
   178→  }
   179→
   180→  update(name: string, update: Partial<DependencyHealth>): void {
   181→    const current = this.dependencies.get(name);
   182→    if (!current) return;
   183→
   184→    const updated: DependencyHealth = {
   185→      ...current,
   186→      ...update,
   187→      lastChecked: new Date().toISOString(),
   188→    };
   189→
   190→    // Track error counts
   191→    if (updated.status === 'unhealthy') {
   192→      updated.errorCount++;
   193→      updated.consecutiveErrors++;
   194→    } else if (updated.status === 'healthy') {
   195→      updated.consecutiveErrors = Math.max(0, updated.consecutiveErrors - 1);
   196→    }
   197→
   198→    this.dependencies.set(name, updated);
   199→  }
   200→
   201→  getAll(): DependencyHealth[] {
   202→    return Array.from(this.dependencies.values());
   203→  }
   204→
   205→  getByName(name: string): DependencyHealth | undefined {
   206→    return this.dependencies.get(name);
   207→  }
   208→
   209→  getByType(type: DependencyHealth['type']): DependencyHealth[] {
   210→    return this.getAll().filter(dep => dep.type === type);
   211→  }
   212→}

   220→class IncidentTracker {
   221→  private activeIncidents: IncidentSummary[] = [];
   222→  private resolvedIncidents: IncidentSummary[] = [];
   223→
   224→  create(incident: Omit<IncidentSummary, 'id' | 'startedAt'>): IncidentSummary {
   225→    const newIncident: IncidentSummary = {
   226→      ...incident,
   227→      id: `inc_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
   228→      startedAt: new Date().toISOString(),
   229→    };
   230→
   231→    this.activeIncidents.push(newIncident);
   232→    return newIncident;
   233→  }
   234→
   235→  resolve(id: string, resolution?: Partial<IncidentSummary>): boolean {
   236→    const index = this.activeIncidents.findIndex(inc => inc.id === id);
   237→    if (index === -1) return false;
   238→
   239→    const incident = this.activeIncidents.splice(index, 1)[0];
   240→    incident.status = 'resolved';
   241→    incident.resolvedAt = new Date().toISOString();
   242→    
   243→    Object.assign(incident, resolution || {});
   244→    this.resolvedIncidents.push(incident);
   245→
   246→    return true;
   247→  }
   248→
   249→  getActive(): IncidentSummary[] {
   250→    return [...this.activeIncidents];
   251→  }
   252→
   253→  getResolved(limit: number = 20): IncidentSummary[] {
   254→    return this.resolvedIncidents.slice(-limit).reverse();
   255→  }
   256→
   257→  getAll(): IncidentSummary[] {
   258→    return [...this.activeIncidents, ...this.resolvedIncidents];
   259→  }
   260→}

   461→/**
   462→ * Disk space check (server-side only)
   463→ */
   464→async function diskSpaceCheck(): Promise<HealthCheckResult> {
   465→  const startTime = Date.now();
   466→  
   467→  if (typeof process === 'undefined') {
   468→    return {
   469→      name: 'disk-space',
   470→      status: 'unknown',
   471→      duration: Date.now() - startTime,
   472→      timestamp: new Date().toISOString(),
   473→      message: 'Cannot check disk space in browser environment',
   474→    };
   475→  }
   476→
   477→  try {
   478→    const fs = await import('fs/promises');
   479→    const path = await import('path');
   480→    
   481→    const stats = await fs.statfs(path.sep);
   482→    const totalSpace = stats.blocks * stats.bsize;
   483→    const freeSpace = stats.bfree * stats.bsize;
   484→    const usedPercent = ((totalSpace - freeSpace) / totalSpace) * 100;
   485→
   486→    let status: HealthStatus = 'healthy';
   487→    if (usedPercent > 95) status = 'unhealthy';
   488→    else if (usedPercent > 85) status = 'degraded';
   489→
   490→    return {
   491→      name: 'disk-space',
   492→      status,
   493→      duration: Date.now() - startTime,
   494→      timestamp: new Date().toISOString(),
   495→      message: `Disk usage: ${usedPercent.toFixed(1)}%`,
   496→      details: {
   497→        totalGB: (totalSpace / (1024 ** 3)).toFixed(1),
   498→        freeGB: (freeSpace / (1024 ** 3)).toFixed(1),
   499→        usedPercent: usedPercent.toFixed(1),
   500→      },
   501→    };
   502→  } catch (error) {
   503→    return {
   504→      name: 'disk-space',
   505→      status: 'unknown',
   506→      duration: Date.now() - startTime,
   507→      timestamp: new Date().toISOString(),
   508→      error: (error as Error).message,
   509→    };
   510→  }
   511→}

   513→/**
   514→ * Memory usage check
   515→ */
   516→async function memoryCheck(): Promise<HealthCheckResult> {
   517→  const startTime = Date.now();
   518→  
   519→  let memUsage: NodeJS.MemoryUsage | null = null;
   520→  
   521→  if (typeof process !== 'undefined') {
   522→    memUsage = process.memoryUsage();
   523→  } else if (typeof performance !== 'undefined') {
   524→    // Browser fallback
   525→    const memory = (performance as any).memory;
   526→    if (memory) {
   527→      memUsage = {
   528→        rss: memory.jsHeapSizeLimit || 0,
   529→        heapUsed: memory.usedJSHeapSize || 0,
   530→        heapTotal: memory.totalJSHeapSize || 0,
   531→        external: 0,
   532→        arrayBuffers: 0,
   533→      };
   534→    }
   535→  }
   536→
   537→  if (!memUsage) {
   538→    return {
   539→      name: 'memory',
   540→      status: 'unknown',
   541→      duration: Date.now() - startTime,
   542→      timestamp: new Date().toISOString(),
   543→      message: 'Memory info not available',
   544→    };
   545→  }
   546→
   547→  const heapUsagePercent = (memUsage.heapUsed / memUsage.heapTotal) * 100;
   548→  let status: HealthStatus = 'healthy';
   549→  
   550→  if (heapUsagePercent > 90) status = 'unhealthy';
   551→  else if (heapUsagePercent > 75) status = 'degraded';
   552→
   553→  return {
   554→    name: 'memory',
   555→    status,
   556→    duration: Date.now() - startTime,
   557→    timestamp: new Date().toISOString(),
   558→    message: `Heap usage: ${heapUsagePercent.toFixed(1)}%`,
   559→    details: {
   560→      heapUsedMB: (memUsage.heapUsed / (1024 ** 2)).toFixed(1),
   561→      heapTotalMB: (memUsage.heapTotal / (1024 ** 2)).toFixed(1),
   562→      rssMB: (memUsage.rss / (1024 ** 2)).toFixed(1),
   563→      heapUsagePercent: heapUsagePercent.toFixed(1),
   564→    },
   565→  };
   566→}

   589→// ===========================================
   590→// Main Health Checker Class
   591→// ===========================================
   592→
   593→export class HealthMonitor {
   594→  private isRunning = false;
   595→  private checkInterval: NodeJS.Timeout | null = null;
   596→  private listeners: Array<(health: SystemHealth) => void> = [];
   597→
   598→  /**
   599→   * Start continuous health monitoring
   600→   */
   601→  start(intervalMs: number = 60000): void {
   602→    if (this.isRunning) return;
   603→    
   604→    this.isRunning = true;
   605→    
   606→    // Run initial check
   607→    this.runAllChecks();
   608→    
   609→    // Set up recurring checks
   610→    this.checkInterval = setInterval(() => {
   611→      this.runAllChecks();
   612→    }, intervalMs);
   613→
   614→    console.log(`🏥 Health monitor started (interval: ${intervalMs}ms)`);
   615→  }
   616→
   617→  /**
   618→   * Stop health monitoring
   619→   */
   620→  stop(): void {
   621→    if (this.checkInterval) {
   622→      clearInterval(this.checkInterval);
   623→      this.checkInterval = null;
   624→    }
   625→    this.isRunning = false;
   626→    console.log('🏥 Health monitor stopped');
   627→  }
   628→
   629→  /**
   630→   * Run all registered health checks
   631→   */
   632→  async runAllChecks(): Promise<SystemHealth> {
   633→    const checks = registry.getChecks();
   634→    const results: HealthCheckResult[] = [];
   635→
   636→    // Run all checks in parallel
   637→    const checkPromises = Array.from(checks.entries()).map(async ([name, { fn }]) => {
   638→      try {
   639→        const result = await fn();
   640→        registry.storeResult(name, result);
   641→        results.push(result);
   642→        return result;
   643→      } catch (error) {
   644→        const errorResult: HealthCheckResult = {
   645→          name,
   646→          status: 'unhealthy',
   647→          duration: 0,
   648→          timestamp: new Date().toISOString(),
   649→          error: (error as Error).message,
   650→        };
   651→        registry.storeResult(name, errorResult);
   652→        results.push(errorResult);
   653→        return errorResult;
   654→      }
   655→    });
   656→
   657→    await Promise.all(checkPromises);
   658→
   659→    // Build system health object
   660→    const systemHealth = this.buildSystemHealth(results);
   661→
   662→    // Notify listeners
   663→    this.listeners.forEach(listener => {
   664→      try {
   665→        listener(systemHealth);
   666→      } catch (e) {
   667→        console.error('Health listener error:', e);
   668→      }
   669→    });
   670→
   671→    // Auto-detect and create incidents
   672→    this.detectIncidents(systemHealth);
   673→
   674→    return systemHealth;
   675→  }
   676→
   677→  /**
   678→   * Run a specific health check by name
   679→   */
   680→  async runCheck(name: string): Promise<HealthCheckResult | null> {
   681→    const check = registry.getChecks().get(name);
   682→    if (!check) return null;
   683→
   684→    try {
   685→      const result = await check.fn();
   686→      registry.storeName(result);
   687→      return result;
   688→    } catch (error) {
   689→      const errorResult: HealthCheckResult = {
   690→        name,
   691→        status: 'unhealthy',
   692→        duration: 0,
   693→        timestamp: new Date().toISOString(),
   694→        error: (error as Error).message,
   695→      };
   696→      registry.storeResult(name, errorResult);
   697→      return errorResult;
   698→    }
   699→  }

   719→   * Get current system health (without running checks)
   720→   */
   721→  async getCurrentHealth(): Promise<SystemHealth> {
   722→    const recentResults = registry.getAllRecentResults(1);
   723→    const results: HealthCheckResult[] = [];
   724→    
   725→    recentResults.forEach((checkResults) => {
   726→      if (checkResults.length > 0) {
   727→        results.push(checkResults[0]);
   728→      }
   729→    });
   730→
   731→    return this.buildSystemHealth(results);
   732→  }

   734→  /**
   735→   * Get uptime statistics
   736→   */
   737→  getUptimeStats(period: UptimeStats['period'] = '24h'): UptimeStats {
   738→    const now = Date.now();
   739→    const periodMs = this.getPeriodMs(period);
   740→    const allResults = registry.getAllRecentResults(1000); // Get lots of data
   741→
   742→    let totalChecks = 0;
   743→    let successfulChecks = 0;
   744→    let failedChecks = 0;
   745→    let totalTime = 0;
   746→    let countForAverage = 0;
   747→
   748→    allResults.forEach((results) => {
   749→      results.forEach((result) => {
   750→        const resultTime = new Date(result.timestamp).getTime();
   751→        
   752→        // Only count results within the period
   753→        if (now - resultTime <= periodMs) {
   754→          totalChecks++;
   755→          totalTime += result.duration;
   756→          
   757→          if (result.status === 'healthy') {
   758→            successfulChecks++;
   759→          } else {
   760→            failedChecks++;
   761→          }
   762→          
   763→          if (result.status !== 'unknown') {
   764→            countForAverage++;
   765→          }
   766→        }
   767→      });
   768→    });
   769→
   770→    const availability = totalChecks > 0 ? (successfulChecks / totalChecks) * 100 : 100;
   771→    const avgResponseTime = countForAverage > 0 ? totalTime / countForAverage : 0;
   772→    const downtimeMinutes = (totalChecks - successfulChecks) * (periodMs / totalChecks) / 60000;
   773→
   774→    // Get incident count for period
   775→    const resolvedInPeriod = incidentTracker.getResolved().filter(
   776→      inc => now - new Date(inc.startedAt!).getTime() <= periodMs
   777→    );
   778→
   779→    return {
   780→      period,
   781→      availability: Math.round(availability * 100) / 100,
   782→      totalChecks,
   783→      successfulChecks,
   784→      failedChecks,
   785→      averageResponseTime: Math.round(avgResponseTime),
   786→      incidents: resolvedInPeriod.length + incidentTracker.getActive().length,
   787→      downtimeMinutes: Math.round(downtimeMinutes * 100) / 100,
   788→    };
   789→  }

   843→  private buildSystemHealth(checks: HealthCheckResult[]): SystemHealth {
   844→    // Determine overall status
   845→    let overallStatus: HealthStatus = 'healthy';
   846→    
   847→    for (const check of checks) {
   848→      if (check.status === 'unhealthy') {
   849→        overallStatus = 'unhealthy';
   850→        break;
   851→      } else if (check.status === 'degraded' && overallStatus !== 'unhealthy') {
   852→        overallStatus = 'degraded';
   853→      }
   854→    }
   855→
   856→    // Calculate metrics (simplified)
   857→    const metrics = this.calculateMetrics();
   858→
   859→    return {
   860→      overallStatus,
   861→      uptime: this.getUptimeStats('24h').availability,
   862→      version: process.env.npm_package_version || '1.0.0',
   863→      environment: process.env.NODE_ENV || 'development',
   864→      timestamp: new Date().toISOString(),
   865→      checks,
   866→      dependencies: dependencyTracker.getAll(),
   867→      metrics,
   868→      incidents: incidentTracker.getActive(),
   869→    };
   870→  }

   875→      return {

   877→        cpuUsage: 0, // Would need additional library
   878→        activeConnections: 0, // Would track from server
   879→        requestsPerSecond: 0, // Would track from middleware
   880→        averageResponseTime: 0, // Would calculate from recent requests
   881→        errorRate: 0, // Would calculate from recent requests

   888→      activeConnections: 0,
   889→      requestsPerSecond: 0,
   890→      averageResponseTime: 0,
   891→      errorRate: 0,

   893→  }

   895→  private detectIncidents(health: SystemHealth): void {
   896→    const activeIncidents = incidentTracker.getActive();
   897→
   898→    // Check for unhealthy components that don't have an active incident
   899→    for (const check of health.checks) {
   900→      if (check.status === 'unhealthy') {
   901→        const hasIncident = activeIncidents.some(
   902→          inc => inc.affectedComponents.includes(check.name)
   903→        );
   904→
   905→        if (!hasIncident) {
   906→          // Create new incident
   907→          incidentTracker.create({
   908→            severity: this.getSeverity(check.name),
   909→            title: `${check.name} is unhealthy`,
   910→            status: 'investigating',
   911→            affectedComponents: [check.name],
   912→          });
   913→        }
   914→      }
   915→    }
   916→
   917→    // Resolve incidents where affected components are healthy
   918→    for (const incident of activeIncidents) {
   919→      const allHealthy = incident.affectedComponents.every(compName =>
   920→        health.checks.find(c => c.name === compName)?.status === 'healthy'
   921→      );
   922→
   923→      if (allHealthy) {
   924→        incidentTracker.resolve(incident.id);
   925→      }
   926→    }
   927→  }
   928→
   929→  private getSeverity(componentName: string): 'critical' | 'major' | 'minor' {
   930→    const criticalComponents = ['database', 'redis', 'disk-space'];
   931→    const majorComponents = ['memory', 'stripe-api'];
   932→
   933→    if (criticalComponents.includes(componentName)) return 'critical';
   934→    if (majorComponents.includes(componentName)) return 'major';
   935→    return 'minor';
   936→  }

   938→  private getPeriodMs(period: UptimeStats['period']): number {
   939→    switch (period) {
   940→      case '24h': return 24 * 60 * 60 * 1000;
   941→      case '7d': return 7 * 24 * 60 * 60 * 1000;
   942→      case '30d': return 30 * 24 * 60 * 60 * 1000;
   943→      case '90d': return 90 * 24 * 60 * 60 * 1000;
   944→      default: return 24 * 60 * 60 * 1000;
   945→    }
   946→  }
