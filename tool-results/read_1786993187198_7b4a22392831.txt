   231→export default function ExhibitionsManagementPage() {
   232→  const [exhibitions, setExhibitions] = useState<Exhibition[]>(sampleExhibitions);
   233→  const [activeTab, setActiveTab] = useState('all');
   234→  const [searchQuery, setSearchQuery] = useState('');
   235→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   236→  
   237→  // Dialog states
   238→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   239→  const [selectedExhibition, setSelectedExhibition] = useState<Exhibition | null>(null);
   240→  
   241→  // Booth management dialog
   242→  const [boothDialogOpen, setBoothDialogOpen] = useState(false);
   243→
   244→  // Filter exhibitions
   245→  const filteredExhibitions = useMemo(() => {
   246→    let result = [...exhibitions];
   247→    
   248→    if (activeTab !== 'all') {
   249→      if (activeTab === 'featured') {
   250→        result = result.filter(e => e.isFeatured);
   251→      } else if (activeTab === 'upcoming') {
   252→        result = result.filter(e => 
   253→          ['DRAFT', 'PUBLISHED', 'REGISTRATION_OPEN', 'REGISTRATION_CLOSED'].includes(e.status)
   254→        );
   255→      } else {
   256→        result = result.filter(e => e.status === activeTab.toUpperCase());
   257→      }
   258→    }
   259→    
   260→    if (statusFilter !== 'ALL') {
   261→      result = result.filter(e => e.status === statusFilter);
   262→    }
   263→    
   264→    if (searchQuery.trim()) {
   265→      const query = searchQuery.toLowerCase();
   266→      result = result.filter(e => 
   267→        e.title.toLowerCase().includes(query) ||
   268→        e.venue.toLowerCase().includes(query) ||
   269→        e.location.toLowerCase().includes(query)
   270→      );
   271→    }
   272→    
   273→    result.sort((a, b) => {
   274→      // Featured first
   275→      if (a.isFeatured && !b.isFeatured) return -1;
   276→      if (!a.isFeatured && b.isFeatured) return 1;
   277→      // Then by date
   278→      return new Date(b.startDate).getTime() - new Date(a.startDate).getTime();
   279→    });
   280→    
   281→    return result;
   282→  }, [exhibitions, activeTab, searchQuery, statusFilter]);
   283→
   284→  // Stats
   285→  const totalExhibitions = exhibitions.length;
   286→  const activeExhibitions = exhibitions.filter(e => 
   287→    ['PUBLISHED', 'REGISTRATION_OPEN', 'REGISTRATION_CLOSED', 'ONGOING'].includes(e.status)
   288→  ).length;
   289→  const totalRegistrations = exhibitions.reduce((sum, e) => sum + e.currentExhibitors, 0);
   290→  const featuredCount = exhibitions.filter(e => e.isFeatured).length;
   291→
   292→  // Handlers
   293→  const formatDate = (dateString: string) => {
   294→    return new Date(dateString).toLocaleDateString('fr-FR', {
   295→      day: '2-digit',
   296→      month: '2-digit',
   297→      year: 'numeric'
   298→    });
   299→  };
   300→
   301→  const openDetailDialog = (exhibition: Exhibition) => {
   302→    setSelectedExhibition(exhibition);
   303→    setDetailDialogOpen(true);
   304→  };
   305→
   306→  const toggleFeatured = async (exhibitionId: string) => {
   307→    setExhibitions(prev => prev.map(e => 
   308→      e.id === exhibitionId ? { ...e, isFeatured: !e.isFeatured } : e
   309→    ));
   310→  };
   311→
   312→  const updateStatus = async (exhibitionId: string, newStatus: ExhibitionStatus) => {
   313→    setExhibitions(prev => prev.map(e => 
   314→      e.id === exhibitionId ? { ...e, status: newStatus } : e
   315→    ));
   316→  };
   317→
   318→  const openBoothDialog = (exhibition: Exhibition) => {
   319→    setSelectedExhibition(exhibition);
   320→    setBoothDialogOpen(true);
   321→  };
   322→
   323→  const exportToCSV = () => {
   324→    const headers = ['Titre', 'Statut', 'Lieu', 'Wilaya', 'Date début', 'Date fin', 'Exposants actuels', 'Max exposants', 'À la une'];
   325→    const csvRows = filteredExhibitions.map(e => [
   326→      e.title,
   327→      statusConfig[e.status].label,
   328→      e.venue,
   329→      e.wilaya,
   330→      formatDate(e.startDate),
   331→      formatDate(e.endDate),
   332→      e.currentExhibitors.toString(),
   333→      e.maxExhibitors.toString(),
   334→      e.isFeatured ? 'Oui' : 'Non'
   335→    ]);
   336→    
   337→    const csvContent = [headers, ...csvRows]
   338→      .map(row => row.map(cell => `"${cell}"`).join(','))
   339→      .join('\n');
   340→    
   341→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   342→    const link = document.createElement('a');
   343→    link.href = URL.createObjectURL(blob);
   344→    link.download = `expositions_${new Date().toISOString().split('T')[0]}.csv`;
   345→    link.click();
   346→  };
   347→
   348→  return (
   349→    <div className="space-y-6">
   350→      {/* Header */}
   351→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   352→        <div>
   353→          <h1 className="text-2xl font-bold text-gray-900">Gestion des Expositions</h1>
   354→          <p className="text-gray-500 mt-1">Créez et gérez les événements et salons</p>
   355→        </div>
   356→        <div className="flex items-center gap-2">
   357→          <Button>
   358→            <Plus className="mr-2 h-4 w-4" /> Nouvelle exposition
   359→          </Button>
   360→          <Button variant="outline" onClick={exportToCSV}>
   361→            <Download className="mr-2 h-4 w-4" /> Exporter CSV
   362→          </Button>
   363→        </div>
   364→      </div>
   365→
   366→      {/* Stats Cards */}
   367→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   368→        <StatsCard
   369→          title="Total expositions"
   370→          value={totalExhibitions}
   371→          description="Tous statuts confondus"
   372→          icon={Calendar}
   373→          iconClassName="bg-blue-100 text-blue-600"
   374→        />
   375→        <StatsCard
   376→          title="Actives"
   377→          value={activeExhibitions}
   378→          description="En cours ou à venir"
   379→          icon={TrendingUp}
   380→          iconClassName="bg-green-100 text-green-600"
   381→        />
   382→        <StatsCard
   383→          title="Total inscriptions"
   384→          value={totalRegistrations}
   385→          description={`${exhibitions.reduce((sum, e) => sum + e.maxExhibitors, 0)} places disponibles`}
   386→          icon={Users}
   387→          iconClassName="bg-purple-100 text-purple-600"
   388→        />
   389→        <StatsCard
   390→          title="À la une"
   391→          value={featuredCount}
   392→          description="Mises en avant"
   393→          icon={Star}
   394→          iconClassName="bg-yellow-100 text-yellow-600"
   395→        />
   396→      </div>
   397→
   398→      {/* Filters */}
   399→      <Card>
   400→        <CardContent className="pt-6">
   401→          <div className="flex flex-col sm:flex-row gap-4">
   402→            <div className="relative flex-1">
   403→              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-gray-400" />
   404→              <Input
   405→                placeholder="Rechercher par titre ou lieu..."
   406→                value={searchQuery}
   407→                onChange={(e) => setSearchQuery(e.target.value)}
   408→                className="pl-10"
   409→              />
   410→            </div>
   411→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   412→              <SelectTrigger className="w-[180px]">
   413→                <Filter className="mr-2 h-4 w-4" />
   414→                <SelectValue placeholder="Statut" />
   415→              </SelectTrigger>
   416→              <SelectContent>
   417→                <SelectItem value="ALL">Tous les statuts</SelectItem>
   418→                <SelectItem value="DRAFT">Brouillon</SelectItem>
   419→                <SelectItem value="PUBLISHED">Publié</SelectItem>
   420→                <SelectItem value="REGISTRATION_OPEN">Inscriptions ouvertes</SelectItem>
   421→                <SelectItem value="ONGOING">En cours</SelectItem>

   425→          </div>
   426→        </CardContent>
   427→      </Card>
   428→
   429→      {/* Tabs and List */}
   430→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   431→        <TabsList>
   432→          <TabsTrigger value="all">Toutes ({exhibitions.length})</TabsTrigger>
   433→          <TabsTrigger value="featured" className="gap-1">
   434→            <Star className="h-3.5 w-3.5 hidden sm:inline" /> À la une ({featuredCount})
   435→          </TabsTrigger>
   436→          <TabsTrigger value="upcoming">À venir</TabsTrigger>
   437→          <TabsTrigger value="ONGOING" className="gap-1">
   438→            <Clock className="h-3.5 w-3.5 hidden sm:inline" /> En cours

   441→        </TabsList>
   442→
   443→        <TabsContent value={activeTab} className="mt-4">

   445→            {filteredExhibitions.length > 0 ? (
   446→              filteredExhibitions.map((exhibition) => (
   447→                <Card key={exhibition.id} className={`overflow-hidden ${exhibition.isFeatured ? 'ring-2 ring-yellow-300' : ''}`}>
   448→                  {/* Cover Image Placeholder */}
   449→                  <div className="h-40 bg-gradient-to-br from-orange-400 to-red-500 relative">
   450→                    {exhibition.isFeatured && (
   451→                      <Badge className="absolute top-3 left-3 bg-yellow-400 text-yellow-900 hover:bg-yellow-400">

   453→                      </Badge>
   454→                    )}
   455→                    <Badge 
   456→                      variant={statusConfig[exhibition.status].variant}
   457→                      className={`absolute bottom-3 right-3 ${statusConfig[exhibition.status].color}`}
   458→                    >
   459→                      {statusConfig[exhibition.status].label}
   460→                    </Badge>
   461→                  </div>
   462→
   463→                  <CardContent className="pt-4 space-y-3">
   464→                    <div>
   465→                      <h3 className="font-semibold text-lg line-clamp-1">{exhibition.title}</h3>
   466→                      <div className="flex items-center gap-1 text-sm text-gray-600 mt-1">
   467→                        <MapPin className="h-3.5 w-3.5 flex-shrink-0" />
   468→                        <span className="truncate">{exhibition.venue}, {exhibition.wilaya}</span>

   470→                    </div>
   471→
   472→                    <div className="flex items-center gap-4 text-sm">
   473→                      <div className="flex items-center gap-1">
   474→                        <Calendar className="h-4 w-4 text-gray-400" />

   483→                    {/* Progress bar */}
