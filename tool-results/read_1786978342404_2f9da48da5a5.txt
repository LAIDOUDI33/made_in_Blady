   216→   216→  /**
   217→   217→   * Process a security event - main entry point
   218→   218→   */
   219→   219→  async recordEvent(eventData: Omit<SecurityEvent, 'id' | 'timestamp' | 'resolved'>): Promise<SecurityEvent> {
   220→   220→    if (!MONITORING_CONFIG.enabled) {
   221→   221→      return {} as SecurityEvent;
   222→   222→    }
   223→   223→
   224→   224→    await this.initialize();
   225→   225→
   226→   226→    const event: SecurityEvent = {
   227→   227→      ...eventData,
   228→   228→      id: `evt_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
   229→   229→      timestamp: new Date(),
   230→   230→      resolved: false,
   231→   231→    };
   232→   232→
   233→   233→    // Add to buffer for real-time analysis
   234→   234→    this.eventBuffer.push(event);
   235→   235→    if (this.eventBuffer.length > 10000) {
   236→   236→      this.eventBuffer = this.eventBuffer.slice(-5000); // Keep last 5000 in memory
   237→   237→    }
   238→   238→
   239→   239→    // Track by IP and user
   240→   240→    if (event.ipAddress) {
   241→   241→      const ipEvents = this.recentEventsByIP.get(event.ipAddress) || [];
   242→   242→      ipEvents.push(event);
   243→   243→      if (ipEvents.length > 1000) ipEvents.shift(); // Keep last 1000 per IP
   244→   244→      this.recentEventsByIP.set(event.ipAddress, ipEvents);
   245→   245→    }
   246→   246→
   247→   247→    if (event.userId) {
   248→   248→      const userEvents = this.recentEventsByUser.get(event.userId) || [];
   249→   249→      userEvents.push(event);
   250→   250→      if (userEvents.length > 500) userEvents.shift();
   251→   251→      this.recentEventsByUser.set(event.userId, userEvents);
   252→   252→    }
   253→   253→
   254→   254→    // Persist to database asynchronously
   255→   255→    this.persistEvent(event);
   256→   256→
   257→   257→    // Evaluate alert rules
   258→   258→    await this.evaluateRules(event);
   259→   259→
   260→   260→    return event;
   261→   261→  }

   263→   263→  /**
   264→   264→   * Calculate risk score for an IP address (0-100)
   265→   265→   */
   266→   266→  calculateIPRiskScore(ipAddress: string): number {
   267→   267→    const events = this.recentEventsByIP.get(ipAddress) || [];
   268→   268→    if (events.length === 0) return 0;
   269→   269→
   270→   270→    let riskScore = 0;
   271→   271→    const now = Date.now();
   272→   272→    const oneHourAgo = now - MONITORING_CONFIG.windows.mediumTerm;
   273→   273→
   274→   274→    // Filter to recent events
   275→   275→    const recentEvents = events.filter(e => e.timestamp.getTime() > oneHourAgo);
   276→   276→
   277→   277→    // Weight different event types
   278→   278→    for (const event of recentEvents) {
   279→   279→      switch (event.eventType) {
   280→   280→        case 'brute_force_detected':
   281→   281→          riskScore += 25;
   282→   282→          break;
   283→   283→        case 'injection_attempt':
   284→   284→          riskScore += 35;
   285→   285→          break;
   286→   286→        case 'xss_attempt':
   287→   287→          riskScore += 20;
   288→   288→          break;
   289→   289→        case 'auth_failure':
   290→   290→          riskScore += 10;
   291→   291→          break;
   292→   292→        case 'rate_limit_exceeded':
   293→   293→          riskScore += 15;
   294→   294→          break;
   295→   295→        case 'payment_fraud_risk':
   296→   296→          riskScore += 30;
   297→   297→          break;
   298→   298→        case 'account_takeover_risk':
   299→   299→          riskScore += 40;
   300→   300→          break;
   301→   301→        default:
   302→   302→          riskScore += 5;
   303→   303→      }
   304→   304→    }
   305→   305→
   306→   306→    // Severity multiplier
   307→   307→    const highSeverityCount = recentEvents.filter(e => 
   308→   308→      e.severity === 'high' || e.severity === 'critical'
   309→   309→    ).length;
   310→   310→    riskScore *= (1 + highSeverityCount * 0.2);
   311→   311→
   312→   312→    // Frequency bonus (rapid attacks are worse)
   313→   313→    if (recentEvents.length > 20) {
   314→   314→      riskScore *= 1.3;
   315→   315→    }
   316→   316→    if (recentEvents.length > 50) {
   317→   317→      riskScore *= 1.5;
   318→   318→    }
   319→   319→
   320→   320→    return Math.min(100, Math.round(riskScore));
   321→   321→  }
   322→   322→
   323→   323→  /**
   324→   324→   * Calculate risk score for a user account (0-100)
   325→   325→   */
   326→   326→  calculateUserRiskScore(userId: string): number {
   327→   327→    const events = this.recentEventsByUser.get(userId) || [];
   328→   328→    if (events.length === 0) return 0;
   329→   329→
   330→   330→    let riskScore = 0;
   331→   331→    const now = Date.now();
   332→   332→    const twentyFourHoursAgo = now - MONITORING_CONFIG.windows.longTerm;
   333→   333→
   334→   334→    const recentEvents = events.filter(e => e.timestamp.getTime() > twentyFourHoursAgo);
   335→   335→
   336→   336→    for (const event of recentEvents) {
   337→   337→      switch (event.eventType) {
   338→   338→        case 'account_takeover_risk':
   339→   339→          riskScore += 40;
   340→   340→          break;
   341→   341→        case 'permission_denied':
   342→   342→          riskScore += 15;
   343→   343→          break;
   344→   344→        case 'data_access_anomaly':
   345→   345→          riskScore += 25;
   346→   346→          break;
   347→   347→        case 'unusual_geolocation':
   348→   348→          riskScore += 20;
   349→   349→          break;
   350→   350→        default:
   351→   351→          riskScore += 5;
   352→   352→      }
   353→   353→    }
   354→   354→
   355→   355→    // Check for multiple IPs (potential account sharing/takeover)
   356→   356→    const uniqueIPs = new Set(recentEvents.map(e => e.ipAddress).filter(Boolean));
   357→   357→    if (uniqueIPs.size > 5) {
   358→   358→      riskScore += 15;
   359→   359→    }
   360→   360→
   361→   361→    return Math.min(100, Math.round(riskScore));
   362→   362→  }

   364→   364→  /**
   365→   365→   * Get current monitoring statistics
   366→   366→   */
   367→   367→  async getStats(): Promise<MonitoringStats> {
   368→   368→    const oneHourAgo = new Date(Date.now() - MONITORING_CONFIG.windows.mediumTerm);
   369→   369→    
   370→   370→    const [totalEvents, eventsByType, blockedIPsCount, recentCritical] = await Promise.all([
   371→   371→      db.securityEvent.count({
   372→   372→        where: { createdAt: { gte: oneHourAgo } },
   373→   373→      }),
   374→   374→      db.securityEvent.groupBy({
   375→   375→        by: ['eventType'],
   376→   376→        _count: { eventType: true },
   377→   377→        where: { createdAt: { gte: oneHourAgo } },
   378→   378→        orderBy: { _count: { eventType: 'desc' } },
   379→   379→        take: 10,
   380→   380→      }),
   381→   381→      db.blockedIP.count({ where: { expiresAt: { gt: new Date() } } }),
   382→   382→      db.securityEvent.findMany({
   383→   383→        where: {
   384→   384→          createdAt: { gte: oneHourAgo },
   385→   385→          severity: { in: ['critical', 'high'] },
   386→   386→          resolved: false,
   387→   387→        },
   388→   388→        orderBy: { createdAt: 'desc' },
   389→   389→        take: 10,
   390→   390→      }),
   391→   391→    ]);
   392→   392→
   393→   393→    // Calculate overall platform risk score
   394→   394→    const highRiskIPs = Array.from(this.recentEventsByIP.entries())
   395→   395→      .filter(([, events]) => this.calculateIPRiskScoreFromEvents(events) > 70);
   396→   396→    
   397→   397→    const overallRisk = Math.min(100, Math.round(
   398→   398→      (highRiskIPs.length * 10) + 
   399→   399→      (recentCritical.length * 15) +
   400→   400→      (totalEvents > 1000 ? 20 : totalEvents / 50)
   401→   401→    ));
   402→   402→
   403→   403→    // Top attacker IPs
   404→   404→    const topAttackers = Array.from(this.recentEventsByIP.entries())
   405→   405→      .map(([ip, events]) => ({ ip, count: events.length }))
   406→   406→      .sort((a, b) => b.count - a.count)
   407→   407→      .slice(0, 10);
   408→   408→
   409→   409→    return {
   410→   410→      totalEvents,
   411→   411→      eventsByType: Object.fromEntries(eventsByType.map(e => [e.eventType, e._count.eventType])),
   412→   412→      eventsBySeverity: {}, // Would need another query
   413→   413→      activeAlerts: recentCritical.length,
   414→   414→      blockedIPs: blockedIPsCount,
   415→   415→      riskScore: overallRisk,
   416→   416→      topAttackerIPs: topAttackers,
   417→   417→      recentCriticalEvents: recentCritical.map(e => ({
   418→   418→        ...e,
   419→   419→        details: typeof e.details === 'string' ? JSON.parse(e.details) : e.details,
   420→   420→      })),
   421→   421→    };
   422→   422→  }

   501→   501→  /**
   502→   502→   * Evaluate alert rules against an event
   503→   503→   */
   504→   504→  private async evaluateRules(event: SecurityEvent): Promise<void> {
   505→   505→    for (const rule of this.rules) {
   506→   506→      if (!rule.enabled) continue;
   507→   507→      if (rule.eventType !== event.eventType) continue;
   508→   508→
   509→   509→      // Check cooldown
   510→   510→      const cooldownKey = `${rule.id}_${event.ipAddress || event.userId || 'global'}`;
   511→   511→      const lastAlert = this.alertCooldowns.get(cooldownKey);
   512→   512→      if (lastAlert && Date.now() - lastAlert < rule.cooldownMs) continue;
   513→   513→
   514→   514→      // Evaluate conditions
   515→   515→      const conditionsMet = rule.conditions.every(condition =>
   516→   516→        this.evaluateCondition(event, condition)
   517→   517→      );
   518→   518→
   519→   519→      if (conditionsMet) {
   520→   520→        // Set cooldown
   521→   521→        this.alertCooldowns.set(cooldownKey, Date.now());
   522→   522→
   523→   523→        // Execute actions
   524→   524→        await this.executeActions(rule, event);
   525→   525→      }
   526→   526→    }
   527→   527→  }
   528→   528→
   529→   529→  /**
   530→   530→   * Evaluate a single condition
   531→   531→   */
   532→   532→  private evaluateCondition(event: SecurityEvent, condition: AlertCondition): boolean {
   533→   533→    const fieldValue = (event as Record<string, unknown>)[condition.field];
   534→   534→    
   535→   535→    switch (condition.operator) {
   536→   536→      case 'equals':
   537→   537→        return fieldValue === condition.value;
   538→   538→      case 'not_equals':
   539→   539→        return fieldValue !== condition.value;
   540→   540→      case 'contains':
   541→   541→        return String(fieldValue).includes(String(condition.value));
   542→   542→      case 'greater_than':
   543→   543→        return Number(fieldValue) > Number(condition.value);
   544→   544→      case 'less_than':
   545→   545→        return Number(fieldValue) < Number(condition.value);
   546→   546→      case 'in':
   547→   547→        return Array.isArray(condition.value) && condition.value.includes(fieldValue);
   548→   548→      default:
   549→   549→        return false;
   550→   550→    }
   551→   551→  }

   553→   553→  /**
   554→   554→   * Execute alert actions
   555→   555→   */
   556→   556→  private async executeActions(rule: AlertRule, event: SecurityEvent): Promise<void> {
   557→   557→    console.log(`[SecurityMonitor] Rule "${rule.name}" triggered by event ${event.id}`);
   558→   558→
   559→   559→    for (const action of rule.actions) {
   560→   560→      if (!action.enabled) continue;
   561→   561→
   562→   562→      try {
   563→   563→        switch (action.type) {
   564→   564→          case 'block_ip':
   565→   565→            if (event.ipAddress) {
   566→   566→              await this.blockIPAddress(event.ipAddress, action.config.durationHours as number || 24);
   567→   567→            }
   568→   568→            break;
   569→   569→
   570→   570→          case 'lock_account':
   571→   571→            if (event.userId) {
   572→   572→              await this.lockUserAccount(event.userId);
   573→   573→            }
   574→   574→            break;
   575→   575→
   576→   576→          case 'email':
   577→   577→            await this.sendAlertEmail(rule.name, event, action.config.recipients as string[]);
   578→   578→            break;
   579→   579→
   580→   580→          case 'webhook':
   581→   581→            await this.sendWebhook(action.config.url as string, rule, event);
   582→   582→            break;
   583→   583→
   584→   584→          case 'log':
   585→   585→            console.log(`[SECURITY ALERT] [${rule.name}]`, {
   586→   586→              eventId: event.id,
   587→   587→              type: event.eventType,
   588→   588→              severity: event.severity,
   589→   589→              ip: event.ipAddress,
   590→   590→              user: event.userId,
   591→   591→              details: event.details,
   592→   592→            });
   593→   593→            break;
   594→   594→
   595→   595→          case 'slack':
   596→   596→            await this.sendSlackNotification(rule.name, event, action.config.webhookUrl as string);
   597→   597→            break;
   598→   598→        }
   599→   599→      } catch (error) {
   600→   600→        console.error(`[SecurityMonitor] Error executing action ${action.type}:`, error);
   601→   601→      }
   602→   602→    }
   603→   603→  }

   695→   695→  /**
   696→   696→   * Helper to calculate risk from events array
   697→   697→   */
   698→   698→  private calculateIPRiskScoreFromEvents(events: SecurityEvent[]): number {
   699→   699→    let risk = 0;
   700→   700→    for (const event of events) {
   701→   701→      switch (event.severity) {
   702→   702→        case 'critical': risk += 30; break;
   703→   703→        case 'high': risk += 20; break;
   704→   704→        case 'medium': risk += 10; break;
   705→   705→        case 'low': risk += 5; break;
   706→   706→      }
   707→   707→    }
   708→   708→    return Math.min(100, risk);
   709→   709→  }
   710→   710→}

   717→   717→/**
   718→   718→ * Record authentication attempt
   719→   719→ */
   720→   720→export async function recordAuthAttempt(params: {
   721→   721→  success: boolean;
   722→   722→  userId?: string;
   723→   723→  ipAddress: string;
   724→   724→  userAgent?: string;
   725→   725→  method?: 'password' | 'oauth' | '2fa';
   726→   726→}): Promise<void> {
   727→   727→  const { success, userId, ipAddress, userAgent, method } = params;
   728→   728→  
   729→   729→  // Check for brute force pattern
   730→   730→  const recentFailures = await db.securityEvent.count({
   731→   731→    where: {
   732→   732→      eventType: 'AUTH_FAILURE',
   733→   733→      ipAddress,
   734→   734→      createdAt: { gte: new Date(Date.now() - 15 * 60 * 1000) }, // Last 15 min
   735→   735→    },
   736→   736→  });
   737→   737→
   738→   738→  if (!success && recentFailures >= 5) {
   739→   739→    // Potential brute force attack
   740→   740→    await securityMonitor.recordEvent({
   741→   741→      eventType: 'brute_force_detected',
   742→   742→      severity: recentFailures >= 10 ? 'critical' : 'high',
   743→   743→      source: 'authentication',
   744→   744→      ipAddress,
   745→   745→      userId,
   746→   746→      details: {
   747→   747→        method,
   748→   748→        failureCount: recentFailures + 1,
   749→   749→        userAgent,
   750→   750→        windowMinutes: 15,
   751→   751→      },
   752→   752→    });
   753→   753→  } else {
   754→   754→    await securityMonitor.recordEvent({
   755→   755→      eventType: success ? 'auth_success' : 'auth_failure',
   756→   756→      severity: success ? 'info' : 'low',
   757→   757→      source: 'authentication',
   758→   758→      ipAddress,
   759→   759→      userId,
   760→   760→      details: { method, userAgent },
   761→   761→    });
   762→   762→  }
   763→   763→}
