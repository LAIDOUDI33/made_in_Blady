   564→                      <TableCell colSpan={7} className="text-center py-12">
   565→                        <div className="text-gray-400">
   566→                          <Search className="h-8 w-8 mx-auto mb-2 opacity-50" />
   567→                          <p>Aucun paiement trouvé</p>
   568→                        </div>
   569→                      </TableCell>
   570→                    </TableRow>
   571→                  )}
   572→                </TableBody>
   573→              </Table>
   574→            </div>
   575→          </CardContent>
   576→        </Card>
   577→      </main>
   578→
   579→      {/* Action Dialog */}
   580→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   581→        <DialogContent className="sm:max-w-md">
   582→          <DialogHeader>
   583→            <DialogTitle>
   584→              {dialogAction === 'verify' && 'Approuver le paiement'}
   585→              {dialogAction === 'reject' && 'Rejeter le paiement'}
   586→              {dialogAction === 'refund' && 'Rembourser le paiement'}
   587→            </DialogTitle>
   588→          </DialogHeader>
   589→          
   590→          {selectedPayment && (
   591→            <div className="space-y-4 py-4">
   592→              {/* Payment Info */}
   593→              <div className="p-3 bg-gray-50 rounded-lg space-y-2">
   594→                <div className="flex justify-between text-sm">
   595→                  <span className="text-gray-500">Commande:</span>
   596→                  <span className="font-mono font-medium">{selectedPayment.orderNumber}</span>
   597→                </div>
   598→                <div className="flex justify-between text-sm">
   599→                  <span className="text-gray-500">Montant:</span>
   600→                  <span className="font-semibold">{formatDZD(selectedPayment.amount)}</span>
   601→                </div>
   602→                <div className="flex justify-between text-sm">
   603→                  <span className="text-gray-500">Client:</span>
   604→                  <span>{selectedPayment.buyerName}</span>
   605→                </div>
   606→              </div>
   607→
   608→              {/* Rejection Reason (only for reject) */}
   609→              {dialogAction === 'reject' && (
   610→                <div className="space-y-2">
   611→                  <Label htmlFor="reason">Motif du rejet *</Label>
   612→                  <Textarea
   613→                    id="reason"
   614→                    placeholder="Expliquez pourquoi ce paiement est rejeté..."
   615→                    value={rejectionReason}
   616→                    onChange={(e) => setRejectionReason(e.target.value)}
   617→                    rows={3}
   618→                  />
   619→                </div>
   620→              )}
   621→
   622→              {/* Notes */}
   623→              <div className="space-y-2">
   624→                <Label htmlFor="notes">Notes internes (optionnel)</Label>
   625→                <Textarea
   626→                  id="notes"
   627→                  placeholder="Ajoutez des notes pour référence interne..."
   628→                  value={notes}
   629→                  onChange={(e) => setNotes(e.target.value)}
   630→                  rows={2}
   631→                />
   632→              </div>
   633→            </div>
   634→          )}
   635→
   636→          <DialogFooter>
   637→            <Button variant="outline" onClick={() => setDialogOpen(false)}>
   638→              Annuler
   639→            </Button>
   640→            <Button
   641→              onClick={handleSubmitAction}
   642→              disabled={isLoading || (dialogAction === 'reject' && !rejectionReason)}
   643→              className={cn(
   644→                dialogAction === 'verify' && "bg-green-600 hover:bg-green-700",
   645→                dialogAction === 'reject' && "bg-red-600 hover:bg-red-700",
   646→                dialogAction === 'refund' && "bg-purple-600 hover:bg-purple-700"
   647→              )}
   648→            >
   649→              {isLoading ? 'Traitement...' : 
   650→               dialogAction === 'verify' ? 'Approuver' :
   651→               dialogAction === 'reject' ? 'Rejeter' : 'Rembourser'}
   652→            </Button>
   653→          </DialogFooter>
   654→        </DialogContent>
   655→      </Dialog>
   656→    </div>
   657→  )
   658→}
