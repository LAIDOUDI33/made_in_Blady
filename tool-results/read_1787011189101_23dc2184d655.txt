   354→export async function mergeContacts(
   355→  primaryId: string,
   356→  secondaryIds: string[]
   357→): Promise<CRMContact> {
   358→  const primary = await db.cRMContact.findUnique({ where: { id: primaryId } })
   359→  if (!primary) throw new Error('Primary contact not found')
   360→  
   361→  // Get all secondary contacts
   362→  const secondaries = await db.cRMContact.findMany({
   363→    where: { id: { in: secondaryIds } },
   364→  })
   365→  
   366→  // Merge tags
   367→  const primaryTags = JSON.parse(primary.tags || '[]') as string[]
   368→  const allTags = new Set(primaryTags)
   369→  
   370→  for (const secondary of secondaries) {
   371→    const secTags = JSON.parse(secondary.tags || '[]') as string[]
   372→    secTags.forEach(tag => allTags.add(tag))
   373→    
   374→    // Merge notes
   375→    if (secondary.notes) {
   376→      primary.notes = primary.notes 
   377→        ? `${primary.notes}\n\n--- Merged from ${secondary.firstName} ${secondary.lastName} ---\n${secondary.notes}`
   378→        : secondary.notes
   379→    }
   380→    
   381→    // Move interactions to primary
   382→    await db.cRMInteraction.updateMany({
   383→      where: { contactId: secondary.id },
   384→      data: { contactId: primaryId },
   385→    })
   386→    
   387→    // Move tasks to primary
   388→    await db.cRMTask.updateMany({
   389→      where: { contactId: secondary.id },
   390→      data: { contactId: primaryId },
   391→    })
   392→    
   393→    // Delete secondary
   394→    await db.cRMContact.delete({ where: { id: secondary.id } })
   395→  }
   396→  
   397→  // Update primary with merged data
   398→  const updated = await db.cRMContact.update({
   399→    where: { id: primaryId },
   400→    data: {
   401→      tags: JSON.stringify(Array.from(allTags)),
   402→      notes: primary.notes,
   403→    },
   404→  })
   405→  
   406→  return mapContactFromDB(updated)
   407→}

   409→export async function searchContacts(
   410→  query: string,
   411→  filters?: {
   412→    companyId?: string
   413→    role?: ContactRole
   414→    tags?: string[]
   415→    hasInteractionsSince?: Date
   416→  },
   417→  options?: {
   418→    page?: number
   419→    pageSize?: number
   420→    sortBy?: string
   421→    sortOrder?: 'asc' | 'desc'
   422→  }
   423→): Promise<PaginatedResult<CRMContact>> {
   424→  const page = options?.page || 1
   425→  const pageSize = options?.pageSize || 20
   426→  const skip = (page - 1) * pageSize
   427→  
   428→  const where: any = {
   429→    OR: [
   430→      { firstName: { contains: query, mode: 'insensitive' } },
   431→      { lastName: { contains: query, mode: 'insensitive' } },
   432→      { email: { contains: query, mode: 'insensitive' } },
   433→      { jobTitle: { contains: query, mode: 'insensitive' } },
   434→      { company: { name: { contains: query, mode: 'insensitive' } } },
   435→    ],
   436→  }
   437→  
   438→  if (filters?.companyId) {
   439→    where.companyId = filters.companyId
   440→  }
   441→  
   442→  if (filters?.role) {
   443→    where.role = filters.role
   444→  }
   445→  
   446→  if (filters?.tags && filters.tags.length > 0) {
   447→    // SQLite JSON search for tags containing any of the filter tags
   448→    where.OR = filters.tags.map(tag => ({
   449→      tags: { contains: tag },
   450→    }))
   451→  }
   452→  
   453→  if (filters?.hasInteractionsSince) {
   454→    where.lastInteractionAt = { gte: filters.hasInteractionsSince }
   455→  }
   456→  
   457→  const [contacts, total] = await Promise.all([
   458→    db.cRMContact.findMany({
   459→      where,
   460→      skip,
   461→      take: pageSize,
   462→      orderBy: {
   463→        [(options?.sortBy || 'createdAt')]: options?.sortOrder || 'desc',
   464→      },
   465→      include: {
   466→        _count: { select: { interactions: true } },
   467→      },
   468→    }),
   469→    db.cRMContact.count({ where }),
   470→  ])
   471→  
   472→  return {
   473→    data: contacts.map(mapContactFromDB),
   474→    total,
   475→    page,
   476→    pageSize,
   477→    totalPages: Math.ceil(total / pageSize),
   478→  }
   479→}

   594→function calculateInitialLeadScore(data: any): number {
   595→  let score = 20 // Base score
   596→  
   597→  // Source scoring
   598→  const sourceScores: Record<string, number> = {
   599→    REFERRAL: 25,
   600→    TRADE_SHOW: 20,
   601→    WEBSITE: 15,
   602→    RFQ: 30,
   603→    EMAIL: 10,
   604→    COLD_CALL: 5,
   605→    SOCIAL_MEDIA: 12,
   606→    PARTNER: 22,
   607→  }
   608→  score += sourceScores[data.source] || 10
   609→  
   610→  // Company size scoring
   611→  const sizeScores: Record<string, number> = {
   612→    '500+': 20,
   613→    '201-500': 18,
   614→    '51-200': 15,
   615→    '11-50': 10,
   616→    '1-10': 5,
   617→  }
   618→  score += sizeScores[data.companySize] || 5
   619→  
   620→  // Value scoring
   621→  if (data.estimatedValue >= 1000000) score += 20
   622→  else if (data.estimatedValue >= 500000) score += 15
   623→  else if (data.estimatedValue >= 100000) score += 10
   624→  else if (data.estimatedValue >= 10000) score += 5
   625→  
   626→  return Math.min(score, 100)
   627→}

   630→  const stageProbabilities: Record<string, number> = {

   634→    PROPOSAL: 50,
   635→    NEGOTIATION: 70,
   636→    WON: 100,
   637→    LOST: 0,
   638→  }
   639→  return stageProbabilities[stage] || 10
   640→}

   655→function getLeadStatusFromStage(stage: string): LeadStatus {
   656→  if (stage === 'WON') return 'WON'
   657→  if (stage === 'LOST') return 'LOST'
   658→  if (['NEGOTIATION', 'PROPOSAL'].includes(stage)) return 'NEGOTIATION'
   659→  if (stage === 'QUALIFIED') return 'QUALIFIED'
   660→  if (stage === 'CONTACTED') return 'CONTACTED'
   661→  return 'NEW'
   662→}

   805→export async function predictConversionProbability(leadId: string): Promise<number> {

   808→    include: { interactions: true, tasks: true },
   809→  })

   814→  // This is a simplified version - in production, use actual ML model
   815→  let probability = lead.probability
   816→  
   817→  // Adjust based on engagement velocity
   818→  const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000)
   819→  const recentInteractions = lead.interactions.filter(i => 
   820→    new Date(i.createdAt) > thirtyDaysAgo
   821→  )
   822→  
   823→  if (recentInteractions.length >= 5) probability += 10
   824→  else if (recentInteractions.length >= 3) probability += 5
   825→  else if (recentInteractions.length === 0) probability -= 10
   826→  
   827→  // Adjust based on task completion rate
   828→  const totalTasks = lead.tasks.length
   829→  const completedTasks = lead.tasks.filter(t => t.status === 'COMPLETED').length
   830→  if (totalTasks > 0) {
   831→    const completionRate = completedTasks / totalTasks
   832→    probability += completionRate * 15
   833→  }
   834→  
   835→  // Time-based decay
   836→  const daysUntilClose = Math.ceil(
   837→    (new Date(lead.expectedCloseDate).getTime() - Date.now()) / (1000 * 60 * 60 * 24)
   838→  )
   839→  
   840→  if (daysUntilClose < 0) {
   841→    probability -= 20 // Overdue
   842→  } else if (daysUntilClose < 7) {
   843→    probability += 5 // Urgency
   844→  }
   845→  
   846→  return Math.max(0, Math.min(100, probability))

  1279→      
  1280→      // Update rule stats

  1284→          lastTriggeredAt: new Date(),

  1292→function evaluateCondition(condition: Condition, payload: Record<string, any>): boolean {
  1293→  const value = getNestedValue(payload, condition.field)
  1294→  
  1295→  switch (condition.operator) {
  1296→    case 'equals':
  1297→      return value === condition.value
  1298→    case 'notEquals':
  1299→      return value !== condition.value
  1300→    case 'contains':
  1301→      return typeof value === 'string' && value.includes(condition.value)
  1302→    case 'greaterThan':
  1303→      return Number(value) > condition.value
  1304→    case 'lessThan':
  1305→      return Number(value) < condition.value
  1306→    case 'in':
  1307→      return Array.isArray(condition.value) && condition.value.includes(value)
  1308→    case 'exists':
  1309→      return value !== undefined && value !== null
  1310→    default:
  1311→      return false
  1312→  }
  1313→}
  1314→
  1315→function getNestedValue(obj: Record<string, any>, path: string): any {
  1316→  return path.split('.').reduce((acc, key) => acc?.[key], obj)
  1317→}

  1319→async function executeAction(action: AutomationAction, payload: Record<string, any>): Promise<void> {
  1320→  switch (action.type) {
  1321→    case 'SEND_EMAIL':
  1322→      // Would integrate with email service
  1323→      console.log('Sending email:', action.config, payload)
  1324→      break
  1325→      
  1326→    case 'CREATE_TASK':
  1327→      if (payload.leadId) {
  1328→        await createTask({
  1329→          leadId: payload.leadId,
  1330→          companyId: payload.companyId || action.config.companyId,
  1331→          title: action.config.title || 'Automated task',
  1332→          description: action.config.description || '',
  1333→          type: action.config.taskType || 'OTHER',
  1334→          priority: action.config.priority || 'MEDIUM',
  1335→          dueDate: new Date(Date.now() + (action.config.dueInHours || 24) * 60 * 60 * 1000),
  1336→          assignedTo: action.config.assignedTo || payload.assignedTo,
  1337→          createdBy: 'automation',
  1338→        })
  1339→      }
  1340→      break
  1341→      
  1342→    case 'UPDATE_LEAD':
  1343→      if (payload.leadId) {
  1344→        await db.cRMLead.update({
  1345→          where: { id: payload.leadId },
  1346→          data: action.config.updates || {},
  1347→        })
  1348→      }
  1349→      break
  1350→      
  1351→    case 'SEND_NOTIFICATION':
  1352→      // Would integrate with notification service
  1353→      console.log('Sending notification:', action.config, payload)
  1354→      break
  1355→      
  1356→    case 'WEBHOOK':
  1357→      // Would make HTTP request to webhook URL
  1358→      console.log('Calling webhook:', action.config.url, payload)
  1359→      break
  1360→      
  1361→    case 'ADD_TAG':
  1362→      if (payload.contactId) {
  1363→        const contact = await db.cRMContact.findUnique({
  1364→          where: { id: payload.contactId },
  1365→        })
  1366→        if (contact) {
  1367→          const tags = JSON.parse(contact.tags || '[]')
  1368→          if (!tags.includes(action.config.tag)) {
  1369→            tags.push(action.config.tag)
  1370→            await db.cRMContact.update({
  1371→              where: { id: payload.contactId },
  1372→              data: { tags: JSON.stringify(tags) },
  1373→            })
  1374→          }
  1375→        }
  1376→      }
  1377→      break
  1378→  }
  1379→}

  1508→export async function getCRMStats(companyId?: string, userId?: string): Promise<{
  1509→  totalContacts: number
  1510→  totalLeads: number
  1511→  activeLeads: number
  1512→  wonLeads: number
  1513→  lostLeads: number
  1514→  conversionRate: number
  1515→  totalPipelineValue: number
  1516→  weightedPipelineValue: number
  1517→  tasksDueToday: number
  1518→  overdueTasks: number
  1519→  interactionsThisWeek: number
  1520→  leadsBySource: Record<string, number>
  1521→  leadsByStage: Record<string, number>
  1522→}> {
  1523→  const where: any = {}
  1524→  if (companyId) where.companyId = companyId
  1525→  if (userId) where.assignedTo = userId
  1526→  
  1527→  const [
  1528→    totalContacts,
  1529→    totalLeads,
  1530→    wonLeads,
  1531→    lostLeads,
  1532→    tasksDueToday,
  1533→    overdueTasks,
  1534→    interactionsThisWeek,
  1535→    leadsBySourceRaw,
  1536→    leadsByStageRaw,
  1537→    pipelineLeads,
  1538→  ] = await Promise.all([
  1539→    db.cRMContact.count({ where: companyId ? { companyId } : {} }),
  1540→    db.cRMLead.count({ where }),
  1541→    db.cRMLead.count({ where: { ...where, status: 'WON' } }),
  1542→    db.cRMLead.count({ where: { ...where, status: 'LOST' } }),
  1543→    db.cRMTask.count({
  1544→      where: {
  1545→        ...(userId ? { assignedTo: userId } : {}),
  1546→        status: { in: ['TODO', 'IN_PROGRESS'] },
  1547→        dueDate: { gte: new Date(new Date().setHours(0, 0, 0, 0)), lte: new Date(new Date().setHours(23, 59, 59, 999)) },
  1548→      },
  1549→    }),
  1550→    db.cRMTask.count({
  1551→      where: {
  1552→        ...(userId ? { assignedTo: userId } : {}),
  1553→        status: { in: ['TODO', 'IN_PROGRESS'] },
  1554→        dueDate: { lt: new Date() },
  1555→      },
  1556→    }),
  1557→    db.cRMInteraction.count({
  1558→      where: {
  1559→        ...(companyId ? { companyId } : {}),
  1560→        createdAt: { gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000) },
  1561→      },
  1562→    }),
  1563→    db.cRMLead.groupBy({
  1564→      by: ['source'],
  1565→      where,
  1566→      _count: true,
  1567→    }),
  1568→    db.cRMLead.groupBy({
  1569→      by: ['pipelineStage'],
  1570→      where: { ...where, status: { notIn: ['WON', 'LOST'] } },
  1571→      _count: true,
  1572→    }),
  1573→    db.cRMLead.findMany({
  1574→      where: { ...where, status: { notIn: ['WON', 'LOST'] } },
  1575→      select: { estimatedValue: true, probability: true },
  1576→    }),
  1577→  ])
  1578→  
  1579→  const activeLeads = totalLeads - wonLeads - lostLeads
  1580→  const conversionRate = totalLeads > 0 ? (wonLeads / totalLeads) * 100 : 0
  1581→  
  1582→  const totalPipelineValue = pipelineLeads.reduce((sum, lead) => sum + (lead.estimatedValue || 0), 0)
  1583→  const weightedPipelineValue = pipelineLeads.reduce(
  1584→    (sum, lead) => sum + ((lead.estimatedValue || 0) * (lead.probability || 0) / 100),
  1585→    0
  1586→  )
  1587→  
  1588→  const leadsBySource: Record<string, number> = {}
  1589→  leadsBySourceRaw.forEach(item => {
  1590→    leadsBySource[item.source] = item._count
  1591→  })
  1592→  
  1593→  const leadsByStage: Record<string, number> = {}
  1594→  leadsByStageRaw.forEach(item => {
  1595→    leadsByStage[item.pipelineStage] = item._count
  1596→  })
  1597→  
  1598→  return {
  1599→    totalContacts,
  1600→    totalLeads,
  1601→    activeLeads,
  1602→    wonLeads,
  1603→    lostLeads,
  1604→    conversionRate: Math.round(conversionRate * 100) / 100,
  1605→    totalPipelineValue: Math.round(totalPipelineValue * 100) / 100,
  1606→    weightedPipelineValue: Math.round(weightedPipelineValue * 100) / 100,
  1607→    tasksDueToday,
  1608→    overdueTasks,
  1609→    interactionsThisWeek,
  1610→    leadsBySource,
  1611→    leadsByStage,
  1612→  }
  1613→}
