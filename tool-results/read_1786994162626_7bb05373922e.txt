   136→function OptionSelector({
   137→  option,
   138→  value,
   139→  onChange,
   140→}: {
   141→  option: CustomizationOption;
   142→  value: any;
   143→  onChange: (value: any) => void;
   144→}) {
   145→  switch (option.type) {
   146→    case 'color':
   147→      return <ColorPicker option={option} value={value} onChange={onChange} />;
   148→    case 'radio':
   149→      return <RadioGroup option={option} value={value} onChange={onChange} />;
   150→    case 'checkbox':
   151→      return <CheckboxGroup option={option} value={value} onChange={onChange} />;
   152→    case 'select':
   153→      return <SelectDropdown option={option} value={value} onChange={onChange} />;
   154→    case 'text':
   155→    case 'number':
   156→      return <TextInputField option={option} value={value} onChange={onChange} />;
   157→    default:
   158→      return null;
   159→  }
   160→}

   162→function ColorPicker({
   163→  option,
   164→  value,
   165→  onChange,
   166→}: {
   167→  option: CustomizationOption;
   168→  value: string;
   169→  onChange: (value: string) => void;
   170→}) {
   171→  return (
   172→    <View style={styles.optionContainer}>
   173→      <View style={styles.optionHeader}>
   174→        <Text style={styles.optionName}>{option.name}</Text>
   175→        {option.required && <Text style={styles.requiredStar}>*</Text>}
   176→      </View>
   177→      
   178→      <View style={styles.colorOptions}>
   179→        {option.values?.map(v => (
   180→          <TouchableOpacity
   181→            key={v.id}
   182→            style={[
   183→              styles.colorSwatch,
   184→              value === v.id && styles.colorSwatchSelected,
   185→              { backgroundColor: v.colorHex || '#ccc' }
   186→            ]}
   187→            onPress={() => onChange(v.id)}
   188→            activeOpacity={0.7}
   189→          >
   190→            {value === v.id && (
   191→              <Ionicons name="checkmark" size={16} color="#fff" />
   192→            )}
   193→            
   194→            {v.priceModifier !== undefined && v.priceModifier > 0 && (
   195→              <View style={styles.priceModifierBadge}>
   196→                <Text style={styles.priceModifierText}>+{v.priceModifier}</Text>
   197→              </View>
   198→            )}
   199→          </TouchableOpacity>
   200→        ))}
   201→      </View>
   202→      
   203→      {option.values?.find(v => v.id === value)?.label && (
   204→        <Text style={styles.selectedValue}>
   205→          {option.values.find(v => v.id === value)?.label}
   206→        </Text>
   207→      )}
   208→    </View>

   261→
   262→function CheckboxGroup({
   263→  option,
   264→  value,
   265→  onChange,
   266→}: {
   267→  option: CustomizationOption;
   268→  value: string[];
   269→  onChange: (value: string[]) => void;
   270→}) {
   271→  const toggleValue = (id: string) => {
   272→    const newValue = value.includes(id)
   273→      ? value.filter(v => v !== id)
   274→      : [...value, id];
   275→    onChange(newValue);
   276→  };
   277→
   278→  return (
   279→    <View style={styles.optionContainer}>
   280→      <View style={styles.optionHeader}>
   281→        <Text style={styles.optionName}>{option.name}</Text>
   282→        {option.required && <Text style={styles.requiredStar}>*</Text>}
   283→      </View>
   284→      
   285→      {option.values?.map(v => (
   286→        <TouchableOpacity
   287→          key={v.id}
   288→          style={styles.checkboxItem}
   289→          onPress={() => toggleValue(v.id)}
   290→          activeOpacity={0.7}
   291→        >
   292→          <View style={[
   293→            styles.checkbox,
   294→            value.includes(v.id) && styles.checkboxSelected
   295→          ]}>
   296→            {value.includes(v.id) && (
   297→              <Ionicons name="checkmark" size={14} color="#fff" />
   298→            )}
   299→          </View>
   300→          
   301→          <Text style={styles.checkboxLabel}>{v.label}</Text>
   302→          
   303→          {v.priceModifier !== undefined && (
   304→            <Text style={styles.optionPriceModifier}>
   305→              +{v.priceModifier.toLocaleString('fr-FR')}
   306→            </Text>
   307→          )}
   308→        </TouchableOpacity>
   309→      ))}
   310→    </View>
   311→  );
   312→}

   387→// Bulk Pricing Table
   388→function BulkPricingTable({ 
   389→  tiers, 
   390→  currency, 
   391→  selectedTier, 
   392→  onSelectTier 
   393→}: { 
   394→  tiers: BulkPricingTier[]; 
   395→  currency: string; 
   396→  selectedTier: string | null;
   397→  onSelectTier: (tierId: string) => void;
   398→}) {
   399→  if (!tiers || tiers.length === 0) return null;
   400→
   401→  return (
   402→    <View style={styles.pricingSection}>
   403→      <View style={styles.sectionHeader}>
   404→        <Ionicons name="pricetags-outline" size={20} color={Colors.primary} />
   405→        <Text style={styles.sectionTitle}>Tarifs par quantité</Text>
   406→      </View>
   407→
   408→      <View style={styles.pricingTable}>
   409→        {/* Header */}
   410→        <View style={styles.pricingHeader}>
   411→          <Text style={styles.pricingHeaderText}>Quantité</Text>
   412→          <Text style={styles.pricingHeaderText}>Prix unitaire</Text>
   413→          <Text style={styles.pricingHeaderText}>Remise</Text>
   414→        </View>
   415→
   416→        {/* Rows */}
   417→        {tiers.map(tier => (
   418→          <TouchableOpacity
   419→            key={tier.id}
   420→            style={[
   421→              styles.pricingRow,
   422→              selectedTier === tier.id && styles.pricingRowSelected,
   423→              tier.isBestValue && styles.pricingRowBestValue,
   424→            ]}
   425→            onPress={() => onSelectTier(tier.id)}
   426→            activeOpacity={0.7}
   427→          >
   428→            <View style={styles.quantityCell}>
   429→              <Text style={styles.quantityText}>
   430→                {tier.minQuantity}+
   431→                {tier.maxQuantity ? ` - ${tier.maxQuantity}` : ''}
   432→              </Text>
   433→              {tier.isBestValue && (
   434→                <View style={styles.bestValueBadge}>
   435→                  <Text style={styles.bestValueText}>Meilleur prix</Text>
   436→                </View>
   437→              )}
   438→            </View>
   439→            
   440→            <Text style={[
   441→              styles.priceCell,
   442→              selectedTier === tier.id && styles.priceCellSelected,
   443→            ]}>
   444→              {tier.unitPrice.toLocaleString('fr-FR')} {currency}
   445→            </Text>
   446→            
   447→            <View style={styles.discountCell}>
   448→              <Text style={styles.discountText}>
   449→                -{tier.discountPercent}%
   450→              </Text>
   451→            </View>
   452→            
   453→            {selectedTier === tier.id && (
   454→              <Ionicons name="checkmark-circle" size={20} color={Colors.primary} />
   455→            )}
   456→          </TouchableOpacity>
   457→        ))}
   458→      </View>
   459→
   460→      {/* Savings info */}
   461→      {selectedTier && tiers.find(t => t.id === selectedTier)?.discountPercent > 0 && (
   462→        <View style={styles.savingsBanner}>
   463→          <Ionicons name="trending-down-outline" size={16} color={Colors.success} />
   464→          <Text style={styles.savingsText}>
   465→            Vous économisez {tiers.find(t => t.id === selectedTier)?.discountPercent}% sur le prix unitaire !
   466→          </Text>
   467→        </View>
   468→      )}
   469→    </View>
   470→  );
   471→}

   473→// Certification Viewer
   474→function CertificationViewer({
   475→  certifications,
   476→  onViewCertificate,
   477→}: {
   478→  certifications: ProductCertification[];
   479→  onViewCertificate: (cert: ProductCertification) => void;
   480→}) {
   481→  if (!certifications || certifications.length === 0) return null;
   482→
   483→  const [showAll, setShowAll] = useState(false);
   484→
   485→  const displayCerts = showAll ? certifications : certifications.slice(0, 3);
   486→
   487→  return (
   488→    <View style={styles.certSection}>
   489→      <View style={styles.sectionHeader}>
   490→        <Ionicons name="shield-checkmark-outline" size={20} color={Colors.success} />
   491→        <Text style={styles.sectionTitle}>Certifications</Text>
   492→        {certifications.length > 3 && (
   493→          <TouchableOpacity onPress={() => setShowAll(!showAll)}>
   494→            <Text style={styles.showAllText}>
   495→              {showAll ? 'Voir moins' : `Voir tout (${certifications.length})`}
   496→            </Text>
   497→          </TouchableOpacity>
   498→        )}
   499→      </View>
   500→
   501→      {displayCerts.map(cert => (
   502→        <TouchableOpacity
   503→          key={cert.id}
   504→          style={[
   505→            styles.certCard,
   506→            !cert.isValid && styles.certCardInvalid,
   507→          ]}
   508→          onPress={() => onViewCertificate(cert)}
   509→          activeOpacity={0.7}
   510→        >
   511→          <View style={[
   512→            styles.certIconContainer,
   513→            { backgroundColor: cert.isValid ? Colors.successLight : Colors.errorLight }
   514→          ]}>
   515→            <Ionicons
   516→              name={(CERTIFICATION_ICONS[cert.name] || CERTIFICATION_ICONS.default) as any}
   517→              size={24}
   518→              color={cert.isValid ? Colors.success : Colors.error}
   519→            />
   520→          </View>
   521→
   522→          <View style={styles.certContent}>
   523→            <Text style={styles.certName}>{cert.name}</Text>
   524→            <Text style={styles.certIssuer}>{cert.issuingBody}</Text>
   525→            <Text style={styles.certNumber}>N° {cert.certificateNumber}</Text>
   526→          </View>
   527→
   528→          <View style={styles.certStatusContainer}>
   529→            <View style={[
   530→              styles.certStatusBadge,
   531→              { backgroundColor: cert.isValid ? Colors.successLight : Colors.errorLight }
   532→            ]}>
   533→              <Text style={[
   534→                styles.certStatusText,
   535→                { color: cert.isValid ? Colors.success : Colors.error }
   536→              ]}>
   537→                {cert.isValid ? 'Valide' : 'Expiré'}
   538→              </Text>
   539→            </View>
   540→            
   541→            <Ionicons name="document-text-outline" size={20} color={Colors.textTertiary} />
   542→          </View>
   543→        </TouchableOpacity>
   544→      ))}
   545→    </View>
   546→  );
   547→}

   549→// Related Products Carousel
   550→function RelatedProductsCarousel({
   551→  products,
   552→  onPressProduct,
   553→}: {
   554→  products: RelatedProduct[];
   555→  onPressProduct: (product: RelatedProduct) => void;
   556→}) {
   557→  if (!products || products.length === 0) return null;
   558→
   559→  return (
   560→    <View style={styles.relatedSection}>
   561→      <View style={styles.sectionHeader}>
   562→        <Ionicons name="link-outline" size={20} color={Colors.info} />
   563→        <Text style={styles.sectionTitle}>Produits associés</Text>
   564→      </View>
   565→
   566→      <FlatList
   567→        data={products}
   568→        keyExtractor={(item) => item.id}
   569→        horizontal
   570→        showsHorizontalScrollIndicator={false}
   571→        contentContainerStyle={styles.relatedList}
   572→        renderItem={({ item }) => {
   573→          const relationConfig = RELATION_TYPE_CONFIG[item.relationType] || RELATION_TYPE_CONFIG.related;
   574→
   575→          return (
   576→            <TouchableOpacity
   577→              style={styles.relatedCard}
   578→              onPress={() => onPressProduct(item)}
   579→              activeOpacity={0.7}
   580→            >
   581→              <View style={styles.relatedImageContainer}>
   582→                {item.image ? (
   583→                  <Image source={{ uri: item.image }} style={styles.relatedImage} />
   584→                ) : (
   585→                  <View style={styles.relatedImagePlaceholder}>
   586→                    <Ionicons name="cube-outline" size={32} color={Colors.textTertiary} />
   587→                  </View>
   588→                )}
   589→                
   590→                <View style={[styles.relationBadge, { backgroundColor: relationConfig.color }]}>
   591→                  <Text style={styles.relationBadgeText}>{relationConfig.label}</Text>
   592→                </View>
   593→              </View>
   594→
   595→              <View style={styles.relatedContent}>
   596→                <Text style={styles.relatedName} numberOfLines={2}>{item.name}</Text>
   597→                <Text style={styles.relatedSupplier}>{item.supplier}</Text>
   598→                <Text style={styles.relatedPrice}>
   599→                  {item.price.toLocaleString('fr-FR')} {item.currency}
   600→                </Text>
   601→              </View>
   602→            </TouchableOpacity>
   603→          );
   604→        }}
   605→      />
   606→    </View>
   607→  );
   608→}

   690→// ============================================
   691→// Main Component
   692→// ============================================
   693→
   694→export default function ProductCustomizer({
   695→  productId,
   696→  basePrice,
   697→  currency,
   698→  customizationOptions = [],
   699→  bulkPricingTiers = [],
   700→  certifications = [],
   701→  relatedProducts = [],
   702→  onSelectionChange,
   703→  onAddToCart,
   704→  onViewCertification,
   705→  onPressRelatedProduct,
   706→  compact = false,
   707→  showHeader = true,
   708→  initialQuantity = 1,
   709→}: ProductCustomizerProps) {
   710→  // State
   711→  const [selections, setSelections] = useState<Record<string, any>>({});
   712→  const [quantity, setQuantity] = useState(initialQuantity);
   713→  const [selectedTier, setSelectedTier] = useState<string | null>(null);
   714→  const [showCertModal, setShowCertModal] = useState(false);
   715→  const [selectedCert, setSelectedCert] = useState<ProductCertification | null>(null);
   716→
   717→  // Calculate total price based on selections and quantity
   718→  const totalPrice = useMemo(() => {
   719→    let price = basePrice;
   720→
   721→    // Apply customization option modifiers
   722→    Object.entries(selections).forEach(([optionId, value]) => {
   723→      const option = customizationOptions.find(o => o.id === optionId);
   724→      if (!option) return;
   725→
   726→      if (Array.isArray(value)) {
   727→        // Checkbox - sum all selected
   728→        value.forEach((valId: string) => {
   729→          const val = option.values?.find(v => v.id === valId);
   730→          if (val?.priceModifier) {
   731→            price += val.priceModifier;
   732→          }
   733→        });
   734→      } else {
   735→        // Single selection
   736→        const val = option.values?.find(v => v.id === value);
   737→        if (val?.priceModifier) {
   738→          price += val.priceModifier;
   739→        }
   740→      }
   741→    });
   742→
   743→    // Apply bulk pricing discount
   744→    if (bulkPricingTiers.length > 0) {
   745→      const applicableTier = bulkPricingTiers.find(tier =>
   746→        quantity >= tier.minQuantity &&
   747→        (!tier.maxQuantity || quantity <= tier.maxQuantity)
   748→      );
   749→
   750→      if (applicableTier) {
   751→        price = applicableTier.unitPrice;
   752→      }
   753→    }
   754→
   755→    return Math.round(price * quantity * 100) / 100;
   756→  }, [selections, quantity, basePrice, customizationOptions, bulkPricingTiers]);
   757→
   758→  // Handle selection change
   759→  const handleSelectionChange = (optionId: string, value: any) => {
   760→    const newSelections = { ...selections, [optionId]: value };
   761→    setSelections(newSelections);
   762→    onSelectionChange?.(newSelections, totalPrice);
   763→  };
   764→
   765→  // Handle add to cart
   766→  const handleAddToCart = () => {
   767→    // Validate required fields
   768→    const missingRequired = customizationOptions
   769→      .filter(opt => opt.required)
   770→      .filter(opt => {
   771→        const value = selections[opt.id];
   772→        if (opt.type === 'checkbox') {
   773→          return !Array.isArray(value) || value.length === 0;
   774→        }
   775→        return !value || (typeof value === 'string' && value.trim() === '');
   776→      });
   777→
   778→    if (missingRequired.length > 0) {
   779→      // Would show error in real app
   780→      console.log('[ProductCustomizer] Missing required:', missingRequired.map(o => o.name));
   781→    }
   782→
   783→    onAddToCart?.(quantity, selections);
   784→  };
   785→
   786→  // Handle view certification
   787→  const handleViewCertification = (cert: ProductCertification) => {
   788→    setSelectedCert(cert);
   789→    setShowCertModal(true);
   790→    onViewCertification?.(cert);
   791→  };
   792→
   793→  // Compact mode - just show price summary
   794→  if (compact) {
   795→    return (
   796→      <View style={styles.compactContainer}>
   797→        <View style={styles.compactPriceRow}>
   798→          <Text style={styles.compactTotalLabel}>Total</Text>
   799→          <Text style={styles.compactTotalPrice}>
   800→            {totalPrice.toLocaleString('fr-FR')} {currency}
   801→          </Text>
   802→        </View>
   803→        
   804→        {bulkPricingTiers.length > 0 && (
   805→          <TouchableOpacity 
   806→            style={styles.bulkInfoButton}
   807→            onPress={() => {} /* Would expand in full mode */}
   808→          >
   809→            <Ionicons name="pricetag-outline" size={14} color={Colors.primary} />
   810→            <Text style={styles.bulkInfoText}>
   811→              Tarif {quantity >= 100 ? 'grossiste' : quantity >= 10 ? 'semi-gros' : 'unitaire'}
   812→            </Text>
   813→          </TouchableOpacity>
   814→        )}
   815→      </View>
   816→    );
   817→  }
   818→
   819→  // Full mode
   820→  return (
   821→    <View style={styles.container}>
   822→      {/* Header */}
   823→      {showHeader && (
   824→        <View style={styles.header}>
   825→          <Text style={styles.headerTitle}>Personnaliser votre commande</Text>
   826→          <Text style={styles.headerSubtitle}>
   827→            Choisissez les options et la quantité souhaitée
   828→          </Text>
   829→        </View>
   830→      )}
   831→
   832→      <ScrollView 
   833→        style={styles.scrollView} 
   834→        showsVerticalScrollIndicator={false}
   835→      >
   836→        {/* Customization Options */}
   837→        {customizationOptions.length > 0 && (
   838→          <View style={styles.optionsSection}>
   839→            <View style={styles.sectionHeader}>
   840→              <Ionicons name="options-outline" size={20} color={Colors.primary} />
   841→              <Text style={styles.sectionTitle}>Options disponibles</Text>
   842→            </View>
   843→
   844→            {customizationOptions
   845→              .sort((a, b) => a.sortOrder - b.sortOrder)
   846→              .map(option => (
   847→                <OptionSelector
   848→                  key={option.id}
   849→                  option={option}
   850→                  value={selections[option.id]}
   851→                  onChange={(value) => handleSelectionChange(option.id, value)}
   852→                />
   853→              ))
   854→            }
   855→          </View>
   856→        )}
   857→
   858→        {/* Quantity Selector */}
   859→        <View style={styles.quantitySection}>
   860→          <View style={styles.sectionHeader}>
   861→            <Ionicons name="cube-outline" size={20} color={Colors.primary} />
   862→            <Text style={styles.sectionTitle}>Quantité</Text>
   863→          </View>
   864→
   865→          <View style={styles.quantityControls}>
   866→            <TouchableOpacity
   867→              style={styles.quantityButton}
   868→              onPress={() => setQuantity(Math.max(1, quantity - 1))}
   869→            >
   870→              <Ionicons name="remove" size={20} color={Colors.text} />
   871→            </TouchableOpacity>
   872→
   873→            <TextInput
   874→              style={styles.quantityInput}
   875→              value={String(quantity)}
   876→              onChangeText={(text) => {
   877→                const num = parseInt(text) || 1;
   878→                setQuantity(Math.max(1, num));
   879→              }}
   880→              keyboardType="number-pad"
   881→              textAlign="center"
   882→            />
   883→
   884→            <TouchableOpacity
   885→              style={styles.quantityButton}
   886→              onPress={() => setQuantity(quantity + 1)}
   887→            >
   888→              <Ionicons name="add" size={20} color={Colors.text} />
   889→            </TouchableOpacity>
   890→          </View>
   891→
   892→          {/* Quick quantity buttons */}
   893→          <View style={styles.quickQuantityButtons}>
   894→            {[10, 25, 50, 100].map(qty => (
   895→              <TouchableOpacity
   896→                key={qty}
   897→                style={[
   898→                  styles.quickQtyButton,
   899→                  quantity === qty && styles.quickQtyButtonSelected,
   900→                ]}
   901→                onPress={() => setQuantity(qty)}
   902→              >
   903→                <Text style={[
   904→                  styles.quickQtyButtonText,
   905→                  quantity === qty && styles.quickQtyButtonTextSelected,
   906→                ]}>
   907→                  {qty}
   908→                </Text>
   909→              </TouchableOpacity>
   910→            ))}
   911→          </View>
   912→        </View>
   913→
   914→        {/* Bulk Pricing */}
   915→        <BulkPricingTable
   916→          tiers={bulkPricingTiers}
   917→          currency={currency}
   918→          selectedTier={selectedTier}
   919→          onSelectTier={setSelectedTier}
   920→        />
   921→
   922→        {/* Certifications */}
   923→        <CertificationViewer
   924→          certifications={certifications}
   925→          onViewCertificate={handleViewCertification}
   926→        />
   927→
   928→        {/* Related Products */}
   929→        <RelatedProductsCarousel
   930→          products={relatedProducts}
   931→          onPressProduct={onPressRelatedProduct || (() => {})}
   932→        />
   933→
   934→        {/* Bottom spacing for sticky footer */}
   935→        <View style={{ height: 120 }} />
   936→      </ScrollView>
   937→
   938→      {/* Sticky Footer */}
   939→      <View style={styles.footer}>
   940→        <View style={styles.footerPriceContainer}>
   941→          <Text style={styles.footerTotalLabel}>Total</Text>
   942→          <Text style={styles.footerTotalPrice}>
   943→            {totalPrice.toLocaleString('fr-FR')} {currency}
   944→          </Text>
   945→          {quantity > 1 && (
   946→            <Text style={styles.footerUnitPrice}>
   947→              ({(totalPrice / quantity).toLocaleString('fr-FR')} {currency}/unité)
   948→            </Text>
   949→          )}
   950→        </View>
   951→
   952→        <TouchableOpacity
   953→          style={styles.addToCartButton}
   954→          onPress={handleAddToCart}
   955→          activeOpacity={0.8}
   956→        >
   957→          <Ionicons name="cart-outline" size={20} color={Colors.white} />
   958→          <Text style={styles.addToCartButtonText}>Ajouter au panier</Text>
   959→        </TouchableOpacity>
   960→      </View>
   961→
   962→      {/* Certificate Modal */}
   963→      <CertificateDetailModal
   964→        certificate={selectedCert}
   965→        visible={showCertModal}
   966→        onClose={() => setShowCertModal(false)}
   967→      />
   968→    </View>
   969→  );
   970→}
