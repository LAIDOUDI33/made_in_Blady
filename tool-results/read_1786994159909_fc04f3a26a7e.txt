   111→// Video Player Modal
   112→function VideoPlayerModal({ 
   113→  video, 
   114→  visible, 
   115→  onClose, 
   116→  onDownload,
   117→  onCast,
   118→}: { 
   119→  video: ProductVideo | null; 
   120→  visible: boolean; 
   121→  onClose: () => void;
   122→  onDownload?: () => void;
   123→  onCast?: () => void;
   124→}) {
   125→  const videoRef = useRef<Video>(null);
   126→  const [isPlaying, setIsPlaying] = useState(false);
   127→  const [isMuted, setIsMuted] = useState(false);
   128→  const [isFullscreen, setIsFullscreen] = useState(false);
   129→  const [playbackPosition, setPlaybackPosition] = useState(0);
   130→  const [duration, setDuration] = useState(0);
   131→  const [isLoading, setIsLoading] = useState(true);
   132→
   133→  const handlePlaybackStatusUpdate = (status: AVPlaybackStatus) => {
   134→    if (!status.isLoaded) return;
   135→    
   136→    setIsPlaying(status.isPlaying);
   137→    setDuration(status.durationMillis || 0);
   138→    setPlaybackPosition(status.positionMillis || 0);
   139→    setIsLoading(false);
   140→    
   141→    if (status.didJustFinish) {
   142→      setIsPlaying(false);
   143→      setPlaybackPosition(0);
   144→    }
   145→  };
   146→
   147→  const togglePlayPause = async () => {
   148→    if (!videoRef.current) return;
   149→    
   150→    if (isPlaying) {
   151→      await videoRef.current.pauseAsync();
   152→    } else {
   153→      await videoRef.current.playAsync();
   154→    }
   155→  };
   156→
   157→  const toggleMute = async () => {
   158→    if (!videoRef.current) return;
   159→    await videoRef.current.setIsMutedAsync(!isMuted);
   160→    setIsMuted(!isMuted);
   161→  };
   162→
   163→  const seekForward = async () => {
   164→    if (!videoRef.current) return;
   165→    const newPosition = Math.min(playbackPosition + 10000, duration);
   166→    await videoRef.current.setPositionAsync(newPosition);
   167→  };
   168→
   169→  const seekBackward = async () => {
   170→    if (!videoRef.current) return;
   171→    const newPosition = Math.max(playbackPosition - 10000, 0);
   172→    await videoRef.current.setPositionAsync(newPosition);
   173→  };
   174→
   175→  const formatTime = (millis: number): string => {
   176→    const totalSeconds = Math.floor(millis / 1000);
   177→    const minutes = Math.floor(totalSeconds / 60);
   178→    const seconds = totalSeconds % 60;
   179→    return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
   180→  };
   181→
   182→  const handleShare = async () => {
   183→    if (!video) return;
   184→    try {
   185→      await Share.share({
   186→        message: `Regardez cette vidéo sur AlgeriaTrade: ${video.title}`,
   187→        url: video.url,
   188→      });
   189→    } catch (error) {
   190→      console.error('[VideoGallery] Error sharing:', error);
   191→    }
   192→  };
   193→
   194→  if (!video) return null;
   195→
   196→  return (
   197→    <Modal
   198→      visible={visible}
   199→      animationType="fade"
   200→      onRequestClose={onClose}
   201→      supportedOrientations={['portrait', 'landscape']}
   202→    >
   203→      <View style={styles.playerContainer}>
   204→        {/* Video */}
   205→        <View style={styles.videoWrapper}>
   206→          <Video
   207→            ref={videoRef}
   208→            source={{ uri: video.url }}
   209→            style={styles.videoPlayer}
   210→            resizeMode={ResizeMode.CONTAIN}
   211→            shouldPlay={false}
   212→            isLooping={false}
   213→            isMuted={isMuted}
   214→            onPlaybackStatusUpdate={handlePlaybackStatusUpdate}
   215→          />
   216→          
   217→          {/* Loading Overlay */}
   218→          {isLoading && (
   219→            <View style={styles.loadingOverlay}>
   220→              <ActivityIndicator size="large" color={Colors.white} />
   221→            </View>
   222→          )}
   223→
   224→          {/* Top Controls */}
   225→          <View style={styles.topControls}>
   226→            <TouchableOpacity style={styles.controlButton} onPress={onClose}>
   227→              <Ionicons name="close" size={28} color={Colors.white} />
   228→            </TouchableOpacity>
   229→            
   230→            <TouchableOpacity style={styles.controlButton} onPress={handleShare}>
   231→              <Ionicons name="share-social-outline" size={24} color={Colors.white} />
   232→            </TouchableOpacity>
   233→          </View>
   234→
   235→          {/* Center Play Button */}
   236→          {!isPlaying && !isLoading && (
   237→            <TouchableOpacity style={styles.centerPlayButton} onPress={togglePlayPause}>
   238→              <Ionicons name="play" size={48} color={Colors.white} />
   239→            </TouchableOpacity>
   240→          )}
   241→        </View>
   242→
   243→        {/* Video Info */}
   244→        <View style={styles.videoInfoContainer}>
   245→          <Text style={styles.videoTitle}>{video.title}</Text>
   246→          
   247→          {video.description && (
   248→            <Text style={styles.videoDescription} numberOfLines={2}>
   249→              {video.description}
   250→            </Text>
   251→          )}
   252→
   253→          <View style={styles.videoMetaRow}>
   254→            <View style={styles.metaItem}>
   255→              <Ionicons name="eye-outline" size={16} color={Colors.textSecondary} />
   256→              <Text style={styles.metaText}>
   257→                {(video.viewCount / 1000).toFixed(1)}K vues
   258→              </Text>
   259→            </View>
   260→            <View style={styles.metaItem}>
   261→              <Ionicons name="time-outline" size={16} color={Colors.textSecondary} />
   262→              <Text style={styles.metaText}>
   263→                {formatTime(video.duration * 1000)}
   264→              </Text>
   265→            </View>
   266→            {VIDEO_TYPE_CONFIG[video.type] && (
   267→              <View style={[styles.typeBadge, { backgroundColor: VIDEO_TYPE_CONFIG[video.type].color + '20' }]}>
   268→                <Ionicons 
   269→                  name={VIDEO_TYPE_CONFIG[video.type].icon as any} 
   270→                  size={12} 
   271→                  color={VIDEO_TYPE_CONFIG[video.type].color} 
   272→                />
   273→                <Text style={[styles.typeBadgeText, { color: VIDEO_TYPE_CONFIG[video.type].color }]}>
   274→                  {VIDEO_TYPE_CONFIG[video.type].label}
   275→                </Text>
   276→              </View>
   277→            )}
   278→          </View>
   279→        </View>
   280→
   281→        {/* Progress Bar */}
   282→        <View style={styles.progressContainer}>
   283→          <View style={styles.progressBarBackground}>
   284→            <View 
   285→              style={[
   286→                styles.progressBarFill,
   287→                { width: `${duration > 0 ? (playbackPosition / duration) * 100 : 0}%` }
   288→              ]} 
   289→            />
   290→          </View>
   291→          <View style={styles.timeLabels}>
   292→            <Text style={styles.timeLabel}>{formatTime(playbackPosition)}</Text>
   293→            <Text style={styles.timeLabel}>{formatTime(duration)}</Text>
   294→          </View>
   295→        </View>
   296→
   297→        {/* Bottom Controls */}
   298→        <View style={styles.bottomControls}>
   299→          <TouchableOpacity style={styles.bottomControlButton} onPress={seekBackward}>
   300→            <Ionicons name="play-back" size={24} color={Colors.text} />
   301→            <Text style={styles.controlLabel}>10s</Text>
   302→          </TouchableOpacity>
   303→
   304→          <TouchableOpacity style={styles.playPauseButton} onPress={togglePlayPause}>
   305→            <Ionicons 
   306→              name={isPlaying ? 'pause' : 'play'} 
   307→              size={32} 
   308→              color={Colors.white} 
   309→            />
   310→          </TouchableOpacity>
   311→
   312→          <TouchableOpacity style={styles.bottomControlButton} onPress={seekForward}>
   313→            <Ionicons name="play-forward" size={24} color={Colors.text} />
   314→            <Text style={styles.controlLabel}>10s</Text>
   315→          </TouchableOpacity>
   316→
   317→          <TouchableOpacity style={styles.bottomControlButton} onPress={toggleMute}>
   318→            <Ionicons 
   319→              name={isMuted ? 'volume-mute' : 'volume-high'} 
   320→              size={24} 
   321→              color={Colors.text} 
   322→            />
   323→          </TouchableOpacity>
   324→
   325→          {onDownload && (
   326→            <TouchableOpacity style={styles.bottomControlButton} onPress={onDownload}>
   327→              <Ionicons name="download-outline" size={24} color={Colors.text} />
   328→            </TouchableOpacity>
   329→          )}
   330→
   331→          {onCast && (
   332→            <TouchableOpacity style={styles.bottomControlButton} onPress={onCast}>
   333→              <Ionicons name="tv-outline" size={24} color={Colors.text} />
   334→            </TouchableOpacity>
   335→          )}
   336→        </View>
   337→      </View>
   338→    </Modal>
   339→  );
   340→}

   342→// Virtual Tour Viewer
   343→function VirtualTourViewer({
   344→  tour,
   345→  visible,
   346→  onClose,
   347→}: {
   348→  tour: VirtualTour | null;
   349→  visible: boolean;
   350→  onClose: () => void;
   351→}) {
   352→  const [currentSceneIndex, setCurrentSceneIndex] = useState(0);
   353→  const [showHotspots, setShowHotspots] = useState(true);
   354→
   355→  if (!tour) return null;
   356→
   357→  // In a real implementation, this would use a 360° viewer library
   358→  // like react-native-360 or similar
   359→  return (
   360→    <Modal
   361→      visible={visible}
   362→      animationType="fade"
   363→      onRequestClose={onClose}
   364→    >
   365→      <View style={styles.tourContainer}>
   366→        {/* Header */}
   367→        <View style={styles.tourHeader}>
   368→          <TouchableOpacity style={styles.tourBackButton} onPress={onClose}>
   369→            <Ionicons name="arrow-back" size={24} color={Colors.text} />
   370→          </TouchableOpacity>
   371→          <View style={styles.tourHeaderInfo}>
   372→            <Text style={styles.tourTitle}>{tour.title}</Text>
   373→            <Text style={styles.tourSubtitle}>
   374→              {TOUR_TYPE_CONFIG[tour.type]?.label || 'Visite virtuelle'}
   375→            </Text>
   376→          </View>
   377→          <TouchableOpacity 
   378→            style={styles.hotspotToggle}
   379→            onPress={() => setShowHotspots(!showHotspots)}
   380→          >
   381→            <Ionicons 
   382→              name={showHotspots ? 'location' : 'location-outline'} 
   383→              size={24} 
   384→              color={showHotspots ? Colors.primary : Colors.textTertiary} 
   385→            />
   386→          </TouchableOpacity>
   387→        </View>
   388→
   389→        {/* 360° View Placeholder */}
   390→        <View style={styles.tourViewer}>
   391→          <Image 
   392→            source={{ uri: tour.coverImage }} 
   393→            style={styles.tourCoverImage}
   394→            resizeMode="cover"
   395→          />
   396→          
   397→          {/* 360° Indicator */}
   398→          <View style={styles.indicator360}>
   399→            <Ionicons name="globe-outline" size={20} color={Colors.white} />
   400→            <Text style={styles.indicator360Text}>Vue 360°</Text>
   401→          </View>
   402→
   403→          {/* Hotspots */}
   404→          {showHotspots && tour.hotspots.map(hotspot => (
   405→            <TouchableOpacity
   406→              key={hotspot.id}
   407→              style={[
   408→                styles.hotspot,
   409→                { left: hotspot.position.x, top: hotspot.position.y }
   410→              ]}
   411→              activeOpacity={0.8}
   412→            >
   413→              <View style={styles.hotspotOuter}>
   414→                <View style={styles.hotspotInner} />
   415→              </View>
   416→              <Text style={styles.hotspotLabel}>{hotspot.title}</Text>
   417→            </TouchableOpacity>
   418→          ))}
   419→
   420→          {/* Navigation Arrows */}
   421→          <TouchableOpacity 
   422→            style={[styles.navArrow, styles.navArrowLeft]}
   423→            onPress={() => setCurrentSceneIndex(Math.max(0, currentSceneIndex - 1))}
   424→          >
   425→            <Ionicons name="chevron-back" size={32} color={Colors.white} />
   426→          </TouchableOpacity>
   427→          
   428→          <TouchableOpacity 
   429→            style={[styles.navArrow, styles.navArrowRight]}
   430→            onPress={() => setCurrentSceneIndex(currentSceneIndex + 1)}
   431→          >
   432→            <Ionicons name="chevron-forward" size={32} color={Colors.white} />
   433→          </TouchableOpacity>
   434→        </View>
   435→
   436→        {/* Tour Info */}
   437→        <View style={styles.tourInfo}>
   438→          {tour.description && (
   439→            <Text style={styles.tourDescription}>{tour.description}</Text>
   440→          )}
   441→          
   442→          <View style={styles.tourStats}>
   443→            <View style={styles.statItem}>
   444→              <Ionicons name="eye-outline" size={18} color={Colors.textSecondary} />
   445→              <Text style={styles.statText}>
   446→                {(tour.viewCount / 1000).toFixed(1)}K vues
   447→              </Text>
   448→            </View>
   449→            {tour.duration && (
   450→              <View style={styles.statItem}>
   451→                <Ionicons name="time-outline" size={18} color={Colors.textSecondary} />
   452→                <Text style={styles.statText}>~{tour.duration} min</Text>
   453→              </View>
   454→            )}
   455→            <View style={styles.statItem}>
   456→              <Ionicons name="navigate-outline" size={18} color={Colors.textSecondary} />
   457→              <Text style={styles.statText}>{tour.hotspots.length} points d'intérêt</Text>
   458→            </View>
   459→          </View>
   460→
   461→          {/* Scene Indicators */}
   462→          <View style={styles.sceneIndicators}>
   463→            {[1, 2, 3, 4, 5].map((_, index) => (
   464→              <View 
   465→                key={index}
   466→                style={[
   467→                  styles.sceneDot,
   468→                  index === currentSceneIndex && styles.sceneDotActive
   469→                ]}
   470→              />
   471→            ))}
   472→          </View>
   473→        </View>
   474→      </View>
   475→    </Modal>
   476→  );
   477→}

   484→  videos,

   491→  autoPlay = false,
   492→}: VideoGalleryProps) {

   495→  const [activeTab, setActiveTab] = useState<'videos' | 'tours'>('videos');

   500→    } else {

   502→    }

   506→    if (onTourPress) {
   507→      onTourPress(tour);
   508→    } else {
   509→      setSelectedTour(tour);
   510→    }
   511→  };
   512→
   513→  const handleDownload = async (video: ProductVideo) => {
   514→    Alert.alert(
   515→      'Téléchargement',
   516→      `Voulez-vous télécharger "${video.title}" pour le visionnage hors ligne ?`,
   517→      [
   518→        { text: 'Annuler', style: 'cancel' },
   519→        { text: 'Télécharger', onPress: () => {
   520→          // In a real app, this would download using FileSystem or similar
   521→          Alert.alert('Succès', 'Le téléchargement va commencer.');
   522→        }}
   523→      ]
   524→    );
   525→  };
   526→
   527→  const handleCast = () => {
   528→    Alert.alert(
   529→      'Casting',
   530→      'Sélectionnez un appareil pour caster la vidéo.',
   531→      [
   532→        { text: 'Annuler', style: 'cancel' },
   533→        { text: 'Chromecast', onPress: () => {
   534→          Alert.alert('Connexion', 'Connexion à Chromecast...');
   535→        }},
   536→        { text: 'Apple TV', onPress: () => {
   537→          Alert.alert('Connexion', 'Connexion à Apple TV...');
   538→        }}
   539→      ]
   540→    );
   541→  };

   543→  const renderVideoItem = ({ item }: { item: ProductVideo }) => {
   544→    const typeConfig = VIDEO_TYPE_CONFIG[item.type];
   545→    
   546→    if (compact) {
   547→      return (
   548→        <TouchableOpacity
   549→          style={styles.compactVideoItem}
   550→          onPress={() => handleVideoPress(item)}
   551→          activeOpacity={0.7}
   552→        >
   553→          <View style={styles.compactThumbnailContainer}>
   554→            <Image source={{ uri: item.thumbnailUrl }} style={styles.compactThumbnail} />
   555→            <View style={styles.playOverlay}>
   556→              <Ionicons name="play-circle" size={36} color={Colors.white} />
   557→            </View>
   558→            <View style={styles.durationBadge}>
   559→              <Text style={styles.durationText}>
   560→                {Math.floor(item.duration / 60)}:{(item.duration % 60).toString().padStart(2, '0')}
   561→              </Text>
   562→            </View>
   563→          </View>
   564→          <View style={styles.compactInfo}>
   565→            <Text style={styles.compactTitle} numberOfLines={2}>{item.title}</Text>
   566→            <Text style={styles.compactViews}>
   567→              {(item.viewCount / 1000).toFixed(1)}K vues
   568→            </Text>
   569→          </View>
   570→        </TouchableOpacity>
   571→      );
   572→    }
   573→
   574→    return (
   575→      <TouchableOpacity
   576→        style={styles.videoCard}
   577→        onPress={() => handleVideoPress(item)}
   578→        activeOpacity={0.7}
   579→      >
   580→        <View style={styles.thumbnailContainer}>
   581→          <Image source={{ uri: item.thumbnailUrl }} style={styles.thumbnail} />
   582→          
   583→          {/* Play Button Overlay */}
   584→          <View style={styles.playOverlay}>
   585→            <Ionicons name="play-circle" size={48} color={Colors.white} />
   586→          </View>
   587→
   588→          {/* Duration Badge */}
   589→          <View style={styles.durationBadge}>
   590→            <Text style={styles.durationText}>
   591→              {Math.floor(item.duration / 60)}:{(item.duration % 60).toString().padStart(2, '0')}
   592→            </Text>
   593→          </View>
   594→
   595→          {/* Primary Badge */}
   596→          {item.isPrimary && (
   597→            <View style={styles.primaryBadge}>
   598→              <Ionicons name="star" size={12} color="#F59E0B" />

   602→
