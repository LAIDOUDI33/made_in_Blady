     1→'use client'
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react'
     4→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
     5→import { Button } from '@/components/ui/button'
     6→import { Badge } from '@/components/ui/badge'
     7→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
     8→import { 
     9→  Select, 
    10→  SelectContent, 
    11→  SelectItem, 
    12→  SelectTrigger, 
    13→  SelectValue 
    14→} from '@/components/ui/select'
    15→import { Input } from '@/components/ui/input'
    16→import { Separator } from '@/components/ui/separator'
    17→import {
    18→  Tooltip,
    19→  TooltipContent,
    20→  TooltipProvider,
    21→  TooltipTrigger,
    22→} from '@/components/ui/tooltip'
    23→import {
    24→  Dialog,
    25→  DialogContent,
    26→  DialogDescription,
    27→  DialogHeader,
    28→  DialogTitle,
    29→  DialogTrigger,
    30→} from '@/components/ui/dialog'
    31→import { useToast } from '@/hooks/use-toast'
    32→import {
    33→  Bitcoin,
    34→  Hexagon,
    35→  DollarSign,
    36→  Copy,
    37→  ExternalLink,
    38→  Clock,
    39→  AlertCircle,
    40→  CheckCircle2,
    41→  Loader2,
    42→  Info,
    43→  QrCode,
    44→  RefreshCw,
    45→  Shield,
    46→  Zap,
    47→  HelpCircle,
    48→} from 'lucide-react'
    49→import { CryptoCurrency, CRYPTO_INFO, getNetworkFeeInfo } from '@/lib/payments/crypto'
    50→
    51→interface CryptoPaymentFormProps {
    52→  orderId: string
    53→  amountDZD: number
    54→  onPaymentComplete?: (paymentId: string) => void
    55→  onPaymentError?: (error: string) => void
    56→}
    57→
    58→interface PaymentData {
    59→  paymentId: string
    60→  depositAddress: string
    61→  expectedAmount: number
    62→  cryptoCurrency: CryptoCurrency
    63→  exchangeRate: number
    64→  expiresAt: string
    65→  qrCodeData: string
    66→}
    67→
    68→interface PaymentStatus {
    69→  status: 'PENDING' | 'PARTIAL' | 'CONFIRMED' | 'EXPIRED' | 'OVERPAID'
    70→  confirmations: number
    71→  requiredConfirmations: number
    72→  actualAmount?: number
    73→  txHash?: string
    74→}
    75→
    76→// Crypto currency icons mapping
    77→const CRYPTO_ICONS: Record<CryptoCurrency, typeof Bitcoin> = {
    78→  BTC: Bitcoin,
    79→  ETH: Hexagon,
    80→  USDT: DollarSign,
    81→  USDC: DollarSign,
    82→}
    83→
    84→export default function CryptoPaymentForm({
    85→  orderId,
    86→  amountDZD,
    87→  onPaymentComplete,
    88→  onPaymentError,
    89→}: CryptoPaymentFormProps) {
    90→  const [selectedCrypto, setSelectedCrypto] = useState<CryptoCurrency>('USDT')
    91→  const [paymentData, setPaymentData] = useState<PaymentData | null>(null)
    92→  const [paymentStatus, setPaymentStatus] = useState<PaymentStatus | null>(null)
    93→  const [exchangeRate, setExchangeRate] = useState<number | null>(null)
    94→  const [isLoading, setIsLoading] = useState(false)
    95→  const [isCreating, setIsCreating] = useState(false)
    96→  const [timeRemaining, setTimeRemaining] = useState<number>(0)
    97→  const [copied, setCopied] = useState(false)
    98→  const [walletAddress, setWalletAddress] = useState('')
    99→  const [showQRCode, setShowQRCode] = useState(true)
   100→  
   101→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   102→  const { toast } = useToast()
   103→
   104→  // Fetch exchange rate when crypto selection changes
   105→  const fetchExchangeRate = useCallback(async (crypto: CryptoCurrency) => {
   106→    setIsLoading(true)
   107→    try {
   108→      const response = await fetch('/api/payments/crypto/rates')
   109→      const data = await response.json()
   110→      
   111→      if (data.success && data.data.rates[crypto]) {
   112→        setExchangeRate(data.data.rates[crypto])
   113→      }
   114→    } catch (error) {
   115→      console.error('Failed to fetch exchange rate:', error)
   116→    } finally {
   117→      setIsLoading(false)
   118→    }
   119→  }, [])
   120→
   121→  useEffect(() => {
   122→    fetchExchangeRate(selectedCrypto)
   123→  }, [selectedCrypto, fetchExchangeRate])
   124→
   125→  // Create crypto payment
   126→  const handleCreatePayment = async () => {
   127→    setIsCreating(true)
   128→    try {
   129→      const response = await fetch('/api/payments/crypto', {
   130→        method: 'POST',
   131→        headers: { 'Content-Type': 'application/json' },
   132→        body: JSON.stringify({
   133→          orderId,
   134→          amount: amountDZD,
   135→          currency: selectedCrypto,
   136→          buyerWalletAddress: walletAddress || undefined,
   137→        }),
   138→      })
   139→
   140→      const data = await response.json()
   141→
   142→      if (data.success) {
   143→        setPaymentData(data.data)
   144→        startCountdown(new Date(data.data.expiresAt))
   145→        startStatusMonitoring(data.data.paymentId)
   146→        
   147→        toast({
   148→          title: 'Payment Created',
   149→          description: `Send ${data.data.expectedAmount} ${selectedCrypto} to the address below.`,
   150→        })
   151→      } else {
   152→        throw new Error(data.error || 'Failed to create payment')
   153→      }
   154→    } catch (error) {
   155→      console.error('Error creating payment:', error)
   156→      const errorMessage = error instanceof Error ? error.message : 'Unknown error'
   157→      onPaymentError?.(errorMessage)
   158→      toast({
   159→        title: 'Error',
   160→        description: errorMessage,
   161→        variant: 'destructive',
   162→      })
   163→    } finally {
   164→      setIsCreating(false)
   165→    }
   166→  }
   167→
   168→  // Countdown timer
   169→  const startCountdown = (expiresAt: Date) => {
   170→    const updateTimer = () => {
   171→      const remaining = expiresAt.getTime() - Date.now()
   172→      setTimeRemaining(Math.max(0, remaining))
   173→      
   174→      if (remaining <= 0) {
   175→        if (intervalRef.current) clearInterval(intervalRef.current)
   176→        setPaymentStatus(prev => prev ? { ...prev, status: 'EXPIRED' } : null)
   177→      }
   178→    }
   179→
   180→    updateTimer()
   181→    intervalRef.current = setInterval(updateTimer, 1000)
   182→  }
   183→
   184→  // Monitor payment status
   185→  const startStatusMonitoring = async (paymentId: string) => {
   186→    const checkStatus = async () => {
   187→      try {
   188→        const response = await fetch(`/api/payments/crypto/${paymentId}/status`)
   189→        const data = await response.json()
   190→
   191→        if (data.success) {
   192→          setPaymentStatus({
   193→            status: data.data.status,
   194→            confirmations: data.data.confirmations,
   195→            requiredConfirmations: data.data.requiredConfirmations,
   196→            actualAmount: data.data.actualAmount,
   197→            txHash: data.data.txHash,
   198→          })
   199→
   200→          // Notify on confirmation
   201→          if (data.data.status === 'CONFIRMED') {
   202→            onPaymentComplete?.(paymentId)
   203→            toast({
   204→              title: 'Payment Confirmed! 🎉',
   205→              description: 'Your cryptocurrency payment has been confirmed.',
   206→            })
   207→            if (intervalRef.current) clearInterval(intervalRef.current)
   208→          }
   209→
   210→          // Stop if expired or final state
   211→          if (['CONFIRMED', 'EXPIRED'].includes(data.data.status)) {
   212→            if (intervalRef.current) clearInterval(intervalRef.current)
   213→          }
   214→        }
   215→      } catch (error) {
   216→        console.error('Error checking status:', error)
   217→      }
   218→    }
   219→
   220→    // Initial check
   221→    await checkStatus()
   222→    
   223→    // Poll every 10 seconds
   224→    const statusInterval = setInterval(checkStatus, 10000)
   225→    
   226→    // Store reference for cleanup
   227→    intervalRef.current = statusInterval
   228→  }
   229→
   230→  // Copy address to clipboard
   231→  const copyAddress = async () => {
   232→    if (!paymentData) return
   233→    
   234→    try {
   235→      await navigator.clipboard.writeText(paymentData.depositAddress)
   236→      setCopied(true)
   237→      toast({
   238→        title: 'Copied!',
   239→        description: 'Address copied to clipboard.',
   240→      })
   241→      setTimeout(() => setCopied(false), 2000)
   242→    } catch (error) {
   243→      console.error('Failed to copy:', error)
   244→    }
   245→  }
   246→
   247→  // Format time remaining
   248→  const formatTime = (ms: number): string => {
   249→    const minutes = Math.floor(ms / 60000)
   250→    const seconds = Math.floor((ms % 60000) / 1000)
   251→    return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`
   252→  }
   253→
   254→  // Get status color
   255→  const getStatusColor = (status: string) => {
   256→    switch (status) {
   257→      case 'PENDING': return 'bg-yellow-500'
   258→      case 'CONFIRMED': return 'bg-green-500'
   259→      case 'EXPIRED': return 'bg-red-500'
   260→      case 'PARTIAL': return 'bg-orange-500'
   261→      case 'OVERPAID': return 'bg-blue-500'
   262→      default: return 'bg-gray-500'
   263→    }
   264→  }
   265→
   266→  // Cleanup on unmount
   267→  useEffect(() => {
   268→    return () => {
   269→      if (intervalRef.current) clearInterval(intervalRef.current)
   270→    }
   271→  }, [])
   272→
   273→  const currentCryptoInfo = CRYPTO_INFO[selectedCrypto]
   274→  const CryptoIcon = CRYPTO_ICONS[selectedCrypto]
   275→  const networkFee = getNetworkFeeInfo(selectedCrypto)
   276→
   277→  return (
   278→    <TooltipProvider>
   279→      <div className="space-y-6">
   280→        {/* Header */}
   281→        <Card>
   282→          <CardHeader>
   283→            <CardTitle className="flex items-center gap-2">
   284→              <Zap className="h-5 w-5 text-yellow-500" />
   285→              Cryptocurrency Payment
   286→            </CardTitle>
   287→            <CardDescription>
   288→              Pay with Bitcoin, Ethereum, Tether, or USD Coin
   289→            </CardDescription>
   290→          </CardHeader>
   291→        </Card>
   292→
   293→        {!paymentData ? (
   294→          /* Step 1: Select Currency & Create Payment */
   295→          <Card>
   296→            <CardContent className="pt-6 space-y-6">
   297→              {/* Amount Display */}
   298→              <div className="text-center p-4 bg-muted rounded-lg">
   299→                <p className="text-sm text-muted-foreground">Amount to Pay</p>
   300→                <p className="text-3xl font-bold">{amountDZD.toLocaleString('fr-DZ')} DZD</p>
   301→              </div>
   302→
   303→              {/* Crypto Selector */}
   304→              <div className="space-y-3">
   305→                <label className="text-sm font-medium flex items-center gap-2">
   306→                  Select Cryptocurrency
   307→                  <Tooltip>
   308→                    <TooltipTrigger asChild>
   309→                      <HelpCircle className="h-4 w-4 text-muted-foreground cursor-help" />
   310→                    </TooltipTrigger>
   311→                    <TooltipContent side="right" className="max-w-xs">
   312→                      <p>{currentCryptoInfo.description.en}</p>
   313→                    </TooltipContent>
   314→                  </Tooltip>
   315→                </label>
   316→
   317→                <Select value={selectedCrypto} onValueChange={(v) => setSelectedCrypto(v as CryptoCurrency)}>
   318→                  <SelectTrigger>
   319→                    <SelectValue placeholder="Select cryptocurrency" />
   320→                  </SelectTrigger>
   321→                  <SelectContent>
   322→                    {(Object.keys(CRYPTO_INFO) as CryptoCurrency[]).map((crypto) => {
   323→                      const info = CRYPTO_INFO[crypto]
   324→                      const Icon = CRYPTO_ICONS[crypto]
   325→                      return (
   326→                        <SelectItem key={crypto} value={crypto}>
   327→                          <div className="flex items-center gap-2">
   328→                            <Icon className="h-4 w-4" style={{ color: info.color }} />
   329→                            <span>{info.fullName}</span>
   330→                          </div>
   331→                        </SelectItem>
   332→                      )
   333→                    })}
   334→                  </SelectContent>
   335→                </Select>
   336→
   337→                {/* Selected Crypto Info */}
   338→                <div className="flex items-center gap-3 p-3 rounded-lg border" style={{ borderColor: currentCryptoInfo.color + '40' }}>
   339→                  <div 
   340→                    className="p-2 rounded-full"
   341→                    style={{ backgroundColor: currentCryptoInfo.color + '20' }}
   342→                  >
   343→                    <CryptoIcon className="h-6 w-6" style={{ color: currentCryptoInfo.color }} />
   344→                  </div>
   345→                  <div className="flex-1">
   346→                    <p className="font-medium">{currentCryptoInfo.name}</p>
   347→                    <p className="text-sm text-muted-foreground line-clamp-1">
   348→                      {currentCryptoInfo.description.en}
   349→                    </p>
   350→                  </div>
   351→                </div>
   352→              </div>
   353→
   354→              {/* Exchange Rate Display */}
   355→              {exchangeRate && (
   356→                <div className="flex items-center justify-between p-3 bg-blue-50 dark:bg-blue-950 rounded-lg">
   357→                  <span className="text-sm text-muted-foreground">Exchange Rate</span>
   358→                  <div className="flex items-center gap-2">
   359→                    <span className="font-mono font-medium">
   360→                      1 {selectedCrypto} = {Math.round(exchangeRate).toLocaleString()} DZD
   361→                    </span>
   362→                    <Button
   363→                      variant="ghost"
   364→                      size="icon"
   365→                      className="h-6 w-6"
   366→                      onClick={() => fetchExchangeRate(selectedCrypto)}
   367→                    >
   368→                      <RefreshCw className={`h-3 w-3 ${isLoading ? 'animate-spin' : ''}`} />
   369→                    </Button>
   370→                  </div>
   371→                </div>
   372→              )}
   373→
   374→              {/* Estimated Amount */}
   375→              {exchangeRate && (
   376→                <div className="text-center p-4 rounded-lg border-2 border-dashed" style={{ borderColor: currentCryptoInfo.color }}>
   377→                  <p className="text-sm text-muted-foreground">You'll send approximately</p>
   378→                  <p className="text-2xl font-bold" style={{ color: currentCryptoInfo.color }}>
   379→                    {(amountDZD / exchangeRate).toFixed(8)} {selectedCrypto}
   380→                  </p>
   381→                </div>
   382→              )}
   383→
   384→              {/* Optional Wallet Address for Refund */}
   385→              <div className="space-y-2">
   386→                <label className="text-sm font-medium flex items-center gap-2">
   387→                  Your Wallet Address (Optional)
   388→                  <Tooltip>
   389→                    <TooltipTrigger asChild>
   390→                      <HelpCircle className="h-4 w-4 text-muted-foreground cursor-help" />
   391→                    </TooltipTrigger>
   392→                    <TooltipContent side="right">
   393→                      <p>Provide your wallet address for potential refunds.</p>
   394→                    </TooltipContent>
   395→                  </Tooltip>
   396→                </label>
   397→                <Input
   398→                  placeholder="0x... or bc1..."
   399→                  value={walletAddress}
   400→                  onChange={(e) => setWalletAddress(e.target.value)}
   401→                />
   402→              </div>
   403→
   404→              {/* Network Fee Disclosure */}
   405→              <div className="p-3 bg-amber-50 dark:bg-amber-950 rounded-lg space- y-2">
   406→                <div className="flex items-start gap-2">
   407→                  <AlertCircle className="h-4 w-4 text-amber-600 mt-0.5 shrink-0" />
   408→                  <div className="text-sm text-amber-800 dark:text-amber-200">
   409→                    <p className="font-medium">Network Fee Notice</p>
   410→                    <p>
   411→                      Additional network fees apply (~{networkFee.estimate} {networkFee.currency}). 
   412→                      This is paid to miners/validators, not AlgeriaTrade.
   413→                    </p>
   414→                  </div>
   415→                </div>
   416→              </div>
   417→
   418→              {/* Create Payment Button */}
   419→              <Button
   420→                onClick={handleCreatePayment}
   421→                disabled={isCreating || !exchangeRate}
   422→                className="w-full"
   423→                size="lg"
   424→              >
   425→                {isCreating ? (
   426→                  <>
   427→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   428→                    Creating Payment...
   429→                  </>
   430→                ) : (
   431→                  <>
   432→                    <Shield className="mr-2 h-4 w-4" />
   433→                    Create Payment Address
   434→                  </>
   435→                )}
   436→              </Button>
   437→            </CardContent>
   438→          </Card>
   439→        ) : (
   440→          /* Step 2: Payment Details & QR Code */
   441→          <div className="space-y-6">
   442→            {/* Status Badge */}
   443→            <Card>
   444→              <CardContent className="pt-6">
   445→                <div className="flex items-center justify-between">
   446→                  <div className="flex items-center gap-3">
   447→                    <Badge 
   448→                      variant="secondary" 
   449→                      className={`${getStatusColor(paymentStatus?.status || 'PENDING')} text-white`}
   450→                    >
   451→                      {paymentStatus?.status || 'PENDING'}
   452→                    </Badge>
   453→                    {paymentStatus?.status === 'PENDING' && (
   454→                      <div className="flex items-center gap-1 text-orange-600">
   455→                        <Clock className="h-4 w-4" />
   456→                        <span className="font-mono font-bold">
   457→                          {formatTime(timeRemaining)}
   458→                        </span>
   459→                      </div>
   460→                    )}
   461→                  </div>
   462→                  
   463→                  {paymentStatus?.txHash && (
   464→                    <a
   465→                      href={`${CRYPTO_INFO[paymentData.cryptoCurrency].explorerUrl}${paymentStatus.txHash}`}
   466→                      target="_blank"
   467→                      rel="noopener noreferrer"
   468→                      className="flex items-center gap-1 text-sm text-blue-600 hover:underline"
   469→                    >
   470→                      View Transaction
   471→                      <ExternalLink className="h-3 w-3" />
   472→                    </a>
   473→                  )}
   474→                </div>
   475→
   476→                {/* Confirmations Progress */}
   477→                {paymentStatus && paymentStatus.confirmations > 0 && (
   478→                  <div className="mt-4 space-y-2">
   479→                    <div className="flex justify-between text-sm">
   480→                      <span>Confirmations</span>
   481→                      <span>
   482→                        {paymentStatus.confirmations} / {paymentStatus.requiredConfirmations}
   483→                      </span>
   484→                    </div>
   485→                    <div className="w-full bg-gray-200 rounded-full h-2">
   486→                      <div 
   487→                        className="bg-green-500 h-2 rounded-full transition-all duration-500"
   488→                        style={{ 
   489→                          width: `${Math.min(100, (paymentStatus.confirmations / paymentStatus.requiredConfirmations) * 100)}%` 
   490→                        }}
   491→                      />
   492→                    </div>
   493→                  </div>
   494→                )}
   495→              </CardContent>
   496→            </Card>
   497→
   498→            {/* QR Code and Address */}
   499→            <Tabs defaultValue="qrcode" className="w-full">
   500→              <TabsList className="grid w-full grid-cols-2">
   501→                <TabsTrigger value="qrcode" className="flex items-center gap-2">
   502→                  <QrCode className="h-4 w-4" />
   503→                  QR Code
   504→                </TabsTrigger>
   505→                <TabsTrigger value="address" className="flex items-center gap-2">
   506→                  <Copy className="h-4 w-4" />
   507→                  Address
   508→                </TabsTrigger>
   509→              </TabsList>
   510→
   511→              <TabsContent value="qrcode" className="space-y-4">
   512→                <Card>
   513→                  <CardContent className="pt-6 flex flex-col items-center">
   514→                    {/* QR Code Placeholder - In production, generate real QR code */}
   515→                    <div className="w-64 h-64 bg-white border-2 rounded-lg p-4 flex items-center justify-center mb-4">
   516→                      <div className="text-center space-y-2">
   517→                        <QrCode className="h-32 w-32 mx-auto text-gray-400" />
   518→                        <p className="text-xs text-gray-500">Scan with your wallet</p>
   519→                      </div>
   520→                    </div>
   521→                    
   522→                    <p className="text-center text-sm text-muted-foreground mb-4">
   523→                      Scan this QR code with your cryptocurrency wallet app
   524→                    </p>
   525→
   526→                    <Button
   527→                      variant="outline"
   528→                      onClick={() => setShowQRCode(!showQRCode)}
   529→                    >
   530→                      {showQRCode ? 'Hide' : 'Show'} QR Code
   531→                    </Button>
   532→                  </CardContent>
   533→                </Card>
   534→              </TabsContent>
   535→
   536→              <TabsContent value="address" className="space-y-4">
   537→                <Card>
   538→                  <CardContent className="pt-6 space-y-4">
   539→                    <div>
   540→                      <label className="text-sm font-medium">Deposit Address</label>
   541→                      <div className="flex gap-2 mt-1">
   542→                        <Input
   543→                          value={paymentData.depositAddress}
   544→                          readOnly
   545→                          className="font-mono text-sm"
   546→                        />
   547→                        <Button
   548→                          variant="outline"
   549→                          size="icon"
   550→                          onClick={copyAddress}
   551→                        >
   552→                          {copied ? (
   553→                            <CheckCircle2 className="h-4 w-4 text-green-600" />
   554→                          ) : (
   555→                            <Copy className="h-4 w-4" />
   556→                          )}
   557→                        </Button>
   558→                      </div>
   559→                    </div>
   560→
   561→                    <Separator />
   562→
   563→                    <div className="grid grid-cols-2 gap-4 text-sm">
   564→                      <div>
   565→                        <p className="text-muted-foreground">Amount</p>
   566→                        <p className="font-semibold font-mono">
   567→                          {paymentData.expectedAmount} {paymentData.cryptoCurrency}
   568→                        </p>
   569→                      </div>
   570→                      <div>
   571→                        <p className="text-muted-foreground">Currency</p>
   572→                        <p className="font-semibold">{CRYPTO_INFO[paymentData.cryptoCurrency].name}</p>
   573→                      </div>
   574→                    </div>
   575→                  </CardContent>
   576→                </Card>
   577→              </TabsContent>
   578→            </Tabs>
   579→
   580→            {/* Important Information */}
   581→            <Card className="border-yellow-200 bg-yellow-50 dark:bg-yellow-950">
   582→              <CardContent className="pt-6 space-y-3">
   583→                <h4 className="font-medium flex items-center gap-2">
   584→                  <AlertCircle className="h-4 w-4 text-yellow-600" />
   585→                  Important
   586→                </h4>
   587→                <ul className="text-sm text-yellow-800 dark:text-yellow-200 space-y-1 list-disc list-inside">
   588→                  <li>Send only <strong>{paymentData.cryptoCurrency}</strong> to this address</li>
   589→                  <li>Minimum confirmations required: {CRYPTO_INFO[paymentData.cryptoCurrency].name === 'Bitcoin' ? '3' : '12'}</li>
   590→                  <li>Payment expires in <strong>{formatTime(timeRemaining)}</strong></li>
   591→                  <li>Do not send from an exchange wallet unless it supports refunds</li>
   592→                </ul>
   593→              </CardContent>
   594→            </Card>
   595→
   596→            {/* Educational Tooltips */}
