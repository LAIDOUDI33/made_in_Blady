     3→
     4→import React, { useState, useEffect, useCallback } from 'react';
     5→import {
     6→  View,
     7→  Text,
     8→  StyleSheet,
     9→  ScrollView,
    10→  TouchableOpacity,
    11→  Alert,
    12→  ActivityIndicator,
    13→  Platform,
    14→  Modal,
    15→  TextInput,
    16→  Image,
    17→} from 'react-native';
    18→import { Ionicons } from '@expo/vector-icons';
    19→import { useNavigation, RouteProp, useRoute } from '@react-navigation/native';
    20→
    21→// Services
    22→import apiService from '../../services/api';
    23→
    24→// Constants
    25→import { Colors, FontFamily, FontSize, Spacing, BorderRadius, Shadows } from '../../utils/constants';
    26→
    27→// Types
    28→import { RootStackParamList } from '../../navigation';
    29→
    30→type EscrowDetailRouteProp = RouteProp<RootStackParamList, 'EscrowDetail'>;
    31→
    32→// ============================================
    33→// Types & Interfaces
    34→// ============================================
    35→
    36→export type EscrowStatus = 
    37→  | 'pending'
    38→  | 'funded'
    39→  | 'in_escrow'
    40→  | 'release_requested'
    41→  | 'released'
    42→  | 'refund_requested'
    43→  | 'refunded'
    44→  | 'disputed'
    45→  | 'cancelled';
    46→
    47→export type DisputeStatus = 
    48→  | 'open'
    49→  | 'investigating'
    50→  | 'mediation'
    51→  | 'arbitration'
    52→  | 'resolved'
    53→  | 'closed';
    54→
    55→export type DisputeReason =
    56→  | 'not_as_described'
    57→  | 'quality_issues'
    58→  | 'shipping_delay'
    59→  | 'damaged_goods'
    60→  | 'wrong_quantity'
    61→  | 'counterfeit'
    62→  | 'other';
    63→
    64→interface EscrowTimelineEvent {
    65→  id: string;
    66→  status: EscrowStatus;
    67→  label: string;
    68→  description?: string;
    69→  timestamp: Date;
    70→  completed: boolean;
    71→}
    72→
    73→interface DisputeMessage {
    74→  id: string;
    75→  senderId: string;
    76→  senderRole: 'buyer' | 'seller' | 'mediator';
    77→  content: string;
    78→  attachments?: Array<{ uri: string; name: string }>;
    79→  createdAt: Date;
    80→}
    81→
    82→interface Dispute {
    83→  id: string;
    84→  reason: DisputeReason;
    85→  status: DisputeStatus;
    86→  description: string;
    87→  requestedAmount: number;
    88→  evidencePhotos: Array<{ uri: string; name: string }>;
    89→  messages: DisputeMessage[];
    90→  mediatorId?: string;
    91→  mediatorName?: string;
    92→  deadline?: Date;
    93→  resolvedAt?: Date;
    94→  resolution?: string;
    95→}
    96→
    97→interface EscrowData {
    98→  id: string;
    99→  orderId: string;
   100→  accountNumber: string;
   101→  amount: number;
   102→  currency: string;
   103→  status: EscrowStatus;
   104→  buyerId: string;
   105→  sellerId: string;
   106→  fundedAt?: Date;
   107→  releaseRequestedAt?: Date;
   108→  releasedAt?: Date;
   109→  refundedAt?: Date;
   110→  autoReleaseDays: number;
   111→  platformFee: number;
   112→  dispute?: Dispute;
   113→  timeline: EscrowTimelineEvent[];
   114→}
   115→
   116→// ============================================
   117→// Constants
   118→// ============================================
   119→

   121→  pending: { label: 'En attente', color: Colors.textTertiary, icon: 'time-outline' },
   122→  funded: { label: 'Financé', color: Colors.info, icon: 'wallet-outline' },
   123→  in_escrow: { label: 'En séquestre', color: Colors.primary, icon: 'shield-checkmark-outline' },
   124→  release_requested: { label: 'Libération demandée', color: Colors.warning, icon: 'arrow-up-outline' },
   125→  released: { label: 'Libéré', color: Colors.success, icon: 'checkmark-circle' },
   126→  refund_requested: { label: 'Remboursement demandé', color: Colors.warning, icon: 'refresh-outline' },
   127→  refunded: { label: 'Remboursé', color: Colors.info, icon: 'arrow-down-outline' },
   128→  disputed: { label: 'En litige', color: Colors.error, icon: 'alert-circle-outline' },
   129→  cancelled: { label: 'Annulé', color: Colors.error, icon: 'close-circle-outline' },
   130→};
   131→
   132→const DISPUTE_REASONS: Array<{ value: DisputeReason; label: string; description: string }> = [
   133→  { value: 'not_as_described', label: 'Produit non conforme', description: 'Le produit ne correspond pas à la description' },
   134→  { value: 'quality_issues', label: 'Problèmes de qualité', description: 'Défauts ou problèmes de qualité constatés' },
   135→  { value: 'shipping_delay', label: 'Retard de livraison', description: 'La livraison est en retard significatif' },
   136→  { value: 'damaged_goods', label: 'Produits endommagés', description: 'Les produits ont été endommagés lors du transport' },
   137→  { value: 'wrong_quantity', label: 'Quantité incorrecte', description: 'La quantité reçue ne correspond pas' },
   138→  { value: 'counterfeit', label: 'Produit contrefait', description: 'Suspicion de contrefaçon' },
   139→  { value: 'other', label: 'Autre raison', description: 'Autre motif de litige' },
   140→];
   141→
   142→// ============================================
   143→// Main Component
   144→// ============================================
   145→
   146→export default function EscrowDetailScreen() {
   147→  const route = useRoute<EscrowDetailRouteProp>();
   148→  const navigation = useNavigation();
   149→  
   150→  const { escrowId } = route.params || {};
   151→  
   152→  // State
   153→  const [loading, setLoading] = useState(true);
   154→  const [escrowData, setEscrowData] = useState<EscrowData | null>(null);
   155→  const [showDisputeModal, setShowDisputeModal] = useState(false);
   156→  const [showFundModal, setShowFundModal] = useState(false);
   157→  const [showChatModal, setShowChatModal] = useState(false);
   158→  const [selectedReason, setSelectedReason] = useState<DisputeReason>('not_as_described');
   159→  const [disputeDescription, setDisputeDescription] = useState('');
   160→  const [fundAmount, setFundAmount] = useState('');
   161→  const [chatMessage, setChatMessage] = useState('');
   162→  const [actionLoading, setActionLoading] = useState(false);
   163→
   164→  // Load escrow data
   165→  const loadEscrowData = useCallback(async () => {
   166→    if (!escrowId) return;
   167→    
   168→    try {
   169→      const data = await apiService.getEscrowDetail(escrowId);
   170→      setEscrowData(data);
   171→    } catch (error) {
   172→      console.error('[EscrowDetailScreen] Error loading data:', error);
   173→      Alert.alert('Erreur', 'Impossible de charger les détails du compte séquestre.');
   174→    } finally {
   175→      setLoading(false);
   176→    }
   177→  }, [escrowId]);
   178→
   179→  useEffect(() => {
   180→    loadEscrowData();
   181→  }, [loadEscrowData]);
   182→
   183→  // Fund escrow account
   184→  const handleFundEscrow = async () => {
   185→    if (!fundAmount || parseFloat(fundAmount) <= 0) {
   186→      Alert.alert('Erreur', 'Veuillez entrer un montant valide.');
   187→      return;
   188→    }
   189→
   190→    setActionLoading(true);
   191→    
   192→    try {
   193→      await apiService.fundEscrow(escrowId!, parseFloat(fundAmount));
   194→      Alert.alert('Succès', 'Le compte a été financé avec succès.', [
   195→        { text: 'OK', onPress: () => {
   196→          setShowFundModal(false);
   197→          loadEscrowData();
   198→        }}
   199→      ]);
   200→    } catch (error) {
   201→      console.error('[EscrowDetailScreen] Error funding escrow:', error);
   202→      Alert.alert('Erreur', 'Impossible de financer le compte. Veuillez réessayer.');
   203→    } finally {
   204→      setActionLoading(false);
   205→    }
   206→  };
   207→
   208→  // Request release
   209→  const handleRequestRelease = async () => {
   210→    Alert.alert(
   211→      'Confirmer la libération',
   212→      'Êtes-vous sûr de vouloir demander la libération des fonds ? Cette action est irréversible.',
   213→      [
   214→        { text: 'Annuler', style: 'cancel' },
   215→        {
   216→          text: 'Confirmer',
   217→          style: 'destructive',
   218→          onPress: async () => {
   219→            setActionLoading(true);
   220→            try {
   221→              await apiService.requestRelease(escrowId!);
   222→              Alert.alert('Succès', 'La demande de libération a été envoyée.', [
   223→                { text: 'OK', onPress: loadEscrowData }
   224→              ]);
   225→            } catch (error) {
   226→              Alert.alert('Erreur', 'Impossible d\'envoyer la demande.');
   227→            } finally {
   228→              setActionLoading(false);
   229→            }
   230→          }
   231→        }
   232→      ]
   233→    );
   234→  };
   235→
   236→  // Request refund
   237→  const handleRequestRefund = async () => {
   238→    Alert.alert(
   239→      'Demander un remboursement',
   240→      'Souhaitez-vous ouvrir un litige pour ce remboursement ?',
   241→      [
   242→        { text: 'Annuler', style: 'cancel' },
   243→        { text: 'Ouvrir un litige', onPress: () => setShowDisputeModal(true) },
   244→        {
   245→          text: 'Remboursement simple',
   246→          onPress: async () => {
   247→            setActionLoading(true);
   248→            try {
   249→              await apiService.requestRefund(escrowId!);
   250→              Alert.alert('Succès', 'La demande de remboursement a été envoyée.', [
   251→                { text: 'OK', onPress: loadEscrowData }
   252→              ]);
   253→            } catch (error) {
   254→              Alert.alert('Erreur', 'Impossible d\'envoyer la demande.');
   255→            } finally {
   256→              setActionLoading(false);
   257→            }
   258→          }
   259→        }
   260→      ]
   261→    );
   262→  };
   263→
   264→  // Submit dispute
   265→  const handleSubmitDispute = async () => {
   266→    if (!disputeDescription.trim()) {
   267→      Alert.alert('Erreur', 'Veuillez décrire le problème en détail.');
   268→      return;
   269→    }
   270→
   271→    setActionLoading(true);
   272→    
   273→    try {
   274→      await apiService.openDispute(escrowId!, {
   275→        reason: selectedReason,
   276→        description: disputeDescription,
   277→      });
   278→      
   279→      Alert.alert('Litige ouvert', 'Votre litige a été soumis. Un médiateur sera assigné sous 24h.', [
   280→        { text: 'OK', onPress: () => {
   281→          setShowDisputeModal(false);
   282→          setDisputeDescription('');
   283→          loadEscrowData();
   284→        }}
   285→      ]);
   286→    } catch (error) {
   287→      console.error('[EscrowDetailScreen] Error opening dispute:', error);
   288→      Alert.alert('Erreur', 'Impossible d\'ouvrir le litige.');
   289→    } finally {
   290→      setActionLoading(false);
   291→    }
   292→  };
   293→
   294→  // Accept resolution
   295→  const handleAcceptResolution = async (type: 'release' | 'refund') => {
   296→    const actionText = type === 'release' ? 'accepter la libération' : 'accepter le remboursement';
   297→    
   298→    Alert.alert(
   299→      `Confirmer`,
   300→      `Êtes-vous sûr de vouloir ${actionText} ?`,
   301→      [
   302→        { text: 'Annuler', style: 'cancel' },
   303→        {
   304→          text: 'Confirmer',
   305→          onPress: async () => {
   306→            setActionLoading(true);
   307→            try {
   308→              if (type === 'release') {
   309→                await apiService.acceptRelease(escrowId!);
   310→              } else {
   311→                await apiService.acceptRefund(escrowId!);
   312→              }
   313→              Alert.alert('Succès', 'La résolution a été acceptée.', [
   314→                { text: 'OK', onPress: loadEscrowData }
   315→              ]);
   316→            } catch (error) {
   317→              Alert.alert('Erreur', 'Une erreur est survenue.');
   318→            } finally {
   319→              setActionLoading(false);
   320→            }
   321→          }
   322→        }
   323→      ]
   324→    );
   325→  };
   326→
   327→  // Send message to mediator
   328→  const handleSendMessage = async () => {
   329→    if (!chatMessage.trim()) return;
   330→
   331→    try {
   332→      await apiService.sendDisputeMessage(escrowId!, chatMessage);
   333→      setChatMessage('');
   334→      loadEscrowData(); // Reload to show new message
   335→    } catch (error) {
   336→      Alert.alert('Erreur', 'Impossible d\'envoyer le message.');
   337→    }
   338→  };
   339→
   340→  // Render loading state
   341→  if (loading) {
   342→    return (
   343→      <View style={styles.loadingContainer}>
   344→        <ActivityIndicator size="large" color={Colors.primary} />
   345→        <Text style={styles.loadingText}>Chargement...</Text>
   346→      </View>
   347→    );
   348→  }
   349→
   350→  if (!escrowData) {
   351→    return (
   352→      <View style={styles.errorContainer}>
   353→        <Ionicons name="alert-circle-outline" size={48} color={Colors.error} />
   354→        <Text style={styles.errorText}>Données non disponibles</Text>
   355→        <TouchableOpacity style={styles.retryButton} onPress={loadEscrowData}>
   356→          <Text style={styles.retryButtonText}>Réessayer</Text>
   357→        </TouchableOpacity>
   358→      </View>
   359→    );
   360→  }
   361→
   362→  const statusConfig = ESCROW_STATUS_CONFIG[escrowData.status];
   363→
   364→  return (
   365→    <View style={styles.container}>
   366→      {/* Header */}
   367→      <View style={styles.header}>
   368→        <TouchableOpacity 
   369→          style={styles.backButton}
   370→          onPress={() => navigation.goBack()}
   371→        >
   372→          <Ionicons name="arrow-back" size={24} color={Colors.text} />
   373→        </TouchableOpacity>
   374→        <Text style={styles.headerTitle}>Trade Assurance</Text>
   375→        <View style={{ width: 40 }} />
   376→      </View>
   377→
   378→      <ScrollView 
   379→        style={styles.scrollView}
   380→        showsVerticalScrollIndicator={false}
   381→      >
   382→        {/* Status Card */}
   383→        <View style={[styles.statusCard, { borderLeftColor: statusConfig.color }]}>
   384→          <View style={styles.statusHeader}>
   385→            <View style={[styles.statusIconContainer, { backgroundColor: statusConfig.color + '20' }]}>
   386→              <Ionicons name={statusConfig.icon as any} size={28} color={statusConfig.color} />
   387→            </View>
   388→            <View style={styles.statusInfo}>
   389→              <Text style={styles.statusLabel}>Statut actuel</Text>
   390→              <Text style={[styles.statusText, { color: statusConfig.color }]}>
   391→                {statusConfig.label}
   392→              </Text>
   393→            </View>
   394→          </View>
   395→
   396→          {/* Amount */}
   397→          <View style={styles.amountSection}>
   398→            <Text style={styles.amountLabel}>Montant en séquestre</Text>
   399→            <Text style={styles.amountValue}>
   400→              {escrowData.amount.toLocaleString('fr-FR', { minimumFractionDigits: 2 })} {escrowData.currency}
   401→            </Text>
   402→          </View>
   403→
   404→          {/* Account Info */}
   405→          <View style={styles.accountInfo}>
   406→            <Text style={styles.accountNumber}>
   407→              N° Compte: {escrowData.accountNumber}
   408→            </Text>
   409→            <Text style={styles.autoReleaseText}>
   410→              Libération automatique dans {escrowData.autoReleaseDays} jours
   411→            </Text>
   412→          </View>
   413→        </View>
   414→
   415→        {/* Timeline */}
   416→        <View style={styles.section}>
   417→          <Text style={styles.sectionTitle}>Historique</Text>
   418→          
   419→          <View style={styles.timeline}>
   420→            {escrowData.timeline.map((event, index) => (
   421→              <View key={event.id} style={styles.timelineItem}>
   422→                <View style={styles.timelineDotContainer}>
   423→                  <View style={[
   424→                    styles.timelineDot,
   425→                    event.completed 
   426→                      ? { backgroundColor: Colors.primary } 
   427→                      : { backgroundColor: Colors.surfaceVariant, borderWidth: 2, borderColor: Colors.border }
   428→                  ]}/>
   429→                  {index < escrowData.timeline.length - 1 && (
   430→                    <View style={styles.timelineLine} />
   431→                  )}
   432→                </View>
   433→                
   434→                <View style={styles.timelineContent}>
   435→                  <Text style={[
   436→                    styles.timelineLabel,
   437→                    !event.completed && styles.timelineLabelPending
   438→                  ]}>
   439→                    {event.label}
   440→                  </Text>
   441→                  <Text style={styles.timelineTime}>
   442→                    {new Date(event.timestamp).toLocaleDateString('fr-FR', {
   443→                      day: 'numeric',
   444→                      month: 'short',
   445→                      hour: '2-digit',
   446→                      minute: '2-digit',
   447→                    })}
   448→                  </Text>
   449→                  {event.description && (
   450→                    <Text style={styles.timelineDescription}>{event.description}</Text>
   451→                  )}
   452→                </View>
   453→              </View>
   454→            ))}
   455→          </View>
   456→        </View>
   457→
   458→        {/* Action Buttons */}
   459→        <View style={styles.actionsSection}>
   460→          {escrowData.status === 'pending' && (
   461→            <TouchableOpacity
   462→              style={styles.actionButton}
   463→              onPress={() => setShowFundModal(true)}
   464→            >
   465→              <Ionicons name="wallet-outline" size={20} color={Colors.white} />
   466→              <Text style={styles.actionButtonText}>Financer le compte</Text>
   467→            </TouchableOpacity>
   468→          )}
   469→
   470→          {(escrowData.status === 'funded' || escrowData.status === 'in_escrow') && (
   471→            <>
   472→              <TouchableOpacity
   473→                style={[styles.actionButton, styles.releaseButton]}
   474→                onPress={handleRequestRelease}
   475→              >
   476→                <Ionicons name="checkmark-circle-outline" size={20} color={Colors.white} />
   477→                <Text style={styles.actionButtonText}>Confirmer la réception</Text>
   478→              </TouchableOpacity>
   479→              
   480→              <TouchableOpacity
   481→                style={[styles.actionButton, styles.refundButton]}
   482→                onPress={handleRequestRefund}
   483→              >
   484→                <Ionicons name="return-down-back-outline" size={20} color={Colors.white} />
   485→                <Text style={styles.actionButtonText}>Demander un remboursement</Text>
   486→              </TouchableOpacity>
   487→            </>
   488→          )}
   489→
   490→          {escrowData.status === 'released' && (
   491→            <View style={styles.completedBanner}>
   492→              <Ionicons name="checkmark-circle" size={24} color={Colors.success} />
   493→              <Text style={styles.completedText}>Les fonds ont été libérés au vendeur</Text>
   494→            </View>
   495→          )}
   496→
   497→          {escrowData.status === 'refunded' && (
   498→            <View style={styles.refundedBanner}>
   499→              <Ionicons name="arrow-back-circle" size={24} color={Colors.info} />
   500→              <Text style={styles.refundedText}>Vous avez été remboursé</Text>
   501→            </View>
   502→          )}
   503→        </View>
   504→
   505→        {/* Dispute Section */}
   506→        {escrowData.dispute && (
   507→          <View style={[styles.disputeCard, { borderLeftColor: Colors.error }]}>
   508→            <View style={styles.disputeHeader}>
   509→              <Ionicons name="alert-triangle" size={24} color={Colors.error} />
   510→              <View style={styles.disputeHeaderInfo}>
   511→                <Text style={styles.disputeTitle}>Litige en cours</Text>
   512→                <Text style={styles.disputeReason}>
   513→                  {DISPUTE_REASONS.find(r => r.value === escrowData.dispute?.reason)?.label}
   514→                </Text>
   515→              </View>
   516→            </View>
   517→
   518→            {/* Dispute Status */}
   519→            <View style={styles.disputeStatusBadge}>
   520→              <Text style={styles.disputeStatusText}>
   521→                Statut: {escrowData.dispute.status.toUpperCase()}
   522→              </Text>
   523→            </View>
   524→
   525→            {/* Mediator Info */}
   526→            {escrowData.dispute.mediatorName && (
   527→              <View style={styles.mediatorInfo}>
   528→                <Ionicons name="person-outline" size={16} color={Colors.textSecondary} />
   529→                <Text style={styles.mediatorText}>
   530→                  Médiateur: {escrowData.dispute.mediatorName}
   531→                </Text>
   532→              </View>
   533→            )}
   534→
   535→            {/* Deadline */}
   536→            {escrowData.dispute.deadline && (
   537→              <View style={styles.deadlineInfo}>
   538→                <Ionicons name="time-outline" size={16} color={Colors.warning} />
   539→                <Text style={styles.deadlineText}>
   540→                  Date limite: {new Date(escrowData.dispute.deadline).toLocaleDateString('fr-FR')}
   541→                </Text>
   542→              </View>
   543→            )}
   544→
   545→            {/* Chat with Mediator */}
   546→            <TouchableOpacity
   547→              style={styles.chatButton}
   548→              onPress={() => setShowChatModal(true)}
   549→            >
   550→              <Ionicons name="chatbubbles-outline" size={18} color={Colors.primary} />
   551→              <Text style={styles.chatButtonText}>Contacter le médiateur</Text>
   552→              <Ionicons name="chevron-forward" size={18} color={Colors.textTertiary} />
   553→            </TouchableOpacity>
   554→
   555→            {/* Messages Preview */}
   556→            {escrowData.dispute.messages.length > 0 && (
   557→              <View style={styles.messagesPreview}>
   558→                {escrowData.dispute.messages.slice(-3).map(msg => (
   559→                  <View 
   560→                    key={msg.id} 
   561→                    style={[
   562→                      styles.messageBubble,
   563→                      msg.senderRole === 'mediator' ? styles.mediatorBubble : styles.userBubble
   564→                    ]}
   565→                  >
   566→                    <Text style={styles.messageSender}>
   567→                      {msg.senderRole === 'mediator' ? 'Médiateur' : 'Vous'}
   568→                    </Text>
   569→                    <Text style={styles.messageContent}>{msg.content}</Text>
   570→                  </View>
   571→                ))}
   572→              </View>
   573→            )}
   574→
   575→            {/* Resolution Actions */}
   576→            {escrowData.dispute.status === 'resolved' && escrowData.dispute.resolution && (
   577→              <View style={styles.resolutionSection}>
   578→                <Text style={styles.resolutionTitle}>Résolution proposée:</Text>
   579→                <Text style={styles.resolutionText}>{escrowData.dispute.resolution}</Text>
   580→                <View style={styles.resolutionActions}>
   581→                  <TouchableOpacity
   582→                    style={[styles.resolutionButton, styles.acceptReleaseButton]}
