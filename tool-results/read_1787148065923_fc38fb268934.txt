   421→// ============================================================================
   422→// CDN Manager Class
   423→// ============================================================================
   424→
   425→export class CDNManager {
   426→  private static instance: CDNManager;
   427→  private configs: Map<CDNProvider, CDNConfig> = new Map();
   428→  private healthScores: Map<CDNProvider, number> = new Map();
   429→  private lastHealthCheck: Date | null = null;
   430→  private purgeHistory: PurgeRequest[] = [];
   431→  private statsHistory: CDNStats[] = [];
   432→
   433→  static getInstance(): CDNManager {
   434→    if (!CDNManager.instance) {
   435→      CDNManager.instance = new CDNManager();
   436→    }
   437→    return CDNManager.instance;
   438→  }
   439→
   440→  constructor() {
   441→    // Initialize with default configs
   442→    Object.entries(DEFAULT_CDN_CONFIGS).forEach(([provider, config]) => {
   443→      this.configs.set(provider as CDNProvider, config);
   444→      this.healthScores.set(provider as CDNProvider, 100); // Start healthy
   445→    });
   446→  }
   447→
   448→  // ==========================================================================
   449→  // Configuration Management
   450→  // ==========================================================================
   451→
   452→  /**
   453→   * Get configuration for a specific provider
   454→   */
   455→  getConfig(provider: CDNProvider): CDNConfig | undefined {
   456→    return this.configs.get(provider);
   457→  }
   458→
   459→  /**
   460→   * Update configuration for a provider
   461→   */
   462→  updateConfig(provider: CDNProvider, updates: Partial<CDNConfig>): void {
   463→    const current = this.configs.get(provider);
   464→    if (current) {
   465→      this.configs.set(provider, { ...current, ...updates });
   466→    }
   467→  }
   468→
   469→  /**
   470→   * Get all enabled providers sorted by priority
   471→   */
   472→  getEnabledProviders(): CDNProvider[] {
   473→    return Array.from(this.configs.entries())
   474→      .filter(([, config]) => config.enabled)
   475→      .sort((a, b) => a[1].priority - b[1].priority)
   476→      .map(([provider]) => provider);
   477→  }
   478→
   479→  /**
   480→   * Get traffic weights for load balancing
   481→   */
   482→  getTrafficWeights(): Record<CDNProvider, number> {
   483→    const weights: Partial<Record<CDNProvider, number>> = {};
   484→    
   485→    for (const [provider, config] of this.configs.entries()) {
   486→      if (config.enabled && this.isHealthy(provider)) {
   487→        weights[provider] = config.weight;
   488→      }
   489→    }
   490→
   491→    return weights as Record<CDNProvider, number>;
   492→  }
   493→
   494→  // ==========================================================================
   495→  // Multi-CDN Fallback Strategy
   496→  // ==========================================================================
   497→
   498→  /**
   499→   * Determine which CDN(s) to use based on health and performance
   500→   */
   501→  selectProvider(requestContext?: {
   502→    ip?: string;
   503→    path?: string;
   504→    userAgent?: string;
   505→  }): FallbackDecision {
   506→    const enabledProviders = this.getEnabledProviders();
   507→    const healthEntries = Array.from(this.healthScores.entries());
   508→
   509→    // Sort providers by health score (descending), then by priority
   510→    const rankedProviders = enabledProviders.sort((a, b) => {
   511→      const healthA = this.healthScores.get(a) || 0;
   512→      const healthB = this.healthScores.get(b) || 0;
   513→      if (healthB !== healthA) return healthB - healthA;
   514→      return (this.configs.get(a)?.priority || 999) - (this.configs.get(b)?.priority || 999);
   515→    });
   516→
   517→    const primaryProvider = rankedProviders[0];
   518→    const fallbackProviders = rankedProviders.slice(1);
   519→
   520→    // Build health scores object
   521→    const healthScores: Record<CDNProvider, number> = {};
   522→    for (const [provider, score] of healthEntries) {
   523→      healthScores[provider] = score;
   524→    }
   525→
   526→    let reason = `Primary: ${primaryProvider} (Health: ${this.healthScores.get(primaryProvider)}%)`;
   527→    
   528→    // Check if primary is degraded
   529→    const primaryHealth = this.healthScores.get(primaryProvider) || 0;
   530→    if (primaryHealth < 80) {
   531→      reason = `${primaryProvider} degraded (${primaryHealth}%), using ${fallbackProviders[0]} as backup`;
   532→    }
   533→
   534→    return {
   535→      primaryProvider,
   536→      fallbackProviders,
   537→      reason,
   538→      healthScores,
   539→    };
   540→  }
   541→
   542→  /**
   543→   * Check if a provider is considered healthy
   544→   */
   545→  isHealthy(provider: CDNProvider): boolean {
   546→    const score = this.healthScores.get(provider) ?? 0;
   547→    return score >= HEALTH_CHECK_THRESHOLDS.minAvailability;
   548→  }
   549→
   550→  /**
   551→   * Perform health checks on all providers
   552→   */
   553→  async performHealthChecks(): Promise<void> {
   554→    const now = new Date();
   555→
   556→    for (const [provider, config] of this.configs.entries()) {
   557→      if (!config.enabled) continue;
   558→
   559→      try {
   560→        const stats = await this.fetchStats(provider);
   561→        
   562→        // Calculate health score based on multiple factors
   563→        const errorRate = stats.errors.total / Math.max(stats.requests.total, 1);
   564→        const availability = 100 - (errorRate * 100);
   565→        const latencyScore = Math.max(0, 100 - ((stats.latency.p95 - 500) / 20));
   566→        const hitRateScore = (stats.requests.hits / Math.max(stats.requests.cached, 1)) * 100;
   567→
   568→        // Weighted average of all factors
   569→        const healthScore = (
   570→          availability * 0.4 +
   571→          latencyScore * 0.35 +
   572→          hitRateScore * 0.25
   573→        );
   574→
   575→        this.healthScores.set(provider, Math.min(100, Math.max(0, healthScore)));
   576→
   577→        // Store stats in history
   578→        this.statsHistory.push({ ...stats, timestamp: now, provider });
   579→        
   580→        // Keep only last 24 hours of history
   581→        const oneDayAgo = new Date(now.getTime() - 86400000);
   582→        this.statsHistory = this.statsHistory.filter(s => s.timestamp >= oneDayAgo);
   583→
   584→      } catch (error) {
   585→        console.error(`Health check failed for ${provider}:`, error);
   586→        // Degrade health score on failure
   587→        const currentScore = this.healthScores.get(provider) || 100;
   588→        this.healthScores.set(provider, Math.max(0, currentScore - 20));
   589→      }
   590→    }
   591→
   592→    this.lastHealthCheck = now;
   593→  }

   599→  /**
   600→   * Apply image transformation rules to a URL
   601→   */
   602→  transformImageUrl(originalUrl: string, options?: {
   603→    width?: number;
   604→    height?: number;
   605→    quality?: number;
   606→    format?: 'webp' | 'avif' | 'jpg' | 'png';
   607→    fit?: 'contain' | 'cover' | 'fill' | 'crop';
   608→  }): string {
   609→    // Find matching rule
   610→    const matchingRule = IMAGE_TRANSFORM_RULES.find(
   611→      rule => rule.enabled && rule.pattern.test(originalUrl)
   612→    );
   613→
   614→    if (!matchingRule && !options) {
   615→      return originalUrl;
   616→    }
   617→
   618→    // Build transformation parameters
   619→    const params: Record<string, string> = {};
   620→
   621→    if (options?.width) params.width = String(options.width);
   622→    if (options?.height) params.height = String(options.height);
   623→    if (options?.quality) params.quality = String(options.quality);
   624→    if (options?.format) params.format = options.format;
   625→    if (options?.fit) params.fit = options.fit;
   626→
   627→    // Merge with rule transformations
   628→    if (matchingRule) {
   629→      for (const transform of matchingRule.transformations) {
   630→        switch (transform.type) {
   631→          case 'format':
   632→            if (!params.format && transform.params.format) {
   633→              params.format = transform.params.format as string;
   634→            }
   635→            break;
   636→          case 'resize':
   637→            if (!params.width && transform.params.width) {
   638→              params.width = String(transform.params.width);
   639→            }
   640→            if (!params.height && transform.params.height) {
   641→              params.height = String(transform.params.height);
   642→            }
   643→            break;
   644→          case 'quality':
   645→            if (!params.quality && transform.params.quality) {
   646→              params.quality = String(transform.params.quality);
   647→            }
   648→            break;
   649→        }
   650→      }
   651→    }
   652→
   653→    // Generate transformed URL based on active CDN
   654→    const activeProvider = this.selectProvider().primaryProvider;
   655→    return this.buildTransformedUrl(activeProvider, originalUrl, params);
   656→  }

   658→  /**
   659→   * Build transformed URL for each CDN provider
   660→   */
   661→  private buildTransformedUrl(
   662→    provider: CDNProvider,
   663→    originalUrl: string,
   664→    params: Record<string, string>
   665→  ): string {
   666→    const searchParams = new URLSearchParams(params);
   667→
   668→    switch (provider) {
   669→      case 'cloudflare':
   670→        // Cloudflare Image Resizing: /cdn-cgi/image/<params>/<url>
   671→        return `/cdn-cgi/image/${searchParams.toString().replace(/&/g, ',')}/${encodeURIComponent(originalUrl)}`;
   672→      
   673→      case 'fastly':
   674→        // Fastly IO: https://<host>/io?<params>&url=<url>
   675→        searchParams.set('url', originalUrl);
   676→        return `https://io.fastly.com?${searchParams.toString()}`;
   677→      
   678→      case 'cloudfront':
   679→        // CloudFront requires Lambda@Edge or separate image service
   680→        searchParams.set('src', originalUrl);
   681→        return `/_image?${searchParams.toString()}`;
   682→      
   683→      default:
   684→        return originalUrl;
   685→    }
   686→  }

   703→  getOptimalFormat(userAgent: string): 'avif' | 'webp' | 'jpg' {
   704→    const ua = userAgent.toLowerCase();

   706→    if (ua.includes('chrome') && !ua.includes('opera')) {

   713→      const firefoxVersion = parseInt(ua.match(/firefox\/(\d+)/)?.[1] || '0');
   714→      if (firefoxVersion >= 113) return 'avif'; // AVIF support from Firefox 113
   715→      return 'webp';
   716→    }

   718→    if (ua.includes('safari') || ua.includes('mac os x')) {
   719→      const safariVersion = parseInt(ua.match(/version\/(\d+)/)?.[1] || '0');
   720→      if (safariVersion >= 16.4) return 'avif'; // AVIF support from Safari 16.4
   721→      return 'webp';
   722→    }
   723→    
   724→    if (ua.includes('edge') || ua.includes('edg/')) {
   725→      return 'avif'; // Edge supports AVIF

   727→
   728→    return 'jpg'; // Fallback for older browsers

   735→  /**
   736→   * Get cache policy for a URL
   737→   */
   738→  getCachePolicy(url: string): CachePolicyConfig | null {
   739→    // Find matching rule (highest priority first)
   740→    const sortedRules = [...CACHE_RULES].sort((a, b) => a.priority - b.priority);
   741→
   742→    for (const rule of sortedRules) {
   743→      if (!rule.enabled) continue;
   744→
   745→      if (rule.pattern instanceof RegExp) {
   746→        if (rule.pattern.test(url)) {
   747→          return rule.cachePolicy;
   748→        }
   749→      } else {
   750→        // String pattern
   751→        if (url.includes(rule.pattern)) {
   752→          return rule.cachePolicy;
   753→        }
   754→      }
   755→    }
   756→
   757→    return null;
   758→  }

   760→  /**
   761→   * Build Cache-Control header value from policy
   762→   */
   763→  buildCacheControlHeader(policy: CachePolicyConfig): string {
   764→    const parts: string[] = [];
   765→
   766→    if (policy.bypassCache) {
   767→      return 'no-store, no-cache, must-revalidate, proxy-revalidate';
   768→    }
   769→
   770→    if (policy.mustRevalidate) {
   771→      parts.push('must-revalidate');
   772→    }
   773→
   774→    if (policy.ttl > 0) {
   775→      parts.push(`max-age=${policy.ttl}`);
   776→    } else {
   777→      parts.push('no-cache');
   778→    }
   779→
   780→    if (policy.staleWhileRevalidate > 0) {
   781→      parts.push(`stale-while-revalidate=${policy.staleWhileRevalidate}`);
   782→    }
   783→
   784→    if (policy.staleIfError > 0) {
   785→      parts.push(`stale-if-error=${policy.staleIfError}`);
   786→    }
   787→
   788→    parts.push('public');
   789→
   790→    return parts.join(', ');
   791→  }

   796→  buildEdgeCacheControlHeader(policy: CachePolicyConfig): string {
