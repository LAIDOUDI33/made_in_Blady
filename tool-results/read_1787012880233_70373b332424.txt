   185→  async triggerSync(
   186→    connectorId: string,
   187→    direction: SyncDirection,
   188→    entityType: string,
   189→    options?: Partial<SyncOptions>
   190→  ): Promise<SyncOperation> {
   191→    const connector = this.connectors.get(connectorId)
   192→    if (!connector) {
   193→      throw new Error(`Connector not found: ${connectorId}`)
   194→    }
   195→    
   196→    const operation: SyncOperation = {
   197→      id: `sync_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
   198→      connectorId,
   199→      entityType,
   200→      direction,
   201→      status: 'QUEUED',
   202→      priority: 10,
   203→      options: {
   204→        conflictResolution: this.config.defaultConflictResolution,
   205→        batchSize: 50,
   206→        ...options,
   207→      },
   208→      createdAt: new Date(),
   209→    }
   210→    
   211→    // Add to queue
   212→    const queueItem: SyncQueueItem = {
   213→      id: operation.id,
   214→      connectorId,
   215→      operation,
   216→      scheduledAt: new Date(),
   217→      attempts: 0,
   218→      maxAttempts: this.config.defaultRetryPolicy.maxRetries + 1,
   219→    }
   220→    
   221→    this.queue.push(queueItem)
   222→    this.sortQueue()
   223→    this.emit('onQueueChange', [...this.queue])
   224→    
   225→    // Log to database
   226→    await this.createSyncLog(operation)
   227→    
   228→    return operation
   229→  }

   251→  async cancelOperation(operationId: string): Promise<boolean> {
   252→    const operation = this.activeOperations.get(operationId)
   253→    if (operation && (operation.status === 'RUNNING' || operation.status === 'QUEUED')) {
   254→      operation.status = 'CANCELLED'
   255→      this.activeOperations.delete(operationId)

   257→      // Remove from queue
   258→      this.queue = this.queue.filter(item => item.id !== operationId)
   259→      this.emit('onQueueChange', [...this.queue])
   260→      
   261→      return true
   262→    }
   263→    
   264→    return false
   265→  }

   279→  private async processQueue(): Promise<void> {
   280→    if (this.isProcessing) return
   281→    
   282→    const activeCount = Array.from(this.activeOperations.values()).filter(
   283→      op => op.status === 'RUNNING'
   284→    ).length
   285→    
   286→    if (activeCount >= this.config.maxConcurrentSyncs) return
   287→    if (this.queue.length === 0) return
   288→    
   289→    this.isProcessing = true
   290→    
   291→    try {
   292→      const nextItem = this.queue.shift()
   293→      if (!nextItem) return
   294→      
   295→      // Check if it's time to execute
   296→      if (nextItem.nextAttemptAt && nextItem.nextAttemptAt > new Date()) {
   297→        this.queue.unshift(nextItem)
   298→        return
   299→      }
   300→      
   301→      await this.executeOperation(nextItem)
   302→    } finally {
   303→      this.isProcessing = false
   304→      this.emit('onQueueChange', [...this.queue])
   305→    }
   306→  }
   307→  
   308→  private async executeOperation(queueItem: SyncQueueItem): Promise<void> {
   309→    const { operation, connectorId } = queueItem
   310→    const connector = this.connectors.get(connectorId)
   311→    
   312→    if (!connector) {
   313→      operation.status = 'FAILED'
   314→      operation.result = {
   315→        success: false,
   316→        recordsProcessed: 0,
   317→        recordsCreated: 0,
   318→        recordsUpdated: 0,
   319→        recordsFailed: 1,
   320→        errors: [{ code: 'CONNECTOR_NOT_FOUND', message: `Connector ${connectorId} not found`, retryable: false }],
   321→        durationMs: 0,
   322→        startedAt: new Date(),
   323→        completedAt: new Date(),
   324→      }
   325→      return
   326→    }
   327→    
   328→    operation.status = 'RUNNING'
   329→    operation.startedAt = new Date()
   330→    this.activeOperations.set(operation.id, operation)
   331→    
   332→    this.emit('onSyncStart', operation)
   333→    
   334→    try {
   335→      let result: SyncResult
   336→      
   337→      switch (operation.direction) {
   338→        case 'PUSH':
   339→          result = await this.executePushSync(connector, operation)
   340→          break
   341→        case 'PULL':
   342→          result = await this.executePullSync(connector, operation)
   343→          break
   344→        case 'BIDIRECTIONAL':
   345→          result = await this.executeBidirectionalSync(connector, operation)
   346→          break
   347→        default:
   348→          throw new Error(`Unknown sync direction: ${operation.direction}`)
   349→      }
   350→      
   351→      operation.status = result.success ? 'COMPLETED' : (result.recordsFailed > 0 ? 'PARTIAL' : 'FAILED')
   352→      operation.result = result
   353→      operation.completedAt = new Date()
   354→      
   355→      // Update database log
   356→      await this.updateSyncLog(operation, result)
   357→      
   358→      this.emit('onSyncComplete', operation)
   359→      operation.options.onComplete?.(result)
   360→      
   361→    } catch (error) {
   362→      operation.status = 'FAILED'
   363→      operation.result = {
   364→        success: false,
   365→        recordsProcessed: 0,
   366→        recordsCreated: 0,
   367→        recordsUpdated: 0,
   368→        recordsFailed: 1,
   369→        errors: [{
   370→          code: 'SYNC_ERROR',
   371→          message: error instanceof Error ? error.message : String(error),
   372→          retryable: true,
   373→        }],
   374→        durationMs: operation.startedAt ? Date.now() - operation.startedAt.getTime() : 0,
   375→        startedAt: operation.startedAt || new Date(),
   376→        completedAt: new Date(),
   377→      }
   378→      operation.completedAt = new Date()
   379→      
   380→      queueItem.attempts++
   381→      
   382→      // Retry logic
   383→      if (this.config.autoRetryOnFailure && queueItem.attempts < queueItem.maxAttempts) {
   384→        const delay = this.calculateRetryDelay(queueItem.attempts)
   385→        queueItem.nextAttemptAt = new Date(Date.now() + delay)
   386→        this.queue.push(queueItem)
   387→        console.log(`Scheduling retry ${queueItem.attempts}/${queueItem.maxAttempts} for operation ${operation.id}`)
   388→      }
   389→      
   390→      this.emit('onSyncError', operation, error instanceof Error ? error : new Error(String(error)))
   391→      operation.options.onError?.(error instanceof Error ? error : new Error(String(error)))
   392→      
   393→      await this.updateSyncLog(operation, operation.result!)
   394→    } finally {
   395→      this.activeOperations.delete(operation.id)
   396→    }
   397→  }

   399→  private async executePushSync(connector: IERPConnector, operation: SyncOperation): Promise<SyncResult> {
   400→    const startTime = Date.now()
   401→    
   402→    switch (operation.entityType) {
   403→      case 'PRODUCTS': {
   404→        // Get local products and push to ERP
   405→        const products = await this.getLocalProducts(operation.options.lastSyncTimestamp)
   406→        
   407→        if (operation.options.dryRun) {
   408→          return this.createDryRunResult(products.length, startTime)
   409→        }
   410→        
   411→        const batches = this.chunkArray(products, operation.options.batchSize || 50)
   412→        let created = 0
   413→        let updated = 0
   414→        let failed = 0
   415→        const errors: any[] = []
   416→        
   417→        for (let i = 0; i < batches.length; i++) {
   418→          operation.options.onProgress?.({
   419→            operationId: operation.id,
   420→            entityType: operation.entityType,
   421→            totalRecords: products.length,
   422→            processedRecords: Math.min((i + 1) * (operation.options.batchSize || 50), products.length),
   423→            percentage: Math.round(((i + 1) / batches.length) * 100),
   424→            currentBatch: i + 1,
   425→            totalBatches: batches.length,
   426→            message: `Pushing batch ${i + 1}/${batches.length}`,
   427→          })
   428→          
   429→          try {
   430→            const result = await connector.syncInventory(batches[i])
   431→            created += result.recordsCreated
   432→            updated += result.recordsUpdated
   433→            failed += result.recordsFailed
   434→            errors.push(...result.errors)
   435→          } catch (error) {
   436→            failed += batches[i].length
   437→            errors.push({
   438→              code: 'BATCH_ERROR',
   439→              message: error instanceof Error ? error.message : String(error),
   440→              retryable: true,
   441→            })
   442→          }
   443→        }
   444→        
   445→        return this.createSyncResult(created + updated + failed, created, updated, failed, errors, startTime)
   446→      }
   447→      
   448→      case 'INVENTORY': {
   449→        const inventoryData = await this.getLocalInventory()
   450→        const result = await connector.syncInventory(inventoryData)
   451→        result.durationMs = Date.now() - startTime
   452→        return result
   453→      }
   454→      
   455→      case 'ORDERS': {
   456→        const orders = await this.getLocalOrders(operation.options.lastSyncTimestamp)
   457→        let created = 0
   458→        let failed = 0
   459→        const errors: any[] = []
   460→        
   461→        for (const order of orders) {
   462→          try {
   463→            await connector.pushOrder(order)
   464→            created++
   465→          } catch (error) {
   466→            failed++
   467→            errors.push({
   468→              recordId: order.id,
   469→              code: 'PUSH_ORDER_ERROR',
   470→              message: error instanceof Error ? error.message : String(error),
   471→              retryable: true,
   472→            })
   473→          }
   474→        }
   475→        
   476→        return this.createSyncResult(orders.length, created, 0, failed, errors, startTime)
   477→      }
   478→      
   479→      default:
   480→        throw new Error(`Push sync not supported for entity type: ${operation.entityType}`)
   481→    }
   482→  }

   484→  private async executePullSync(connector: IERPConnector, operation: SyncOperation): Promise<SyncResult> {
   485→    const startTime = Date.now()
   486→    
   487→    switch (operation.entityType) {
   488→      case 'PRODUCTS': {
   489→        const fetchOptions = operation.options.fullSync 
   490→          ? undefined 
   491→          : { modifiedSince: operation.options.lastSyncTimestamp }
   492→        
   493→        const products = await connector.fetchProducts(fetchOptions)
   494→        
   495→        let created = 0
   496→        let updated = 0
   497→        let failed = 0
   498→        const errors: any[] = []
   499→        
   500→        for (let i = 0; i < products.length; i++) {
   501→          operation.options.onProgress?.({
   502→            operationId: operation.id,
   503→            entityType: operation.entityType,
   504→            totalRecords: products.length,
   505→            processedRecords: i + 1,
   506→            percentage: Math.round(((i + 1) / products.length) * 100),
   507→            currentBatch: Math.floor(i / (operation.options.batchSize || 50)) + 1,
   508→            totalBatches: Math.ceil(products.length / (operation.options.batchSize || 50)),
   509→            message: `Processing product ${i + 1}/${products.length}`,
   510→          })
   511→          
   512→          try {
   513→            const transformed = await connector.transformData(products[i], 'fromERP')
   514→            await this.upsertLocalProduct(transformed, operation.options.conflictResolution)
   515→            
   516→            // Check if product exists to determine create vs update
   517→            const existing = await this.findLocalProductBySKU(products[i].sku)
   518→            if (existing) updated++
   519→            else created++
   520→          } catch (error) {
   521→            failed++
   522→            errors.push({
   523→              recordId: products[i].id,
   524→              code: 'PULL_PRODUCT_ERROR',
   525→              message: error instanceof Error ? error.message : String(error),
   526→              retryable: true,
   527→            })
   528→          }
   529→        }
   530→        
   531→        return this.createSyncResult(products.length, created, updated, failed, errors, startTime)
   532→      }
   533→      
   534→      case 'INVENTORY': {
   535→        const updates = await connector.fetchInventoryUpdates()
   536→        
   537→        for (const update of updates) {
   538→          await this.updateLocalInventory(update)
   539→        }
   540→        
   541→        return this.createSyncResult(updates.length, 0, updates.length, 0, [], startTime)
   542→      }
   543→      
   544→      case 'ORDERS': {
   545→        const orders = await connector.pullOrders()
   546→        
   547→        for (const order of orders) {
   548→          await this.upsertLocalOrder(order)
   549→        }
   550→        
   551→        return this.createSyncResult(orders.length, orders.length, 0, 0, [], startTime)
   552→      }
   553→      
   554→      default:
   555→        throw new Error(`Pull sync not supported for entity type: ${operation.entityType}`)
   556→    }
   557→  }

   559→  private async executeBidirectionalSync(connector: IERPConnector, operation: SyncOperation): Promise<SyncResult> {
   560→    const startTime = Date.now()
   561→    
   562→    // Execute both push and pull, then merge results
   563→    const pushResult = await this.executePushSync(connector, operation)
   564→    const pullResult = await this.executePullSync(connector, operation)
   565→    
   566→    return {
   567→      success: pushResult.success && pullResult.success,
   568→      recordsProcessed: pushResult.recordsProcessed + pullResult.recordsProcessed,
   569→      recordsCreated: pushResult.recordsCreated + pullResult.recordsCreated,
   570→      recordsUpdated: pushResult.recordsUpdated + pullResult.recordsUpdated,
   571→      recordsFailed: pushResult.recordsFailed + pullResult.recordsFailed,
   572→      errors: [...pushResult.errors, ...pullResult.errors],
   573→      durationMs: Date.now() - startTime,
   574→      startedAt: pushResult.startedAt,
   575→      completedAt: new Date(),
   576→    }
   577→  }

   581→  // ============================================
   582→  
   583→  async resolveConflict(
   584→    localData: any,
   585→    erpData: any,
   586→    strategy: ConflictResolution = this.config.defaultConflictResolution
   587→  ): Promise<any> {
   588→    switch (strategy) {
   589→      case 'LAST_WRITE_WINS':
   590→        const localDate = new Date(localData.updatedAt || localData.updated_at || 0)
   591→        const erpDate = new Date(erpData.updatedAt || erpData.updated_at || 0)
   592→        return localDate > erpDate ? localData : erpData
   593→        
   594→      case 'PLATFORM_WINS':
   595→        return localData
   596→        
   597→      case 'ERP_WINS':
   598→        return erpData
   599→        
   600→      case 'MERGE':
   601→        return this.mergeObjects(localData, erpData)
   602→        
   603→      case 'MANUAL':
   604→        // Return both for manual review
   605→        return {
   606→          _conflict: true,
   607→          _local: localData,
   608→          _erp: erpData,
   609→        }
   610→        
   611→      default:
   612→        return erpData
   613→    }
   614→  }
   615→  
   616→  private mergeObjects(target: any, source: any): any {
   617→    const result = { ...target }
   618→    
   619→    for (const key of Object.keys(source)) {
   620→      if (typeof source[key] === 'object' && source[key] !== null && !Array.isArray(source[key])) {
   621→        result[key] = this.mergeObjects(result[key] || {}, source[key])
   622→      } else if (source[key] !== undefined && source[key] !== null) {
   623→        result[key] = source[key]
   624→      }
   625→    }
   626→    
   627→    return result
   628→  }

   651→  private async updateSyncLog(operation: SyncOperation, result: SyncResult): Promise<void> {
   652→    try {
   653→      await db.erpSyncLogNew.update({

   659→          recordsFailed: result.recordsFailed,
   660→          durationSeconds: Math.round(result.durationMs / 1000),
   661→          completedAt: result.completedAt,
   662→          errorMessage: result.errors.length > 0 
   663→            ? JSON.stringify(result.errors.slice(0, 5))

   751→  // SCHEDULED SYNC
   752→  // ============================================
   753→  
   754→  startScheduledSync(connectorId: string, frequency: SyncFrequency, entityTypes?: string[]): void {
   755→    const intervalMs = this.getIntervalFromFrequency(frequency)
   756→    
   757→    if (intervalMs > 0) {
   758→      setInterval(async () => {
   759→        const types = entityTypes || ['PRODUCTS', 'INVENTORY']
   760→        for (const entityType of types) {
   761→          await this.triggerSync(connectorId, 'BIDIRECTIONAL', entityType)
   762→        }
   763→      }, intervalMs)
   764→    }
   765→  }
   766→  
   767→  private getIntervalFromFrequency(frequency: SyncFrequency): number {
   768→    switch (frequency) {
   769→      case 'REALTIME':
   770→      case 'EVERY_5_MIN':
   771→        return 5 * 60 * 1000
   772→      case 'EVERY_15_MIN':
   773→        return 15 * 60 * 1000
   774→      case 'EVERY_30_MIN':
   775→        return 30 * 60 * 1000
   776→      case 'HOURLY':
   777→        return 60 * 60 * 1000
   778→      case 'DAILY':
   779→        return 24 * 60 * 60 * 1000
   780→      case 'WEEKLY':
   781→        return 7 * 24 * 60 * 60 * 1000
   782→      case 'MANUAL':
   783→        return 0
   784→      default:
   785→        return 0
   786→    }
   787→  }

   799→  async performHealthCheck(): Promise<{
   800→    healthy: number
   801→    degraded: number
   802→    unhealthy: number
   803→    details: Map<string, any>
   804→  }> {
   805→    const details = new Map<string, any>()
   806→    let healthy = 0
   807→    let degraded = 0
   808→    let unhealthy = 0
   809→    
   810→    for (const [id, connector] of this.connectors) {
   811→      try {
   812→        const check = await connector.healthCheck()
   813→        details.set(id, check)
   814→        
   815→        switch (check.status) {
   816→          case 'healthy': healthy++; break
   817→          case 'degraded': degraded++; break
   818→          case 'unhealthy': unhealthy++; break
   819→        }
   820→      } catch (error) {
   821→        unhealthy++
   822→        details.set(id, { status: 'unhealthy', error: error instanceof Error ? error.message : String(error) })
   823→      }
   824→    }
   825→    
   826→    return { healthy, degraded, unhealthy, details }
   827→  }

   844→  on(event: 'onSyncStart' | 'onSyncComplete' | 'onSyncError' | 'onQueueChange', callback: Function): void {
   845→    this.listeners[event].push(callback as any)
   846→  }
   847→  
   848→  off(event: 'onSyncStart' | 'onSyncComplete' | 'onSyncError' | 'onQueueChange', callback: Function): void {
   849→    const index = this.listeners[event].indexOf(callback as any)
   850→    if (index > -1) {
   851→      this.listeners[event].splice(index, 1)
   852→    }
   853→  }
   854→  
   855→  private emit(event: keyof SyncEngine['listeners'], data: any): void {
   856→    this.listeners[event].forEach(callback => {
   857→      try {
   858→        callback(data)
   859→      } catch (error) {
   860→        console.error(`Event listener error (${event}):`, error)
   861→      }
   862→    })
   863→  }

   881→  private sortQueue(): void {
   882→    this.queue.sort((a, b) => a.operation.priority - b.operation.priority)

   885→  private calculateRetryDelay(attempt: number): number {
   886→    const baseDelay = this.config.defaultRetryPolicy.baseDelayMs
   887→    const multiplier = this.config.defaultRetryPolicy.backoffMultiplier
   888→    const maxDelay = this.config.defaultRetryPolicy.maxDelayMs
   889→    
   890→    return Math.min(baseDelay * Math.pow(multiplier, attempt), maxDelay)
   891→  }

   897→  private chunkArray<T>(array: T[], size: number): T[][] {
   898→    const chunks: T[][] = []
   899→    for (let i = 0; i < array.length; i += size) {
   900→      chunks.push(array.slice(i, i + size))
   901→    }
   902→    return chunks
   903→  }
