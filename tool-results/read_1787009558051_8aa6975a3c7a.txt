     1→'use client'
     2→
     3→import React, { useState, useCallback, useEffect } from 'react'
     4→import { 
     5→  Lock, 
     6→  ShieldCheck, 
     7→  CreditCard, 
     8→  CheckCircle2,
     9→  AlertCircle,
    10→  Loader2,
    11→  Globe2,
    12→  Save,
    13→  ChevronDown
    14→} from 'lucide-react'
    15→import { Button } from '@/components/ui/button'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Checkbox } from '@/components/ui/checkbox'
    19→import { Card, CardContent, CardHeader, CardTitle, CardDescription, CardFooter } from '@/components/ui/card'
    20→import { cn, formatDZD } from '@/lib/utils'
    21→import { SUPPORTED_CURRENCIES, formatStripeAmount, convertFromDZD } from '@/lib/payments/stripe'
    22→
    23→// ============================================
    24→// TYPES
    25→// ============================================
    26→
    27→interface StripeFormProps {
    28→  orderId: string
    29→  orderNumber: string
    30→  amountDZD: number // Amount in DZD (will be converted)
    31→  customerEmail?: string
    32→  customerName?: string
    33→  onPaymentSuccess?: (result: StripePaymentResult) => void
    34→  onPaymentError?: (error: string) => void
    35→  isProcessing?: boolean
    36→}
    37→
    38→export interface StripePaymentResult {
    39→  success: boolean
    40→  paymentIntentId?: string
    41→  clientSecret?: string
    42→  message?: string
    43→  error?: string
    44→}
    45→
    46→interface FormErrors {
    47→  cardNumber?: string
    48→  expiryDate?: string
    49→  cvv?: string
    50→  cardholderName?: string
    51→  email?: string
    52→  name?: string
    53→  addressLine1?: string
    54→  city?: string
    55→  country?: string
    56→  postalCode?: string
    57→}
    58→
    59→type FormStep = 'details' | 'payment' | 'processing' | 'success' | 'error'
    60→
    61→interface CurrencyOption {
    62→  code: string
    63→  name: string
    64→  symbol: string
    65→  flag: string
    66→}
    67→
    68→// ============================================
    69→// COMPONENT
    70→// ============================================
    71→
    72→export function StripeForm({
    73→  orderId,
    74→  orderNumber,
    75→  amountDZD,
    76→  customerEmail: initialEmail = '',
    77→  customerName: initialName = '',
    78→  onPaymentSuccess,
    79→  onPaymentError,
    80→  isProcessing: externalProcessing = false,
    81→}: StripeFormProps) {
    82→  // Form state
    83→  const [selectedCurrency, setSelectedCurrency] = useState<CurrencyOption>(SUPPORTED_CURRENCIES[0])
    84→  const [showCurrencyDropdown, setShowCurrencyDropdown] = useState(false)
    85→  
    86→  // Customer info
    87→  const [email, setEmail] = useState(initialEmail)
    88→  const [name, setName] = useState(initialName)
    89→  
    90→  // Card details
    91→  const [cardNumber, setCardNumber] = useState('')
    92→  const [expiryDate, setExpiryDate] = useState('')
    93→  const [cvv, setCvv] = useState('')
    94→  const [cardholderName, setCardholderName] = useState('')
    95→  const [saveCard, setSaveCard] = useState(false)
    96→  
    97→  // Address (for international orders)
    98→  const [addressLine1, setAddressLine1] = useState('')
    99→  const [city, setCity] = useState('')
   100→  const [postalCode, setPostalCode] = useState('')
   101→  const [country, setCountry] = useState('US')
   102→  
   103→  // UI state
   104→  const [errors, setErrors] = useState<FormErrors>({})
   105→  const [isProcessing, setIsProcessing] = useState(false)
   106→  const [currentStep, setCurrentStep] = useState<FormStep>('details')
   107→  const [paymentResult, setPaymentResult] = useState<StripePaymentResult | null>(null)
   108→  const [clientSecret, setClientSecret] = useState<string | null>(null)
   109→
   110→  // Calculate converted amount
   111→  const convertedAmount = convertFromDZD(amountDZD, selectedCurrency.code)
   112→
   113→  // Format card number with spaces
   114→  const formatCardNumber = useCallback((value: string) => {
   115→    const cleaned = value.replace(/\s/g, '').replace(/\D/g, '')
   116→    const groups = cleaned.match(/.{1,4}/g)
   117→    return groups ? groups.join(' ') : ''
   118→  }, [])
   119→
   120→  // Format expiry date
   121→  const formatExpiryDate = useCallback((value: string) => {
   122→    const cleaned = value.replace(/\D/g, '')
   123→    if (cleaned.length >= 2) {
   124→      return cleaned.slice(0, 2) + '/' + cleaned.slice(2, 4)
   125→    }
   126→    return cleaned
   127→  }, [])
   128→
   129→  // Detect card type
   130→  const detectCardType = (number: string): 'visa' | 'mastercard' | 'amex' | 'unknown' => {
   131→    const cleaned = number.replace(/\s/g, '')
   132→    
   133→    if (/^4/.test(cleaned)) return 'visa'
   134→    if (/^5[1-5]/.test(cleaned) || /^2[2-7]/.test(cleaned)) return 'mastercard'
   135→    if (/^3[47]/.test(cleaned)) return 'amex'
   136→    
   137→    return 'unknown'
   138→  }
   139→
   140→  const cardType = detectCardType(cardNumber)
   141→
   142→  // Validate step 1: Details
   143→  const validateDetails = (): boolean => {
   144→    const newErrors: FormErrors = {}
   145→
   146→    if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   147→      newErrors.email = 'Adresse email invalide'
   148→    }
   149→
   150→    if (!name || name.trim().length < 3) {
   151→      newErrors.name = 'Nom requis (minimum 3 caractères)'
   152→    }
   153→
   154→    setErrors(newErrors)
   155→    return Object.keys(newErrors).length === 0
   156→  }
   157→
   158→  // Validate step 2: Payment
   159→  const validatePayment = (): boolean => {
   160→    const newErrors: FormErrors = {}
   161→
   162→    // Card number validation (basic Luhn check would go here)
   163→    const cleanedCard = cardNumber.replace(/\s/g, '')
   164→    if (!cleanedCard || cleanedCard.length < 13 || cleanedCard.length > 19) {
   165→      newErrors.cardNumber = 'Numéro de carte invalide'
   166→    }
   167→
   168→    // Expiry validation
   169→    if (!expiryDate || !/^\d{2}\/\d{2}$/.test(expiryDate)) {
   170→      newErrors.expiryDate = 'Date invalide (MM/YY)'
   171→    } else {
   172→      const [month, year] = expiryDate.split('/').map(Number)
   173→      if (month < 1 || month > 12) {
   174→        newErrors.expiryDate = 'Mois invalide'
   175→      } else {
   176→        const expDate = new Date(2000 + year, month - 1)
   177→        if (expDate < new Date()) {
   178→          newErrors.expiryDate = 'Carte expirée'
   179→        }
   180→      }
   181→    }
   182→
   183→    // CVV validation
   184→    if (!cvv || !/^\d{3,4}$/.test(cvv)) {
   185→      newErrors.cvv = 'CVV invalide (3-4 chiffres)'
   186→    }
   187→
   188→    // Name validation
   189→    if (!cardholderName || cardholderName.trim().length < 3) {
   190→      newErrors.cardholderName = 'Nom du titulaire requis'
   191→    }
   192→
   193→    setErrors(newErrors)
   194→    return Object.keys(newErrors).length === 0
   195→  }
   196→
   197→  // Handle continue to payment step
   198→  const handleContinueToPayment = async () => {
   199→    if (!validateDetails()) return
   200→    
   201→    setCurrentStep('payment')
   202→  }
   203→
   204→  // Handle back to details
   205→  const handleBackToDetails = () => {
   206→    setCurrentStep('details')
   207→  }
   208→
   209→  // Create Payment Intent and process payment
   210→  const handlePaymentSubmit = async (e: React.FormEvent) => {
   211→    e.preventDefault()
   212→    
   213→    if (!validatePayment()) return
   214→
   215→    setIsProcessing(true)
   216→    setCurrentStep('processing')
   217→
   218→    try {
   219→      // Step 1: Create Payment Intent
   220→      const response = await fetch('/api/payments/stripe', {
   221→        method: 'POST',
   222→        headers: { 'Content-Type': 'application/json' },
   223→        body: JSON.stringify({
   224→          orderId,
   225→          amount: convertedAmount.toString(),
   226→          currency: selectedCurrency.code,
   227→          customerEmail: email,
   228→          customerName: name,
   229→        }),
   230→      })
   231→
   232→      const result = await response.json()
   233→
   234→      if (!response.ok || !result.success) {
   235→        throw new Error(result.error || 'Failed to create payment')
   236→      }
   237→
   238→      setClientSecret(result.payment.clientSecret)
   239→
   240→      // Step 2: In a real implementation, you'd use Stripe.js Elements here
   241→      // For this demo, we'll simulate successful confirmation after creating intent
   242→      
   243→      // Simulate payment processing delay
   244→      await new Promise(resolve => setTimeout(resolve, 2000))
   245→
   246→      // Confirm the payment intent
   247→      const confirmResponse = await fetch(`/api/payments/stripe/${result.payment.paymentIntentId}/confirm`, {
   248→        method: 'POST',
   249→        headers: { 'Content-Type': 'application/json' },
   250→      })
   251→
   252→      const confirmResult = await confirmResponse.json()
   253→
   254→      if (confirmResult.success && confirmResult.stripeStatus === 'succeeded') {
   255→        setCurrentStep('success')
   256→        setPaymentResult({
   257→          success: true,
   258→          paymentIntentId: result.payment.paymentIntentId,
   259→          clientSecret: result.payment.clientSecret,
   260→          message: 'Paiement réussi!',
   261→        })
   262→        
   263→        onPaymentSuccess?.({
   264→          success: true,
   265→          paymentIntentId: result.payment.paymentIntentId,
   266→          message: 'Paiement confirmé avec succès',
   267→        })
   268→      } else {
   269→        throw new Error(confirmResult.error || 'Payment confirmation failed')
   270→      }
   271→    } catch (error) {
   272→      console.error('Stripe Payment error:', error)
   273→      setCurrentStep('error')
   274→      setPaymentResult({
   275→        success: false,
   276→        error: error instanceof Error ? error.message : 'Erreur de paiement',
   277→      })
   278→      onPaymentError?.(error instanceof Error ? error.message : 'Erreur de paiement')
   279→    } finally {
   280→      setIsProcessing(false)
   281→    }
   282→  }
   283→
   284→  // Currency selector handler
   285→  const handleCurrencySelect = (currency: CurrencyOption) => {
   286→    setSelectedCurrency(currency)
   287→    setShowCurrencyDropdown(false)
   288→  }
   289→
   290→  return (
   291→    <Card className="w-full max-w-lg mx-auto overflow-hidden">
   292→      {/* Header */}
   293→      <CardHeader className="bg-gradient-to-r from-indigo-600 to-purple-600 text-white">
   294→        <div className="flex items-center gap-3">
   295→          <div className="p-2 bg-white/10 rounded-lg">
   296→            <Globe2 className="h-6 w-6" />
   297→          </div>
   298→          <div>
   299→            <CardTitle className="text-xl font-bold">Paiement International</CardTitle>
   300→            <CardDescription className="text-white/80 text-sm">
   301→              Cartes Visa, Mastercard via Stripe
   302→            </CardDescription>
   303→          </div>
   304→        </div>
   305→      </CardHeader>
   306→
   307→      <CardContent className="p-6">
   308→        {/* Amount Display with Currency Selector */}
   309→        <div className="mb-6 p-4 bg-gray-50 rounded-lg border border-gray-100">
   310→          <div className="flex justify-between items-center">
   311→            <div>
   312→              <p className="text-sm text-gray-500">Montant à payer</p>
   313→              <p className="text-xs text-gray-400 mt-1">
   314→                Commande: {orderNumber} • {formatDZD(amountDZD)}
   315→              </p>
   316→            </div>
   317→            <div className="flex items-center gap-2">
   318→              <p className="text-2xl font-bold text-indigo-600">
   319→                {formatStripeAmount(convertedAmount, selectedCurrency.code)}
   320→              </p>
   321→              
   322→              {/* Currency Dropdown */}
   323→              <div className="relative">
   324→                <button
   325→                  type="button"
   326→                  onClick={() => setShowCurrencyDropdown(!showCurrencyDropdown)}
   327→                  className="flex items-center gap-1 px-2 py-1 bg-white border border-gray-200 rounded-md text-sm hover:bg-gray-50"
   328→                  disabled={isProcessing || currentStep !== 'details'}
   329→                >
   330→                  <span>{selectedCurrency.flag}</span>
   331→                  <span>{selectedCurrency.code}</span>
   332→                  <ChevronDown className="h-3 w-3" />
   333→                </button>
   334→                
   335→                {showCurrencyDropdown && (
   336→                  <div className="absolute right-0 top-full mt-1 w-40 bg-white border border-gray-200 rounded-lg shadow-lg z-10 py-1">
   337→                    {SUPPORTED_CURRENCIES.map((currency) => (
   338→                      <button
   339→                        key={currency.code}
   340→                        type="button"
   341→                        onClick={() => handleCurrencySelect(currency)}
   342→                        className={cn(
   343→                          "w-full flex items-center gap-2 px-3 py-2 text-left hover:bg-indigo-50",
   344→                          selectedCurrency.code === currency.code && "bg-indigo-50 text-indigo-700"
   345→                        )}
   346→                      >
   347→                        <span>{currency.flag}</span>
   348→                        <span>{currency.code}</span>
   349→                        <span className="text-xs text-gray-400">{currency.symbol}</span>
   350→                      </button>
   351→                    ))}
   352→                  </div>
   353→                )}
   354→              </div>
   355→            </div>
   356→          </div>
   357→          
   358→          {/* Exchange rate note */}
   359→          <p className="text-xs text-gray-400 mt-2">
   360→            Taux approximatif: 1 DZD ≈ {(convertFromDZD(1, selectedCurrency.code)).toFixed(4)} {selectedCurrency.code}
   361→          </p>
   362→        </div>
   363→
   364→        {/* Step 1: Customer Details */}
   365→        {currentStep === 'details' && (
   366→          <form onSubmit={(e) => { e.preventDefault(); handleContinueToPayment(); }} className="space-y-5">
   367→            <h3 className="font-semibold text-lg">Informations de contact</h3>
   368→            
   369→            {/* Email */}
   370→            <div className="space-y-2">
   371→              <Label htmlFor="stripe-email" className="text-sm font-medium">
   372→                Email *
   373→              </Label>
   374→              <Input
   375→                id="stripe-email"
   376→                type="email"
   377→                placeholder="votre@email.com"
   378→                value={email}
   379→                onChange={(e) => {
   380→                  setEmail(e.target.value)
   381→                  if (errors.email) setErrors(prev => ({ ...prev, email: undefined }))
   382→                }}
   383→                className={cn("h-12", errors.email && "border-red-500")}
   384→                disabled={isProcessing || externalProcessing}
   385→              />
   386→              {errors.email && (
   387→                <p className="text-xs text-red-500 flex items-center gap-1">
   388→                  <AlertCircle className="w-3 h-3" />
   389→                  {errors.email}
   390→                </p>
   391→              )}
   392→            </div>
   393→
   394→            {/* Name */}
   395→            <div className="space-y-2">
   396→              <Label htmlFor="stripe-name" className="text-sm font-medium">
   397→                Nom complet *
   398→              </Label>
   399→              <Input
   400→                id="stripe-name"
   401→                type="text"
   402→                placeholder="John Doe"
   403→                value={name}
   404→                onChange={(e) => {
   405→                  setName(e.target.value)
   406→                  if (errors.name) setErrors(prev => ({ ...prev, name: undefined }))
   407→                }}
   408→                className={cn("h-12", errors.name && "border-red-500")}
   409→                disabled={isProcessing || externalProcessing}
   410→              />
   411→              {errors.name && (
   412→                <p className="text-xs text-red-500 flex items-center gap-1">
   413→                  <AlertCircle className="w-3 h-3" />
   414→                  {errors.name}
   415→                </p>
   416→              )}
   417→            </div>
   418→
   419→            {/* Continue button */}
   420→            <Button
   421→              type="submit"
   422→              className="w-full h-12 bg-indigo-600 hover:bg-indigo-700 text-white font-semibold"
   423→              disabled={isProcessing || externalProcessing}
   424→            >
   425→              Continuer vers le paiement
   426→              <ChevronDown className="ml-2 h-4 w-4 rotate-90" />
   427→            </Button>
   428→          </form>
   429→        )}
   430→
   431→        {/* Step 2: Card Payment */}
   432→        {currentStep === 'payment' && (
   433→          <form onSubmit={handlePaymentSubmit} className="space-y-5">
   434→            {/* Back button */}
   435→            <button
   436→              type="button"
   437→              onClick={handleBackToDetails}
   438→              className="text-sm text-gray-500 hover:text-gray-700 flex items-center gap-1"
   439→            >
   440→              ← Modifier les informations
   441→            </button>
   442→
   443→            <h3 className="font-semibold text-lg">Informations de carte</h3>
   444→
   445→            {/* Card Number */}
   446→            <div className="space-y-2">
   447→              <Label htmlFor="card-number" className="text-sm font-medium">
   448→                Numéro de carte *
   449→              </Label>
   450→              <div className="relative">
   451→                <Input
   452→                  id="card-number"
   453→                  type="text"
   454→                  placeholder="0000 0000 0000 0000"
   455→                  value={cardNumber}
   456→                  onChange={(e) => {
   457→                    const formatted = formatCardNumber(e.target.value)
   458→                    if (formatted.replace(/\s/g, '').length <= 19) {
   459→                      setCardNumber(formatted)
   460→                      if (errors.cardNumber) setErrors(prev => ({ ...prev, cardNumber: undefined }))
   461→                    }
   462→                  }}
   463→                  className={cn(
   464→                    "h-12 text-lg tracking-wider pr-16",
   465→                    errors.cardNumber && "border-red-500",
   466→                    cardType === 'visa' && "border-blue-400",
   467→                    cardType === 'mastercard' && "border-orange-400",
   468→                    cardType === 'amex' && "border-blue-300"
   469→                  )}
   470→                  disabled={isProcessing || externalProcessing}
   471→                />
   472→                <div className="absolute right-3 top-1/2 -translate-y-1/2">
   473→                  {cardType === 'visa' && <VisaLogo />}
   474→                  {cardType === 'mastercard' && <MastercardLogo />}
   475→                  {cardType === 'amex' && <AmexLogo />}
   476→                  {!cardNumber && <CreditCardIcon />}
   477→                </div>
   478→              </div>
   479→              {errors.cardNumber && (
   480→                <p className="text-xs text-red-500">{errors.cardNumber}</p>
   481→              )}
   482→            </div>
   483→
   484→            {/* Expiry & CVV */}
   485→            <div className="grid grid-cols-2 gap-4">
   486→              <div className="space-y-2">
   487→                <Label htmlFor="expiry" className="text-sm font-medium">
   488→                  Expiration *
   489→                </Label>
   490→                <Input
   491→                  id="expiry"
   492→                  type="text"
   493→                  placeholder="MM/YY"
   494→                  value={expiryDate}
   495→                  onChange={(e) => {
   496→                    setExpiryDate(formatExpiryDate(e.target.value))
   497→                    if (errors.expiryDate) setErrors(prev => ({ ...prev, expiryDate: undefined }))
   498→                  }}
   499→                  maxLength={5}
   500→                  className={cn("h-12", errors.expiryDate && "border-red-500")}
   501→                  disabled={isProcessing || externalProcessing}
   502→                />
   503→                {errors.expiryDate && (
   504→                  <p className="text-xs text-red-500">{errors.expiryDate}</p>
   505→                )}
   506→              </div>
   507→
   508→              <div className="space-y-2">
   509→                <Label htmlFor="cvv" className="text-sm font-medium">
   510→                  CVV *
   511→                </Label>
   512→                <div className="relative">
   513→                  <Input
   514→                    id="cvv"
   515→                    type="password"
   516→                    placeholder="•••"
   517→                    value={cvv}
   518→                    onChange={(e) => {
   519→                      const val = e.target.value.replace(/\D/g, '').slice(0, 4)
   520→                      setCvv(val)
   521→                      if (errors.cvv) setErrors(prev => ({ ...prev, cvv: undefined }))
   522→                    }}
   523→                    className={cn("h-12 pr-10", errors.cvv && "border-red-500")}
   524→                    disabled={isProcessing || externalProcessing}
   525→                  />
   526→                  <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   527→                </div>
   528→                {errors.cvv && (
   529→                  <p className="text-xs text-red-500">{errors.cvv}</p>
   530→                )}
   531→              </div>
   532→            </div>
   533→
   534→            {/* Cardholder Name */}
   535→            <div className="space-y-2">
   536→              <Label htmlFor="cardholder-name" className="text-sm font-medium">
   537→                Nom du titulaire *
   538→              </Label>
   539→              <Input
   540→                id="cardholder-name"
   541→                type="text"
   542→                placeholder="NOM sur la carte"
   543→                value={cardholderName}
   544→                onChange={(e) => {
   545→                  setCardholderName(e.target.value.toUpperCase())
   546→                  if (errors.cardholderName) setErrors(prev => ({ ...prev, cardholderName: undefined }))
   547→                }}
   548→                className={cn("h-12 uppercase", errors.cardholderName && "border-red-500")}
   549→                disabled={isProcessing || externalProcessing}
   550→              />
   551→              {errors.cardholderName && (
   552→                <p className="text-xs text-red-500">{errors.cardholderName}</p>
   553→              )}
   554→            </div>
   555→
   556→            {/* Save Card Option */}
   557→            <div className="flex items-center space-x-2">
   558→              <Checkbox
   559→                id="save-card-stripe"
   560→                checked={saveCard}
   561→                onCheckedChange={(checked) => setSaveCard(checked === true)}
   562→                disabled={isProcessing || externalProcessing}
   563→              />
   564→              <Label htmlFor="save-card-stripe" className="text-sm text-gray-600 cursor-pointer">
   565→                Enregistrer cette carte pour mes prochains paiements
   566→              </Label>
   567→            </div>
   568→
   569→            {/* Submit Button */}
   570→            <Button
   571→              type="submit"
   572→              className="w-full h-14 bg-indigo-600 hover:bg-indigo-700 text-white font-semibold text-base transition-all duration-200"
   573→              disabled={isProcessing || externalProcessing}
   574→            >
   575→              {isProcessing || externalProcessing ? (
   576→                <span className="flex items-center gap-2">
   577→                  <Loader2 className="h-5 w-5 animate-spin" />
   578→                  Traitement en cours...
   579→                </span>
   580→              ) : (
   581→                <span className="flex items-center justify-center gap-2">
   582→                  <Lock className="h-5 w-5" />
   583→                  Payer {formatStripeAmount(convertedAmount, selectedCurrency.code)}
   584→                </span>
   585→              )}
   586→            </Button>
   587→
   588→            {/* Security Note */}
   589→            <div className="flex items-center justify-center gap-2 pt-2 text-xs text-gray-400">
   590→              <ShieldCheck className="h-4 w-4" />
   591→              <span>Paiement sécurisé par Stripe avec authentification 3D Secure</span>
   592→            </div>
   593→          </form>
   594→        )}
   595→
   596→        {/* Processing State */}
   597→        {currentStep === 'processing' && (
   598→          <div className="py-8 text-center space-y-4">
   599→            <div className="mx-auto w-16 h-16 bg-indigo-50 rounded-full flex items-center justify-center">
   600→              <Loader2 className="h-8 w-8 text-indigo-600 animate-spin" />
   601→            </div>
   602→            <div>
   603→              <h3 className="font-semibold text-lg text-indigo-700">
   604→                Traitement du paiement...
   605→              </h3>
   606→              <p className="text-sm text-gray-500 mt-2">
   607→                Veuillez ne pas fermer cette page pendant le traitement.
   608→              </p>
   609→            </div>
   610→            
   611→            {/* Progress steps */}
   612→            <div className="max-w-xs mx-auto space-y-2">
   613→              <div className="flex items-center gap-2 text-sm">
   614→                <CheckCircle2 className="h-4 w-4 text-green-500" />
   615→                <span>Création du paiement</span>
   616→              </div>
   617→              <div className="flex items-center gap-2 text-sm">
   618→                <Loader2 className="h-4 w-4 text-indigo-500 animate-spin" />
   619→                <span>Authentification bancaire</span>
   620→              </div>
   621→              <div className="flex items-center gap-2 text-sm text-gray-400">
   622→                <div className="h-4 w-4 rounded-full border border-gray-300" />
   623→                <span>Confirmation finale</span>
   624→              </div>
   625→            </div>
   626→          </div>
   627→        )}
   628→
   629→        {/* Success State */}
   630→        {currentStep === 'success' && (
   631→          <div className="py-8 text-center space-y-4">
   632→            <div className="mx-auto w-16 h-16 bg-green-50 rounded-full flex items-center justify-center">
   633→              <CheckCircle2 className="h-10 w-10 text-green-600" />
   634→            </div>
   635→            <div>
   636→              <h3 className="font-semibold text-lg text-green-700">
   637→                Paiement réussi !
   638→              </h3>
   639→              <p className="text-sm text-gray-500 mt-2">
   640→                Votre paiement a été traité avec succès.
   641→              </p>
   642→              {paymentResult?.paymentIntentId && (
   643→                <p className="text-xs text-gray-400 mt-2">
   644→                  Transaction: {paymentResult.paymentIntentId}
   645→                </p>
   646→              )}
   647→            </div>
   648→            <Button
   649→              onClick={() => onPaymentSuccess?.(paymentResult!)}
   650→              className="bg-green-600 hover:bg-green-700"
   651→            >
   652→              Continuer
   653→            </Button>
   654→          </div>
   655→        )}
   656→
   657→        {/* Error State */}
   658→        {currentStep === 'error' && (
   659→          <div className="py-8 text-center space-y-4">
   660→            <div className="mx-auto w-16 h-16 bg-red-50 rounded-full flex items-center justify-center">
   661→              <AlertCircle className="h-10 w-10 text-red-600" />
   662→            </div>
   663→            <div>
   664→              <h3 className="font-semibold text-lg text-red-700">
   665→                Erreur de paiement
   666→              </h3>
   667→              <p className="text-sm text-gray-500 mt-2">
   668→                {paymentResult?.error || 'Une erreur est survenue lors du traitement.'}
   669→              </p>
   670→            </div>
   671→            <div className="flex gap-3 justify-center">
   672→              <Button
   673→                variant="outline"
   674→                onClick={() => setCurrentStep('payment')}
   675→              >
   676→                Réessayer
   677→              </Button>
   678→              <Button
   679→                variant="ghost"
   680→                onClick={() => onPaymentError?.('Payment cancelled')}
   681→              >
   682→                Annuler
   683→              </Button>
   684→            </div>
   685→          </div>
   686→        )}
   687→      </CardContent>
   688→
   689→      {/* Footer */}
   690→      {(currentStep === 'details' || currentStep === 'payment') && (
   691→        <CardFooter className="bg-gray-50 px-6 py-4 border-t">
   692→          <div className="w-full flex items-center justify-between text-xs text-gray-400">
   693→            <div className="flex items-center gap-2">
   694→              <Globe2 className="h-4 w-4" />
   695→              <span>Sécurisé par Stripe</span>
   696→            </div>
   697→            <a 
   698→              href="https://stripe.com/security" 
   699→              target="_blank" 
   700→              rel="noopener noreferrer"
   701→              className="hover:text-indigo-600"
   702→            >
   703→              Sécurité →
   704→            </a>
   705→          </div>
   706→        </CardFooter>
   707→      )}
   708→    </Card>
   709→  )
   710→}
