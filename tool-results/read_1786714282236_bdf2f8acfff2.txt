   249→  /**
   250→   * Get or create a user session
   251→   */
   252→  getOrCreateSession(sessionId: string, context?: Partial<SessionAnalytics['context']>): SessionAnalytics {
   253→    let session = this.sessions.get(sessionId);
   254→    
   255→    if (!session || this.isSessionExpired(session)) {
   256→      session = {
   257→        sessionId,
   258→        userId: undefined,
   259→        startTime: new Date().toISOString(),
   260→        endTime: undefined,
   261→        duration: 0,
   262→        pageViews: 0,
   263→        events: 0,
   264→        bounce: true,
   265→        exitPage: '',
   266→        entryPage: context?.url || '',
   267→        pagesVisited: [],
   268→        conversions: [],
   269→        deviceType: context?.deviceType || this.detectDeviceType(),
   270→        country: context?.country,
   271→        source: context?.source,
   272→        medium: context?.medium,
   273→        campaign: context?.campaign,
   274→      };
   275→      
   276→      this.sessions.set(sessionId, session);
   277→    }
   278→
   279→    return session;
   280→  }
   281→
   282→  /**
   283→   * Update session with page view
   284→   */
   285→  recordPageView(sessionId: string, url: string): void {
   286→    const session = this.getOrCreateSession(sessionId, { url });
   287→    
   288→    session.pageViews++;
   289→    session.pagesVisited.push(url);
   290→    session.exitPage = url;
   291→    session.bounce = session.pageViews === 1;
   292→    session.events++;
   293→    
   294→    if (session.pageViews === 1) {
   295→      session.entryPage = url;
   296→    }
   297→  }
   298→
   299→  /**
   300→   * End a session
   301→   */
   302→  endSession(sessionId: string): void {
   303→    const session = this.sessions.get(sessionId);
   304→    if (session) {
   305→      session.endTime = new Date().toISOString();
   306→      session.duration = Math.round(
   307→        (new Date(session.endTime).getTime() - new Date(session.startTime).getTime()) / 1000
   308→      );
   309→    }
   310→  }

   316→  /**
   317→   * Track a business event
   318→   */
   319→  trackEvent(
   320→    eventType: UserBehaviorEvent['eventType'],
   321→    data: Record<string, any>,
   322→    context: Partial<UserBehaviorEvent['context']> = {}
   323→  ): void {
   324→    // Apply sampling
   325→    if (Math.random() > this.config.sampleRate) {
   326→      return;
   327→    }
   328→
   329→    const sessionId = data.sessionId || this.generateSessionId();
   330→    const session = this.getOrCreateSession(sessionId, context);
   331→
   332→    const event: UserBehaviorEvent = {
   333→      id: `evt_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
   334→      userId: data.userId,
   335→      sessionId,
   336→      eventType,
   337→      timestamp: new Date().toISOString(),
   338→      data,
   339→      context: {
   340→        url: context.url || typeof window !== 'undefined' ? window.location.href : '',
   341→        referrer: context.referrer || (typeof document !== 'undefined' ? document.referrer : ''),
   342→        userAgent: context.userAgent || (typeof navigator !== 'undefined' ? navigator.userAgent : ''),
   343→        ip: this.config.anonymizeIp ? undefined : context.ip,
   344→        country: context.country,
   345→        deviceType: context.deviceType || this.detectDeviceType(),
   346→        tenantId: context.tenantId,
   347→      },
   348→    };
   349→
   350→    // Store event
   351→    this.events.push(event);
   352→    session.events++;
   353→
   354→    // Update specific metrics based on event type
   355→    this.updateMetricsFromEvent(event);
   356→
   357→    // Trim old events
   358→    this.trimOldEvents();
   359→  }
   360→
   361→  /**
   362→   * Track multiple events in batch
   363→   */
   364→  trackBatch(events: Array<{
   365→    eventType: UserBehaviorEvent['eventType'];
   366→    data: Record<string, any>;
   367→    context?: Partial<UserBehaviorEvent['context']>;
   368→  }>): void {
   369→    for (const event of events) {
   370→      this.trackEvent(event.eventType, event.data, event.context || {});
   371→    }
   372→  }

   378→  /**
   379→   * Define or update a conversion funnel
   380→   */
   381→  defineFunnel(funnelId: string, name: string, steps: string[]): ConversionFunnel {
   382→    const now = new Date();
   383→    const periodStart = new Date(now.getTime() - 24 * 60 * 60 * 1000); // Last 24 hours
   384→
   385→    const funnelSteps: FunnelStep[] = steps.map((stepName, index) => {
   386→      // Count users who completed this step
   387→      const count = this.countUsersInFunnelStep(funnelId, stepName, periodStart, now);
   388→      
   389→      // Calculate percentages
   390→      const prevCount = index > 0 ? 
   391→        this.countUsersInFunnelStep(funnelId, steps[index - 1], periodStart, now) :
   392→        count;
   393→      
   394→      const firstStepCount = this.countUsersInFunnelStep(funnelId, steps[0], periodStart, now);
   395→
   396→      return {
   397→        name: stepName,
   398→        count,
   399→        percentage: prevCount > 0 ? (count / prevCount) * 100 : 100,
   400→        overallPercentage: firstStepCount > 0 ? (count / firstStepCount) * 100 : 100,
   401→        dropoffRate: prevCount > 0 ? ((prevCount - count) / prevCount) * 100 : 0,
   402→        avgTimeToComplete: this.calculateAvgTimeToComplete(funnelId, steps[index - 1], stepName),
   403→      };
   404→    });
   405→
   406→    const totalConversions = funnelSteps[funnelSteps.length - 1]?.count || 0;
   407→    const firstStepUsers = funnelSteps[0]?.count || 1;
   408→
   409→    const funnel: ConversionFunnel = {
   410→      id: funnelId,
   411→      name,
   412→      period: {
   413→        start: periodStart.toISOString(),
   414→        end: now.toISOString(),
   415→      },
   416→      steps: funnelSteps,
   417→      totalConversions,
   418→      conversionRate: (totalConversions / firstStepUsers) * 100,
   419→      revenueAttributed: this.calculateFunnelRevenue(funnelId, periodStart, now),
   420→    };
   421→
   422→    this.funnels.set(funnelId, funnel);
   423→    return funnel;
   424→  }

   469→  /**
   470→   * Get metrics for a specific product
   471→   */
   472→  getProductMetrics(productId: string, days: number = 30): ProductMetrics {
   473→    const cached = this.productMetrics.get(productId);
   474→    
   475→    if (cached && !this.isCacheStale(cached.period.end, days)) {
   476→      return cached;
   477→    }
   478→
   479→    const now = new Date();
   480→    const periodStart = new Date(now.getTime() - days * 24 * 60 * 60 * 1000);
   481→
   482→    // Filter product-related events
   483→    const productEvents = this.events.filter(e => 
   484→      e.data.productId === productId &&
   485→      new Date(e.timestamp) >= periodStart
   486→    );
   487→
   488→    // Calculate metrics
   489→    const metrics: ProductMetrics = {
   490→      productId,
   491→      period: {
   492→        start: periodStart.toISOString(),
   493→        end: now.toISOString(),
   494→      },
   495→      
   496→      // Views & Engagement
   497→      views: productEvents.filter(e => e.eventType === 'product_view').length,
   498→      uniqueViewers: new Set(productEvents.filter(e => e.eventType === 'product_view').map(e => e.userId)).size,
   499→      avgTimeOnPage: this.calculateAvgTimeOnPage(productId, productEvents),
   500→      addToCartCount: productEvents.filter(e => e.eventType === 'add_to_cart').length,
   501→      favoriteCount: productEvents.filter(e => e.eventType === 'favorite_add').length,
   502→      shares: productEvents.filter(e => e.eventType === 'share').length,
   503→      
   504→      // Conversion
   505→      rfqsReceived: productEvents.filter(e => e.eventType === 'rfq_create').length,
   506→      quotationsSent: productEvents.filter(e => e.eventType === 'quote_send').length,
   507→      ordersPlaced: productEvents.filter(e => e.eventType === 'order_place').length,
   508→      conversionRate: 0, // Calculated below
   509→      
   510→      // Search & Discovery
   511→      searchAppearances: 0, // Would need search log integration
   512→      clickThroughRate: 0,
   513→      positionInSearch: { avg: 0, median: 0, p90: 0 },
   514→      
   515→      // Reviews & Trust
   516→      reviewCount: productEvents.filter(e => e.eventType === 'review_submit').length,
   517→      avgRating: this.calculateAvgRating(productId),
   518→      responseRate: 0, // Would need supplier data
   519→      avgResponseTime: 0,
   520→      
   521→      // Revenue
   522→      revenueGenerated: this.calculateProductRevenue(productId, productEvents),
   523→      revenuePerView: 0, // Calculated below
   524→    };
   525→
   526→    // Calculate derived metrics
   527→    metrics.conversionRate = metrics.views > 0 ? (metrics.ordersPlaced / metrics.views) * 100 : 0;
   528→    metrics.revenuePerView = metrics.views > 0 ? metrics.revenueGenerated / metrics.views : 0;
   529→
   530→    // Cache result
   531→    this.productMetrics.set(productId, metrics);
   532→    return metrics;
   533→  }

   538→  getTopProducts(metric: keyof Pick<ProductMetrics, 'views' | 'conversionRate' | 'revenueGenerated'>, limit: number = 10): ProductMetrics[] {
   539→    const allProducts = Array.from(this.productMetrics.values());

   541→    return allProducts
   542→      .sort((a, b) => (b[metric] as number) - (a[metric] as number))
   543→      .slice(0, limit);

   619→  /**
   620→   * Record a revenue event
   621→   */
   622→  recordRevenue(data: {
   623→    amount: number;
   624→    currency: string;
   625→    paymentMethod: string;
   626→    orderId: string;
   627→    userId: string;
   628→    category?: string;
   629→    isRecurring?: boolean;

   631→    couponCode?: string;
   632→  }): void {
   633→    const today = new Date().toISOString().split('T')[0];
   634→    
   635→    let dayMetric = this.revenueHistory.find(r => r.timestamp.startsWith(today));
   636→    
   637→    if (!dayMetric) {
   638→      dayMetric = {
   639→        timestamp: today,
   640→        period: 'daily',
   641→        grossRevenue: 0,
   642→        netRevenue: 0,
   643→        taxCollected: 0,
   644→        discountsGiven: 0,
   645→        byPaymentMethod: {},
   646→        byCategory: {},
   647→        byCountry: {},
   648→        byPlan: {},
   649→        transactionCount: 0,
   650→        avgTransactionValue: 0,
   651→        recurringRevenue: 0,
   652→        newCustomerRevenue: 0,
   653→        returningCustomerRevenue: 0,
   654→        projectedMonthEnd: 0,
   655→        growthRate: 0,
   656→      };
   657→      this.revenueHistory.push(dayMetric);
   658→    }
   659→
   660→    // Update totals
   661→    dayMetric.grossRevenue += data.amount;
   662→    dayMetric.transactionCount++;
   663→    
   664→    if (data.isRecurring) {
   665→      dayMetric.recurringRevenue += data.amount;
   666→    } else if (data.isNewCustomer) {
   667→      dayMetric.newCustomerRevenue += data.amount;
   668→    } else {
   669→      dayMetric.returningCustomerRevenue += data.amount;
   670→    }
   671→
   672→    // Update breakdowns
   673→    dayMetric.byPaymentMethod[data.paymentMethod] = 
   674→      (dayMetric.byPaymentMethod[data.paymentMethod] || 0) + data.amount;
   675→    
   676→    if (data.category) {
   677→      dayMetric.byCategory[data.category] = 
   678→        (dayMetric.byCategory[data.category] || 0) + data.amount;
   679→    }
   680→
   681→    // Recalculate averages
   682→    dayMetric.avgTransactionValue = dayMetric.grossRevenue / dayMetric.transactionCount;
   683→  }

   685→  /**
   686→   * Get revenue metrics for a period
   687→   */
   688→  getRevenueMetrics(period: 'daily' | 'weekly' | 'monthly', periods: number = 7): RevenueMetric[] {
   689→    const sorted = [...this.revenueHistory]
   690→      .sort((a, b) => b.timestamp.localeCompare(a.timestamp))
   691→      .slice(0, periods)
   692→      .reverse();
   693→
   694→    // Calculate growth rates and projections
   695→    return sorted.map((metric, index) => {
   696→      if (index > 0) {
   697→        const prevMetric = sorted[index - 1];
   698→        metric.growthRate = prevMetric.grossRevenue > 0 
   699→          ? ((metric.grossRevenue - prevMetric.grossRevenue) / prevMetric.grossRevenue) * 100 
   700→          : 0;
   701→        
   702→        // Simple linear projection for month end
   703→        const daysPassed = index + 1;
   704→        const daysInMonth = new Date(new Date().getFullYear(), new Date().getMonth() + 1, 0).getDate();
   705→        metric.projectedMonthEnd = metric.grossRevenue * (daysInMonth / daysPassed);
   706→      }
   707→
   708→      return metric;
   709→    });
   710→  }

   742→      const lastActivity = s.endTime ? new Date(s.endTime).getTime() : now.getTime();

   744→    });
   745→
   746→    // Bounce rate

   748→    const bounceRate = sessionsArray.length > 0 ? (bouncedSessions.length / sessionsArray.length) * 100 : 0;
   749→
   750→    // Average session duration
   751→    const totalDuration = sessionsArray.reduce((sum, s) => sum + s.duration, 0);
   752→    const avgDuration = sessionsArray.length > 0 ? totalDuration / sessionsArray.length : 0;
   753→
   754→    // Get funnels
   755→    const ecommerceFunnel = this.getEcommerceFunnel();
   756→    const rfqFunnel = this.getRfqFunnel();
   757→    const activationFunnel = this.getActivationFunnel();
   758→
   759→    return {
   760→      overview: {
   761→        totalSessions: sessionsArray.length,
   762→        activeUsers: activeSessions.length,
   763→        totalEvents: this.events.length,
   764→        bounceRate: Math.round(bounceRate * 100) / 100,
   765→        avgSessionDuration: Math.round(avgDuration),
   766→        conversionRate: ecommerceFunnel.conversionRate,
   767→      },
   768→      funnels: [ecommerceFunnel, rfqFunnel, activationFunnel],
   769→      topProducts: this.getTopProducts('revenueGenerated', 5),

   903→  private aggregateEventsByDay(days: number): Array<{ date: string; count: number }> {
   904→    const result: Array<{ date: string; count: number }> = [];
   905→    const now = new Date();
   906→
   907→    for (let i = days - 1; i >= 0; i--) {
   908→      const date = new Date(now);
   909→      date.setDate(date.getDate() - i);
   910→      const dateStr = date.toISOString().split('T')[0];
   911→
   912→      const count = this.events.filter(e => 
   913→        e.timestamp.startsWith(dateStr)
   914→      ).length;
   915→
   916→      result.push({ date: dateStr, count });
   917→    }
   918→
   919→    return result;
   920→  }
