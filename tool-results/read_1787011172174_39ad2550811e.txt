   683→// MAIN EXPORT FUNCTION
   684→// ============================================
   685→
   686→export function getContractTemplate(
   687→  type: ContractType,
   688→  language: ContractLanguage = 'BILINGUAL'
   689→): ContractTemplate {
   690→  switch (type) {
   691→    case 'SALES_AGREEMENT':
   692→      return createSalesAgreementTemplate(language);
   693→    case 'SUPPLY_CONTRACT':
   694→      return createSupplyContractTemplate(language);
   695→    case 'SERVICE_AGREEMENT':
   696→      return createServiceAgreementTemplate(language);
   697→    case 'NON_DISCLOSURE':
   698→      return createNDATemplate(language);
   699→    case 'DISTRIBUTION_AGREEMENT':
   700→      return createDistributionAgreementTemplate(language);
   701→    case 'EXCLUSIVITY':
   702→      return createExclusivityTemplate(language);
   703→    case 'FRAMEWORK_AGREEMENT':
   704→      return createFrameworkAgreementTemplate(language);
   705→    default:
   706→      return createSalesAgreementTemplate(language);
   707→  }
   708→}
