   233→  // Dialog states

   241→  const [isLoading, setIsLoading] = useState(false);
   242→
   243→  // Filter content
   244→  const filteredContent = useMemo(() => {
   245→    let result = [...content];
   246→    
   247→    if (activeTab !== 'all') {
   248→      if (activeTab === 'flagged') {
   249→        result = result.filter(c => c.status === 'FLAGGED');
   250→      } else {
   251→        result = result.filter(c => c.status === activeTab.toUpperCase());
   252→      }
   253→    }
   254→    
   255→    if (typeFilter !== 'ALL') {
   256→      result = result.filter(c => c.type === typeFilter);
   257→    }
   258→    
   259→    if (searchQuery.trim()) {
   260→      const query = searchQuery.toLowerCase();
   261→      result = result.filter(c => 
   262→        c.title.toLowerCase().includes(query) ||
   263→        c.authorName.toLowerCase().includes(query) ||
   264→        c.companyName?.toLowerCase().includes(query)
   265→      );
   266→    }
   267→    
   268→    result.sort((a, b) => {
   269→      if (a.status === 'FLAGGED') return -1;
   270→      if (b.status === 'FLAGGED') return 1;
   271→      if (a.status === 'PENDING' && b.status !== 'PENDING') return -1;
   272→      if (a.status !== 'PENDING' && b.status === 'PENDING') return 1;
   273→      return new Date(b.submittedAt).getTime() - new Date(a.submittedAt).getTime();
   274→    });
   275→    
   276→    return result;
   277→  }, [content, activeTab, searchQuery, typeFilter]);
   278→
   279→  // Stats
   280→  const pendingCount = content.filter(c => c.status === 'PENDING').length;
   281→  const flaggedCount = content.filter(c => c.status === 'FLAGGED').length;
   282→  const approvedToday = content.filter(c => 
   283→    c.status === 'APPROVED' && 
   284→    c.reviewedAt && 
   285→    new Date(c.reviewedAt).toDateString() === new Date().toDateString()
   286→  ).length;
   287→  const totalViews = content.reduce((sum, c) => sum + c.views, 0);
   288→
   289→  // Handlers
   290→  const formatDuration = (seconds: number) => {
   291→    const mins = Math.floor(seconds / 60);
   292→    const secs = seconds % 60;
   293→    return `${mins}:${secs.toString().padStart(2, '0')}`;
   294→  };
   295→
   296→  const formatDate = (dateString: string) => {
   297→    return new Date(dateString).toLocaleDateString('fr-FR', {
   298→      day: '2-digit',
   299→      month: '2-digit',
   300→      year: 'numeric'
   301→    });
   302→  };
   303→
   304→  const openPreviewDialog = (item: ContentItem) => {
   305→    setSelectedContent(item);
   306→    setPreviewDialogOpen(true);
   307→  };
   308→
   309→  const openActionDialog = (item: ContentItem, action: 'approve' | 'reject') => {
   310→    setSelectedContent(item);
   311→    setActionType(action);
   312→    setActionReason('');
   313→    setActionDialogOpen(true);
   314→  };
   315→
   316→  const handleAction = async () => {
   317→    if (!selectedContent) return;
   318→    
   319→    setIsLoading(true);
   320→    await new Promise(resolve => setTimeout(resolve, 800));
   321→    
   322→    setContent(prev => prev.map(c => 
   323→      c.id === selectedContent.id 
   324→        ? { 
   325→            ...c, 
   326→            status: actionType === 'approve' ? 'APPROVED' as const : 'REJECTED' as const,
   327→            reviewedAt: new Date().toISOString(),
   328→            reviewedBy: 'Admin Current',
   329→            rejectionReason: actionType === 'reject' ? actionReason : undefined,
   330→          }
   331→        : c
   332→    ));
   333→    
   334→    setIsLoading(false);
   335→    setActionDialogOpen(false);
   336→    setPreviewDialogOpen(false);
   337→  };
   338→
   339→  const handleClearFlag = async (itemId: string) => {
   340→    setContent(prev => prev.map(c => 
   341→      c.id === itemId ? { ...c, status: 'APPROVED' as const, flagReason: undefined, flagNotes: undefined } : c
   342→    ));
   343→  };
   344→
   345→  const handleRemoveContent = async (itemId: string) => {
   346→    setContent(prev => prev.map(c => 
   347→      c.id === itemId ? { ...c, status: 'REMOVED' as const } : c
   348→    ));
   349→  };
   350→
   351→  const exportToCSV = () => {
   352→    const headers = ['ID', 'Titre', 'Type', 'Auteur', 'Entreprise', 'Durée', 'Statut', 'Vues', 'Date soumission'];
   353→    const csvRows = filteredContent.map(c => [
   354→      c.id,

   356→      typeConfig[c.type].label,
   357→      c.authorName,
   358→      c.companyName || '',
   359→      formatDuration(c.duration),
   360→      statusConfig[c.status].label,
   361→      c.views.toString(),
   362→      formatDate(c.submittedAt)
   363→    ]);
   364→    
   365→    const csvContent = [headers, ...csvRows]
   366→      .map(row => row.map(cell => `"${cell}"`).join(','))
   367→      .join('\n');
   368→    
   369→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   370→    const link = document.createElement('a');
   371→    link.href = URL.createObjectURL(blob);
   372→    link.download = `contenu_${new Date().toISOString().split('T')[0]}.csv`;
   373→    link.click();
   374→  };

   378→      {/* Header */}

   380→        <div>

   391→        <StatsCard
   392→          title="En attente"
   393→          value={pendingCount}
   394→          description="À modérer"
   395→          icon={Clock}
   396→          iconClassName={pendingCount > 0 ? "bg-yellow-100 text-yellow-600 animate-pulse" : "bg-gray-100 text-gray-600"}
   397→        />
   398→        <StatsCard
   399→          title="Signalés"
   400→          value={flaggedCount}
   401→          description="Nécessitent attention"

   407→          value={approvedToday}
   408→          description="Contenu validé"
   409→          icon={CheckCircle2}
   410→          iconClassName="bg-green-100 text-green-600"
   411→        />
   412→        <StatsCard
   413→          title="Total vues"
   414→          value={totalViews.toLocaleString()}
   415→          description="Toutes vidéos"
   416→          icon={Play}
   417→          iconClassName="bg-purple-100 text-purple-600"
   418→        />
   419→      </div>
   420→
   421→      {/* Filters */}
   422→      <Card>
   423→        <CardContent className="pt-6">
   424→          <div className="flex flex-col sm:flex-row gap-4">
   425→            <div className="relative flex-1">
   426→              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-gray-400" />
   427→              <Input
   428→                placeholder="Rechercher par titre ou auteur..."
   429→                value={searchQuery}
   430→                onChange={(e) => setSearchQuery(e.target.value)}
   431→                className="pl-10"
   432→              />
   433→            </div>
   434→            <Select value={typeFilter} onValueChange={setTypeFilter}>
   435→              <SelectTrigger className="w-[180px]">
   436→                <Filter className="mr-2 h-4 w-4" />
   437→                <SelectValue placeholder="Type" />
   438→              </SelectTrigger>
   439→              <SelectContent>
   440→                <SelectItem value="ALL">Tous les types</SelectItem>
   441→                <SelectItem value="PRODUCT_VIDEO">Vidéo produit</SelectItem>
   442→                <SelectItem value="COMPANY_TOUR">Visite entreprise</SelectItem>
   443→                <SelectItem value="PROMO_VIDEO">Vidéo promo</SelectItem>

   453→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   454→        <TabsList>
   455→          <TabsTrigger value="pending" className="gap-1">
   456→            <Clock className="h-3.5 w-3.5 hidden sm:inline" /> En attente ({pendingCount})
   457→          </TabsTrigger>
   458→          <TabsTrigger value="flagged" className="gap-1">
   459→            <Flag className="h-3.5 w-3.5 hidden sm:inline" /> Signalés ({flaggedCount})
   460→          </TabsTrigger>
   461→          <TabsTrigger value="approved" className="gap-1">
   462→            <CheckCircle2 className="h-3.5 w-3.5 hidden sm:inline" /> Approuvés
   463→          </TabsTrigger>
   464→          <TabsTrigger value="rejected" className="gap-1">
   465→            <XCircle className="h-3.5 w-3.5 hidden sm:inline" /> Rejetés

   467→          <TabsTrigger value="all">Tout</TabsTrigger>
   468→        </TabsList>
   469→
   470→        <TabsContent value={activeTab} className="mt-4">
   471→          <Card>
   472→            <CardContent className="p-0">
   473→              {filteredContent.length > 0 ? (
   474→                <div className="overflow-x-auto">
   475→                  <Table>
   476→                    <TableHeader>

   487→                      </TableRow>
   488→                    </TableHeader>
   489→                    <TableBody>
   490→                      {filteredContent.map((item) => {
   491→                        const status = statusConfig[item.status];
   492→                        const type = typeConfig[item.type];
   493→                        
   494→                        return (
   495→                          <TableRow 
   496→                            key={item.id}
   497→                            className={
   498→                              item.status === 'FLAGGED' 
   499→                                ? 'bg-orange-50/30 hover:bg-orange-50/50' 
   500→                                : item.status === 'PENDING'
   501→                                  ? 'bg-yellow-50/30 hover:bg-yellow-50/50'
   502→                                  : ''
   503→                            }
   504→                          >
   505→                            <TableCell>
   506→                              <div className="w-16 h-10 rounded-lg bg-gray-200 flex items-center justify-center overflow-hidden">

   508→                                  <img src={item.thumbnailUrl} alt="" className="w-full h-full object-cover" />

   511→                                )}
   512→                              </div>
   513→                            </TableCell>
   514→                            <TableCell>
   515→                              <div className="max-w-[200px]">
   516→                                <p className="font-medium truncate">{item.title}</p>
   517→                                {item.companyName && (
   518→                                  <p className="text-xs text-gray-500 truncate">{item.companyName}</p>

   520→                              </div>
   521→                            </TableCell>
   522→                            <TableCell className="hidden md:table-cell">

   527→                            </TableCell>
   528→                            <TableCell>

   533→                            </TableCell>
   534→                            <TableCell>
