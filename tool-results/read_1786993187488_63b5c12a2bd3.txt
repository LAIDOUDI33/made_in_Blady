   365→export default function ShippingManagementPage() {
   366→  const [carriers] = useState<Carrier[]>(sampleCarriers);
   367→  const [rates, setRates] = useState<ShippingRate[]>(sampleRates);
   368→  const [incoterms] = useState<IncotermConfig[]>(sampleIncoterms);
   369→  const [shipments] = useState<ShipmentTracking[]>(sampleShipments);
   370→  
   371→  const [activeTab, setActiveTab] = useState('tracking');
   372→  const [searchQuery, setSearchQuery] = useState('');
   373→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   374→  
   375→  // Dialog states
   376→  const [rateDialogOpen, setRateDialogOpen] = useState(false);
   377→  const [editingRate, setEditingRate] = useState<ShippingRate | null>(null);
   378→  
   379→  // Shipment detail dialog
   380→  const [shipmentDetailOpen, setShipmentDetailOpen] = useState(false);
   381→  const [selectedShipment, setSelectedShipment] = useState<ShipmentTracking | null>(null);
   382→
   383→  // Filter shipments
   384→  const filteredShipments = useMemo(() => {
   385→    let result = [...shipments];
   386→    
   387→    if (statusFilter !== 'ALL') {
   388→      result = result.filter(s => s.status === statusFilter);
   389→    }
   390→    
   391→    if (searchQuery.trim()) {
   392→      const query = searchQuery.toLowerCase();
   393→      result = result.filter(s => 
   394→        s.trackingNumber.toLowerCase().includes(query) ||
   395→        s.orderId.toLowerCase().includes(query) ||
   396→        s.customerName.toLowerCase().includes(query)
   397→      );
   398→    }
   399→    
   400→    result.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   401→    
   402→    return result;
   403→  }, [shipments, searchQuery, statusFilter]);
   404→
   405→  // Stats
   406→  const inTransitCount = shipments.filter(s => s.status === 'IN_TRANSIT').length;
   407→  const outForDelivery = shipments.filter(s => s.status === 'OUT_FOR_DELIVERY').length;
   408→  const deliveredToday = shipments.filter(s => 
   409→    s.status === 'DELIVERED' && 
   410→    s.actualDelivery &&
   411→    new Date(s.actualDelivery).toDateString() === new Date().toDateString()
   412→  ).length;
   413→  const avgOnTimeRate = carriers.length > 0 
   414→    ? Math.round(carriers.reduce((sum, c) => sum + c.onTimeRate, 0) / carriers.length)
   415→    : 0;
   416→
   417→  // Handlers
   418→  const formatCurrency = (amount: number) => {
   419→    return new Intl.NumberFormat('fr-DZ', { style: 'currency', currency: 'DZD' }).format(amount);
   420→  };
   421→
   422→  const formatDate = (dateString: string) => {
   423→    return new Date(dateString).toLocaleDateString('fr-FR', {
   424→      day: '2-digit',
   425→      month: '2-digit',
   426→      year: 'numeric'
   427→    });
   428→  };
   429→
   430→  const formatDateTime = (dateString: string) => {
   431→    return new Date(dateString).toLocaleDateString('fr-FR', {
   432→      day: '2-digit',
   433→      month: '2-digit',
   434→      year: 'numeric',
   435→      hour: '2-digit',
   436→      minute: '2-digit'
   437→    });
   438→  };
   439→
   440→  const openRateDialog = (rate?: ShippingRate) => {
   441→    setEditingRate(rate || {
   442→      id: '',
   443→      origin: '',
   444→      originName: '',
   445→      destination: '',
   446→      destinationName: '',
   447→      carrierId: '',
   448→      carrierName: '',
   449→      method: 'GROUND',
   450→      basePrice: 0,
   451→      currency: 'DZD',
   452→      pricePerKg: 0,
   453→      estimatedDays: { min: 1, max: 3 },
   454→      isActive: true,
   455→    });
   456→    setRateDialogOpen(true);
   457→  };
   458→
   459→  const handleSaveRate = async () => {
   460→    if (!editingRate) return;
   461→    await new Promise(resolve => setTimeout(resolve, 500));
   462→    
   463→    if (editingRate.id) {
   464→      setRates(prev => prev.map(r => r.id === editingRate.id ? editingRate : r));
   465→    } else {
   466→      setRates(prev => [...prev, { ...editingRate, id: `RATE${Date.now()}` }]);
   467→    }
   468→    setRateDialogOpen(false);
   469→  };
   470→
   471→  const openShipmentDetail = (shipment: ShipmentTracking) => {
   472→    setSelectedShipment(shipment);
   473→    setShipmentDetailOpen(true);
   474→  };
   475→
   476→  const exportToCSV = () => {
   477→    const headers = ['Tracking', 'Commande', 'Client', 'Transporteur', 'Origine', 'Destination', 'Statut', 'Poids', 'Coût', 'Est. livraison'];
   478→    const csvRows = filteredShipments.map(s => [
   479→      s.trackingNumber,
   480→      s.orderId,
   481→      s.customerName,
   482→      s.carrierName,
   483→      s.origin,
   484→      s.destination,
   485→      statusConfig[s.status].label,
   486→      `${s.weight} kg`,
   487→      formatCurrency(s.cost),
   488→      s.estimatedDelivery
   489→    ]);
   490→    
   491→    const csvContent = [headers, ...csvRows]
   492→      .map(row => row.map(cell => `"${cell}"`).join(','))
   493→      .join('\n');
   494→    
   495→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   496→    const link = document.createElement('a');
   497→    link.href = URL.createObjectURL(blob);
   498→    link.download = `expeditions_${new Date().toISOString().split('T')[0]}.csv`;
   499→    link.click();
   500→  };
   501→
   502→  return (
   503→    <div className="space-y-6">
   504→      {/* Header */}
   505→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   506→        <div>
   507→          <h1 className="text-2xl font-bold text-gray-900">Configuration Expédition</h1>
   508→          <p className="text-gray-500 mt-1">Gérez les transporteurs, tarifs et suivez les expéditions</p>
   509→        </div>
   510→        <Button variant="outline" onClick={exportToCSV}>
   511→          <Download className="mr-2 h-4 w-4" /> Exporter CSV
   512→        </Button>
   513→      </div>
   514→
   515→      {/* Stats Cards */}
   516→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   517→        <StatsCard
   518→          title="En transit"
   519→          value={inTransitCount}
   520→          description="Expéditions en cours"
   521→          icon={Truck}
   522→          iconClassName="bg-purple-100 text-purple-600"
   523→        />
   524→        <StatsCard
   525→          title="En livraison"
   526→          value={outForDelivery}
   527→          description="Aujourd'hui"
   528→          icon={Package}
   529→          iconClassName="bg-green-100 text-green-600"
   530→        />
   531→        <StatsCard
   532→          title="Livrées aujourd'hui"
   533→          value={deliveredToday}
   534→          description="Avec succès"
   535→          icon={CheckCircle2}
   536→          iconClassName="bg-blue-100 text-blue-600"
   537→        />
   538→        <StatsCard
   539→          title="Taux ponctualité"
   540→          value={`${avgOnTimeRate}%`}
   541→          description="Moyenne transporteurs"
   542→          icon={TrendingUp}
   543→          iconClassName="bg-orange-100 text-orange-600"
   544→        />
   545→      </div>
   546→
   547→      {/* Main Tabs */}
   548→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   549→        <TabsList>
