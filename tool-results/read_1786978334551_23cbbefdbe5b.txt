   187→  /**
   188→   * Initialize the monitoring system
   189→   */
   190→  async initialize(): Promise<void> {
   191→    if (this.isInitialized) return;
   192→

   194→      // Load custom rules from database if available

   196→        where: { enabled: true },

   198→      
   199→      if (customRules.length > 0) {
   200→        this.rules = [...DEFAULT_ALERT_RULES, ...customRules.map(r => ({
   201→          ...r,
   202→          conditions: JSON.parse(r.conditions as string),
   203→          actions: JSON.parse(r.actions as string),
   204→        } as AlertRule))];
   205→      }
   206→      
   207→      this.isInitialized = true;
   208→      console.log('[SecurityMonitor] Initialized with', this.rules.length, 'active rules');
   209→    } catch (error) {
   210→      console.error('[SecurityMonitor] Initialization error:', error);
   211→      // Continue with default rules
   212→      this.isInitialized = true;
   213→    }
   214→  }

   216→  /**
   217→   * Process a security event - main entry point
   218→   */
   219→  async recordEvent(eventData: Omit<SecurityEvent, 'id' | 'timestamp' | 'resolved'>): Promise<SecurityEvent> {
   220→    if (!MONITORING_CONFIG.enabled) {
   221→      return {} as SecurityEvent;
   222→    }
   223→
   224→    await this.initialize();
   225→
   226→    const event: SecurityEvent = {
   227→      ...eventData,
   228→      id: `evt_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
   229→      timestamp: new Date(),
   230→      resolved: false,
   231→    };
   232→
   233→    // Add to buffer for real-time analysis
   234→    this.eventBuffer.push(event);
   235→    if (this.eventBuffer.length > 10000) {
   236→      this.eventBuffer = this.eventBuffer.slice(-5000); // Keep last 5000 in memory
   237→    }
   238→
   239→    // Track by IP and user
   240→    if (event.ipAddress) {
   241→      const ipEvents = this.recentEventsByIP.get(event.ipAddress) || [];
   242→      ipEvents.push(event);
   243→      if (ipEvents.length > 1000) ipEvents.shift(); // Keep last 1000 per IP
   244→      this.recentEventsByIP.set(event.ipAddress, ipEvents);
   245→    }
   246→
   247→    if (event.userId) {
   248→      const userEvents = this.recentEventsByUser.get(event.userId) || [];
   249→      userEvents.push(event);
   250→      if (userEvents.length > 500) userEvents.shift();
   251→      this.recentEventsByUser.set(event.userId, userEvents);
   252→    }
   253→
   254→    // Persist to database asynchronously
   255→    this.persistEvent(event);
   256→
   257→    // Evaluate alert rules
   258→    await this.evaluateRules(event);
   259→
   260→    return event;
   261→  }

   263→  /**
   264→   * Calculate risk score for an IP address (0-100)
   265→   */
   266→  calculateIPRiskScore(ipAddress: string): number {
   267→    const events = this.recentEventsByIP.get(ipAddress) || [];
   268→    if (events.length === 0) return 0;
   269→
   270→    let riskScore = 0;
   271→    const now = Date.now();
   272→    const oneHourAgo = now - MONITORING_CONFIG.windows.mediumTerm;
   273→
   274→    // Filter to recent events
   275→    const recentEvents = events.filter(e => e.timestamp.getTime() > oneHourAgo);
   276→
   277→    // Weight different event types
   278→    for (const event of recentEvents) {
   279→      switch (event.eventType) {
   280→        case 'brute_force_detected':
   281→          riskScore += 25;
   282→          break;
   283→        case 'injection_attempt':
   284→          riskScore += 35;
   285→          break;
   286→        case 'xss_attempt':
   287→          riskScore += 20;
   288→          break;
   289→        case 'auth_failure':
   290→          riskScore += 10;
   291→          break;
   292→        case 'rate_limit_exceeded':
   293→          riskScore += 15;
   294→          break;
   295→        case 'payment_fraud_risk':
   296→          riskScore += 30;
   297→          break;
   298→        case 'account_takeover_risk':
   299→          riskScore += 40;
   300→          break;
   301→        default:
   302→          riskScore += 5;
   303→      }
   304→    }
   305→
   306→    // Severity multiplier
   307→    const highSeverityCount = recentEvents.filter(e => 
   308→      e.severity === 'high' || e.severity === 'critical'
   309→    ).length;
   310→    riskScore *= (1 + highSeverityCount * 0.2);
   311→
   312→    // Frequency bonus (rapid attacks are worse)
   313→    if (recentEvents.length > 20) {
   314→      riskScore *= 1.3;
   315→    }
   316→    if (recentEvents.length > 50) {
   317→      riskScore *= 1.5;
   318→    }
   319→
   320→    return Math.min(100, Math.round(riskScore));
   321→  }
   322→
   323→  /**
   324→   * Calculate risk score for a user account (0-100)
   325→   */
   326→  calculateUserRiskScore(userId: string): number {
   327→    const events = this.recentEventsByUser.get(userId) || [];
   328→    if (events.length === 0) return 0;
   329→
   330→    let riskScore = 0;
   331→    const now = Date.now();
   332→    const twentyFourHoursAgo = now - MONITORING_CONFIG.windows.longTerm;
   333→
   334→    const recentEvents = events.filter(e => e.timestamp.getTime() > twentyFourHoursAgo);
   335→
   336→    for (const event of recentEvents) {
   337→      switch (event.eventType) {
   338→        case 'account_takeover_risk':
   339→          riskScore += 40;
   340→          break;
   341→        case 'permission_denied':
   342→          riskScore += 15;
   343→          break;
   344→        case 'data_access_anomaly':
   345→          riskScore += 25;
   346→          break;
   347→        case 'unusual_geolocation':
   348→          riskScore += 20;
   349→          break;
   350→        default:
   351→          riskScore += 5;
   352→      }
   353→    }
   354→
   355→    // Check for multiple IPs (potential account sharing/takeover)
   356→    const uniqueIPs = new Set(recentEvents.map(e => e.ipAddress).filter(Boolean));
   357→    if (uniqueIPs.size > 5) {
   358→      riskScore += 15;
   359→    }
   360→
   361→    return Math.min(100, Math.round(riskScore));
   362→  }

   364→  /**
   365→   * Get current monitoring statistics
   366→   */
   367→  async getStats(): Promise<MonitoringStats> {
   368→    const oneHourAgo = new Date(Date.now() - MONITORING_CONFIG.windows.mediumTerm);
   369→    
   370→    const [totalEvents, eventsByType, blockedIPsCount, recentCritical] = await Promise.all([
   371→      db.securityEvent.count({
   372→        where: { createdAt: { gte: oneHourAgo } },
   373→      }),
   374→      db.securityEvent.groupBy({
   375→        by: ['eventType'],
   376→        _count: { eventType: true },
   377→        where: { createdAt: { gte: oneHourAgo } },
   378→        orderBy: { _count: { eventType: 'desc' } },
   379→        take: 10,
   380→      }),
   381→      db.blockedIP.count({ where: { expiresAt: { gt: new Date() } } }),
   382→      db.securityEvent.findMany({
   383→        where: {
   384→          createdAt: { gte: oneHourAgo },
   385→          severity: { in: ['critical', 'high'] },
   386→          resolved: false,
   387→        },
   388→        orderBy: { createdAt: 'desc' },
   389→        take: 10,
   390→      }),
   391→    ]);
   392→
   393→    // Calculate overall platform risk score
   394→    const highRiskIPs = Array.from(this.recentEventsByIP.entries())
   395→      .filter(([, events]) => this.calculateIPRiskScoreFromEvents(events) > 70);
   396→    
   397→    const overallRisk = Math.min(100, Math.round(
   398→      (highRiskIPs.length * 10) + 
   399→      (recentCritical.length * 15) +
   400→      (totalEvents > 1000 ? 20 : totalEvents / 50)
   401→    ));
   402→
   403→    // Top attacker IPs
   404→    const topAttackers = Array.from(this.recentEventsByIP.entries())
   405→      .map(([ip, events]) => ({ ip, count: events.length }))
   406→      .sort((a, b) => b.count - a.count)
   407→      .slice(0, 10);
   408→
   409→    return {
   410→      totalEvents,
   411→      eventsByType: Object.fromEntries(eventsByType.map(e => [e.eventType, e._count.eventType])),
   412→      eventsBySeverity: {}, // Would need another query
   413→      activeAlerts: recentCritical.length,
   414→      blockedIPs: blockedIPsCount,
   415→      riskScore: overallRisk,
   416→      topAttackerIPs: topAttackers,
   417→      recentCriticalEvents: recentCritical.map(e => ({
   418→        ...e,
   419→        details: typeof e.details === 'string' ? JSON.parse(e.details) : e.details,
   420→      })),
   421→    };
   422→  }

   501→  /**
   502→   * Evaluate alert rules against an event
   503→   */
   504→  private async evaluateRules(event: SecurityEvent): Promise<void> {
   505→    for (const rule of this.rules) {
   506→      if (!rule.enabled) continue;
   507→      if (rule.eventType !== event.eventType) continue;
   508→
   509→      // Check cooldown
   510→      const cooldownKey = `${rule.id}_${event.ipAddress || event.userId || 'global'}`;
   511→      const lastAlert = this.alertCooldowns.get(cooldownKey);
   512→      if (lastAlert && Date.now() - lastAlert < rule.cooldownMs) continue;
   513→
   514→      // Evaluate conditions
   515→      const conditionsMet = rule.conditions.every(condition =>
   516→        this.evaluateCondition(event, condition)
   517→      );
   518→
   519→      if (conditionsMet) {
   520→        // Set cooldown
   521→        this.alertCooldowns.set(cooldownKey, Date.now());
   522→
   523→        // Execute actions
   524→        await this.executeActions(rule, event);
   525→      }
   526→    }
   527→  }
   528→
   529→  /**
   530→   * Evaluate a single condition
   531→   */
   532→  private evaluateCondition(event: SecurityEvent, condition: AlertCondition): boolean {
   533→    const fieldValue = (event as Record<string, unknown>)[condition.field];
   534→    
   535→    switch (condition.operator) {
   536→      case 'equals':
   537→        return fieldValue === condition.value;
   538→      case 'not_equals':
   539→        return fieldValue !== condition.value;
   540→      case 'contains':
   541→        return String(fieldValue).includes(String(condition.value));
   542→      case 'greater_than':
   543→        return Number(fieldValue) > Number(condition.value);
   544→      case 'less_than':
   545→        return Number(fieldValue) < Number(condition.value);
   546→      case 'in':
   547→        return Array.isArray(condition.value) && condition.value.includes(fieldValue);
   548→      default:
   549→        return false;
   550→    }
   551→  }

   553→  /**
   554→   * Execute alert actions
   555→   */
   556→  private async executeActions(rule: AlertRule, event: SecurityEvent): Promise<void> {
   557→    console.log(`[SecurityMonitor] Rule "${rule.name}" triggered by event ${event.id}`);
   558→
   559→    for (const action of rule.actions) {
   560→      if (!action.enabled) continue;
   561→
   562→      try {
   563→        switch (action.type) {
   564→          case 'block_ip':
   565→            if (event.ipAddress) {
   566→              await this.blockIPAddress(event.ipAddress, action.config.durationHours as number || 24);
   567→            }
   568→            break;
   569→
   570→          case 'lock_account':
   571→            if (event.userId) {
   572→              await this.lockUserAccount(event.userId);
   573→            }
   574→            break;
   575→
   576→          case 'email':
   577→            await this.sendAlertEmail(rule.name, event, action.config.recipients as string[]);
   578→            break;
   579→
   580→          case 'webhook':
   581→            await this.sendWebhook(action.config.url as string, rule, event);
   582→            break;
   583→
   584→          case 'log':
   585→            console.log(`[SECURITY ALERT] [${rule.name}]`, {
   586→              eventId: event.id,
   587→              type: event.eventType,
   588→              severity: event.severity,
   589→              ip: event.ipAddress,
   590→              user: event.userId,
   591→              details: event.details,
   592→            });
   593→            break;
   594→
   595→          case 'slack':
   596→            await this.sendSlackNotification(rule.name, event, action.config.webhookUrl as string);
   597→            break;
   598→        }
   599→      } catch (error) {
   600→        console.error(`[SecurityMonitor] Error executing action ${action.type}:`, error);
   601→      }
   602→    }
   603→  }

   616→        expiresAt,
   617→        reason: 'Automated block by security monitor',

   624→  /**
   625→   * Lock a user account
   626→   */
   627→  private async lockUserAccount(userId: string): Promise<void> {
   628→    await db.user.update({

   695→  /**
   696→   * Helper to calculate risk from events array
   697→   */
   698→  private calculateIPRiskScoreFromEvents(events: SecurityEvent[]): number {
   699→    let risk = 0;
   700→    for (const event of events) {
   701→      switch (event.severity) {
   702→        case 'critical': risk += 30; break;
   703→        case 'high': risk += 20; break;
   704→        case 'medium': risk += 10; break;
   705→        case 'low': risk += 5; break;
   706→      }
   707→    }
   708→    return Math.min(100, risk);
   709→  }
   710→}

   717→/**
   718→ * Record authentication attempt
   719→ */
   720→export async function recordAuthAttempt(params: {
   721→  success: boolean;
   722→  userId?: string;
   723→  ipAddress: string;
   724→  userAgent?: string;
   725→  method?: 'password' | 'oauth' | '2fa';
   726→}): Promise<void> {
   727→  const { success, userId, ipAddress, userAgent, method } = params;
   728→  
   729→  // Check for brute force pattern
   730→  const recentFailures = await db.securityEvent.count({
   731→    where: {
   732→      eventType: 'AUTH_FAILURE',
   733→      ipAddress,
   734→      createdAt: { gte: new Date(Date.now() - 15 * 60 * 1000) }, // Last 15 min
   735→    },
   736→  });
   737→
   738→  if (!success && recentFailures >= 5) {
   739→    // Potential brute force attack
   740→    await securityMonitor.recordEvent({
   741→      eventType: 'brute_force_detected',
   742→      severity: recentFailures >= 10 ? 'critical' : 'high',
   743→      source: 'authentication',
   744→      ipAddress,
   745→      userId,
   746→      details: {
   747→        method,
   748→        failureCount: recentFailures + 1,
   749→        userAgent,
   750→        windowMinutes: 15,
   751→      },
   752→    });
   753→  } else {
   754→    await securityMonitor.recordEvent({
   755→      eventType: success ? 'auth_success' : 'auth_failure',
   756→      severity: success ? 'info' : 'low',
   757→      source: 'authentication',
   758→      ipAddress,
   759→      userId,
   760→      details: { method, userAgent },
   761→    });
   762→  }
   763→}

   789→/**
   790→ * Record injection attempt
   791→ */
   792→export async function recordInjectionAttempt(params: {
   793→  type: 'sql' | 'nosql' | 'xss' | 'command';
   794→  ipAddress: string;
   795→  userId?: string;
   796→  payload?: string;
   797→  endpoint?: string;
   798→}): Promise<void> {
   799→  await securityMonitor.recordEvent({
   800→    eventType: params.type === 'xss' ? 'xss_attempt' : 'injection_attempt',
   801→    severity: 'high',
   802→    source: 'waf',
   803→    ipAddress: params.ipAddress,
   804→    userId: params.userId,
   805→    details: {
   806→      injectionType: params.type,
   807→      payload: params.payload?.substring(0, 500), // Truncate for storage
   808→      endpoint: params.endpoint,
   809→    },
   810→  });
   811→
   812→  // Auto-block for injection attempts
   813→  const riskScore = securityMonitor.calculateIPRiskScore(params.ipAddress);
   814→  if (riskScore >= MONITORING_CONFIG.thresholds.autoBlockIP) {
   815→    await securityMonitor.blockIPAddress(params.ipAddress, 48); // 48 hours for injection
   816→  }
   817→}
