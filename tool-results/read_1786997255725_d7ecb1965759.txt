   131→export abstract class BaseERPClient implements ERPClient {
   132→  abstract config: ERPConfig
   133→
   134→  abstract testConnection(): Promise<boolean>
   135→  abstract pullEntities(entityType: EntityType, options?: SyncOptions): Promise<SyncResult>
   136→  abstract pushEntities(entityType: EntityType, data: any[], options?: SyncOptions): Promise<SyncResult>
   137→  abstract getEntity(entityType: EntityType, externalId: string): Promise<any>
   138→  abstract createEntity(entityType: EntityType, data: any): Promise<string>
   139→  abstract updateEntity(entityType: EntityType, externalId: string, data: any): Promise<boolean>
   140→  abstract deleteEntity(entityType: EntityType, externalId: string): Promise<boolean>
   141→
   142→  // Common utility methods
   143→  protected transformData(data: any, mappings: FieldMapping[], direction: 'toERP' | 'fromERP'): any {
   144→    const result: any = {}
   145→    
   146→    for (const mapping of mappings) {
   147→      const sourceField = direction === 'toERP' ? mapping.localField : mapping.erpField
   148→      const targetField = direction === 'toERP' ? mapping.erpField : mapping.localField
   149→      
   150→      let value = this.getNestedValue(data, sourceField)
   151→      
   152→      if (value === undefined || value === null) {
   153→        if (mapping.defaultValue !== undefined) {
   154→          value = mapping.defaultValue
   155→        } else {
   156→          continue
   157→        }
   158→      }
   159→      
   160→      // Apply transformation if specified
   161→      if (mapping.transform) {
   162→        value = this.applyTransform(value, mapping.transform)
   163→      }
   164→      
   165→      this.setNestedValue(result, targetField, value)
   166→    }
   167→    
   168→    return result
   169→  }
   170→
   171→  protected getNestedValue(obj: any, path: string): any {
   172→    return path.split('.').reduce((acc, key) => acc?.[key], obj)
   173→  }
   174→
   175→  protected setNestedValue(obj: any, path: string, value: any): void {
   176→    const keys = path.split('.')
   177→    let current = obj
   178→    
   179→    for (let i = 0; i < keys.length - 1; i++) {
   180→      if (!(keys[i] in current) || typeof current[keys[i]] !== 'object') {
   181→        current[keys[i]] = {}
   182→      }
   183→      current = current[keys[i]]
   184→    }
   185→    
   186→    current[keys[keys.length - 1]] = value
   187→  }
   188→
   189→  protected applyTransform(value: any, transformName: string): any {
   190→    const transforms: Record<string, (v: any) => any> = {
   191→      'uppercase': (v) => String(v).toUpperCase(),
   192→      'lowercase': (v) => String(v).toLowerCase(),
   193→      'trim': (v) => String(v).trim(),
   194→      'toString': (v) => String(v),
   195→      'toNumber': (v) => Number(v),
   196→      'toFloat': (v) => parseFloat(v),
   197→      'toInt': (v) => parseInt(v, 10),
   198→      'toBoolean': (v) => Boolean(v),
   199→      'toDate': (v) => new Date(v),
   200→      'toISODate': (v) => new Date(v).toISOString(),
   201→      'encodeBase64': (v) => Buffer.from(String(v)).toString('base64'),
   202→      'decodeBase64': (v) => Buffer.from(String(v), 'base64').toString(),
   203→      'extractNumbers': (v) => String(v).replace(/[^0-9.-]/g, ''),
   204→      'formatPrice': (v) => parseFloat(v).toFixed(2),
   205→      'formatDate': (v) => new Date(v).toISOString().split('T')[0],
   206→      'mapCurrency': (v) => {
   207→        const currencyMap: Record<string, string> = { 'DZD': 'DZ', 'EUR': 'EUR', 'USD': 'USD' }
   208→        return currencyMap[String(v)] || v
   209→      },
   210→      'mapLanguage': (v) => {
   211→        const langMap: Record<string, string> = { 'FR': 'fr_FR', 'AR': 'ar_DZ', 'EN': 'en_US' }
   212→        return langMap[String(v).toUpperCase()] || v
   213→      },
   214→    }
   215→    
   216→    const transform = transforms[transformName]
   217→    if (!transform) {
   218→      console.warn(`Unknown transform: ${transformName}`)
   219→      return value
   220→    }
   221→    
   222→    try {
   223→      return transform(value)
   224→    } catch (error) {
   225→      console.error(`Error applying transform ${transformName}:`, error)
   226→      return value
   227→    }
   228→  }
   229→
   230→  protected async createSyncLog(
   231→    erpConfigId: string,
   232→    entityType: EntityType,
   233→    direction: 'PUSH' | 'PULL',
   234→    status: SyncStatus,
   235→    stats: Partial<SyncLog>
   236→  ): Promise<SyncLog> {
   237→    const log = await db.eRPSyncLog.create({
   238→      data: {
   239→        id: uuidv4(),
   240→        erpConfigId,
   241→        entityType,
   242→        direction,
   243→        status,
   244→        recordsProcessed: stats.processed || 0,
   245→        recordsSuccess: stats.success ? stats.created + stats.updated : 0,
   246→        recordsFailed: stats.failed || 0,
   247→        errors: JSON.stringify(stats.errors || []),
   248→        startedAt: stats.startedAt || new Date(),
   249→        completedAt: status !== 'PENDING' ? new Date() : undefined,
   250→        durationMs: stats.durationMs,
   251→      },
   252→    })
   253→    
   254→    return mapSyncLogFromDB(log)
   255→  }
   256→
   257→  protected buildAuthHeaders(): Record<string, string> {
   258→    const headers: Record<string, string> = {
   259→      'Content-Type': 'application/json',
   260→    }
   261→    
   262→    switch (this.config.authType) {
   263→      case 'API_KEY':
   264→        if (this.config.apiKey) {
   265→          headers['Authorization'] = `Bearer ${this.config.apiKey}`
   266→          headers['X-API-Key'] = this.config.apiKey
   267→        }
   268→        break
   269→        
   270→      case 'BASIC':
   271→        if (this.config.username && this.config.password) {
   272→          const credentials = Buffer.from(`${this.config.username}:${this.config.password}`).toString('base64')
   273→          headers['Authorization'] = `Basic ${credentials}`
   274→        }
   275→        break
   276→        
   277→      case 'OAUTH2':
   278→        // OAuth2 token should be obtained and cached separately
   279→        headers['Authorization'] = `Bearer ${this.config.apiKey || ''}`
   280→        break
   281→        
   282→      case 'CERTIFICATE':
   283→        // Certificate auth is handled at transport layer
   284→        break
   285→    }
   286→    
   287→    return headers
   288→  }
   289→}

   295→const clientCache = new Map<string, ERPClient>()
   296→
   297→export function getERPClient(config: ERPConfig): ERPClient {
   298→  const cacheKey = `${config.type}-${config.id}`
   299→  
   300→  if (clientCache.has(cacheKey)) {
   301→    return clientCache.get(cacheKey)!
   302→  }
   303→  
   304→  let client: ERPClient
   305→  
   306→  switch (config.type) {
   307→    case 'SAP':
   308→      // Import dynamically to avoid circular dependencies
   309→      const { SAPConnector } = require('./sap-connector')
   310→      client = new SAPConnector(config)
   311→      break
   312→      
   313→    case 'ODOO':
   314→      const { OdooConnector } = require('./odoo-connector')
   315→      client = new OdooConnector(config)
   316→      break
   317→      
   318→    case 'DYNAMICS':
   319→      // Dynamics connector would be implemented similarly
   320→      throw new Error('Microsoft Dynamics connector not yet implemented')
   321→      
   322→    case 'CUSTOM':
   323→      // Custom connector would use user-provided configuration
   324→      throw new Error('Custom ERP connectors must be implemented per integration')
   325→      
   326→    default:
   327→      throw new Error(`Unsupported ERP type: ${config.type}`)
   328→  }
   329→  
   330→  clientCache.set(cacheKey, client)
   331→  return client
   332→}

   360→export async function testConnection(erpConfigId: string): Promise<{success: boolean; message: string; latencyMs?: number}> {
   361→  const config = await db.eRPConfig.findUnique({ where: { id: erpConfigId } })
   362→  
   363→  if (!config) {
   364→    throw new Error('ERP configuration not found')
   365→  }
   366→  
   367→  const mappedConfig = mapERPConfigFromDB(config)
   368→  const client = getERPClient(mappedConfig)
   369→  
   370→  const startTime = Date.now()
   371→  
   372→  try {
   373→    const success = await client.testConnection()
   374→    const latencyMs = Date.now() - startTime
   375→    
   376→    // Update connection status
   377→    await db.eRPConfig.update({
   378→      where: { id: erpConfigId },
   379→      data: {
   380→        connectionStatus: success ? 'CONNECTED' : 'ERROR',
   381→        lastSyncAt: success ? new Date() : undefined,
   382→      },
   383→    })
   384→    
   385→    return {
   386→      success,
   387→      message: success 
   388→        ? 'Connexion établie avec succès' 
   389→        : 'Échec de la connexion - vérifiez vos paramètres',
   390→      latencyMs,
   391→    }
   392→  } catch (error: any) {
   393→    await db.eRPConfig.update({
   394→      where: { id: erpConfigId },
   395→      data: { connectionStatus: 'ERROR' },
   396→    })
   397→    
   398→    return {
   399→      success: false,
   400→      message: error.message || 'Erreur de connexion inattendue',
   401→      latencyMs: Date.now() - startTime,
   402→    }
   403→  }
   404→}

   406→export async function syncEntity(
   407→  erpConfigId: string,
   408→  entityType: EntityType,
   409→  options?: SyncOptions
   410→): Promise<SyncLog> {
   411→  const config = await db.eRPConfig.findUnique({ where: { id: erpConfigId } })
   412→  
   413→  if (!config) {
   414→    throw new Error('ERP configuration not found')
   415→  }
   416→  
   417→  if (!config.enabled) {
   418→    throw new Error('ERP configuration is disabled')
   419→  }
   420→  
   421→  const mappedConfig = mapERPConfigFromDB(config)
   422→  const client = getERPClient(mappedConfig)
   423→  
   424→  const directions = JSON.parse(config.syncDirections || '{}')
   425→  const direction = (directions[entityType] || 'PULL') as 'PUSH' | 'PULL'
   426→  
   427→  // Create pending sync log
   428→  const syncLog = await client.createSyncLog(erpConfigId, entityType, direction, 'PENDING', {
   429→    startedAt: new Date(),
   430→  })
   431→  
   432→  const startTime = Date.now()
   433→  
   434→  try {
   435→    let result: SyncResult
   436→    
   437→    if (direction === 'PULL') {
   438→      result = await client.pullEntities(entityType, options)
   439→    } else {
   440→      // For PUSH, we need to get local data first
   441→      const localData = await getLocalEntitiesForSync(entityType, options?.entityIds)
   442→      result = await client.pushEntities(entityType, localData, options)
   443→    }
   444→    
   445→    const durationMs = Date.now() - startTime
   446→    
   447→    // Update sync log with results
   448→    const finalStatus = result.failed > 0 && result.success === 0 
   449→      ? 'FAILED' 
   450→      : result.failed > 0 
   451→        ? 'PARTIAL' 
   452→        : 'SUCCESS'
   453→    
   454→    await db.eRPSyncLog.update({
   455→      where: { id: syncLog.id },
   456→      data: {
   457→        status: finalStatus,
   458→        recordsProcessed: result.processed,
   459→        recordsSuccess: result.created + result.updated,
   460→        recordsFailed: result.failed,
   461→        errors: JSON.stringify(result.errors || []),
   462→        completedAt: new Date(),
   463→        durationMs,
   464→      },
   465→    })
   466→    
   467→    // Update last sync timestamp on config
   468→    await db.eRPConfig.update({
   469→      where: { id: erpConfigId },
   470→      data: { lastSyncAt: new Date() },
   471→    })
   472→    
   473→    return {
   474→      ...syncLog,
   475→      status: finalStatus,
   476→      recordsProcessed: result.processed,
   477→      recordsSuccess: result.created + result.updated,
   478→      recordsFailed: result.failed,
   479→      errors: result.errors || [],
   480→      completedAt: new Date(),
   481→      durationMs,
   482→    }
   483→  } catch (error: any) {
   484→    const durationMs = Date.now() - startTime
   485→    
   486→    await db.eRPSyncLog.update({
   487→      where: { id: syncLog.id },
   488→      data: {
   489→        status: 'FAILED',
   490→        errors: JSON.stringify([{
   491→          message: error.message || 'Sync failed unexpectedly',
   492→          code: 'SYNC_ERROR',
   493→          details: error.stack,
   494→        }]),
   495→        completedAt: new Date(),
   496→        durationMs,
   497→      },
   498→    })
   499→    
   500→    return {
   501→      ...syncLog,
   502→      status: 'FAILED',
   503→      errors: [{
   504→        message: error.message || 'Sync failed unexpectedly',
   505→        code: 'SYNC_ERROR',
   506→      }],
   507→      completedAt: new Date(),
   508→      durationMs,
   509→    }
   510→  }
   511→}

   513→export async function syncAll(erpConfigId: string): Promise<SyncLog[]> {
   514→  const config = await db.eRPConfig.findUnique({ where: { id: erpConfigId } })
   515→  
   516→  if (!config) {
   517→    throw new Error('ERP configuration not found')
   518→  }
   519→  
   520→  const directions = JSON.parse(config.syncDirections || '{}')
   521→  const entitiesToSync = Object.entries(directions)
   522→    .filter(([, dir]) => dir !== 'NONE')
   523→    .map(([entity]) => entity as EntityType)
   524→  
   525→  const logs: SyncLog[] = []
   526→  
   527→  for (const entityType of entitiesToSync) {
   528→    try {
   529→      const log = await syncEntity(erpConfigId, entityType)
   530→      logs.push(log)
   531→    } catch (error) {
   532→      console.error(`Failed to sync ${entityType}:`, error)
   533→    }
   534→  }
   535→  
   536→  return logs
   537→}

   659→export async function handleERPWebhook(
   660→  erpConfigId: string,
   661→  payload: unknown,
   662→  signature: string
   663→): Promise<void> {
   664→  const config = await db.eRPConfig.findUnique({ where: { id: erpConfigId } })
   665→  
   666→  if (!config) {
   667→    throw new Error('ERP configuration not found')
   668→  }
   669→  
   670→  if (!config.webhookSecret) {
   671→    throw new Error('Webhook secret not configured')
   672→  }
   673→  
   674→  // Verify signature (simplified - implement proper HMAC verification in production)
   675→  const expectedSignature = generateWebhookSignature(payload, config.webhookSecret)
   676→  if (signature !== expectedSignature) {
   677→    throw new Error('Invalid webhook signature')
   678→  }
   679→  
   680→  // Process webhook payload
   681→  const data = payload as Record<string, any>
   682→  const entityType = (data.entity_type || data.entityType)?.toUpperCase() as EntityType
   683→  const action = data.action || data.event // create, update, delete
   684→  
   685→  if (!entityType || !action) {
   686→    throw new Error('Invalid webhook payload: missing entity_type or action')
   687→  }
   688→  
   689→  // Handle based on action
   690→  switch (action.toLowerCase()) {
   691→    case 'create':
   692→    case 'created':
   693→      await handleWebhookCreate(erpConfigId, entityType, data)
   694→      break
   695→      
   696→    case 'update':
   697→    case 'updated':
   698→      await handleWebhookUpdate(erpConfigId, entityType, data)
   699→      break
   700→      
   701→    case 'delete':
   702→    case 'deleted':
   703→      await handleWebhookDelete(erpConfigId, entityType, data)
   704→      break
   705→      
   706→    default:
   707→      console.log(`Unhandled webhook action: ${action}`)
   708→  }
   709→}

   748→export async function getSyncHistory(
   749→  erpConfigId: string,
   750→  filters?: {
   751→    entityType?: EntityType
   752→    status?: SyncStatus
   753→    dateFrom?: Date
   754→    dateTo?: Date
   755→    limit?: number
   756→    offset?: number
   757→  }
   758→): Promise<{data: SyncLog[], total: number}> {
   759→  const where: any = { erpConfigId }
   760→  
   761→  if (filters?.entityType) where.entityType = filters.entityType
   762→  if (filters?.status) where.status = filters.status
   763→  if (filters?.dateFrom || filters?.dateTo) {
   764→    where.startedAt = {}
   765→    if (filters.dateFrom) where.startedAt.gte = filters.dateFrom
   766→    if (filters.dateTo) where.startedAt.lte = filters.dateTo
   767→  }
   768→  
   769→  const [logs, total] = await Promise.all([
   770→    db.eRPSyncLog.findMany({
   771→      where,
   772→      orderBy: { startedAt: 'desc' },
   773→      take: filters?.limit || 50,
   774→      skip: filters?.offset || 0,
   775→    }),
   776→    db.eRPSyncLog.count({ where }),
   777→  ])
   778→  
   779→  return {
   780→    data: logs.map(mapSyncLogFromDB),
   781→    total,
   782→  }
   783→}
