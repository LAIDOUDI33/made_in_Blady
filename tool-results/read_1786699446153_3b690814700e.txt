   545→/**

   551→
   552→/**
   553→ * Get all unique categories
   554→ */
   555→export function getCategories(): { name: ApiCategory; count: number }[] {
   556→  const categoryMap = new Map<ApiCategory, number>();
   557→  
   558→  for (const endpoint of API_ENDPOINTS) {
   559→    const current = categoryMap.get(endpoint.category) || 0;
   560→    categoryMap.set(endpoint.category, current + 1);
   561→  }
   562→
   563→  return Array.from(categoryMap.entries()).map(([name, count]) => ({ name, count }));
   564→}

   903→/**
   904→ * Extract OpenAPI parameters from endpoint definition
   905→ */
   906→function extractParameters(endpoint: ApiEndpoint): any[] {
   907→  const params: any[] = [];
   908→  
   909→  // Path parameters
   910→  if (endpoint.path.includes('{slug}')) {
   911→    params.push({
   912→      name: 'slug',
   913→      in: 'path',
   914→      required: true,
   915→      description: 'Resource identifier (slug)',
   916→      schema: { type: 'string' },
   917→      example: 'panneau-solaire-300w',
   918→    });
   919→  }
   920→  
   921→  if (endpoint.path.includes('{id}')) {
   922→    params.push({
   923→      name: 'id',
   924→      in: 'path',
   925→      required: true,
   926→      description: 'Resource ID',
   927→      schema: { type: 'string' },
   928→    });
   929→  }
   930→
   931→  // Query parameters for GET requests
   932→  if (endpoint.method === 'GET' && endpoint.requestSchema?.properties) {
   933→    for (const [name, schema] of Object.entries(endpoint.requestSchema.properties as any)) {
   934→      params.push({
   935→        name,
   936→        in: 'query',
   937→        required: endpoint.requestSchema.required?.includes(name) || false,
   938→        description: (schema as any).description,
   939→        schema: {
   940→          type: (schema as any).type,
   941→          enum: (schema as any).enum,
   942→          default: (schema as any).default,
   943→        },
   944→      });
   945→    }
   946→  }
   947→  
   948→  return params;
   949→}
