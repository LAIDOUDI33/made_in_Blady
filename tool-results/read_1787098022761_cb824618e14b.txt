   363→    
   364→    if (avgScore >= 3.5) return 'Excellent'

   382→        call.callerCompany.toLowerCase().includes(query) ||

   418→        return <Badge className="bg-red-100 text-red-700 border-red-200">Poor</Badge>
   419→      default:
   420→        return <Badge variant="secondary">{quality}</Badge>
   421→    }
   422→  }
   423→
   424→  const getQualityColor = (quality: CallRecord['quality']) => {
   425→    switch (quality) {
   426→      case 'EXCELLENT': return 'text-emerald-500'
   427→      case 'GOOD': return 'text-blue-500'
   428→      case 'FAIR': return 'text-yellow-500'
