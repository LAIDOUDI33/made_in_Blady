   256→export default function EscrowManagementPage() {
   257→  const [escrows, setEscrows] = useState<EscrowAccount[]>(sampleEscrows);
   258→  const [disputes] = useState<Dispute[]>(sampleDisputes);
   259→  const [activeTab, setActiveTab] = useState('escrows');
   260→  const [searchQuery, setSearchQuery] = useState('');
   261→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   262→  
   263→  // Detail dialog states
   264→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   265→  const [selectedEscrow, setSelectedEscrow] = useState<EscrowAccount | null>(null);
   266→  
   267→  // Action dialog states
   268→  const [actionDialogOpen, setActionDialogOpen] = useState(false);
   269→  const [actionType, setActionType] = useState<'release' | 'refund'>('release');
   270→  const [actionReason, setActionReason] = useState('');
   271→  const [isLoading, setIsLoading] = useState(false);
   272→
   273→  // Dispute detail dialog
   274→  const [disputeDialogOpen, setDisputeDialogOpen] = useState(false);
   275→  const [selectedDispute, setSelectedDispute] = useState<Dispute | null>(null);
   276→
   277→  // Filter escrows
   278→  const filteredEscrows = useMemo(() => {
   279→    let result = [...escrows];
   280→    
   281→    if (statusFilter !== 'ALL') {
   282→      result = result.filter(e => e.status === statusFilter);
   283→    }
   284→    
   285→    if (searchQuery.trim()) {
   286→      const query = searchQuery.toLowerCase();
   287→      result = result.filter(e => 
   288→        e.id.toLowerCase().includes(query) ||
   289→        e.orderId.toLowerCase().includes(query) ||
   290→        e.buyerName.toLowerCase().includes(query) ||
   291→        e.sellerName.toLowerCase().includes(query)
   292→      );
   293→    }
   294→    
   295→    result.sort((a, b) => {
   296→      if (a.status === 'DISPUTED') return -1;
   297→      if (b.status === 'DISPUTED') return 1;
   298→      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   299→    });
   300→    
   301→    return result;
   302→  }, [escrows, searchQuery, statusFilter]);
   303→
   304→  // Stats
   305→  const totalInEscrow = escrows
   306→    .filter(e => e.status === 'ACTIVE' || e.status === 'PENDING_RELEASE' || e.status === 'DISPUTED')
   307→    .reduce((sum, e) => sum + e.amount, 0);
   308→  
   309→  const pendingReleases = escrows.filter(e => e.status === 'PENDING_RELEASE').length;
   310→  const activeDisputes = disputes.filter(d => d.status === 'OPEN' || d.status === 'INVESTIGATING').length;
   311→  const totalReleasedThisMonth = escrows
   312→    .filter(e => e.status === 'RELEASED' && e.releaseDate && 
   313→      new Date(e.releaseDate).getMonth() === new Date().getMonth())
   314→    .reduce((sum, e) => sum + e.amount, 0);
   315→
   316→  // Handlers
   317→  const formatCurrency = (amount: number) => {
   318→    return new Intl.NumberFormat('fr-DZ', { style: 'currency', currency: 'DZD' }).format(amount);
   319→  };
   320→
   321→  const formatDate = (dateString: string) => {
   322→    return new Date(dateString).toLocaleDateString('fr-FR', {
   323→      day: '2-digit',
   324→      month: '2-digit',
   325→      year: 'numeric'
   326→    });
   327→  };
   328→
   329→  const openDetailDialog = (escrow: EscrowAccount) => {
   330→    setSelectedEscrow(escrow);
   331→    setDetailDialogOpen(true);
   332→  };
   333→
   334→  const openActionDialog = (escrow: EscrowAccount, action: 'release' | 'refund') => {
   335→    setSelectedEscrow(escrow);
   336→    setActionType(action);
   337→    setActionReason('');
   338→    setActionDialogOpen(true);
   339→  };
   340→
   341→  const handleAction = async () => {
   342→    if (!selectedEscrow) return;
   343→    
   344→    setIsLoading(true);
   345→    await new Promise(resolve => setTimeout(resolve, 800));
   346→    
   347→    setEscrows(prev => prev.map(e => 
   348→      e.id === selectedEscrow.id 
   349→        ? { 
   350→            ...e, 
   351→            status: actionType === 'release' ? 'RELEASED' as const : 'REFUNDED' as const,
   352→            releaseDate: new Date().toISOString(),
   353→          }
   354→        : e
   355→    ));
   356→    
   357→    setIsLoading(false);
   358→    setActionDialogOpen(false);
   359→    setDetailDialogOpen(false);
   360→  };
   361→
   362→  const openDisputeDialog = (dispute: Dispute) => {
   363→    setSelectedDispute(dispute);
   364→    setDisputeDialogOpen(true);
   365→  };
   366→
   367→  const exportToCSV = () => {
   368→    const headers = ['ID', 'Commande', 'Acheteur', 'Vendeur', 'Montant', 'Statut', 'Date création'];
   369→    const csvRows = filteredEscrows.map(e => [
   370→      e.id,
   371→      e.orderId,
   372→      e.buyerName,
   373→      e.sellerName,
   374→      formatCurrency(e.amount),
   375→      statusConfig[e.status].label,
   376→      formatDate(e.createdAt)
   377→    ]);
   378→    
   379→    const csvContent = [headers, ...csvRows]
   380→      .map(row => row.map(cell => `"${cell}"`).join(','))
   381→      .join('\n');
   382→    
   383→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   384→    const link = document.createElement('a');
   385→    link.href = URL.createObjectURL(blob);
   386→    link.download = `escrows_${new Date().toISOString().split('T')[0]}.csv`;
   387→    link.click();
   388→  };
   389→
   390→  return (
   391→    <div className="space-y-6">
   392→      {/* Header */}
   393→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   394→        <div>
   395→          <h1 className="text-2xl font-bold text-gray-900">Gestion des Comptes Escrow</h1>
   396→          <p className="text-gray-500 mt-1">Surveillez les transactions sécurisées et gérez les litiges</p>
   397→        </div>
   398→        <Button variant="outline" onClick={exportToCSV}>
   399→          <Download className="mr-2 h-4 w-4" /> Exporter CSV
   400→        </Button>
   401→      </div>
   402→
   403→      {/* Stats Cards */}
   404→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   405→        <StatsCard
   406→          title="Total en escrow"
   407→          value={formatCurrency(totalInEscrow)}
   408→          description={`${escrows.filter(e => e.status === 'ACTIVE' || e.status === 'PENDING_RELEASE' || e.status === 'DISPUTED').length} comptes actifs`}
   409→          icon={Lock}
   410→          iconClassName="bg-blue-100 text-blue-600"
   411→        />
   412→        <StatsCard
   413→          title="Libérations en attente"
   414→          value={pendingReleases}
   415→          description="À traiter"
   416→          icon={Clock}
   417→          iconClassName={pendingReleases > 0 ? "bg-yellow-100 text-yellow-600 animate-pulse" : "bg-gray-100 text-gray-600"}
   418→        />
   419→        <StatsCard
   420→          title="Litiges actifs"
   421→          value={activeDisputes}
   422→          description={`${disputes.length} total ce mois`}
   423→          icon={Gavel}
   424→          iconClassName={activeDisputes > 0 ? "bg-red-100 text-red-600" : "bg-gray-100 text-gray-600"}
   425→        />
   426→        <StatsCard
   427→          title="Libéré ce mois"
   428→          value={formatCurrency(totalReleasedThisMonth)}
   429→          description="Volume total"
   430→          icon={TrendingUp}
   431→          iconClassName="bg-green-100 text-green-600"
   432→        />
   433→      </div>
   434→
   435→      {/* Main Tabs */}
   436→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   437→        <TabsList>
   438→          <TabsTrigger value="escrows" className="gap-1">
   439→            <ShieldCheck className="h-4 w-4" /> Comptes Escrow ({escrows.length})
   440→          </TabsTrigger>
   441→          <TabsTrigger value="disputes" className="gap-1">
   442→            <Gavel className="h-4 w-4" /> Litiges ({disputes.length})
   443→            {activeDisputes > 0 && (
   444→              <Badge variant="destructive" className="ml-1 h-5 min-w-5 px-1.5 text-xs">
   445→                {activeDisputes}
   446→              </Badge>
   447→            )}
   448→          </TabsTrigger>
   449→        </TabsList>
   450→
   451→        {/* Escrows Tab */}
   452→        <TabsContent value="escrows" className="mt-4 space-y-4">
   453→          {/* Filters */}
   454→          <Card>
   455→            <CardContent className="pt-6">
   456→              <div className="flex flex-col sm:flex-row gap-4">
   457→                <div className="relative flex-1">
   458→                  <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-gray-400" />
   459→                  <Input
   460→                    placeholder="Rechercher par ID, commande ou nom..."
   461→                    value={searchQuery}
   462→                    onChange={(e) => setSearchQuery(e.target.value)}
   463→                    className="pl-10"
   464→                  />
   465→                </div>
   466→                <Select value={statusFilter} onValueChange={setStatusFilter}>
   467→                  <SelectTrigger className="w-[180px]">
   468→                    <Filter className="mr-2 h-4 w-4" />
   469→                    <SelectValue placeholder="Statut" />
   470→                  </SelectTrigger>
   471→                  <SelectContent>
   472→                    <SelectItem value="ALL">Tous les statuts</SelectItem>
   473→                    <SelectItem value="ACTIVE">Actif</SelectItem>
   474→                    <SelectItem value="PENDING_RELEASE">En attente libération</SelectItem>
   475→                    <SelectItem value="DISPUTED">Litigieux</SelectItem>
   476→                    <SelectItem value="RELEASED">Libéré</SelectItem>
   477→                    <SelectItem value="REFUNDED">Remboursé</SelectItem>
   478→                  </SelectContent>
   479→                </Select>
   480→              </div>
   481→            </CardContent>
   482→          </Card>
   483→
   484→          {/* Escrows Table */}
   485→          <Card>
   486→            <CardContent className="p-0">
   487→              {filteredEscrows.length > 0 ? (
   488→                <div className="overflow-x-auto">
   489→                  <Table>
   490→                    <TableHeader>
   491→                      <TableRow className="bg-gray-50 hover:bg-gray-50">
   492→                        <TableHead>ID</TableHead>
   493→                        <TableHead>Commande</TableHead>
   494→                        <TableHead>Acheteur</TableHead>
   495→                        <TableHead className="hidden lg:table-cell">Vendeur</TableHead>
   496→                        <TableHead>Montant</TableHead>
   497→                        <TableHead>Statut</TableHead>
   498→                        <TableHead className="hidden md:table-cell">Créé le</TableHead>
   499→                        <TableHead className="w-12">Actions</TableHead>
   500→                      </TableRow>
   501→                    </TableHeader>
   502→                    <TableBody>
   503→                      {filteredEscrows.map((escrow) => {
   504→                        const status = statusConfig[escrow.status];
   505→                        
   506→                        return (
   507→                          <TableRow 
   508→                            key={escrow.id}
   509→                            className={
   510→                              escrow.status === 'DISPUTED' 
   511→                                ? 'bg-red-50/30 hover:bg-red-50/50' 
   512→                                : escrow.status === 'PENDING_RELEASE'
   513→                                  ? 'bg-yellow-50/30 hover:bg-yellow-50/50'
   514→                                  : ''
   515→                            }
   516→                          >
   517→                            <TableCell>
   518→                              <code className="text-xs bg-gray-100 px-1.5 py-0.5 rounded">{escrow.id}</code>
   519→                            </TableCell>
   520→                            <TableCell>
   521→                              <span className="font-medium">{escrow.orderId}</span>
   522→                            </TableCell>
   523→                            <TableCell>
   524→                              <div>
   525→                                <div className="font-medium text-sm">{escrow.buyerName}</div>
   526→                                <div className="text-xs text-gray-500 truncate max-w-[120px]">{escrow.buyerEmail}</div>
   527→                              </div>
   528→                            </TableCell>
   529→                            <TableCell className="hidden lg:table-cell">
   530→                              <div>
   531→                                <div className="text-sm">{escrow.sellerName}</div>
   532→                                <div className="text-xs text-gray-500 truncate max-w-[120px]">{escrow.sellerEmail}</div>
   533→                              </div>
