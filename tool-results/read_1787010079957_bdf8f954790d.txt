    77→    77→const CRYPTO_ICONS: Record<CryptoCurrency, typeof Bitcoin> = {

    83→    83→
    84→    84→export default function CryptoPaymentForm({
    85→    85→  orderId,
    86→    86→  amountDZD,
    87→    87→  onPaymentComplete,
    88→    88→  onPaymentError,
    89→    89→}: CryptoPaymentFormProps) {
    90→    90→  const [selectedCrypto, setSelectedCrypto] = useState<CryptoCurrency>('USDT')
    91→    91→  const [paymentData, setPaymentData] = useState<PaymentData | null>(null)
    92→    92→  const [paymentStatus, setPaymentStatus] = useState<PaymentStatus | null>(null)
    93→    93→  const [exchangeRate, setExchangeRate] = useState<number | null>(null)
    94→    94→  const [isLoading, setIsLoading] = useState(false)
    95→    95→  const [isCreating, setIsCreating] = useState(false)
    96→    96→  const [timeRemaining, setTimeRemaining] = useState<number>(0)
    97→    97→  const [copied, setCopied] = useState(false)
    98→    98→  const [walletAddress, setWalletAddress] = useState('')
    99→    99→  const [showQRCode, setShowQRCode] = useState(true)
   100→   100→  
   101→   101→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   102→   102→  const { toast } = useToast()
   103→   103→
   104→   104→  // Fetch exchange rate when crypto selection changes
   105→   105→  const fetchExchangeRate = useCallback(async (crypto: CryptoCurrency) => {
   106→   106→    setIsLoading(true)
   107→   107→    try {
   108→   108→      const response = await fetch('/api/payments/crypto/rates')
   109→   109→      const data = await response.json()
   110→   110→      
   111→   111→      if (data.success && data.data.rates[crypto]) {
   112→   112→        setExchangeRate(data.data.rates[crypto])
   113→   113→      }
   114→   114→    } catch (error) {
   115→   115→      console.error('Failed to fetch exchange rate:', error)
   116→   116→    } finally {
   117→   117→      setIsLoading(false)
   118→   118→    }
   119→   119→  }, [])
   120→   120→
   121→   121→  useEffect(() => {
   122→   122→    fetchExchangeRate(selectedCrypto)
   123→   123→  }, [selectedCrypto, fetchExchangeRate])
   124→   124→
   125→   125→  // Create crypto payment
   126→   126→  const handleCreatePayment = async () => {
   127→   127→    setIsCreating(true)
   128→   128→    try {
   129→   129→      const response = await fetch('/api/payments/crypto', {
   130→   130→        method: 'POST',
   131→   131→        headers: { 'Content-Type': 'application/json' },
   132→   132→        body: JSON.stringify({
   133→   133→          orderId,
   134→   134→          amount: amountDZD,
   135→   135→          currency: selectedCrypto,
   136→   136→          buyerWalletAddress: walletAddress || undefined,
   137→   137→        }),
   138→   138→      })
   139→   139→
   140→   140→      const data = await response.json()
   141→   141→
   142→   142→      if (data.success) {
   143→   143→        setPaymentData(data.data)
   144→   144→        startCountdown(new Date(data.data.expiresAt))
   145→   145→        startStatusMonitoring(data.data.paymentId)
   146→   146→        
   147→   147→        toast({
   148→   148→          title: 'Payment Created',
   149→   149→          description: `Send ${data.data.expectedAmount} ${selectedCrypto} to the address below.`,
   150→   150→        })
   151→   151→      } else {
   152→   152→        throw new Error(data.error || 'Failed to create payment')
   153→   153→      }
   154→   154→    } catch (error) {
   155→   155→      console.error('Error creating payment:', error)
   156→   156→      const errorMessage = error instanceof Error ? error.message : 'Unknown error'
   157→   157→      onPaymentError?.(errorMessage)
   158→   158→      toast({
   159→   159→        title: 'Error',
   160→   160→        description: errorMessage,
   161→   161→        variant: 'destructive',
   162→   162→      })
   163→   163→    } finally {
   164→   164→      setIsCreating(false)
   165→   165→    }
   166→   166→  }
   167→   167→
   168→   168→  // Countdown timer
   169→   169→  const startCountdown = (expiresAt: Date) => {
   170→   170→    const updateTimer = () => {
   171→   171→      const remaining = expiresAt.getTime() - Date.now()
   172→   172→      setTimeRemaining(Math.max(0, remaining))
   173→   173→      
   174→   174→      if (remaining <= 0) {
   175→   175→        if (intervalRef.current) clearInterval(intervalRef.current)
   176→   176→        setPaymentStatus(prev => prev ? { ...prev, status: 'EXPIRED' } : null)
   177→   177→      }
   178→   178→    }
   179→   179→
   180→   180→    updateTimer()
   181→   181→    intervalRef.current = setInterval(updateTimer, 1000)
   182→   182→  }
   183→   183→
   184→   184→  // Monitor payment status
   185→   185→  const startStatusMonitoring = async (paymentId: string) => {
   186→   186→    const checkStatus = async () => {
   187→   187→      try {
   188→   188→        const response = await fetch(`/api/payments/crypto/${paymentId}/status`)
   189→   189→        const data = await response.json()
   190→   190→
   191→   191→        if (data.success) {
   192→   192→          setPaymentStatus({
   193→   193→            status: data.data.status,
   194→   194→            confirmations: data.data.confirmations,
   195→   195→            requiredConfirmations: data.data.requiredConfirmations,
   196→   196→            actualAmount: data.data.actualAmount,
   197→   197→            txHash: data.data.txHash,
   198→   198→          })
   199→   199→
   200→   200→          // Notify on confirmation
   201→   201→          if (data.data.status === 'CONFIRMED') {
   202→   202→            onPaymentComplete?.(paymentId)
   203→   203→            toast({
   204→   204→              title: 'Payment Confirmed! 🎉',
   205→   205→              description: 'Your cryptocurrency payment has been confirmed.',
   206→   206→            })
   207→   207→            if (intervalRef.current) clearInterval(intervalRef.current)
   208→   208→          }
   209→   209→
   210→   210→          // Stop if expired or final state
   211→   211→          if (['CONFIRMED', 'EXPIRED'].includes(data.data.status)) {
   212→   212→            if (intervalRef.current) clearInterval(intervalRef.current)
   213→   213→          }
   214→   214→        }
   215→   215→      } catch (error) {
   216→   216→        console.error('Error checking status:', error)
   217→   217→      }
   218→   218→    }
   219→   219→
   220→   220→    // Initial check
   221→   221→    await checkStatus()
   222→   222→    
   223→   223→    // Poll every 10 seconds
   224→   224→    const statusInterval = setInterval(checkStatus, 10000)
   225→   225→    
   226→   226→    // Store reference for cleanup
   227→   227→    intervalRef.current = statusInterval
   228→   228→  }
   229→   229→
   230→   230→  // Copy address to clipboard
   231→   231→  const copyAddress = async () => {
   232→   232→    if (!paymentData) return
   233→   233→    
   234→   234→    try {
   235→   235→      await navigator.clipboard.writeText(paymentData.depositAddress)
   236→   236→      setCopied(true)
   237→   237→      toast({
   238→   238→        title: 'Copied!',
   239→   239→        description: 'Address copied to clipboard.',
   240→   240→      })
   241→   241→      setTimeout(() => setCopied(false), 2000)
   242→   242→    } catch (error) {
   243→   243→      console.error('Failed to copy:', error)
   244→   244→    }
   245→   245→  }
   246→   246→
   247→   247→  // Format time remaining
   248→   248→  const formatTime = (ms: number): string => {
   249→   249→    const minutes = Math.floor(ms / 60000)
   250→   250→    const seconds = Math.floor((ms % 60000) / 1000)
   251→   251→    return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`
   252→   252→  }
   253→   253→
   254→   254→  // Get status color
   255→   255→  const getStatusColor = (status: string) => {
   256→   256→    switch (status) {
   257→   257→      case 'PENDING': return 'bg-yellow-500'
   258→   258→      case 'CONFIRMED': return 'bg-green-500'
   259→   259→      case 'EXPIRED': return 'bg-red-500'
   260→   260→      case 'PARTIAL': return 'bg-orange-500'
   261→   261→      case 'OVERPAID': return 'bg-blue-500'
   262→   262→      default: return 'bg-gray-500'
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  // Cleanup on unmount
   267→   267→  useEffect(() => {
   268→   268→    return () => {
   269→   269→      if (intervalRef.current) clearInterval(intervalRef.current)
   270→   270→    }
   271→   271→  }, [])
   272→   272→
   273→   273→  const currentCryptoInfo = CRYPTO_INFO[selectedCrypto]
   274→   274→  const CryptoIcon = CRYPTO_ICONS[selectedCrypto]
   275→   275→  const networkFee = getNetworkFeeInfo(selectedCrypto)
   276→   276→
   277→   277→  return (
   278→   278→    <TooltipProvider>
   279→   279→      <div className="space-y-6">
   280→   280→        {/* Header */}
   281→   281→        <Card>
   282→   282→          <CardHeader>
   283→   283→            <CardTitle className="flex items-center gap-2">
   284→   284→              <Zap className="h-5 w-5 text-yellow-500" />
   285→   285→              Cryptocurrency Payment
   286→   286→            </CardTitle>
   287→   287→            <CardDescription>
   288→   288→              Pay with Bitcoin, Ethereum, Tether, or USD Coin
   289→   289→            </CardDescription>
   290→   290→          </CardHeader>
   291→   291→        </Card>
   292→   292→
   293→   293→        {!paymentData ? (
   294→   294→          /* Step 1: Select Currency & Create Payment */
   295→   295→          <Card>
   296→   296→            <CardContent className="pt-6 space-y-6">
   297→   297→              {/* Amount Display */}
   298→   298→              <div className="text-center p-4 bg-muted rounded-lg">
   299→   299→                <p className="text-sm text-muted-foreground">Amount to Pay</p>
   300→   300→                <p className="text-3xl font-bold">{amountDZD.toLocaleString('fr-DZ')} DZD</p>
   301→   301→              </div>
   302→   302→
   303→   303→              {/* Crypto Selector */}
   304→   304→              <div className="space-y-3">
   305→   305→                <label className="text-sm font-medium flex items-center gap-2">
   306→   306→                  Select Cryptocurrency
   307→   307→                  <Tooltip>
   308→   308→                    <TooltipTrigger asChild>
   309→   309→                      <HelpCircle className="h-4 w-4 text-muted-foreground cursor-help" />
   310→   310→                    </TooltipTrigger>
   311→   311→                    <TooltipContent side="right" className="max-w-xs">
   312→   312→                      <p>{currentCryptoInfo.description.en}</p>
   313→   313→                    </TooltipContent>
   314→   314→                  </Tooltip>
   315→   315→                </label>
   316→   316→
   317→   317→                <Select value={selectedCrypto} onValueChange={(v) => setSelectedCrypto(v as CryptoCurrency)}>
   318→   318→                  <SelectTrigger>
   319→   319→                    <SelectValue placeholder="Select cryptocurrency" />
   320→   320→                  </SelectTrigger>
   321→   321→                  <SelectContent>
   322→   322→                    {(Object.keys(CRYPTO_INFO) as CryptoCurrency[]).map((crypto) => {
   323→   323→                      const info = CRYPTO_INFO[crypto]
   324→   324→                      const Icon = CRYPTO_ICONS[crypto]
   325→   325→                      return (
   326→   326→                        <SelectItem key={crypto} value={crypto}>
   327→   327→                          <div className="flex items-center gap-2">
   328→   328→                            <Icon className="h-4 w-4" style={{ color: info.color }} />
   329→   329→                            <span>{info.fullName}</span>
   330→   330→                          </div>
   331→   331→                        </SelectItem>
   332→   332→                      )
   333→   333→                    })}
   334→   334→                  </SelectContent>
   335→   335→                </Select>
   336→   336→
   337→   337→                {/* Selected Crypto Info */}
   338→   338→                <div className="flex items-center gap-3 p-3 rounded-lg border" style={{ borderColor: currentCryptoInfo.color + '40' }}>
   339→   339→                  <div 
   340→   340→                    className="p-2 rounded-full"
   341→   341→                    style={{ backgroundColor: currentCryptoInfo.color + '20' }}
   342→   342→                  >
   343→   343→                    <CryptoIcon className="h-6 w-6" style={{ color: currentCryptoInfo.color }} />
   344→   344→                  </div>
   345→   345→                  <div className="flex-1">
   346→   346→                    <p className="font-medium">{currentCryptoInfo.name}</p>
   347→   347→                    <p className="text-sm text-muted-foreground line-clamp-1">
   348→   348→                      {currentCryptoInfo.description.en}
   349→   349→                    </p>
   350→   350→                  </div>
   351→   351→                </div>
   352→   352→              </div>
   353→   353→
   354→   354→              {/* Exchange Rate Display */}
   355→   355→              {exchangeRate && (
   356→   356→                <div className="flex items-center justify-between p-3 bg-blue-50 dark:bg-blue-950 rounded-lg">
   357→   357→                  <span className="text-sm text-muted-foreground">Exchange Rate</span>
   358→   358→                  <div className="flex items-center gap-2">
   359→   359→                    <span className="font-mono font-medium">
   360→   360→                      1 {selectedCrypto} = {Math.round(exchangeRate).toLocaleString()} DZD
   361→   361→                    </span>
   362→   362→                    <Button
   363→   363→                      variant="ghost"
   364→   364→                      size="icon"
   365→   365→                      className="h-6 w-6"
   366→   366→                      onClick={() => fetchExchangeRate(selectedCrypto)}
   367→   367→                    >
   368→   368→                      <RefreshCw className={`h-3 w-3 ${isLoading ? 'animate-spin' : ''}`} />
   369→   369→                    </Button>
   370→   370→                  </div>
   371→   371→                </div>
   372→   372→              )}
   373→   373→
   374→   374→              {/* Estimated Amount */}
   375→   375→              {exchangeRate && (
   376→   376→                <div className="text-center p-4 rounded-lg border-2 border-dashed" style={{ borderColor: currentCryptoInfo.color }}>
   377→   377→                  <p className="text-sm text-muted-foreground">You'll send approximately</p>
   378→   378→                  <p className="text-2xl font-bold" style={{ color: currentCryptoInfo.color }}>
   379→   379→                    {(amountDZD / exchangeRate).toFixed(8)} {selectedCrypto}
   380→   380→                  </p>
   381→   381→                </div>
   382→   382→              )}
   383→   383→
   384→   384→              {/* Optional Wallet Address for Refund */}
   385→   385→              <div className="space-y-2">
   386→   386→                <label className="text-sm font-medium flex items-center gap-2">
   387→   387→                  Your Wallet Address (Optional)
   388→   388→                  <Tooltip>
   389→   389→                    <TooltipTrigger asChild>
   390→   390→                      <HelpCircle className="h-4 w-4 text-muted-foreground cursor-help" />
   391→   391→                    </TooltipTrigger>
   392→   392→                    <TooltipContent side="right">
   393→   393→                      <p>Provide your wallet address for potential refunds.</p>
   394→   394→                    </TooltipContent>
   395→   395→                  </Tooltip>
   396→   396→                </label>
   397→   397→                <Input
   398→   398→                  placeholder="0x... or bc1..."
   399→   399→                  value={walletAddress}
   400→   400→                  onChange={(e) => setWalletAddress(e.target.value)}
   401→   401→                />
   402→   402→              </div>
   403→   403→
   404→   404→              {/* Network Fee Disclosure */}
   405→   405→              <div className="p-3 bg-amber-50 dark:bg-amber-950 rounded-lg space- y-2">
   406→   406→                <div className="flex items-start gap-2">
   407→   407→                  <AlertCircle className="h-4 w-4 text-amber-600 mt-0.5 shrink-0" />
   408→   408→                  <div className="text-sm text-amber-800 dark:text-amber-200">
   409→   409→                    <p className="font-medium">Network Fee Notice</p>
   410→   410→                    <p>
   411→   411→                      Additional network fees apply (~{networkFee.estimate} {networkFee.currency}). 
   412→   412→                      This is paid to miners/validators, not AlgeriaTrade.
   413→   413→                    </p>
   414→   414→                  </div>
   415→   415→                </div>
   416→   416→              </div>
   417→   417→
   418→   418→              {/* Create Payment Button */}
   419→   419→              <Button
   420→   420→                onClick={handleCreatePayment}
   421→   421→                disabled={isCreating || !exchangeRate}
   422→   422→                className="w-full"
   423→   423→                size="lg"
   424→   424→              >
   425→   425→                {isCreating ? (
   426→   426→                  <>
   427→   427→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   428→   428→                    Creating Payment...
   429→   429→                  </>
   430→   430→                ) : (
   431→   431→                  <>
   432→   432→                    <Shield className="mr-2 h-4 w-4" />
   433→   433→                    Create Payment Address
   434→   434→                  </>
   435→   435→                )}
   436→   436→              </Button>
   437→   437→            </CardContent>
   438→   438→          </Card>
   439→   439→        ) : (
   440→   440→          /* Step 2: Payment Details & QR Code */
   441→   441→          <div className="space-y-6">
   442→   442→            {/* Status Badge */}
   443→   443→            <Card>
   444→   444→              <CardContent className="pt-6">
   445→   445→                <div className="flex items-center justify-between">
   446→   446→                  <div className="flex items-center gap-3">
   447→   447→                    <Badge 
   448→   448→                      variant="secondary" 
   449→   449→                      className={`${getStatusColor(paymentStatus?.status || 'PENDING')} text-white`}
   450→   450→                    >
   451→   451→                      {paymentStatus?.status || 'PENDING'}
   452→   452→                    </Badge>
   453→   453→                    {paymentStatus?.status === 'PENDING' && (
   454→   454→                      <div className="flex items-center gap-1 text-orange-600">
   455→   455→                        <Clock className="h-4 w-4" />
   456→   456→                        <span className="font-mono font-bold">
   457→   457→                          {formatTime(timeRemaining)}
   458→   458→                        </span>
   459→   459→                      </div>
   460→   460→                    )}
   461→   461→                  </div>
   462→   462→                  
   463→   463→                  {paymentStatus?.txHash && (
   464→   464→                    <a
   465→   465→                      href={`${CRYPTO_INFO[paymentData.cryptoCurrency].explorerUrl}${paymentStatus.txHash}`}
   466→   466→                      target="_blank"
   467→   467→                      rel="noopener noreferrer"
   468→   468→                      className="flex items-center gap-1 text-sm text-blue-600 hover:underline"
   469→   469→                    >
   470→   470→                      View Transaction
   471→   471→                      <ExternalLink className="h-3 w-3" />
   472→   472→                    </a>
   473→   473→                  )}
   474→   474→                </div>
   475→   475→
   476→   476→                {/* Confirmations Progress */}
   477→   477→                {paymentStatus && paymentStatus.confirmations > 0 && (
   478→   478→                  <div className="mt-4 space-y-2">
   479→   479→                    <div className="flex justify-between text-sm">
   480→   480→                      <span>Confirmations</span>
   481→   481→                      <span>
   482→   482→                        {paymentStatus.confirmations} / {paymentStatus.requiredConfirmations}
   483→   483→                      </span>
   484→   484→                    </div>
   485→   485→                    <div className="w-full bg-gray-200 rounded-full h-2">
   486→   486→                      <div 
   487→   487→                        className="bg-green-500 h-2 rounded-full transition-all duration-500"
   488→   488→                        style={{ 
   489→   489→                          width: `${Math.min(100, (paymentStatus.confirmations / paymentStatus.requiredConfirmations) * 100)}%` 
   490→   490→                        }}
   491→   491→                      />
   492→   492→                    </div>
   493→   493→                  </div>
   494→   494→                )}
   495→   495→              </CardContent>
   496→   496→            </Card>
